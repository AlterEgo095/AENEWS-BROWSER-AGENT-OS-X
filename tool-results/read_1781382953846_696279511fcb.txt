   285→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   286→    const startTime = Date.now();
   287→    // Bridge: use LLM for strategic planning, goal decomposition, and task prioritization
   288→    if (this.bridge) {
   289→      try {
   290→        const llmResult = await this.bridge.callLLM({
   291→          systemPrompt: `You are the ${this.config.name} agent in the Meta-Intelligence cluster. Analyze the following task and provide detailed strategic planning, goal decomposition, and task prioritization.`,
   292→          userPrompt: JSON.stringify(input.payload),
   293→          temperature: 0.3,
   294→          maxTokens: 2048,
   295→        });
   296→
   297→        const analysis = llmResult.content;
   298→
   299→        return this.createAgentOutput(
   300→          input.taskId,
   301→          true,
   302→          { analysis, costUsd: llmResult.costUsd, tokensUsed: llmResult.tokenCount },
   303→          undefined,
   304→          startTime,
   305→        );
   306→      } catch (error) {
   307→        this.logger.warn(`Bridge LLM failed, fallback: ${(error as Error).message}`);
   308→      }
   309→    }
   310→
   311→    const { action, ...params } = input.payload;
   312→
   313→    if (!action) {
   314→      return this.createAgentOutput(
   315→        input.taskId,
   316→        false,
   317→        null,
   318→        'Missing required parameter: action',
   319→        startTime,
   320→      );
   321→    }
   322→
   323→    const supportedActions = [
   324→      'createPlan',
   325→      'decomposeGoal',
   326→      'prioritizeTasks',
   327→      'estimateEffort',
   328→      'identifyDependencies',
   329→      'optimizePlan',
   330→    ];
   331→
   332→    if (!supportedActions.includes(action)) {
   333→      return this.createAgentOutput(
   334→        input.taskId,
   335→        false,
   336→        null,
   337→        `Unknown planner action: ${action}. Supported: ${supportedActions.join(', ')}`,
   338→        startTime,
   339→      );
   340→    }
   341→
   342→    try {
   343→      const tool = this.getTool(action);
   344→      if (!tool) {
   345→        return this.createAgentOutput(
   346→          input.taskId,
   347→          false,
   348→          null,
   349→          `Tool not found: ${action}`,
   350→          startTime,
   351→        );
   352→      }
   353→
   354→      const result = await tool.execute(params);
   355→      await this.storeInWorkingMemory(
   356→        `planner:last:${action}`,
   357→        { params, result, timestamp: new Date() },
   358→        300000,
   359→      );
   360→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   361→    } catch (error) {
   362→      const msg = (error as Error).message;
   363→      this.logger.error(`MetaPlanner execution failed for ${action}: ${msg}`);
   364→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   365→    }
   366→  }

   375→  private async createPlan(params: {
   376→    goal: string;
   377→    constraints?: Record<string, any>;
   378→    timeframe?: string;
   379→  }): Promise<{
   380→    planId: string;
   381→    steps: Array<{ id: string; description: string; order: number; estimatedDurationMs: number }>;
   382→    estimatedDurationMs: number;
   383→    resourceRequirements: Record<string, number>;
   384→  }> {
   385→    const { goal, constraints = {}, timeframe } = params;
   386→
   387→    if (!goal || typeof goal !== 'string') {
   388→      throw new Error('Valid goal string is required for plan creation');
   389→    }
   390→
   391→    const planId = this.generateId();
   392→    const wordCount = goal.split(/\s+/).length;
   393→    const complexityFactor = Math.min(wordCount / 20, 3);
   394→
   395→    const steps: PlanStepRecord[] = [
   396→      {
   397→        id: `${planId}-step-0`,
   398→        description: `Analyze goal: ${goal.substring(0, 80)}`,
   399→        order: 0,
   400→        estimatedDurationMs: 3000 + complexityFactor * 1000,
   401→        dependsOn: [],
   402→        status: 'pending',
   403→        priority: 3,
   404→      },
   405→      {
   406→        id: `${planId}-step-1`,
   407→        description: 'Decompose goal into subtasks',
   408→        order: 1,
   409→        estimatedDurationMs: 5000 + complexityFactor * 1500,
   410→        dependsOn: [`${planId}-step-0`],
   411→        status: 'pending',
   412→        priority: 3,
   413→      },
   414→      {
   415→        id: `${planId}-step-2`,
   416→        description: 'Identify dependencies and execution order',
   417→        order: 2,
   418→        estimatedDurationMs: 4000 + complexityFactor * 1000,
   419→        dependsOn: [`${planId}-step-1`],
   420→        status: 'pending',
   421→        priority: 2,
   422→      },
   423→      {
   424→        id: `${planId}-step-3`,
   425→        description: 'Estimate effort and resource requirements',
   426→        order: 3,
   427→        estimatedDurationMs: 3000 + complexityFactor * 800,
   428→        dependsOn: [`${planId}-step-1`],
   429→        status: 'pending',
   430→        priority: 2,
   431→      },
   432→      {
   433→        id: `${planId}-step-4`,
   434→        description: 'Finalize and validate plan',
   435→        order: 4,
   436→        estimatedDurationMs: 2000 + complexityFactor * 500,
   437→        dependsOn: [`${planId}-step-2`, `${planId}-step-3`],
   438→        status: 'pending',
   439→        priority: 3,
   440→      },
   441→    ];
   442→
   443→    const estimatedDurationMs = steps.reduce((sum, s) => sum + s.estimatedDurationMs, 0);
   444→
   445→    const record: PlanRecord = {
   446→      id: planId,
   447→      goal,
   448→      steps,
   449→      status: 'draft',
   450→      createdAt: new Date(),
   451→      estimatedDurationMs,
   452→    };
   453→
   454→    this.plans.set(planId, record);
   455→
   456→    const resourceRequirements: Record<string, number> = {
   457→      planner: 1,
   458→      critic: 1,
   459→      executor: Math.max(1, Math.ceil(complexityFactor)),
   460→    };
   461→
   462→    if (constraints.maxAgents) {
   463→      resourceRequirements.executor = Math.min(
   464→        resourceRequirements.executor,
   465→        constraints.maxAgents,
   466→      );
   467→    }
   468→
   469→    this.logger.log(
   470→      `Plan created: planId=${planId}, steps=${steps.length}, duration=${estimatedDurationMs}ms`,
   471→    );
   472→
   473→    return {
   474→      planId,
   475→      steps: steps.map((s) => ({
   476→        id: s.id,
   477→        description: s.description,
   478→        order: s.order,
   479→        estimatedDurationMs: s.estimatedDurationMs,
   480→      })),
   481→      estimatedDurationMs,
   482→      resourceRequirements,
   483→    };
   484→  }

   486→  private async decomposeGoal(params: {
   487→    goal: string;
   488→    maxDepth?: number;
   489→    strategy?: string;
   490→  }): Promise<{
   491→    subtasks: Array<{ id: string; description: string; depth: number; parentTaskId?: string }>;
   492→    dependencyGraph: Record<string, string[]>;
   493→    executionOrder: string[][];
   494→  }> {
   495→    const { goal, maxDepth = 2, strategy = 'hybrid' } = params;
   496→
   497→    if (!goal || typeof goal !== 'string') {
   498→      throw new Error('Valid goal string is required for decomposition');
   499→    }
   500→
   501→    const subtasks: Array<{
   502→      id: string;
   503→      description: string;
   504→      depth: number;
   505→      parentTaskId?: string;
   506→    }> = [];
   507→    const dependencyGraph: Record<string, string[]> = {};
   508→
   509→    // Level 0: Main goal decomposition
   510→    const mainComponents = this.extractMainComponents(goal);
   511→    for (let i = 0; i < mainComponents.length; i++) {
   512→      const id = `subtask-0-${i}`;
   513→      subtasks.push({
   514→        id,
   515→        description: mainComponents[i],
   516→        depth: 0,
   517→      });
   518→      dependencyGraph[id] = [];
   519→
   520→      // Level 1: Further decomposition
   521→      if (maxDepth >= 1) {
   522→        const subComponents = this.extractSubComponents(mainComponents[i]);
   523→        for (let j = 0; j < subComponents.length; j++) {
   524→          const subId = `subtask-1-${i}-${j}`;
   525→          subtasks.push({
   526→            id: subId,
   527→            description: subComponents[j],
   528→            depth: 1,
   529→            parentTaskId: id,
   530→          });
   531→          dependencyGraph[subId] = [id];
   532→        }
   533→      }
   534→    }
   535→
   536→    // Build execution order based on strategy
   537→    const executionOrder: string[][] = [];
   538→
   539→    if (strategy === 'sequential') {
   540→      for (const subtask of subtasks) {
   541→        executionOrder.push([subtask.id]);
   542→      }
   543→    } else if (strategy === 'parallel') {
   544→      const byDepth = new Map<number, string[]>();
   545→      for (const subtask of subtasks) {
   546→        const depthTasks = byDepth.get(subtask.depth) || [];
   547→        depthTasks.push(subtask.id);
   548→        byDepth.set(subtask.depth, depthTasks);
   549→      }
   550→      for (const [, ids] of byDepth) {
   551→        executionOrder.push(ids);
   552→      }
   553→    } else {
   554→      // hybrid: group by depth but respect dependencies
   555→      const depth0 = subtasks.filter((s) => s.depth === 0).map((s) => s.id);
   556→      executionOrder.push(depth0.slice(0, 2));
   557→      if (depth0.length > 2) executionOrder.push(depth0.slice(2));
   558→      const depth1 = subtasks.filter((s) => s.depth === 1).map((s) => s.id);
   559→      if (depth1.length > 0) {
   560→        executionOrder.push(depth1.slice(0, 3));
   561→        if (depth1.length > 3) executionOrder.push(depth1.slice(3));
   562→      }
   563→    }
   564→
   565→    this.logger.log(
   566→      `Goal decomposed: goal="${goal.substring(0, 50)}", subtasks=${subtasks.length}, strategy=${strategy}`,
   567→    );
   568→
   569→    return { subtasks, dependencyGraph, executionOrder };
   570→  }

   572→  private async prioritizeTasks(params: {
   573→    tasks: Array<{ id: string; description: string; priority?: number }>;
   574→    criteria?: string[];
   575→    weights?: Record<string, number>;
   576→  }): Promise<{
   577→    prioritizedTasks: Array<{ id: string; description: string; score: number; rank: number }>;
   578→    ranking: Array<{ id: string; score: number; factors: Record<string, number> }>;
   579→    rationale: string;
   580→  }> {
   581→    const {
   582→      tasks,
   583→      criteria = ['urgency', 'impact', 'effort', 'dependencies'],
   584→      weights = {},
   585→    } = params;
   586→
   587→    if (!tasks || !Array.isArray(tasks) || tasks.length === 0) {
   588→      throw new Error('Non-empty tasks array is required for prioritization');
   589→    }
   590→
   591→    const defaultWeights: Record<string, number> = {
   592→      urgency: 0.3,
   593→      impact: 0.3,
   594→      effort: 0.2,
   595→      dependencies: 0.2,
   596→    };
   597→
   598→    const effectiveWeights = { ...defaultWeights, ...weights };
   599→
   600→    const ranked = tasks.map((task) => {
   601→      const factors: Record<string, number> = {};
   602→      let totalScore = 0;
   603→
   604→      for (const criterion of criteria) {
   605→        const weight = effectiveWeights[criterion] || 0.1;
   606→        let score: number;
   607→
   608→        switch (criterion) {
   609→          case 'urgency':
   610→            score = task.priority ?? 1;
   611→            break;
   612→          case 'impact':
   613→            score = this.estimateImpact(task.description);
   614→            break;
   615→          case 'effort':
   616→            score = 5 - this.estimateEffortScore(task.description);
   617→            break;
   618→          case 'dependencies':
   619→            score = Math.random() * 5;
   620→            break;
   621→          default:
   622→            score = 3;
   623→        }
   624→
   625→        factors[criterion] = Math.round(score * 100) / 100;
   626→        totalScore += score * weight;
   627→      }
   628→
   629→      return {
   630→        id: task.id,
   631→        description: task.description,
   632→        score: Math.round(totalScore * 100) / 100,
   633→        factors,
   634→      };
   635→    });
   636→
   637→    ranked.sort((a, b) => b.score - a.score);
   638→
   639→    const prioritizedTasks = ranked.map((r, i) => ({
   640→      id: r.id,
   641→      description: r.description,
   642→      score: r.score,
   643→      rank: i + 1,
   644→    }));
   645→
   646→    const ranking = ranked.map((r) => ({
   647→      id: r.id,
   648→      score: r.score,
   649→      factors: r.factors,
   650→    }));
   651→
   652→    const rationale = this.generatePrioritizationRationale(criteria, effectiveWeights, ranked);
   653→
   654→    this.logger.log(`Tasks prioritized: count=${tasks.length}, criteria=${criteria.join(',')}`);
   655→
   656→    return { prioritizedTasks, ranking, rationale };
   657→  }

   659→  private async estimateEffort(params: {
   660→    taskDescription: string;
   661→    complexity?: string;
   662→    resources?: Record<string, any>;
   663→  }): Promise<{
   664→    estimatedHours: number;
   665→    confidence: number;
   666→    breakdown: Record<string, number>;
   667→    riskFactors: string[];
   668→  }> {
   669→    const { taskDescription, complexity = 'medium', resources = {} } = params;
   670→
   671→    if (!taskDescription || typeof taskDescription !== 'string') {
   672→      throw new Error('Valid taskDescription string is required for effort estimation');
   673→    }
   674→
   675→    const complexityMultiplier: Record<string, number> = {
   676→      low: 0.6,
   677→      medium: 1.0,
   678→      high: 1.6,
   679→      critical: 2.2,
   680→    };
   681→
   682→    const baseHours = this.estimateBaseHours(taskDescription);
   683→    const multiplier = complexityMultiplier[complexity] || 1.0;
   684→    const estimatedHours = Math.round(baseHours * multiplier * 10) / 10;
   685→
   686→    const confidence = this.calculateConfidence(taskDescription, resources);
   687→
   688→    const breakdown: Record<string, number> = {
   689→      analysis: Math.round(estimatedHours * 0.2 * 10) / 10,
   690→      implementation: Math.round(estimatedHours * 0.45 * 10) / 10,
   691→      testing: Math.round(estimatedHours * 0.2 * 10) / 10,
   692→      review: Math.round(estimatedHours * 0.1 * 10) / 10,
   693→      buffer: Math.round(estimatedHours * 0.05 * 10) / 10,
   694→    };
   695→
   696→    const riskFactors = this.identifyEffortRiskFactors(taskDescription, complexity, resources);
   697→
   698→    this.logger.log(
   699→      `Effort estimated: hours=${estimatedHours}, complexity=${complexity}, confidence=${confidence}`,
   700→    );
   701→
   702→    return { estimatedHours, confidence, breakdown, riskFactors };
   703→  }

   705→  private async identifyDependencies(params: {
   706→    tasks: Array<{ id: string; description: string; requires?: string[] }>;
   707→    context?: Record<string, any>;
   708→  }): Promise<{
   709→    dependencies: Array<{ taskId: string; dependsOn: string[]; type: string }>;
   710→    criticalPath: string[];
   711→    parallelGroups: string[][];
   712→  }> {
   713→    const { tasks, context = {} } = params;
   714→
   715→    if (!tasks || !Array.isArray(tasks) || tasks.length === 0) {
   716→      throw new Error('Non-empty tasks array is required for dependency identification');
   717→    }
   718→
   719→    const dependencies: Array<{ taskId: string; dependsOn: string[]; type: string }> = [];
   720→
   721→    // Process explicit dependencies
   722→    for (const task of tasks) {
   723→      const explicitDeps = task.requires || [];
   724→
   725→      // Also infer implicit dependencies from descriptions
   726→      const implicitDeps = this.inferImplicitDependencies(task, tasks);
   727→
   728→      const allDeps = [...new Set([...explicitDeps, ...implicitDeps])];
   729→      const depType =
   730→        allDeps.length === 0 ? 'independent' : allDeps.length <= 2 ? 'sequential' : 'complex';
   731→
   732→      dependencies.push({
   733→        taskId: task.id,
   734→        dependsOn: allDeps,
   735→        type: depType,
   736→      });
   737→    }
   738→
   739→    // Calculate critical path (simplified)
   740→    const criticalPath = this.calculateCriticalPath(tasks, dependencies);
   741→
   742→    // Identify parallel groups
   743→    const parallelGroups = this.identifyParallelGroups(tasks, dependencies);
   744→
   745→    this.logger.log(
   746→      `Dependencies identified: tasks=${tasks.length}, criticalPath=${criticalPath.length}, parallelGroups=${parallelGroups.length}`,
   747→    );
   748→
   749→    return { dependencies, criticalPath, parallelGroups };
   750→  }

   752→  private async optimizePlan(params: {
   753→    planId: string;
   754→    optimizationGoal: string;
   755→    constraints?: Record<string, any>;
   756→  }): Promise<{
   757→    originalEstimate: number;
   758→    optimizedEstimate: number;
   759→    improvements: Array<{ description: string; savingsMs: number; tradeoff: string }>;
   760→    tradeoffs: string[];
   761→  }> {
   762→    const { planId, optimizationGoal, constraints = {} } = params;
   763→
   764→    if (!planId || typeof planId !== 'string') {
   765→      throw new Error('Valid planId string is required');
   766→    }
   767→    if (!['speed', 'cost', 'quality', 'balanced'].includes(optimizationGoal)) {
   768→      throw new Error(
   769→        `Invalid optimization goal: ${optimizationGoal}. Must be one of: speed, cost, quality, balanced`,
   770→      );
   771→    }
   772→
   773→    const plan = this.plans.get(planId);
   774→    if (!plan) {
   775→      throw new Error(`Plan not found: ${planId}`);
   776→    }
   777→
   778→    const originalEstimate = plan.estimatedDurationMs;
   779→    const improvements: Array<{ description: string; savingsMs: number; tradeoff: string }> = [];
   780→    const tradeoffs: string[] = [];
   781→
   782→    let optimizedEstimate = originalEstimate;
   783→
   784→    switch (optimizationGoal) {
   785→      case 'speed':
   786→        // Parallelize independent steps
   787→        const parallelSavings = Math.round(originalEstimate * 0.25);
   788→        optimizedEstimate -= parallelSavings;
   789→        improvements.push({
   790→          description: 'Parallelized independent execution steps',
   791→          savingsMs: parallelSavings,
   792→          tradeoff: 'Increased resource consumption due to parallel execution',
   793→        });
   794→        tradeoffs.push('Higher resource usage during parallel execution phases');
   795→        break;
   796→
   797→      case 'cost':
   798→        // Reduce resource usage, may increase duration
   799→        const costSavings = Math.round(originalEstimate * 0.15);
   800→        optimizedEstimate += Math.round(originalEstimate * 0.1);
   801→        improvements.push({
   802→          description: 'Consolidated tasks to reduce agent count',
   803→          savingsMs: -Math.round(originalEstimate * 0.1),
   804→          tradeoff: 'Reduced cost but longer execution time',
   805→        });
   806→        tradeoffs.push('Sequential execution increases total duration by ~10%');
   807→        break;
   808→
   809→      case 'quality':
   810→        // Add review steps
   811→        optimizedEstimate += Math.round(originalEstimate * 0.2);
   812→        improvements.push({
   813→          description: 'Added quality review gates between phases',
   814→          savingsMs: -Math.round(originalEstimate * 0.2),
   815→          tradeoff: 'Higher quality output but 20% longer execution',
   816→        });
   817→        tradeoffs.push('Additional review steps increase duration by ~20%');
   818→        break;
   819→
   820→      case 'balanced':
   821→        // Moderate optimization
   822→        const balancedSavings = Math.round(originalEstimate * 0.1);
   823→        optimizedEstimate -= balancedSavings;
   824→        improvements.push({
   825→          description: 'Optimized critical path and parallelized where safe',
   826→          savingsMs: balancedSavings,
   827→          tradeoff: 'Minor quality review reduction for time savings',
   828→        });
   829→        tradeoffs.push('Slightly reduced review coverage for faster execution');
   830→        break;
   831→    }
   832→
   833→    optimizedEstimate = Math.max(1000, optimizedEstimate);
   834→
   835→    this.logger.log(
   836→      `Plan optimized: planId=${planId}, goal=${optimizationGoal}, original=${originalEstimate}ms, optimized=${optimizedEstimate}ms`,
   837→    );
   838→
   839→    return { originalEstimate, optimizedEstimate, improvements, tradeoffs };
   840→  }

   876→  private estimateImpact(description: string): number {
   877→    const highImpactWords = ['critical', 'essential', 'key', 'primary', 'core', 'vital'];
   878→    const lowImpactWords = ['minor', 'optional', 'nice-to-have', 'low-priority'];
   879→    const lower = description.toLowerCase();
   880→
   881→    if (highImpactWords.some((w) => lower.includes(w))) return 4 + Math.random();
   882→    if (lowImpactWords.some((w) => lower.includes(w))) return 1 + Math.random();
   883→    return 2.5 + Math.random() * 1.5;
   884→  }

   886→  private estimateEffortScore(description: string): number {
   887→    const words = description.split(/\s+/).length;
   888→    if (words > 20) return 4 + Math.random();
   889→    if (words > 10) return 3 + Math.random();
   890→    return 1.5 + Math.random() * 1.5;
   891→  }

   893→  private estimateBaseHours(description: string): number {
   894→    const words = description.split(/\s+/).length;
   895→    const base = 2 + words * 0.1;
   896→    return Math.max(1, Math.min(40, base));
   897→  }

   899→  private calculateConfidence(description: string, resources: Record<string, any>): number {
   900→    let confidence = 0.7;
   901→    const words = description.split(/\s+/);
   902→    if (words.length > 5) confidence -= 0.05;
   903→    if (words.length > 15) confidence -= 0.1;
   904→    if (resources.experienced) confidence += 0.1;
   905→    if (resources.tooling) confidence += 0.05;
   906→    return Math.round(Math.max(0.3, Math.min(0.95, confidence)) * 100) / 100;
   907→  }

   930→  private inferImplicitDependencies(
   931→    task: { id: string; description: string; requires?: string[] },
   932→    allTasks: Array<{ id: string; description: string }>,
   933→  ): string[] {
   934→    const implicit: string[] = [];
   935→    const lower = task.description.toLowerCase();
   936→
   937→    for (const other of allTasks) {
   938→      if (other.id === task.id) continue;
   939→      const otherLower = other.description.toLowerCase();
   940→
   941→      if (lower.includes('verify') && otherLower.includes('implement')) {
   942→        implicit.push(other.id);
   943→      }
   944→      if (lower.includes('test') && otherLower.includes('implement')) {
   945→        implicit.push(other.id);
   946→      }
   947→      if (lower.includes('deploy') && otherLower.includes('test')) {
   948→        implicit.push(other.id);
   949→      }
   950→    }
   951→
   952→    return [...new Set(implicit)];
   953→  }

   955→  private calculateCriticalPath(
   956→    tasks: Array<{ id: string }>,
   957→    dependencies: Array<{ taskId: string; dependsOn: string[] }>,
   958→  ): string[] {
   959→    const depMap = new Map(dependencies.map((d) => [d.taskId, d.dependsOn]));
   960→    const visited = new Set<string>();
   961→    const path: string[] = [];
   962→
   963→    const visit = (id: string) => {
   964→      if (visited.has(id)) return;
   965→      visited.add(id);
   966→      const deps = depMap.get(id) || [];
   967→      for (const dep of deps) {
   968→        visit(dep);
   969→      }
   970→      path.push(id);
   971→    };
   972→
   973→    for (const task of tasks) {
   974→      visit(task.id);
   975→    }
   976→
   977→    return path;
   978→  }

   980→  private identifyParallelGroups(
   981→    tasks: Array<{ id: string }>,
   982→    dependencies: Array<{ taskId: string; dependsOn: string[]; type: string }>,
   983→  ): string[][] {
   984→    const groups: string[][] = [];
   985→    const independent = dependencies.filter((d) => d.type === 'independent').map((d) => d.taskId);
   986→    if (independent.length > 0) groups.push(independent);
   987→
   988→    const sequential = dependencies.filter((d) => d.type === 'sequential').map((d) => d.taskId);
   989→    if (sequential.length > 0) groups.push(sequential);
   990→
   991→    const complex = dependencies.filter((d) => d.type === 'complex').map((d) => d.taskId);
   992→    if (complex.length > 0) groups.push(complex);
   993→
   994→    return groups.length > 0 ? groups : [tasks.map((t) => t.id)];
   995→  }

  1006→      `Top priority goes to task "${topTask.id}" with a composite score of ${topTask.score.toFixed(2)}. ` +

  1008→    );
