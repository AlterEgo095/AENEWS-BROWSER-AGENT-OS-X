    78→// Status icon helper
    79→function StatusIcon({ status }: { status: string }) {
    80→  switch (status) {
    81→    case 'healthy':
    82→    case 'connected':
    83→    case 'completed':
    84→    case 'active':
    85→      return <CheckCircle2 className="h-4 w-4 text-emerald-400" />;
    86→    case 'degraded':
    87→    case 'initializing':
    88→    case 'coordinating':
    89→    case 'running':
    90→      return <AlertTriangle className="h-4 w-4 text-amber-400" />;
    91→    case 'critical':
    92→    case 'error':
    93→    case 'failed':
    94→      return <XCircle className="h-4 w-4 text-red-400" />;
    95→    case 'offline':
    96→    case 'disconnected':
    97→    case 'passive':
    98→      return <Unplug className="h-4 w-4 text-gray-400" />;
    99→    default:
   100→      return <Activity className="h-4 w-4 text-slate-400" />;
   101→  }

   296→// ---------- History Item ----------

   304→    completed: 'text-emerald-400',

   307→    coordinating: 'text-amber-400',

   314→          <p className="text-sm font-medium text-foreground truncate">{item.description}</p>
   315→        </div>
   316→        <div className="flex items-center gap-2 mt-1">
   317→          <span className={cn('rounded-full border px-2 py-0.5 text-[9px] font-semibold uppercase', typeColors[item.type])}>
   318→            {item.type}
   319→          </span>
   320→          {item.pattern && (
   321→            <span className={cn('rounded-full border px-2 py-0.5 text-[9px] font-semibold uppercase', collaborationPatternColors[item.pattern])}>
   322→              {collaborationPatternIcons[item.pattern]} {patternLabels[item.pattern]}
   323→            </span>
   324→          )}
   325→          <span className={cn('text-[10px] font-medium', statusColorMap[item.status] || 'text-slate-400')}>
   326→            {item.status}
   327→          </span>
   328→        </div>
   329→      </div>
   330→      <span className="text-[10px] text-muted-foreground whitespace-nowrap shrink-0">
   331→        {formatRelativeTime(item.createdAt)}
   332→      </span>
   333→    </div>
   334→  );
   335→}

   357→// ---------- Main Page ----------
   358→export default function OrchestrationPage() {
   359→  // Data states
   360→  const [clusterHealth, setClusterHealth] = useState<ClusterHealthInfo[]>([]);
   361→  const [connectors, setConnectors] = useState<UnifiedConnectorInfo[]>([]);
   362→  const [stats, setStats] = useState<OrchestrationStatistics | null>(null);
   363→  const [history, setHistory] = useState<OrchestrationHistoryItem[]>([]);
   364→  const [loading, setLoading] = useState(true);
   365→
   366→  // Form states
   367→  const [decomposeDesc, setDecomposeDesc] = useState('');
   368→  const [decomposeObjectives, setDecomposeObjectives] = useState('');
   369→  const [decomposeResult, setDecomposeResult] = useState<DecompositionResult | null>(null);
   370→  const [decomposing, setDecomposing] = useState(false);
   371→
   372→  const [collabPattern, setCollabPattern] = useState<CollaborationPattern>('pipeline');
   373→  const [collabDesc, setCollabDesc] = useState('');
   374→  const [collabObjectives, setCollabObjectives] = useState('');
   375→  const [collabResult, setCollabResult] = useState<CollaborationResult | null>(null);
   376→  const [launchingCollab, setLaunchingCollab] = useState(false);
   377→
   378→  // Active section for tab-like navigation

   380→
   381→  // WebSocket for live updates
   382→  const { subscribe, unsubscribe, lastEvent } = useWebSocket();
   383→
   384→  const handleOrchestrationEvent = useCallback((event: { type: string; payload: Record<string, unknown> }) => {
   385→    if (event.type === 'orchestration:collaboration' || event.type === 'orchestration:coordination') {
   386→      // Refresh history on live events
   387→      fetchHistory();
   388→    }
   389→  }, []);
   390→
   391→  useEffect(() => {
   392→    subscribe('orchestration:collaboration', handleOrchestrationEvent);
   393→    subscribe('orchestration:coordination', handleOrchestrationEvent);
   394→    return () => {
   395→      unsubscribe('orchestration:collaboration', handleOrchestrationEvent);
   396→      unsubscribe('orchestration:coordination', handleOrchestrationEvent);
   397→    };
   398→  }, [subscribe, unsubscribe, handleOrchestrationEvent]);
   399→
   400→  // Log last event for debugging
   401→  useEffect(() => {
   402→    if (lastEvent?.type?.startsWith('orchestration:')) {
   403→      // Live event received — could show toast
   404→    }
   405→  }, [lastEvent]);
   406→
   407→  useEffect(() => {
   408→    fetchAllData();
   409→  }, []);
   410→
   411→  async function fetchAllData() {
   412→    setLoading(true);
   413→    try {
   414→      const results = await Promise.allSettled([
   415→        api.orchestration.getClusterHealth(),
   416→        api.orchestration.getConnectors(),
   417→        api.orchestration.getStatistics(),
   418→        api.orchestration.getHistory(undefined, 10),
   419→      ]);
   420→
   421→      if (results[0].status === 'fulfilled') setClusterHealth(results[0].value);
   422→
   423→      if (results[1].status === 'fulfilled') setConnectors(results[1].value);
   424→
   425→      if (results[2].status === 'fulfilled') setStats(results[2].value);
   426→
   427→      if (results[3].status === 'fulfilled') setHistory(results[3].value);
   428→    } catch {
   429→    } finally {
   430→      setLoading(false);
   431→    }
   432→  }
   433→
   434→  async function fetchHistory() {
   435→    try {
   436→      const result = await api.orchestration.getHistory(undefined, 10);
   437→      setHistory(result);
   438→    } catch {
   439→      // keep existing
   440→    }
   441→  }
   442→
   443→  async function handleDecompose() {
   444→    if (!decomposeDesc.trim()) return;
   445→    setDecomposing(true);
   446→    setDecomposeResult(null);
   447→    try {
   448→      const objectives = decomposeObjectives
   449→        .split('\n')
   450→        .map((o) => o.trim())
   451→        .filter(Boolean);
   452→      const result = await api.orchestration.decompose(null, decomposeDesc, objectives);
   453→      setDecomposeResult(result);
   454→    } catch {
   455→      // Decomposition failed
   456→      setDecomposeResult(null);
   457→    } finally {
   458→      setDecomposing(false);
   459→    }
   460→  }
   461→
   462→  async function handleCollaborate() {
   463→    if (!collabDesc.trim()) return;
   464→    setLaunchingCollab(true);
   465→    setCollabResult(null);
   466→    try {
   467→      const objectives = collabObjectives
   468→        .split('\n')
   469→        .map((o) => o.trim())
   470→        .filter(Boolean);
   471→      const result = await api.orchestration.collaborate(collabPattern, collabDesc, objectives);
   472→      setCollabResult(result);
   473→    } catch {
   474→      // Collaboration failed
   475→      setCollabResult(null);
   476→    } finally {
   477→      setLaunchingCollab(false);
   478→    }
   479→  }
   480→
   481→  const healthyClusters = clusterHealth.filter((c) => c.status === 'healthy').length;
   482→  const connectedConnectors = connectors.filter((c) => c.status === 'connected').length;
   483→  const maxPatternUsage = stats ? Math.max(...Object.values(stats.patternUsage)) : 1;
   484→
   485→  if (loading) {
   486→    return (
   487→      <div className="space-y-6">
   488→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
   489→          {[1, 2, 3, 4].map((i) => (
   490→            <div key={i} className="h-32 animate-shimmer rounded-xl" />
   491→          ))}
   492→        </div>
   493→        <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
   494→          {[1, 2, 3].map((i) => (
   495→            <div key={i} className="h-64 animate-shimmer rounded-xl" />
   496→          ))}
   497→        </div>
   498→      </div>
   499→    );
   500→  }
   501→
   502→  return (
   503→    <div className="space-y-6 animate-slide-in">
   504→      {/* Header */}
   505→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   506→        <div>
   507→          <h1 className="text-2xl font-bold text-foreground">Intelligent Orchestration</h1>
   508→          <p className="text-sm text-muted-foreground">
   509→            Phase 8 — Cluster health, connector management & multi-agent collaboration
   510→          </p>
   511→        </div>
   512→        <button
   513→          onClick={fetchAllData}
   514→          className="flex items-center gap-2 rounded-lg border border-border bg-card px-4 py-2 text-sm font-medium text-foreground transition-colors hover:bg-white/5"
   515→        >
   516→          <RefreshCw className={cn('h-4 w-4', loading && 'animate-spin')} />
   517→          Refresh
   518→        </button>
   519→      </div>
   520→
   521→      {/* Section Tabs */}
   522→      <div className="flex gap-1 rounded-lg border border-border bg-card p-1">
   523→        {([
   524→          { key: 'overview', label: 'Overview', icon: Network },
   525→          { key: 'decompose', label: 'Decompose', icon: GitBranch },
   526→          { key: 'collaborate', label: 'Collaborate', icon: Layers },
   527→        ] as const).map((tab) => (
   528→          <button
   529→            key={tab.key}
   530→            onClick={() => setActiveSection(tab.key)}
   531→            className={cn(
   532→              'flex items-center gap-2 rounded-md px-4 py-2 text-sm font-medium transition-all',
   533→              activeSection === tab.key
   534→                ? 'bg-primary/15 text-primary'
   535→                : 'text-muted-foreground hover:bg-white/5 hover:text-foreground'
