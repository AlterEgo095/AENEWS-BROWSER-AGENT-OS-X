    50→    // Reuse in-flight refresh promise to prevent concurrent refreshes
    51→    if (!this.refreshing) {
    52→      this.refreshing = useAuthStore.getState().refreshAuth().finally(() => {
    53→        this.refreshing = null;
    54→      });
    55→    }
    56→    return this.refreshing;
    57→  }
    58→
    59→  /**
    60→   * Handle session expiry — try refreshing before redirecting to login.
    61→   */
    62→  private async handleUnauthorized(originalEndpoint: string, originalOptions?: RequestInit): Promise<boolean> {
    63→    const newToken = await this.tryRefresh();
    64→
    65→    if (newToken) {
    66→      // Token refreshed successfully — the caller should retry the request
    67→      return true;
    68→    }
    69→
    70→    // Refresh also failed — redirect to login
    71→    if (typeof window !== 'undefined' && !window.location.pathname.startsWith('/login')) {
    72→      window.location.href = '/login';
    73→    }
    74→    return false;
    75→  }

    77→  /**
    78→   * Determine if a request should be retried.
    79→   * - 5xx: retry with backoff
    80→   * - 4xx: never retry (client error)
    81→   * - Network error: retry
    82→   */
    83→  private shouldRetry(status: number | null, attempt: number): boolean {
    84→    if (attempt >= MAX_RETRIES) return false;
    85→    // 4xx = client error, never retry (except 401 which is handled separately)
    86→    if (status !== null && status >= 400 && status < 500) return false;
    87→    // 5xx or network failure: retry
    88→    return true;
    89→  }
    90→
    91→  private async request<T>(endpoint: string, options?: RequestInit, _attempt: number = 0, _isRetryAfterRefresh: boolean = false): Promise<T> {
    92→    const url = `${this.baseUrl}${endpoint}`;
    93→    const controller = new AbortController();
    94→    const timeoutId = setTimeout(() => controller.abort(), REQUEST_TIMEOUT_MS);
    95→
    96→    try {
    97→      const response = await fetch(url, {
    98→        ...options,
    99→        signal: controller.signal,
   100→        credentials: 'include', // Required for sending httpOnly cookies
   101→        headers: {
   102→          ...this.getHeaders(),
   103→          ...options?.headers,
   104→        },
   105→      });
   106→
   107→      if (response.status === 401) {
   108→        // On 401, try refreshing the token before giving up
   109→        if (!_isRetryAfterRefresh) {
   110→          const refreshed = await this.handleUnauthorized(endpoint, options);
   111→          if (refreshed) {
   112→            // Retry the original request with the new token
   113→            return this.request<T>(endpoint, options, 0, true);
   114→          }
   115→        }
   116→        throw new Error('Session expired. Please log in again.');
   117→      }
   118→
   119→      if (!response.ok) {
   120→        // Retry on 5xx
   121→        if (this.shouldRetry(response.status, _attempt)) {
   122→          const delay = Math.min(1000 * Math.pow(2, _attempt), 5000);
   123→          await new Promise((resolve) => setTimeout(resolve, delay));
   124→          return this.request<T>(endpoint, options, _attempt + 1, _isRetryAfterRefresh);
   125→        }
   126→
   127→        const error = await response.json().catch(() => ({ message: 'Request failed' }));
   128→        throw new Error(error.message || `HTTP ${response.status}`);
   129→      }
   130→
   131→      // Handle 204 No Content
   132→      if (response.status === 204) {
   133→        return undefined as T;
   134→      }
   135→
   136→      const json = await response.json();
   137→
   138→      // Unwrap the TransformInterceptor response: { success, data, meta }
   139→      // If the response has a `data` field at the top level and a `success` field,
   140→      // return just the `data` portion. Otherwise, return the full response.
   141→      if (json && typeof json === 'object' && 'success' in json && 'data' in json) {
   142→        return json.data as T;
   143→      }
   144→
   145→      return json as T;
   146→    } catch (err) {
   147→      // Retry on network errors (abort, fetch failure) unless it's an auth redirect
   148→      if (err instanceof Error && err.message === 'Session expired. Please log in again.') {
   149→        throw err;
   150→      }
   151→
   152→      if (err instanceof DOMException && err.name === 'AbortError') {
   153→        if (this.shouldRetry(null, _attempt)) {
   154→          const delay = Math.min(1000 * Math.pow(2, _attempt), 5000);
   155→          await new Promise((resolve) => setTimeout(resolve, delay));
   156→          return this.request<T>(endpoint, options, _attempt + 1, _isRetryAfterRefresh);
   157→        }
   158→        throw new Error('Request timed out. Please try again.');
   159→      }
   160→
   161→      // Network error — retry
   162→      if (err instanceof TypeError && this.shouldRetry(null, _attempt)) {
   163→        const delay = Math.min(1000 * Math.pow(2, _attempt), 5000);
   164→        await new Promise((resolve) => setTimeout(resolve, delay));
   165→        return this.request<T>(endpoint, options, _attempt + 1, _isRetryAfterRefresh);
   166→      }
   167→
   168→      throw err;
   169→    } finally {
   170→      clearTimeout(timeoutId);
   171→    }
   172→  }
