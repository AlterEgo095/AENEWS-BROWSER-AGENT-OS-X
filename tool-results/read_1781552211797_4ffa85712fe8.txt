   464→  // ── Pattern-Based Analysis Fallback ───────────────────────────────
   465→
   466→  private patternBasedAnalysis(errorMessage: string, stackTrace: string) {
   467→    const msg = errorMessage.toLowerCase();
   468→    let category: string = 'unknown';
   469→    let type: string = 'unknown';
   470→    let severity: string = 'medium';
   471→    let isRecoverable = true;
   472→
   473→    if (msg.includes('timeout') || msg.includes('timed out')) {
   474→      category = 'timeout'; type = 'timeout'; severity = 'high'; isRecoverable = true;
   475→    } else if (msg.includes('connection') || msg.includes('econnrefused') || msg.includes('network')) {
   476→      category = 'network'; type = 'connection-refused'; severity = 'high'; isRecoverable = true;
   477→    } else if (msg.includes('auth') || msg.includes('unauthorized') || msg.includes('forbidden')) {
   478→      category = 'permission'; type = 'auth-failure'; severity = 'critical'; isRecoverable = false;
   479→    } else if (msg.includes('memory') || msg.includes('oom') || msg.includes('out of memory')) {
   480→      category = 'resource'; type = 'out-of-memory'; severity = 'critical'; isRecoverable = false;
   481→    } else if (msg.includes('config') || msg.includes('env') || msg.includes('missing')) {
   482→      category = 'configuration'; type = 'misconfiguration'; severity = 'high'; isRecoverable = true;
   483→    } else if (msg.includes('null') || msg.includes('undefined') || msg.includes('cannot read')) {
   484→      category = 'application'; type = 'null-pointer'; severity = 'high'; isRecoverable = false;
   485→    }
   486→
   487→    return {
   488→      rootCause: {
   489→        identified: category !== 'unknown',
   490→        category,
   491→        type,
   492→        description: `Pattern-based diagnosis: ${errorMessage.slice(0, 200)}`,
   493→        confidence: category !== 'unknown' ? 0.7 : 0.3,
   494→        evidence: [errorMessage.slice(0, 500)],
   495→        location: stackTrace ? stackTrace.split('\n')[0] || '' : '',
   496→      },
   497→      errorCategory: isRecoverable ? 'transient' : 'persistent',
   498→      severity,
   499→      isRecoverable,
   500→      estimatedRecoveryEffort: isRecoverable ? 'moderate' : 'significant',
   501→      affectedComponents: [],
   502→      errorPattern: {
   503→        isFirstOccurrence: true,
   504→        occurrenceCount: 0,
   505→        frequency: 'one-time',
   506→        trendingDirection: 'unknown',
   507→      },
   508→      suggestedRemediation: [],
   509→      context: {
   510→        systemState: {},
   511→        recentChanges: [],
   512→        relatedErrors: [],
   513→        resourceUtilization: { cpu: 0, memory: 0, disk: 0, network: 0 },
   514→      },
   515→    };
   516→  }

   518→  private regexBasedClassification(errorMessage: string, errorCode: string) {
   519→    const msg = (errorMessage || '').toLowerCase();
   520→    let type = 'unknown';
   521→    let category = 'transient';
   522→    let severity = 'medium';
   523→    let isRecoverable = true;
   524→    let isRetryable = false;
   525→
   526→    if (msg.includes('timeout')) { type = 'timeout'; isRetryable = true; }
   527→    else if (msg.includes('connection') || msg.includes('network')) { type = 'network'; isRetryable = true; }
   528→    else if (msg.includes('auth') || msg.includes('unauthorized')) { type = 'authentication'; severity = 'critical'; isRecoverable = false; }
   529→    else if (msg.includes('null') || msg.includes('undefined')) { type = 'runtime'; severity = 'high'; }
   530→    else if (msg.includes('syntax') || msg.includes('parse')) { type = 'syntax'; severity = 'high'; isRecoverable = false; }
   531→    else if (msg.includes('memory') || msg.includes('oom')) { type = 'resource'; severity = 'critical'; }
   532→
   533→    return {
   534→      type,
   535→      category,
   536→      severity,
   537→      isRecoverable,
   538→      isRetryable,
   539→      requiresHumanIntervention: !isRecoverable,
   540→      tags: [type, category, severity],
   541→      taxonomy: { domain: 'application', subdomain: type, errorFamily: category, errorClass: type },
   542→    };
   543→  }

   545→  private parseStackTrace(errorMessage: string, stackTrace: string) {
   546→    const lines = stackTrace.split('\n').filter((l) => l.trim());
   547→    const causalChain = lines.slice(0, 5).map((line, i) => ({
   548→      step: i + 1,
   549→      component: line.trim().split(' ')[0] || 'unknown',
   550→      event: line.trim().slice(0, 100),
   551→      timestamp: new Date().toISOString(),
   552→      type: (i === 0 ? 'trigger' : i === lines.length - 1 ? 'manifestation' : 'propagation') as 'trigger' | 'propagation' | 'amplification' | 'manifestation',
   553→      details: line.trim(),
   554→      confidence: 0.5,
   555→    }));
   556→
   557→    return {
   558→      rootCause: {
   559→        identified: causalChain.length > 0,
   560→        category: 'application',
   561→        description: `Stack trace analysis: ${errorMessage.slice(0, 200)}`,
   562→        confidence: 0.4,
   563→        evidence: lines.slice(0, 3),
   564→        location: lines[0] || '',
   565→        component: causalChain[0]?.component || '',
   566→        firstOccurrence: new Date().toISOString(),
   567→      },
   568→      causalChain,
   569→      timeline: causalChain.map((c) => ({
   570→        timestamp: c.timestamp,
   571→        event: c.event,
   572→        component: c.component,
   573→        type: 'cause' as const,
   574→        severity: 'error' as const,
   575→        details: c.details,
   576→      })),
   577→      dependencyGraph: {
   578→        nodes: causalChain.map((c) => ({
   579→          id: c.component,
   580→          type: 'component',
   581→          name: c.component,
   582→          status: 'unknown' as const,
   583→        })),
   584→        edges: causalChain.slice(1).map((c, i) => ({
   585→          from: causalChain[i].component,
   586→          to: c.component,
   587→          type: 'calls' as const,
   588→          status: 'degraded' as const,
   589→        })),
   590→        failedNodes: [causalChain[0]?.component].filter(Boolean),
   591→      },
   592→      contributingFactors: [],
   593→      blastRadius: {
   594→        directlyAffected: causalChain.map((c) => c.component),
   595→        indirectlyAffected: [],
   596→        potentiallyAffected: [],
   597→        totalComponentsAtRisk: causalChain.length,
   598→      },
   599→    };
   600→  }
