     1→/**
     2→ * AENEWS Software Factory — Development Connector
     3→ *
     4→ * Maps dev.* capabilities to real tool invocations:
     5→ *   dev.architecture  → LLM: analyze & design
     6→ *   dev.frontend      → LLM: generate HTML/CSS/JS + write files
     7→ *   dev.backend       → LLM: generate server code + write files
     8→ *   dev.database      → LLM: generate schema + write files
     9→ *   dev.api           → LLM: generate API code + write files
    10→ *   dev.devops        → LLM: generate CI/CD configs + write files
    11→ *   dev.docker        → Generate Dockerfile + docker-compose
    12→ *   dev.kubernetes    → Generate K8s manifests
    13→ *   dev.qa            → Shell: run tests + LLM: analyze
    14→ *   dev.test          → LLM: generate test code + write files
    15→ *   dev.debug         → LLM: analyze errors + suggest fixes
    16→ *   dev.documentation → LLM: generate docs + write files
    17→ *
    18→ * Tools: z-ai-web-dev-sdk (LLM), child_process (shell), fs (files)
    19→ */
    20→
    21→import { Injectable, Logger } from '@nestjs/common';
    22→import * as fs from 'fs';
    23→import * as path from 'path';
    24→import { execSync } from 'child_process';
    25→import {
    26→  CapabilityId,
    27→  CapabilityPack,
    28→  DevCapability,
    29→} from '../interfaces';
    30→import {
    31→  ICapabilityConnector,
    32→  ConnectorInput,
    33→  ConnectorOutput,

    35→} from './connector.interface';
    36→import { LLMHelper } from './llm-helper';
    37→
    38→@Injectable()
    39→export class DevelopmentConnector implements ICapabilityConnector {
    40→  readonly supportedPack = CapabilityPack.DEVELOPMENT;
    41→  private readonly logger = new Logger(DevelopmentConnector.name);
    42→  private readonly llm = new LLMHelper();
    43→
    44→  private static readonly DEV_CAPABILITIES = new Set<string>(Object.values(DevCapability));
    45→
    46→  supports(capabilityId: CapabilityId): boolean {
    47→    return DevelopmentConnector.DEV_CAPABILITIES.has(capabilityId as string);
    48→  }
    49→
    50→  async execute(capabilityId: CapabilityId, input: ConnectorInput): Promise<ConnectorOutput> {
    51→    const startTime = Date.now();
    52→    const capId = capabilityId as DevCapability;
    53→
    54→    this.logger.log(`Dev connector executing: ${capId} for mission ${input.missionId}`);
    55→
    56→    // Ensure workspace directories exist
    57→    this.ensureWorkspace(input.workspaceDir);
    58→
    59→    try {
    60→      let result: ConnectorOutput;
    61→
    62→      switch (capId) {
    63→        case DevCapability.ARCHITECTURE:
    64→          result = await this.executeArchitecture(input);
    65→          break;
    66→        case DevCapability.FRONTEND:
    67→          result = await this.executeFrontend(input);
    68→          break;
    69→        case DevCapability.BACKEND:
    70→          result = await this.executeBackend(input);
    71→          break;
    72→        case DevCapability.DATABASE:
    73→          result = await this.executeDatabase(input);
    74→          break;
    75→        case DevCapability.API:
    76→          result = await this.executeApi(input);
    77→          break;
    78→        case DevCapability.DEVOPS:
    79→          result = await this.executeDevOps(input);
    80→          break;
    81→        case DevCapability.DOCKER:
    82→          result = await this.executeDocker(input);
    83→          break;
    84→        case DevCapability.KUBERNETES:
    85→          result = await this.executeKubernetes(input);
    86→          break;
    87→        case DevCapability.QA:
    88→          result = await this.executeQA(input);
    89→          break;
    90→        case DevCapability.TEST:
    91→          result = await this.executeTest(input);
    92→          break;
    93→        case DevCapability.DEBUG:
    94→          result = await this.executeDebug(input);
    95→          break;
    96→        case DevCapability.DOCUMENTATION:
    97→          result = await this.executeDocumentation(input);
    98→          break;
    99→        default:
   100→          result = await this.executeGenericDev(capId, input);
   101→      }
   102→
   103→      result.durationMs = Date.now() - startTime;
   104→      return result;
   105→    } catch (error: any) {
   106→      this.logger.error(`Dev connector failed for ${capId}: ${error.message}`);
   107→      return {
   108→        success: false,
   109→        artifacts: [],
   110→        output: { error: error.message },
   111→        costUsd: 0,
   112→        durationMs: Date.now() - startTime,
   113→        error: error.message,
   114→      };
   115→    }
   116→  }

   122→  private async executeArchitecture(input: ConnectorInput): Promise<ConnectorOutput> {
   123→    const systemPrompt = `You are a senior software architect. Analyze the mission and produce a complete technical architecture document.
   124→Include: tech stack, project structure, component diagram, data models, API design, security considerations.
   125→Output in markdown format.`;
   126→
   127→    const userPrompt = `Design the architecture for: "${input.instruction}"
   128→${input.parameters.context ? `Context: ${JSON.stringify(input.parameters.context)}` : ''}
   129→Produce a detailed ARCHITECTURE.md with all design decisions.`;
   130→
   131→    const llmResult = await this.llm.call({ systemPrompt, userPrompt, maxTokens: 4096 });
   132→
   133→    const filePath = path.join(input.workspaceDir, 'docs', 'ARCHITECTURE.md');
   134→    fs.mkdirSync(path.dirname(filePath), { recursive: true });
   135→    fs.writeFileSync(filePath, llmResult.content, 'utf-8');
   136→
   137→    return {
   138→      success: true,
   139→      artifacts: [this.makeArtifact('ARCHITECTURE.md', 'document', filePath, llmResult.content)],
   140→      output: { architecture: llmResult.content.substring(0, 1000) },
   141→      costUsd: llmResult.costUsd,
   142→      durationMs: 0,
   143→    };
   144→  }

   150→  private async executeFrontend(input: ConnectorInput): Promise<ConnectorOutput> {
   151→    const systemPrompt = `You are an expert frontend developer. Generate complete, working, production-ready code.
   152→For each file, use this format:
   153→===FILE: path/to/file===
   154→(file content)
   155→===ENDFILE===
   156→
   157→Generate ALL frontend files: HTML, CSS, JavaScript. Make it beautiful and functional.`;
   158→
   159→    const userPrompt = `Build the frontend for: "${input.instruction}"
   160→${this.getExistingFilesContext(input)}
   161→Create complete HTML/CSS/JS that works when opened in a browser.`;
   162→
   163→    const llmResult = await this.llm.call({ systemPrompt, userPrompt, maxTokens: 4096 });
   164→    const files = this.llm.parseGeneratedFiles(llmResult.content);
   165→    const artifacts = this.writeFiles(files, input.workspaceDir);
   166→
   167→    // If LLM didn't produce parseable files, generate a template
   168→    if (artifacts.length === 0) {
   169→      const template = this.generateFrontendTemplate(input.instruction);
   170→      const templateFiles = this.llm.parseGeneratedFiles(template);
   171→      this.writeFiles(templateFiles, input.workspaceDir);
   172→      artifacts.push(...this.writeFiles(templateFiles, input.workspaceDir));
   173→    }
   174→
   175→    return {
   176→      success: artifacts.length > 0,
   177→      artifacts,
   178→      output: { filesGenerated: artifacts.map(a => a.name) },
   179→      costUsd: llmResult.costUsd,
   180→      durationMs: 0,
   181→    };
   182→  }

   394→  // ═══════════════════════════════════════════════════════════
   395→
   396→  private async executeQA(input: ConnectorInput): Promise<ConnectorOutput> {
   397→    const results: any[] = [];
   398→    let allPassed = true;
   399→
   400→    // Try running existing test files
   401→    const testDir = path.join(input.workspaceDir, 'tests');
   402→    if (fs.existsSync(testDir)) {
   403→      const testFiles = fs.readdirSync(testDir).filter(f => f.endsWith('.js') || f.endsWith('.mjs'));
   404→      for (const testFile of testFiles.slice(0, 5)) {
   405→        try {
   406→          const output = execSync(`node "${path.join(testDir, testFile)}" 2>&1`, {
   407→            timeout: 30000,
   408→            cwd: input.workspaceDir,
   409→          }).toString();
   410→          results.push({ file: testFile, passed: true, output: output.slice(0, 500) });
   411→        } catch (err: any) {
   412→          results.push({ file: testFile, passed: false, output: (err.stdout || err.message || '').toString().slice(0, 500) });
   413→          allPassed = false;
   414→        }
   415→      }
   416→    }
   417→
   418→    // LLM-based quality analysis
   419→    let llmAnalysis = '';
   420→    try {
   421→      const srcFiles = this.collectSourceFiles(input.workspaceDir);
   422→      if (srcFiles.length > 0) {
   423→        const llmResult = await this.llm.call({
   424→          systemPrompt: 'You are a QA engineer. Analyze code for bugs and quality issues.',
   425→          userPrompt: `Analyze this code for quality:\n${srcFiles.slice(0, 5).map(f => `--- ${f.name} ---\n${f.content?.slice(0, 500) || ''}`).join('\n\n')}\n\nReply in JSON: {"passed": true/false, "analysis": "brief analysis", "bugs": []}`,
   426→          maxTokens: 2048,
   427→        });
   428→        llmAnalysis = llmResult.content;
   429→      }
   430→    } catch { /* optional */ }
   431→
   432→    // Write QA report
   433→    const reportPath = path.join(input.workspaceDir, 'docs', 'QA-REPORT.md');
   434→    fs.mkdirSync(path.dirname(reportPath), { recursive: true });
   435→    const report = `# QA Report\n\n## Test Results\n${results.map(r => `- ${r.file}: ${r.passed ? 'PASS' : 'FAIL'}`).join('\n')}\n\n## LLM Analysis\n${llmAnalysis || 'N/A'}\n`;
   436→    fs.writeFileSync(reportPath, report, 'utf-8');
   437→
   438→    return {
   439→      success: allPassed,
   440→      artifacts: [this.makeArtifact('QA-REPORT.md', 'report', reportPath, report)],
   441→      output: { results, llmAnalysis: llmAnalysis.substring(0, 500) },
   442→      costUsd: 0.02,
   443→      durationMs: 0,
   444→    };
   445→  }

   521→  private async executeDocumentation(input: ConnectorInput): Promise<ConnectorOutput> {

   539→      fs.writeFileSync(readmePath, readme, 'utf-8');
   540→      artifacts.push(this.makeArtifact('README.md', 'document', readmePath, readme));
   541→    }
   542→
   543→    return {
   544→      success: artifacts.length > 0,

   584→  private ensureWorkspace(workspaceDir: string): void {
   585→    const dirs = ['src', 'tests', 'docs', 'db', 'k8s'];
   586→    for (const dir of dirs) {
   587→      fs.mkdirSync(path.join(workspaceDir, dir), { recursive: true });
   588→    }
   589→  }
   590→
   591→  private writeFiles(files: Map<string, string>, workspaceDir: string): GeneratedArtifact[] {
   592→    const artifacts: GeneratedArtifact[] = [];
   593→    files.forEach((content, filePath) => {
   594→      const fullPath = path.join(workspaceDir, filePath);
   595→      fs.mkdirSync(path.dirname(fullPath), { recursive: true });
   596→      fs.writeFileSync(fullPath, content, 'utf-8');
   597→
   598→      let type: GeneratedArtifact['type'] = 'source';
   599→      if (filePath.includes('test') || filePath.includes('spec')) type = 'test';
   600→      else if (filePath.endsWith('.md') || filePath.endsWith('.txt')) type = 'document';
   601→      else if (filePath.endsWith('.json') || filePath.endsWith('.yml') || filePath.endsWith('.yaml') || filePath.endsWith('Dockerfile') || filePath.includes('.config')) type = 'config';
   602→
   603→      artifacts.push(this.makeArtifact(path.basename(filePath), type, fullPath, content));
   604→      this.logger.log(`  Created: ${filePath} (${Buffer.byteLength(content)} bytes)`);
   605→    });
   606→    return artifacts;
   607→  }
   608→
   609→  private makeArtifact(name: string, type: GeneratedArtifact['type'], fullPath: string, content: string): GeneratedArtifact {
   610→    return {
   611→      name,
   612→      type,
   613→      path: fullPath,
   614→      size: Buffer.byteLength(content),
   615→      content: content.substring(0, 500),
   616→    };
   617→  }
   618→
   619→  private getExistingFilesContext(input: ConnectorInput): string {
   620→    const existing = input.previousResults;
   621→    if (!existing || existing.size === 0) return '';
   622→
   623→    const summaries: string[] = [];
   624→    existing.forEach((output, capId) => {
   625→      if (output.artifacts?.length > 0) {
   626→        summaries.push(`${capId}: ${output.artifacts.map((a: any) => a.name).join(', ')}`);
   627→      }
   628→    });
   629→    return summaries.length > 0 ? `Previously generated: ${summaries.join('; ')}` : '';
   630→  }

   632→  private collectSourceFiles(workspaceDir: string): { name: string; content?: string }[] {
   633→    const files: { name: string; content?: string }[] = [];
   634→    const extensions = ['.js', '.ts', '.html', '.css', '.py'];
   635→
   636→    const walkDir = (dir: string) => {
   637→      if (!fs.existsSync(dir)) return;
   638→      for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
   639→        const fullPath = path.join(dir, entry.name);
   640→        if (entry.isDirectory() && !entry.name.startsWith('.') && entry.name !== 'node_modules') {
   641→          walkDir(fullPath);
   642→        } else if (entry.isFile() && extensions.some(ext => entry.name.endsWith(ext))) {
   643→          try {
   644→            const content = fs.readFileSync(fullPath, 'utf-8').slice(0, 1000);
   645→            files.push({ name: path.relative(workspaceDir, fullPath), content });
   646→          } catch { /* skip */ }
   647→        }
   648→      }
   649→    };
   650→
   651→    walkDir(workspaceDir);
   652→    return files;
   653→  }
