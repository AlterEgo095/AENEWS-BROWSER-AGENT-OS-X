   218→   218→  // ─── Lifecycle: Execute ──────────────────────────────────────────
   219→   219→
   220→   220→  async execute(input: AgentInput): Promise<AgentOutput> {
   221→   221→    // 1. Check status
   222→   222→    if (this.status !== AgentStatus.RUNNING && this.status !== AgentStatus.IDLE) {
   223→   223→      throw new AgentError(
   224→   224→        `Agent cannot execute in ${this.status} state`,
   225→   225→        AgentErrorCode.NOT_RUNNING,
   226→   226→        this.config.id,
   227→   227→        input.taskId,
   228→   228→      );
   229→   229→    }
   230→   230→
   231→   231→    // 2. Check circuit breaker
   232→   232→    if (this.circuitBreaker.state === 'open') {
   233→   233→      throw new AgentError(
   234→   234→        'Circuit breaker is open; agent is not accepting tasks',
   235→   235→        AgentErrorCode.CIRCUIT_BREAKER_OPEN,
   236→   236→        this.config.id,
   237→   237→        input.taskId,
   238→   238→      );
   239→   239→    }
   240→   240→
   241→   241→    // 3. Check max concurrent tasks
   242→   242→    if (this.currentTasks.size >= this.config.maxConcurrentTasks) {
   243→   243→      throw new AgentError(
   244→   244→        `Max concurrent tasks (${this.config.maxConcurrentTasks}) reached`,
   245→   245→        AgentErrorCode.MAX_CONCURRENT_TASKS,
   246→   246→        this.config.id,
   247→   247→        input.taskId,
   248→   248→      );
   249→   249→    }
   250→   250→
   251→   251→    // 4. Check permissions
   252→   252→    await this.checkPermission(PermissionAction.EXECUTE, PermissionResource.TASK);
   253→   253→
   254→   254→    // 5. Validate input
   255→   255→    const isValid = await this.onValidateInput(input);
   256→   256→    if (!isValid) {
   257→   257→      throw new AgentError(
   258→   258→        'Invalid input provided',
   259→   259→        AgentErrorCode.INVALID_INPUT,
   260→   260→        this.config.id,
   261→   261→        input.taskId,
   262→   262→      );
   263→   263→    }
   264→   264→
   265→   265→    // 6. Set up execution context
   266→   266→    this.correlationId = input.context?.correlationId || uuidv4();
   267→   267→    this.currentTasks.add(input.taskId);
   268→   268→    this.lastActivity = new Date();
   269→   269→
   270→   270→    // 7. Record start time
   271→   271→    const startTime = Date.now();
   272→   272→    const startMemory = process.memoryUsage().heapUsed;
   273→   273→
   274→   274→    this.executeLifecyclePhase(LifecyclePhase.PRE_EXECUTE, { taskId: input.taskId });
   275→   275→    this.emitEvent(AgentEventType.TASK_STARTED, {
   276→   276→      taskId: input.taskId,
   277→   277→      agentId: this.config.id,
   278→   278→      priority: input.priority || TaskPriority.NORMAL,
   279→   279→    });
   280→   280→
   281→   281→    try {
   282→   282→      // Auto-start if idle
   283→   283→      if (this.status === AgentStatus.IDLE) {
   284→   284→        await this.start();
   285→   285→      }
   286→   286→
   287→   287→      // 8. Execute with retry logic (exponential backoff)
   288→   288→      let retryCount = 0;
   289→   289→      let lastError: Error | null = null;
   290→   290→
   291→   291→      const output = await this.executeWithRetry<AgentOutput>(async () => {
   292→   292→        try {
   293→   293→          // Execute with timeout
   294→   294→          const result = await this.executeWithTimeout(input);
   295→   295→          return result;
   296→   296→        } catch (execError) {
   297→   297→          retryCount++;
   298→   298→          lastError = execError as Error;
   299→   299→
   300→   300→          if (retryCount > this.config.retryPolicy.maxRetries) {
   301→   301→            throw execError;
   302→   302→          }
   303→   303→
   304→   304→          const backoffMs = this.calculateBackoff(retryCount);
   305→   305→          this.logger.warn(
   306→   306→            `Task ${input.taskId} failed (attempt ${retryCount}/${this.config.retryPolicy.maxRetries}), ` +
   307→   307→            `retrying in ${backoffMs}ms: ${lastError.message}`,
   308→   308→          );
   309→   309→          await this.sleep(backoffMs);
   310→   310→
   311→   311→          // Re-throw so executeWithRetry can handle it
   312→   312→          throw execError;
   313→   313→        }
   314→   314→      });
   315→   315→
   316→   316→      // 9. Collect metrics and emit events on success
   317→   317→      this.currentTasks.delete(input.taskId);
   318→   318→      this.completedTaskCount++;
   319→   319→      this.lastActivity = new Date();
   320→   320→
   321→   321→      this.recordCircuitBreakerSuccess();
   322→   322→
   323→   323→      // Store result in memory
   324→   324→      await this.storeExecutionResult(input, output);
   325→   325→
   326→   326→      // Lifecycle and event hooks
   327→   327→      this.executeLifecyclePhase(LifecyclePhase.POST_EXECUTE, { taskId: input.taskId });
   328→   328→      this.emitEvent(AgentEventType.TASK_COMPLETED, {
   329→   329→        taskId: input.taskId,
   330→   330→        agentId: this.config.id,
   331→   331→        success: true,
   332→   332→        executionTimeMs: Date.now() - startTime,
   333→   333→        result: output.result,
   334→   334→      });
   335→   335→
   336→   336→      await this.onSuccess(input, output);
   337→   337→
   338→   338→      // 10. Return output
   339→   339→      return output;
   340→   340→    } catch (error) {
   341→   341→      this.currentTasks.delete(input.taskId);
   342→   342→      this.failedTaskCount++;
   343→   343→      this.recordCircuitBreakerFailure();
   344→   344→      this.lastActivity = new Date();
   345→   345→
   346→   346→      const agentError = error as AgentError;
   347→   347→      this.logger.error(
   348→   348→        `Task ${input.taskId} execution error: ${agentError.message}`,
   349→   349→        agentError.stack,
   350→   350→      );
   351→   351→
   352→   352→      this.emitEvent(AgentEventType.AGENT_ERROR, {
   353→   353→        errorCode: agentError.code || AgentErrorCode.EXECUTION_FAILED,
   354→   354→        errorMessage: agentError.message,
   355→   355→        taskId: input.taskId,
   356→   356→        recoverable: true,
   357→   357→      } as AgentErrorPayload);
   358→   358→
   359→   359→      this.emitEvent(AgentEventType.TASK_FAILED, {
   360→   360→        taskId: input.taskId,
   361→   361→        agentId: this.config.id,
   362→   362→        success: false,
   363→   363→        executionTimeMs: Date.now() - startTime,
   364→   364→        error: agentError.message,
   365→   365→      });
   366→   366→
   367→   367→      await this.onFailure(input, agentError);
   368→   368→
   369→   369→      // 10. Return output (with error info)
   370→   370→      return {
   371→   371→        taskId: input.taskId,
   372→   372→        success: false,
   373→   373→        result: null,
   374→   374→        error: agentError.message,
   375→   375→        metrics: this.collectMetrics(startTime, startMemory),
   376→   376→        timestamp: new Date(),
   377→   377→      };
   378→   378→    }
   379→   379→  }

   413→   413→  // ─── Lifecycle: Stop ─────────────────────────────────────────────
   414→   414→
   415→   415→  async stop(): Promise<void> {
   416→   416→    this.executeLifecyclePhase(LifecyclePhase.PRE_STOP);
   417→   417→
   418→   418→    // Clear all active timeouts
   419→   419→    for (const [taskId, timeout] of this.activeTimeouts) {
   420→   420→      clearTimeout(timeout);
   421→   421→      this.activeTimeouts.delete(taskId);
   422→   422→    }
   423→   423→
   424→   424→    // Wait for current tasks to finish (with a grace period)
   425→   425→    const gracePeriodMs = 5000;
   426→   426→    const startTime = Date.now();
   427→   427→    while (this.currentTasks.size > 0 && Date.now() - startTime < gracePeriodMs) {
   428→   428→      await this.sleep(100);
   429→   429→    }
   430→   430→
   431→   431→    // Force-clear remaining tasks
   432→   432→    if (this.currentTasks.size > 0) {
   433→   433→      this.logger.warn(
   434→   434→        `Force-stopping with ${this.currentTasks.size} active tasks`,
   435→   435→      );
   436→   436→      this.currentTasks.clear();
   437→   437→    }
   438→   438→
   439→   439→    this.stopHealthMonitoring();
   440→   440→    this.stopMetricsCollection();
   441→   441→
   442→   442→    await this.transitionTo(AgentStatus.STOPPED);
   443→   443→    this.executeLifecyclePhase(LifecyclePhase.POST_STOP);
   444→   444→
   445→   445→    this.logger.log(`Agent stopped: ${this.config.id}`);
   446→   446→    this.emitEvent(AgentEventType.AGENT_STOPPED, {
   447→   447→      agentId: this.config.id,
   448→   448→      completedTasks: this.completedTaskCount,
   449→   449→      failedTasks: this.failedTaskCount,
   450→   450→    });
   451→   451→  }

   469→   469→    this.executeLifecyclePhase(LifecyclePhase.POST_DESTROY);
   470→   470→
   471→   471→    this.logger.log(`Agent destroyed: ${this.config.id}`);
   472→   472→    this.emitEvent(AgentEventType.AGENT_DESTROYED, {
   473→   473→      agentId: this.config.id,
   474→   474→      cluster: this.config.cluster,
   475→   475→    });
   476→   476→  }

   514→   514→  private initializeState(): AgentState {

   633→   633→  /**
   634→   634→   * Execute agent task with a timeout.
   635→   635→   */
   636→   636→  protected async executeWithTimeout(input: AgentInput): Promise<AgentOutput> {
   637→   637→    const timeoutMs = input.context?.timeout || this.config.timeout;
   638→   638→
   639→   639→    return new Promise<AgentOutput>((resolve, reject) => {
   640→   640→      const timeoutId = setTimeout(() => {
   641→   641→        this.activeTimeouts.delete(input.taskId);

   643→   643→          new AgentError(

   646→   646→            this.config.id,
   647→   647→            input.taskId,
   648→   648→          ),
   649→   649→        );
   650→   650→      }, timeoutMs);
   651→   651→
   652→   652→      this.activeTimeouts.set(input.taskId, timeoutId);
   653→   653→
   654→   654→      this.onExecute(input)
   655→   655→        .then((output) => {
   656→   656→          clearTimeout(timeoutId);
   657→   657→          this.activeTimeouts.delete(input.taskId);
   658→   658→          resolve(output);
   659→   659→        })
   660→   660→        .catch((error) => {
   661→   661→          clearTimeout(timeoutId);
   662→   662→          this.activeTimeouts.delete(input.taskId);
   663→   663→          reject(error);
   664→   664→        });
   665→   665→    });
   666→   666→  }

   683→   683→

   687→   687→          resolve(result);

   689→   689→        .catch((error) => {
   690→   690→          clearTimeout(timeoutId);
   691→   691→          reject(error);

   696→   696→  // ─── Retry Logic ─────────────────────────────────────────────────
   697→   697→
   698→   698→  /**
   699→   699→   * Calculate exponential backoff delay for a given retry attempt.
   700→   700→   */
   701→   701→  protected calculateBackoff(retryCount: number): number {
   702→   702→    const { backoffMs, exponentialBackoff } = this.config.retryPolicy;
   703→   703→    if (exponentialBackoff) {
   704→   704→      return backoffMs * Math.pow(2, retryCount - 1) + Math.random() * 100;
   705→   705→    }
   706→   706→    return backoffMs;
   707→   707→  }
   708→   708→
   709→   709→  /**
   710→   710→   * Execute a function with retry logic and exponential backoff.
   711→   711→   * Retries up to maxRetries times on failure.
   712→   712→   */
   713→   713→  protected async executeWithRetry<T>(fn: () => Promise<T>): Promise<T> {
   714→   714→    const { maxRetries, backoffMs, exponentialBackoff } = this.config.retryPolicy;
   715→   715→    let lastError: Error | null = null;
   716→   716→
   717→   717→    for (let attempt = 0; attempt <= maxRetries; attempt++) {
   718→   718→      try {
   719→   719→        return await fn();
   720→   720→      } catch (error) {
   721→   721→        lastError = error as Error;
   722→   722→
   723→   723→        if (attempt >= maxRetries) {
   724→   724→          this.logger.error(
   725→   725→            `All ${maxRetries} retries exhausted: ${lastError.message}`,
   726→   726→          );
   727→   727→          throw new AgentError(
   728→   728→            `Retry exhausted after ${maxRetries} attempts: ${lastError.message}`,
   729→   729→            AgentErrorCode.RETRY_EXHAUSTED,
   730→   730→            this.config.id,
   731→   731→            undefined,
   732→   732→            lastError,
   733→   733→          );
   734→   734→        }
   735→   735→
   736→   736→        const delay = exponentialBackoff
   737→   737→          ? backoffMs * Math.pow(2, attempt) + Math.random() * 100
   738→   738→          : backoffMs;
   739→   739→
   740→   740→        this.logger.warn(
   741→   741→          `Attempt ${attempt + 1}/${maxRetries} failed, retrying in ${Math.round(delay)}ms: ${lastError.message}`,
   742→   742→        );
   743→   743→
   744→   744→        await this.sleep(delay);
   745→   745→      }
   746→   746→    }
   747→   747→
   748→   748→    // This should never be reached, but TypeScript needs it
   749→   749→    throw lastError!;
   750→   750→  }
