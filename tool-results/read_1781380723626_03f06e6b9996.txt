   267→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   268→    const startTime = Date.now();
   269→    const { action, ...params } = input.payload;
   270→
   271→    if (!action) {
   272→      return this.createAgentOutput(
   273→        input.taskId,
   274→        false,
   275→        null,
   276→        'Missing required parameter: action',
   277→        startTime,
   278→      );
   279→    }
   280→
   281→    const supportedActions = [
   282→      'generateUnitTests',
   283→      'generateIntegrationTests',
   284→      'runTests',
   285→      'analyzeCoverage',
   286→      'generateFixtures',
   287→    ];
   288→
   289→    if (!supportedActions.includes(action)) {
   290→      return this.createAgentOutput(
   291→        input.taskId,
   292→        false,
   293→        null,
   294→        `Unknown testing action: ${action}. Supported: ${supportedActions.join(', ')}`,
   295→        startTime,
   296→      );
   297→    }
   298→
   299→    try {
   300→      const tool = this.getTool(action);
   301→      if (!tool) {
   302→        return this.createAgentOutput(
   303→          input.taskId,
   304→          false,
   305→          null,
   306→          `Tool not found: ${action}`,
   307→          startTime,
   308→        );
   309→      }
   310→
   311→      const result = await tool.execute(params);
   312→
   313→      await this.storeInWorkingMemory(
   314→        `testing:last:${action}`,
   315→        { params, result, timestamp: new Date() },
   316→        300000,
   317→      );
   318→
   319→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   320→    } catch (error) {
   321→      const msg = (error as Error).message;
   322→      this.logger.error(`Testing execution failed for ${action}: ${msg}`);
   323→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   324→    }
   325→  }

   335→  private async generateUnitTests(params: {
   336→    code: string;
   337→    language: string;
   338→    framework?: string;
   339→    filePath?: string;
   340→  }): Promise<{
   341→    testCode: string;
   342→    testFilePath: string;
   343→    testCases: TestCase[];
   344→    estimatedCoverage: number;
   345→  }> {
   346→    const { code, language, framework = 'jest', filePath } = params;
   347→
   348→    if (!code || typeof code !== 'string') {
   349→      throw new Error('Source code is required for test generation');
   350→    }
   351→    if (!language || typeof language !== 'string') {
   352→      throw new Error('Programming language is required');
   353→    }
   354→
   355→    // Extract function signatures from source code
   356→    const functions = this.extractFunctionSignatures(code, language);
   357→
   358→    if (functions.length === 0) {
   359→      throw new Error('No testable functions found in the provided code');
   360→    }
   361→
   362→    // Generate test cases for each function
   363→    const testCases: TestCase[] = [];
   364→    for (const func of functions) {
   365→      testCases.push(...this.generateTestCasesForFunction(func));
   366→    }
   367→
   368→    // Generate the test code
   369→    const testCode = this.renderUnitTestCode(testCases, functions, framework, language, filePath);
   370→
   371→    // Determine test file path
   372→    const testFilePath = filePath
   373→      ? filePath.replace(/\.\w+$/, `.spec.${this.getExtension(language)}`)
   374→      : `generated.spec.${this.getExtension(language)}`;
   375→
   376→    // Estimate coverage
   377→    const estimatedCoverage = Math.min(
   378→      100,
   379→      Math.round((testCases.length / (functions.length * 3)) * 80),
   380→    );
   381→
   382→    this.logger.log(
   383→      `Generated ${testCases.length} unit test(s) for ${functions.length} function(s), estimated coverage: ${estimatedCoverage}%`,
   384→    );
   385→
   386→    return { testCode, testFilePath, testCases, estimatedCoverage };
   387→  }

   390→    code: string;

   392→    framework?: string;
   393→    endpoints?: Array<{ method: string; path: string; body?: any }>;
   394→  }): Promise<{
   395→    testCode: string;
   396→    testFilePath: string;
   397→    testSuites: Array<{ name: string; tests: string[] }>;
   398→  }> {
   399→    const { code, language, framework = 'jest', endpoints } = params;
   400→
   401→    if (!code || typeof code !== 'string') {
   402→      throw new Error('Source code is required for integration test generation');
   403→    }
   404→    if (!language || typeof language !== 'string') {
   405→      throw new Error('Programming language is required');
   406→    }
   407→
   408→    // Extract endpoints from code if not provided
   409→    const extractedEndpoints = endpoints || this.extractEndpoints(code, language);
   410→
   411→    if (extractedEndpoints.length === 0) {
   412→      throw new Error('No endpoints found for integration test generation');
   413→    }
   414→
   415→    const testSuites: Array<{ name: string; tests: string[] }> = [];
   416→
   417→    for (const endpoint of extractedEndpoints) {
   418→      const suiteName = `${endpoint.method.toUpperCase()} ${endpoint.path}`;
   419→      const tests = this.generateEndpointTests(endpoint);
   420→      testSuites.push({ name: suiteName, tests });
   421→    }
   422→
   423→    // Render integration test code
   424→    const testCode = this.renderIntegrationTestCode(testSuites, framework, language);
   425→
   426→    const testFilePath = `integration.spec.${this.getExtension(language)}`;
   427→
   428→    this.logger.log(
   429→      `Generated integration tests: ${testSuites.length} suite(s), ${testSuites.reduce((sum, s) => sum + s.tests.length, 0)} test(s)`,
   430→    );
   431→
   432→    return { testCode, testFilePath, testSuites };
   433→  }

   435→  private async runTests(params: {
   436→    testPath: string;
   437→    framework?: string;
   438→    coverage?: boolean;
   439→    watch?: boolean;
   440→  }): Promise<{
   441→    passed: number;
   442→    failed: number;
   443→    skipped: number;
   444→    totalDuration: number;
   445→    coveragePercent: number | null;
   446→    results: TestResult[];
   447→  }> {
   448→    const { testPath, framework = 'jest', coverage = false, watch = false } = params;
   449→
   450→    if (!testPath || typeof testPath !== 'string') {
   451→      throw new Error('Test path is required');
   452→    }
   453→
   454→    // Simulate test execution
   455→    // In a real implementation, this would spawn a test runner process
   456→    const results = this.simulateTestExecution(testPath, framework);
   457→    const passed = results.filter((r) => r.status === 'passed').length;
   458→    const failed = results.filter((r) => r.status === 'failed').length;
   459→    const skipped = results.filter((r) => r.status === 'skipped').length;
   460→    const totalDuration = results.reduce((sum, r) => sum + r.duration, 0);
   461→
   462→    let coveragePercent: number | null = null;
   463→    if (coverage) {
   464→      coveragePercent = this.simulateCoverageResult(passed, failed, results.length);
   465→      this.coverageHistory.push({ timestamp: new Date(), coverage: coveragePercent });
   466→    }
   467→
   468→    // Store results
   469→    this.testResults.set(testPath, results);
   470→
   471→    // Store in memory for later reference
   472→    await this.storeInWorkingMemory(
   473→      `testing:results:${testPath}`,
   474→      { passed, failed, skipped, coveragePercent, timestamp: new Date() },
   475→      300000,
   476→    );
   477→
   478→    this.logger.log(
   479→      `Test run complete: ${passed} passed, ${failed} failed, ${skipped} skipped, ${totalDuration}ms`,
   480→    );
   481→
   482→    return { passed, failed, skipped, totalDuration, coveragePercent, results };
   483→  }

   485→  private async analyzeCoverage(params: {
   486→    coverageData: Record<string, any>;
   487→    sourceCode?: string;
   488→    threshold?: number;
   489→  }): Promise<{
   490→    lineCoverage: number;
   491→    branchCoverage: number;
   492→    functionCoverage: number;
   493→    statementCoverage: number;
   494→    uncoveredPaths: UncoveredPath[];
   495→    meetsThreshold: boolean;
   496→  }> {
   497→    const { coverageData, sourceCode, threshold = 80 } = params;
   498→
   499→    if (!coverageData || typeof coverageData !== 'object') {
   500→      throw new Error('Coverage data is required');
   501→    }
   502→
   503→    // Extract or simulate coverage metrics
   504→    const lineCoverage = coverageData.lines?.pct ?? this.simulateMetric(60, 95);
   505→    const branchCoverage = coverageData.branches?.pct ?? this.simulateMetric(40, 90);
   506→    const functionCoverage = coverageData.functions?.pct ?? this.simulateMetric(50, 95);
   507→    const statementCoverage = coverageData.statements?.pct ?? this.simulateMetric(55, 95);
   508→
   509→    // Identify uncovered paths
   510→    const uncoveredPaths: UncoveredPath[] = [];
   511→
   512→    if (sourceCode) {
   513→      const lines = sourceCode.split('\n');
   514→      this.identifyUncoveredPaths(
   515→        lines,
   516→        lineCoverage,
   517→        branchCoverage,
   518→        functionCoverage,
   519→        uncoveredPaths,
   520→      );
   521→    } else {
   522→      // Generate generic uncovered paths from coverage data
   523→      this.generateGenericUncoveredPaths(coverageData, uncoveredPaths);
   524→    }
   525→
   526→    const meetsThreshold =
   527→      lineCoverage >= threshold && branchCoverage >= threshold && functionCoverage >= threshold;
   528→
   529→    this.logger.log(
   530→      `Coverage analysis: lines=${lineCoverage}%, branches=${branchCoverage}%, functions=${functionCoverage}%, meets ${threshold}% threshold: ${meetsThreshold}`,
   531→    );
   532→
   533→    return {
   534→      lineCoverage,
   535→      branchCoverage,
   536→      functionCoverage,
   537→      statementCoverage,
   538→      uncoveredPaths,
   539→      meetsThreshold,
   540→    };
   541→  }

   557→    }

   561→
   562→    // Generate fixture data based on schema
   563→    const fixtures: Record<string, any>[] = [];
   564→    for (let i = 0; i < count; i++) {

   567→    }
   568→
   569→    // Render fixtures in the requested format
   570→    let fixtureCode: string;
   571→    let filePath: string;
   572→
   573→    switch (format) {
   574→      case 'typescript':
   575→        fixtureCode = this.renderTypeScriptFixtures(fixtures, schema);
   576→        filePath = 'fixtures.ts';
   577→        break;
   578→      case 'python':
   579→        fixtureCode = this.renderPythonFixtures(fixtures);
   580→        filePath = 'fixtures.py';
   581→        break;
   582→      case 'json':
   583→      default:
   584→        fixtureCode = JSON.stringify(fixtures, null, 2);
   585→        filePath = 'fixtures.json';
   586→        break;
   587→    }
   588→
   589→    this.logger.log(`Generated ${count} fixture(s) in ${format} format`);
   590→
   591→    return { fixtures, fixtureCode, filePath };
   592→  }

   594→  // ─── Function Signature Extraction ─────────────────────────────
   595→
   596→  private extractFunctionSignatures(code: string, language: string): FunctionSignature[] {
   597→    const functions: FunctionSignature[] = [];
   598→
   599→    if (language === 'typescript' || language === 'javascript') {
   600→      // Named function declarations
   601→      const funcDeclRegex =
   602→        /(?:export\s+)?(?:async\s+)?function\s+(\w+)\s*\(([^)]*)\)(?:\s*:\s*(\w+))?/g;
   603→      let match: RegExpExecArray | null;
   604→      while ((match = funcDeclRegex.exec(code)) !== null) {
   605→        functions.push({
   606→          name: match[1],
   607→          params: this.parseParams(match[2]),
   608→          returnType: match[3] || 'void',
   609→          isAsync: code.substring(match.index - 10, match.index).includes('async'),
   610→          isExported: code.substring(match.index - 10, match.index).includes('export'),
   611→        });
   612→      }
   613→
   614→      // Arrow functions / const declarations
   615→      const arrowRegex =
   616→        /(?:export\s+)?(?:const|let)\s+(\w+)\s*=\s*(?:async\s+)?(?:\(([^)]*)\)|(\w+))\s*(?::\s*(\w+))?\s*=>/g;
   617→      while ((match = arrowRegex.exec(code)) !== null) {
   618→        functions.push({
   619→          name: match[1],
   620→          params: this.parseParams(match[2] || match[3] || ''),
   621→          returnType: match[4] || 'void',
   622→          isAsync: code.substring(match.index, match.index + 30).includes('async'),
   623→          isExported: code.substring(match.index - 10, match.index).includes('export'),
   624→        });
   625→      }
   626→
   627→      // Class methods
   628→      const methodRegex =
   629→        /(?:public|private|protected)?\s*(?:async\s+)?(\w+)\s*\(([^)]*)\)(?:\s*:\s*(\w+))?\s*{/g;
   630→      while ((match = methodRegex.exec(code)) !== null) {
   631→        if (!['constructor', 'if', 'for', 'while', 'switch', 'catch'].includes(match[1])) {
   632→          functions.push({
   633→            name: match[1],
   634→            params: this.parseParams(match[2]),
   635→            returnType: match[3] || 'void',
   636→            isAsync: code.substring(match.index, match.index + 30).includes('async'),
   637→            isExported: false,
   638→          });
   639→        }
   640→      }
   641→    }
   642→
   643→    if (language === 'python') {
   644→      const funcRegex = /def\s+(\w+)\s*\(([^)]*)\)(?:\s*->\s*(\w+))?:/g;
   645→      let match: RegExpExecArray | null;
   646→      while ((match = funcRegex.exec(code)) !== null) {
   647→        functions.push({
   648→          name: match[1],
   649→          params: this.parsePythonParams(match[2]),
   650→          returnType: match[3] || 'None',
   651→          isAsync: code.substring(match.index - 10, match.index).includes('async'),
   652→          isExported: false,
   653→        });
   654→      }
   655→    }
   656→
   657→    return functions;
   658→  }
   659→
   660→  private parseParams(paramStr: string): Array<{ name: string; type: string }> {
   661→    if (!paramStr.trim()) return [];
   662→
   663→    return paramStr
   664→      .split(',')
   665→      .map((param) => {
   666→        const trimmed = param.trim();
   667→        const parts = trimmed.split(':');
   668→        const name = parts[0].replace(/\?.*$/, '').trim();
   669→        const type = parts[1] ? parts[1].replace(/\s*=\s*.*$/, '').trim() : 'any';
   670→        return { name, type };
   671→      })
   672→      .filter((p) => p.name.length > 0);
   673→  }
   674→
   675→  private parsePythonParams(paramStr: string): Array<{ name: string; type: string }> {
   676→    if (!paramStr.trim()) return [];
   677→
   678→    return paramStr
   679→      .split(',')
   680→      .map((param) => {
   681→        const trimmed = param.trim();
   682→        if (trimmed.startsWith('self') || trimmed.startsWith('cls')) {
   683→          return { name: trimmed, type: 'self' };
   684→        }
   685→        const parts = trimmed.split(':');
   686→        const name = parts[0].replace(/\s*=\s*.*$/, '').trim();
   687→        const type = parts[1] ? parts[1].replace(/\s*=\s*.*$/, '').trim() : 'Any';
   688→        return { name, type };
   689→      })
   690→      .filter((p) => p.name.length > 0 && p.type !== 'self');
   691→  }

   695→  private generateTestCasesForFunction(func: FunctionSignature): TestCase[] {
   696→    const cases: TestCase[] = [];
   697→
   698→    // Positive test case
   699→    cases.push({
   700→      name: `${func.name} - should execute successfully with valid input`,
   701→      description: `Valid input test for ${func.name}`,
   702→      input: this.generateValidInput(func.params),
   703→      expectedOutput: this.generateExpectedOutput(func.returnType),
   704→      type: 'positive',
   705→    });
   706→
   707→    // Negative test case
   708→    if (func.params.length > 0) {
   709→      cases.push({
   710→        name: `${func.name} - should handle invalid input gracefully`,
   711→        description: `Invalid input test for ${func.name}`,
   712→        input: this.generateInvalidInput(func.params),
   713→        expectedOutput: null,
   714→        type: 'negative',
   715→      });
   716→
   717→      // Edge cases for each parameter
   718→      for (const param of func.params) {
   719→        cases.push({
   720→          name: `${func.name} - should handle edge case for ${param.name}`,
   721→          description: `Edge case test for parameter ${param.name} of type ${param.type}`,
   722→          input: this.generateEdgeCaseInput(func.params, param),
   723→          expectedOutput: null,
   724→          type: 'edge',
   725→        });
   726→      }
   727→    }
   728→
   729→    // Boundary test for numeric params
   730→    const numericParams = func.params.filter((p) =>
   731→      ['number', 'number', 'int', 'float', 'double'].includes(p.type.toLowerCase()),
   732→    );
   733→    for (const param of numericParams) {
   734→      cases.push({
   735→        name: `${func.name} - should handle boundary values for ${param.name}`,
   736→        description: `Boundary test for ${param.name}`,
   737→        input: { ...this.generateValidInput(func.params), [param.name]: 0 },
   738→        expectedOutput: null,
   739→        type: 'boundary',
   740→      });
   741→    }
   742→
   743→    // Null/undefined test
   744→    if (func.params.length > 0) {
   745→      cases.push({
   746→        name: `${func.name} - should handle null/undefined input`,
   747→        description: `Null/undefined input test for ${func.name}`,
   748→        input: this.generateNullInput(func.params),
   749→        expectedOutput: null,
   750→        type: 'negative',
   751→      });
   752→    }
   753→
   754→    return cases;
   755→  }

   912→

   914→      imports.length > 0
   915→        ? `import { ${imports.join(', ')} } from '${importPath}';`

   919→      .map((tc) => {

   925→    const input = ${inputStr};

   927→      tc.type === 'negative'
   928→        ? `await expect(${tc.name.split(' - ')[0].split('.').pop()}(input)).rejects.toThrow();`

   931→    }
   932→  });`;
   933→      })

   935→
   936→    return `${importStatement}
   937→

   940→});
   941→`;

   995→  private extractEndpoints(
   996→    code: string,
   997→    language: string,
   998→  ): Array<{ method: string; path: string; body?: any }> {
   999→    const endpoints: Array<{ method: string; path: string; body?: any }> = [];
  1000→
  1001→    if (language === 'typescript' || language === 'javascript') {
  1002→      const routeRegex = /@(Get|Post|Put|Delete|Patch)\(['"]([^'"]+)['"]\)/g;
  1003→      let match: RegExpExecArray | null;
  1004→      while ((match = routeRegex.exec(code)) !== null) {
  1005→        endpoints.push({ method: match[1].toLowerCase(), path: match[2] });
  1006→      }
  1007→
  1008→      // Express-style routes
  1009→      const expressRegex = /(?:router|app)\.(get|post|put|delete|patch)\(['"]([^'"]+)['"]/g;
  1010→      while ((match = expressRegex.exec(code)) !== null) {
  1011→        endpoints.push({ method: match[1].toLowerCase(), path: match[2] });
  1012→      }
  1013→    }
  1014→
  1015→    if (language === 'python') {
  1016→      const flaskRegex = /@(app|bp)\.route\(['"]([^'"]+)['"]\],\s*methods\s*=\s*\[(\w+)\]/g;
  1017→      let match: RegExpExecArray | null;
  1018→      while ((match = flaskRegex.exec(code)) !== null) {
  1019→        endpoints.push({ method: match[3].toLowerCase(), path: match[2] });
  1020→      }
  1021→    }
  1022→
  1023→    return endpoints;
  1024→  }

  1026→  private generateEndpointTests(endpoint: { method: string; path: string; body?: any }): string[] {
  1027→    const tests: string[] = [];

  1031→    tests.push(`should return 200 for ${methodName} ${endpoint.path}`);

  1033→    // Error tests

  1044→    // Not found test (for parameterized routes)

  1046→      tests.push(`should return 404 for ${methodName} ${endpoint.path} with non-existent resource`);

  1117→  private identifyUncoveredPaths(
  1118→    lines: string[],
  1119→    lineCoverage: number,
  1120→    branchCoverage: number,
  1121→    functionCoverage: number,
  1122→    uncoveredPaths: UncoveredPath[],
  1123→  ): void {
  1124→    // Find functions that might not be covered
  1125→    if (functionCoverage < 100) {
  1126→      for (let i = 0; i < lines.length; i++) {
  1127→        const funcMatch = lines[i].match(/(?:function\s+(\w+)|(?:const|let)\s+(\w+)\s*=)/);
  1128→        if (funcMatch) {
  1129→          uncoveredPaths.push({
  1130→            file: 'source',
  1131→            lineStart: i + 1,
  1132→            lineEnd: i + 1,
  1133→            type: 'function',
  1134→            description: `Function "${funcMatch[1] || funcMatch[2]}" may not be covered by tests`,
  1135→            suggestion: `Add test cases that call this function directly`,
  1136→          });
  1137→        }
  1138→      }
  1139→    }
  1140→
  1141→    // Find branches that might not be covered
  1142→    if (branchCoverage < 100) {
  1143→      for (let i = 0; i < lines.length; i++) {
  1144→        if (lines[i].includes('if (') || lines[i].includes('if(')) {
  1145→          uncoveredPaths.push({
  1146→            file: 'source',
  1147→            lineStart: i + 1,
  1148→            lineEnd: i + 1,
  1149→            type: 'branch',
  1150→            description: `Conditional branch at line ${i + 1} may not be fully covered`,
  1151→            suggestion: `Add test cases for both true and false branches of this condition`,
  1152→          });
  1153→        }
  1154→      }
  1155→    }
  1156→
  1157→    // Limit uncovered paths reported
  1158→    if (uncoveredPaths.length > 20) {
  1159→      uncoveredPaths.splice(20);
  1160→    }
  1161→  }

  1211→  private generateValueFromSchema(
  1212→    propSchema: Record<string, any>,
  1213→    key: string,
  1214→    index: number,
  1215→  ): any {
  1216→    if (propSchema.enum) {
  1217→      return propSchema.enum[index % propSchema.enum.length];
  1218→    }
  1219→
  1220→    switch (propSchema.type) {
  1221→      case 'string':
  1222→        if (key.toLowerCase().includes('email')) return `user${index}@example.com`;
  1223→        if (key.toLowerCase().includes('name')) return `Test Name ${index}`;
  1224→        if (key.toLowerCase().includes('url') || key.toLowerCase().includes('link'))
  1225→          return `https://example.com/${index}`;
  1226→        if (key.toLowerCase().includes('id')) return `${1000 + index}`;
  1227→        if (key.toLowerCase().includes('date') || key.toLowerCase().includes('at'))
  1228→          return new Date(Date.now() + index * 86400000).toISOString();
  1229→        return `test-${key}-${index}`;
  1230→      case 'number':
  1231→      case 'integer':
  1232→        if (key.toLowerCase().includes('price') || key.toLowerCase().includes('amount'))
  1233→          return Math.round((10 + index * 5.99) * 100) / 100;
  1234→        if (key.toLowerCase().includes('age')) return 20 + (index % 60);
  1235→        return index + 1;
  1236→      case 'boolean':
  1237→        return index % 2 === 0;
  1238→      case 'array':
  1239→        return [];
  1240→      case 'object':
  1241→        return {};
  1242→      default:
  1243→        return `fixture-${key}-${index}`;
  1244→    }
  1245→  }

  1247→  private generateValueByTypeHint(typeHint: string, key: string, index: number): any {
  1248→    const lower = typeHint.toLowerCase();
  1249→    if (lower.includes('string')) return `test-${key}-${index}`;
  1250→    if (lower.includes('number') || lower.includes('int')) return index + 1;
  1251→    if (lower.includes('bool')) return index % 2 === 0;
  1252→    if (lower.includes('date')) return new Date().toISOString();
  1253→    return `fixture-${key}-${index}`;
  1254→  }

  1266→        const type =
  1267→          prop.type === 'string'
  1268→            ? 'string'
  1269→            : prop.type === 'number' || prop.type === 'integer'
  1270→              ? 'number'
  1271→              : prop.type === 'boolean'
  1272→                ? 'boolean'
  1273→                : prop.type === 'array'

  1275→                  : 'any';

  1280→    const fixtureData = fixtures

  1284→      )
