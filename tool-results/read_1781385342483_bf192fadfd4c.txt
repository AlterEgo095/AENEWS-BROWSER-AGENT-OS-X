    16→/**
    17→ * Represents a payload submitted when an agent registers its capabilities
    18→ * with the Mission OS capability registry.
    19→ */
    20→export interface CapabilityRegistration {
    21→  /** Unique identifier of the registering agent */
    22→  agentId: string;
    23→  /** List of capabilities this agent provides */
    24→  capabilities: CapabilityDescriptor[];
    25→  /** Timestamp of the registration event */
    26→  registeredAt: string;
    27→  /** Optional agent metadata (name, version, runtime, etc.) */
    28→  agentMetadata?: Record<string, unknown>;
    29→}
    30→
    31→/**
    32→ * Full description of a single capability exposed by an agent,
    33→ * including schemas, versioning, and cost/latency estimates.
    34→ */
    35→export interface CapabilityDescriptor {
    36→  /** Human-readable capability name (e.g., "web-search", "code-generation") */
    37→  name: string;
    38→  /** Detailed description of what this capability does */
    39→  description: string;
    40→  /** JSON Schema describing the expected input shape */
    41→  inputSchema: Record<string, unknown>;
    42→  /** JSON Schema describing the output shape */
    43→  outputSchema: Record<string, unknown>;
    44→  /** Semantic version of this capability (e.g., "1.2.0") */
    45→  version: string;
    46→  /** Whether this capability is deprecated and slated for removal */
    47→  deprecated: boolean;
    48→  /** Estimated cost per invocation (in USD or internal credits) */
    49→  costEstimate: number;
    50→  /** Estimated latency in milliseconds for a typical invocation */
    51→  latencyEstimate: number;
    52→  /** Optional tags for categorization and discovery */
    53→  tags?: string[];
    54→  /** Optional deprecation message with migration guidance */
    55→  deprecationMessage?: string;
    56→  /** Required permissions or roles to invoke this capability */
    57→  requiredPermissions?: string[];
    58→}
