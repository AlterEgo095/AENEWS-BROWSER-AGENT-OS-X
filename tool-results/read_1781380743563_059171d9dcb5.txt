   384→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   385→    const startTime = Date.now();
   386→    const { action, ...params } = input.payload;
   387→
   388→    if (!action) {
   389→      return this.createAgentOutput(
   390→        input.taskId,
   391→        false,
   392→        null,
   393→        'Missing required parameter: action',
   394→        startTime,
   395→      );
   396→    }
   397→
   398→    const supportedActions = [
   399→      'createTask',
   400→      'updateTask',
   401→      'assignTask',
   402→      'trackProgress',
   403→      'generateReport',
   404→      'setDeadline',
   405→      'prioritizeTask',
   406→    ];
   407→
   408→    if (!supportedActions.includes(action)) {
   409→      return this.createAgentOutput(
   410→        input.taskId,
   411→        false,
   412→        null,
   413→        `Unknown task management action: ${action}. Supported: ${supportedActions.join(', ')}`,
   414→        startTime,
   415→      );
   416→    }
   417→
   418→    try {
   419→      const tool = this.getTool(action);
   420→      if (!tool) {
   421→        return this.createAgentOutput(
   422→          input.taskId,
   423→          false,
   424→          null,
   425→          `Tool not found: ${action}`,
   426→          startTime,
   427→        );
   428→      }
   429→
   430→      const result = await tool.execute(params);
   431→
   432→      await this.storeInWorkingMemory(
   433→        `task-mgmt:last:${action}`,
   434→        { params, result, timestamp: new Date() },
   435→        300000,
   436→      );
   437→
   438→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   439→    } catch (error) {
   440→      const msg = (error as Error).message;
   441→      this.logger.error(`TaskManagement execution failed for ${action}: ${msg}`);
   442→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   443→    }
   444→  }

   455→  private async createTask(params: {
   456→    title: string;
   457→    description?: string;
   458→    projectId?: string;
   459→    assignee?: string;
   460→    priority?: string;
   461→    deadline?: string;
   462→    tags?: string[];
   463→    subtasks?: Array<{ title: string }>;
   464→    estimatedHours?: number;
   465→  }): Promise<{
   466→    taskId: string;
   467→    status: string;
   468→    createdAt: string;
   469→  }> {
   470→    const {
   471→      title,
   472→      description = '',
   473→      projectId = '',
   474→      assignee = '',
   475→      priority = 'medium',
   476→      deadline,
   477→      tags = [],
   478→      subtasks = [],
   479→      estimatedHours = 0,
   480→    } = params;
   481→
   482→    if (!title || typeof title !== 'string') {
   483→      throw new Error('A valid task title is required');
   484→    }
   485→
   486→    const validPriorities: TaskPriority[] = ['low', 'medium', 'high', 'critical'];
   487→    if (!validPriorities.includes(priority as TaskPriority)) {
   488→      throw new Error(`Invalid priority: ${priority}. Supported: ${validPriorities.join(', ')}`);
   489→    }
   490→
   491→    let deadlineDate: Date | null = null;
   492→    if (deadline) {
   493→      deadlineDate = new Date(deadline);
   494→      if (isNaN(deadlineDate.getTime())) {
   495→        throw new Error('Invalid deadline: must be a valid ISO date string');
   496→      }
   497→    }
   498→
   499→    const taskId = this.generateTaskId();
   500→    const taskSubtasks: Subtask[] = subtasks.map((st, i) => ({
   501→      id: `${taskId}-sub-${i + 1}`,
   502→      title: st.title,
   503→      completed: false,
   504→    }));
   505→
   506→    const task: TaskItem = {
   507→      id: taskId,
   508→      title,
   509→      description,
   510→      projectId,
   511→      assignee,
   512→      priority: priority as TaskPriority,
   513→      status: 'todo',
   514→      progress: 0,
   515→      deadline: deadlineDate,
   516→      tags,
   517→      subtasks: taskSubtasks,
   518→      estimatedHours,
   519→      actualHours: 0,
   520→      comments: [],
   521→      createdAt: new Date(),
   522→      updatedAt: new Date(),
   523→      completedAt: null,
   524→    };
   525→
   526→    this.tasks.set(taskId, task);
   527→
   528→    // Add to project if specified
   529→    if (projectId) {
   530→      const project = this.projects.get(projectId);
   531→      if (project) {
   532→        project.taskIds.push(taskId);
   533→      } else {
   534→        // Auto-create project
   535→        this.projects.set(projectId, {
   536→          id: projectId,
   537→          name: projectId,
   538→          description: `Project ${projectId}`,
   539→          taskIds: [taskId],
   540→          createdAt: new Date(),
   541→        });
   542→      }
   543→    }
   544→
   545→    this.logger.log(`Created task: ${taskId}, title="${title}", priority=${priority}`);
   546→
   547→    return {
   548→      taskId,
   549→      status: 'todo',
   550→      createdAt: task.createdAt.toISOString(),
   551→    };
   552→  }

   554→  private async updateTask(params: {
   555→    taskId: string;
   556→    title?: string;
   557→    description?: string;
   558→    status?: string;
   559→    assignee?: string;
   560→    priority?: string;
   561→    progress?: number;
   562→    tags?: string[];
   563→    actualHours?: number;
   564→  }): Promise<{
   565→    taskId: string;
   566→    updatedFields: string[];
   567→    status: string;
   568→  }> {
   569→    const { taskId, title, description, status, assignee, priority, progress, tags, actualHours } =
   570→      params;
   571→
   572→    if (!taskId || typeof taskId !== 'string') {
   573→      throw new Error('A valid taskId is required');
   574→    }
   575→
   576→    const task = this.tasks.get(taskId);
   577→    if (!task) {
   578→      throw new Error(`Task not found: ${taskId}`);
   579→    }
   580→
   581→    const updatedFields: string[] = [];
   582→
   583→    if (title !== undefined) {
   584→      task.title = title;
   585→      updatedFields.push('title');
   586→    }
   587→    if (description !== undefined) {
   588→      task.description = description;
   589→      updatedFields.push('description');
   590→    }
   591→    if (status !== undefined) {
   592→      const validStatuses: TaskStatus[] = [
   593→        'todo',
   594→        'in_progress',
   595→        'review',
   596→        'done',
   597→        'blocked',
   598→        'cancelled',
   599→      ];
   600→      if (!validStatuses.includes(status as TaskStatus)) {
   601→        throw new Error(`Invalid status: ${status}. Supported: ${validStatuses.join(', ')}`);
   602→      }
   603→      task.status = status as TaskStatus;
   604→      updatedFields.push('status');
   605→
   606→      // Auto-set progress and completion
   607→      if (status === 'done') {
   608→        task.progress = 100;
   609→        task.completedAt = new Date();
   610→      } else if (status === 'in_progress' && task.progress === 0) {
   611→        task.progress = 10;
   612→      }
   613→    }
   614→    if (assignee !== undefined) {
   615→      task.assignee = assignee;
   616→      updatedFields.push('assignee');
   617→    }
   618→    if (priority !== undefined) {
   619→      const validPriorities: TaskPriority[] = ['low', 'medium', 'high', 'critical'];
   620→      if (!validPriorities.includes(priority as TaskPriority)) {
   621→        throw new Error(`Invalid priority: ${priority}. Supported: ${validPriorities.join(', ')}`);
   622→      }
   623→      task.priority = priority as TaskPriority;
   624→      updatedFields.push('priority');
   625→    }
   626→    if (progress !== undefined) {
   627→      if (progress < 0 || progress > 100) {
   628→        throw new Error('Progress must be between 0 and 100');
   629→      }
   630→      task.progress = progress;
   631→      updatedFields.push('progress');
   632→
   633→      // Auto-update status based on progress
   634→      if (progress === 100 && task.status !== 'done') {
   635→        task.status = 'done';
   636→        task.completedAt = new Date();
   637→      } else if (progress > 0 && task.status === 'todo') {
   638→        task.status = 'in_progress';
   639→      }
   640→    }
   641→    if (tags !== undefined) {
   642→      task.tags = tags;
   643→      updatedFields.push('tags');
   644→    }
   645→    if (actualHours !== undefined) {
   646→      task.actualHours = actualHours;
   647→      updatedFields.push('actualHours');
   648→    }
   649→
   650→    task.updatedAt = new Date();
   651→
   652→    this.logger.log(`Updated task: ${taskId}, fields=[${updatedFields.join(',')}]`);
   653→
   654→    return {
   655→      taskId,
   656→      updatedFields,
   657→      status: task.status,
   658→    };
   659→  }

   663→    assignee: string;
   664→    message?: string;

   718→  private async trackProgress(params: {
   719→    projectId?: string;
   720→    assignee?: string;
   721→    status?: string;
   722→    dateFrom?: string;
   723→    dateTo?: string;
   724→  }): Promise<{
   725→    totalTasks: number;
   726→    completedTasks: number;
   727→    inProgressTasks: number;
   728→    overdueTasks: number;
   729→    averageProgress: number;
   730→    tasks: TaskItem[];
   731→  }> {
   732→    const { projectId, assignee, status, dateFrom, dateTo } = params;
   733→
   734→    let filteredTasks = Array.from(this.tasks.values());
   735→
   736→    // Filter by project
   737→    if (projectId) {
   738→      const project = this.projects.get(projectId);
   739→      if (project) {
   740→        const projectTaskIds = new Set(project.taskIds);
   741→        filteredTasks = filteredTasks.filter((t) => projectTaskIds.has(t.id));
   742→      }
   743→    }
   744→
   745→    // Filter by assignee
   746→    if (assignee) {
   747→      filteredTasks = filteredTasks.filter((t) => t.assignee === assignee);
   748→    }
   749→
   750→    // Filter by status
   751→    if (status) {
   752→      filteredTasks = filteredTasks.filter((t) => t.status === status);
   753→    }
   754→
   755→    // Filter by date range
   756→    if (dateFrom) {
   757→      const from = new Date(dateFrom);
   758→      if (!isNaN(from.getTime())) {
   759→        filteredTasks = filteredTasks.filter((t) => t.createdAt >= from);
   760→      }
   761→    }
   762→    if (dateTo) {
   763→      const to = new Date(dateTo);
   764→      if (!isNaN(to.getTime())) {
   765→        filteredTasks = filteredTasks.filter((t) => t.createdAt <= to);
   766→      }
   767→    }
   768→
   769→    // Calculate metrics
   770→    const totalTasks = filteredTasks.length;
   771→    const completedTasks = filteredTasks.filter((t) => t.status === 'done').length;
   772→    const inProgressTasks = filteredTasks.filter(
   773→      (t) => t.status === 'in_progress' || t.status === 'review',
   774→    ).length;
   775→
   776→    const now = new Date();
   777→    const overdueTasks = filteredTasks.filter(
   778→      (t) => t.deadline && t.deadline < now && t.status !== 'done' && t.status !== 'cancelled',
   779→    ).length;
   780→
   781→    const averageProgress =
   782→      totalTasks > 0
   783→        ? Math.round(filteredTasks.reduce((sum, t) => sum + t.progress, 0) / totalTasks)
   784→        : 0;
   785→
   786→    this.logger.log(
   787→      `Tracked progress: total=${totalTasks}, done=${completedTasks}, in-progress=${inProgressTasks}, overdue=${overdueTasks}`,
   788→    );
   789→
   790→    return {
   791→      totalTasks,
   792→      completedTasks,
   793→      inProgressTasks,
   794→      overdueTasks,
   795→      averageProgress,
   796→      tasks: filteredTasks,
   797→    };
   798→  }

   800→  private async generateReport(params: {
   801→    projectId?: string;
   802→    reportType: string;
   803→    assignee?: string;
   804→    dateFrom?: string;
   805→    dateTo?: string;
   806→  }): Promise<{
   807→    reportType: string;
   808→    generatedAt: string;
   809→    data: any;
   810→  }> {
   811→    const { projectId, reportType, assignee, dateFrom, dateTo } = params;
   812→
   813→    if (!reportType || typeof reportType !== 'string') {
   814→      throw new Error('A valid reportType is required');
   815→    }
   816→
   817→    const validReportTypes = ['summary', 'detailed', 'burndown', 'velocity'];
   818→    if (!validReportTypes.includes(reportType)) {
   819→      throw new Error(
   820→        `Invalid reportType: ${reportType}. Supported: ${validReportTypes.join(', ')}`,
   821→      );
   822→    }
   823→
   824→    // Get filtered tasks
   825→    let filteredTasks = Array.from(this.tasks.values());
   826→
   827→    if (projectId) {
   828→      const project = this.projects.get(projectId);
   829→      if (project) {
   830→        const projectTaskIds = new Set(project.taskIds);
   831→        filteredTasks = filteredTasks.filter((t) => projectTaskIds.has(t.id));
   832→      }
   833→    }
   834→
   835→    if (assignee) {
   836→      filteredTasks = filteredTasks.filter((t) => t.assignee === assignee);
   837→    }
   838→
   839→    if (dateFrom) {
   840→      const from = new Date(dateFrom);
   841→      if (!isNaN(from.getTime())) {
   842→        filteredTasks = filteredTasks.filter((t) => t.createdAt >= from);
   843→      }
   844→    }
   845→
   846→    if (dateTo) {
   847→      const to = new Date(dateTo);
   848→      if (!isNaN(to.getTime())) {
   849→        filteredTasks = filteredTasks.filter((t) => t.createdAt <= to);
   850→      }
   851→    }
   852→
   853→    const now = new Date();
   854→    const generatedAt = now.toISOString();
   855→
   856→    let data: any;
   857→
   858→    switch (reportType) {
   859→      case 'summary': {
   860→        const byStatus: Record<string, number> = {};
   861→        const byPriority: Record<string, number> = {};
   862→        const byAssignee: Record<string, number> = {};
   863→
   864→        for (const task of filteredTasks) {
   865→          byStatus[task.status] = (byStatus[task.status] || 0) + 1;
   866→          byPriority[task.priority] = (byPriority[task.priority] || 0) + 1;
   867→          if (task.assignee) {
   868→            byAssignee[task.assignee] = (byAssignee[task.assignee] || 0) + 1;
   869→          }
   870→        }
   871→
   872→        const totalEstimated = filteredTasks.reduce((s, t) => s + t.estimatedHours, 0);
   873→        const totalActual = filteredTasks.reduce((s, t) => s + t.actualHours, 0);
   874→        const overdue = filteredTasks.filter(
   875→          (t) => t.deadline && t.deadline < now && t.status !== 'done' && t.status !== 'cancelled',
   876→        ).length;
   877→
   878→        data = {
   879→          totalTasks: filteredTasks.length,
   880→          byStatus,
   881→          byPriority,
   882→          byAssignee,
   883→          totalEstimatedHours: totalEstimated,
   884→          totalActualHours: totalActual,
   885→          varianceHours: totalActual - totalEstimated,
   886→          overdueTasks: overdue,
   887→          completionRate:
   888→            filteredTasks.length > 0
   889→              ? Math.round(((byStatus['done'] || 0) / filteredTasks.length) * 100)
   890→              : 0,
   891→        };
   892→        break;
   893→      }
   894→      case 'detailed': {
   895→        data = {
   896→          tasks: filteredTasks.map((t) => ({
   897→            id: t.id,
   898→            title: t.title,
   899→            status: t.status,
   900→            priority: t.priority,
   901→            assignee: t.assignee,
   902→            progress: t.progress,
   903→            deadline: t.deadline?.toISOString() || null,
   904→            isOverdue: t.deadline
   905→              ? t.deadline < now && t.status !== 'done' && t.status !== 'cancelled'
   906→              : false,
   907→            estimatedHours: t.estimatedHours,
   908→            actualHours: t.actualHours,
   909→            subtaskCompletion:
   910→              t.subtasks.length > 0
   911→                ? `${t.subtasks.filter((s) => s.completed).length}/${t.subtasks.length}`
   912→                : 'N/A',
   913→            createdAt: t.createdAt.toISOString(),
   914→            completedAt: t.completedAt?.toISOString() || null,
   915→          })),
   916→        };
   917→        break;
   918→      }
   919→      case 'burndown': {
   920→        // Generate daily burndown data
   921→        const completedTasksByDay: Record<string, number> = {};
   922→        const createdTasksByDay: Record<string, number> = {};
   923→
   924→        for (const task of filteredTasks) {
   925→          const createdDay = task.createdAt.toISOString().split('T')[0];
   926→          createdTasksByDay[createdDay] = (createdTasksByDay[createdDay] || 0) + 1;
   927→
   928→          if (task.completedAt) {
   929→            const completedDay = task.completedAt.toISOString().split('T')[0];
   930→            completedTasksByDay[completedDay] = (completedTasksByDay[completedDay] || 0) + 1;
   931→          }
   932→        }
   933→
   934→        data = {
   935→          totalTasks: filteredTasks.length,
   936→          completedTasks: filteredTasks.filter((t) => t.status === 'done').length,
   937→          remainingTasks: filteredTasks.filter(
   938→            (t) => t.status !== 'done' && t.status !== 'cancelled',
   939→          ).length,
   940→          createdTasksByDay,
   941→          completedTasksByDay,
   942→          idealBurndownRate:
   943→            filteredTasks.length > 0
   944→              ? filteredTasks.length /
   945→                Math.max(
   946→                  1,
   947→                  Math.ceil(
   948→                    (now.getTime() - Math.min(...filteredTasks.map((t) => t.createdAt.getTime()))) /
   949→                      86400000,
   950→                  ),
   951→                )
   952→              : 0,
   953→        };
   954→        break;
   955→      }
   956→      case 'velocity': {
   957→        // Calculate weekly velocity
   958→        const weeklyCompleted: Record<string, number> = {};
   959→
   960→        for (const task of filteredTasks) {
   961→          if (task.completedAt) {
   962→            const weekStart = this.getWeekStart(task.completedAt);
   963→            const weekKey = weekStart.toISOString().split('T')[0];
   964→            weeklyCompleted[weekKey] = (weeklyCompleted[weekKey] || 0) + 1;
   965→          }
   966→        }
   967→
   968→        const completedCounts = Object.values(weeklyCompleted);
   969→        const avgVelocity =
   970→          completedCounts.length > 0
   971→            ? completedCounts.reduce((a, b) => a + b, 0) / completedCounts.length
   972→            : 0;
   973→
   974→        const totalRemaining = filteredTasks.filter(
   975→          (t) => t.status !== 'done' && t.status !== 'cancelled',
   976→        ).length;
   977→
   978→        data = {
   979→          weeklyCompleted,
   980→          averageWeeklyVelocity: Math.round(avgVelocity * 10) / 10,
   981→          totalRemaining,
   982→          estimatedWeeksToComplete:
   983→            avgVelocity > 0 ? Math.ceil(totalRemaining / avgVelocity) : Infinity,
   984→        };
   985→        break;
   986→      }
   987→    }
   988→
   989→    this.logger.log(`Generated ${reportType} report: ${filteredTasks.length} task(s) analyzed`);
   990→
   991→    return {
   992→      reportType,
   993→      generatedAt,
   994→      data,
   995→    };
   996→  }
