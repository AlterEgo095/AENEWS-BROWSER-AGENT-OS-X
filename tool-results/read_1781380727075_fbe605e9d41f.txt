   320→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   321→    const startTime = Date.now();
   322→    const { action, ...params } = input.payload;
   323→
   324→    if (!action) {
   325→      return this.createAgentOutput(
   326→        input.taskId,
   327→        false,
   328→        null,
   329→        'Missing required parameter: action',
   330→        startTime,
   331→      );
   332→    }
   333→
   334→    const supportedActions = [
   335→      'build',
   336→      'compile',
   337→      'bundle',
   338→      'cleanBuild',
   339→      'configureBuild',
   340→      'getBuildInfo',
   341→    ];
   342→
   343→    if (!supportedActions.includes(action)) {
   344→      return this.createAgentOutput(
   345→        input.taskId,
   346→        false,
   347→        null,
   348→        `Unknown build action: ${action}. Supported: ${supportedActions.join(', ')}`,
   349→        startTime,
   350→      );
   351→    }
   352→
   353→    try {
   354→      const tool = this.getTool(action);
   355→      if (!tool) {
   356→        return this.createAgentOutput(
   357→          input.taskId,
   358→          false,
   359→          null,
   360→          `Tool not found: ${action}`,
   361→          startTime,
   362→        );
   363→      }
   364→
   365→      const result = await tool.execute(params);
   366→
   367→      await this.storeInWorkingMemory(
   368→        `build:last:${action}`,
   369→        { params, result, timestamp: new Date() },
   370→        300000,
   371→      );
   372→
   373→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   374→    } catch (error) {
   375→      const msg = (error as Error).message;
   376→      this.logger.error(`Build execution failed for ${action}: ${msg}`);
   377→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   378→    }
   379→  }

   389→  private async build(params: {
   390→    projectPath: string;
   391→    environment?: string;
   392→    target?: string;
   393→    watch?: boolean;
   394→    verbose?: boolean;
   395→  }): Promise<{
   396→    success: boolean;
   397→    duration: number;
   398→    outputFiles: BuildOutputFile[];
   399→    errors: string[];
   400→    warnings: string[];
   401→    environment: string;
   402→  }> {
   403→    const {
   404→      projectPath,
   405→      environment = 'production',
   406→      target = 'node',
   407→      watch = false,
   408→      verbose = false,
   409→    } = params;
   410→
   411→    if (!projectPath || typeof projectPath !== 'string') {
   412→      throw new Error('Project path is required');
   413→    }
   414→
   415→    const buildStart = Date.now();
   416→    const errors: string[] = [];
   417→    const warnings: string[] = [];
   418→
   419→    // Simulate build process
   420→    const outputFiles: BuildOutputFile[] = [];
   421→
   422→    // Step 1: Validation
   423→    if (verbose) this.logger.log('Validating project structure...');
   424→
   425→    // Step 2: Compilation
   426→    if (verbose) this.logger.log('Compiling TypeScript...');
   427→    const compileResult = this.simulateCompilation(projectPath);
   428→    if (compileResult.errors.length > 0) {
   429→      errors.push(...compileResult.errors.map((e) => `${e.file}:${e.line} - ${e.message}`));
   430→    }
   431→    if (compileResult.warnings.length > 0) {
   432→      warnings.push(...compileResult.warnings.map((w) => `${w.file}:${w.line} - ${w.message}`));
   433→    }
   434→
   435→    // Step 3: Bundling
   436→    if (verbose) this.logger.log('Bundling output...');
   437→    outputFiles.push(...this.simulateBuildOutput(projectPath, environment));
   438→
   439→    // Step 4: Asset optimization (production only)
   440→    if (environment === 'production') {
   441→      if (verbose) this.logger.log('Optimizing assets...');
   442→      warnings.push('Consider enabling gzip compression for production assets');
   443→    }
   444→
   445→    const duration = Date.now() - buildStart;
   446→    const success = errors.length === 0;
   447→
   448→    // Record build history
   449→    this.buildHistory.push({ timestamp: new Date(), success, duration, environment });
   450→
   451→    this.lastBuildInfo = {
   452→      projectPath,
   453→      environment,
   454→      target,
   455→      success,
   456→      duration,
   457→      outputCount: outputFiles.length,
   458→      timestamp: new Date(),
   459→    };
   460→
   461→    // Store build info in memory
   462→    await this.storeInWorkingMemory('build:lastResult', this.lastBuildInfo, 300000);
   463→
   464→    this.logger.log(
   465→      `Build ${success ? 'succeeded' : 'failed'}: ${duration}ms, ${outputFiles.length} output file(s), ${errors.length} error(s), ${warnings.length} warning(s)`,
   466→    );
   467→
   468→    return { success, duration, outputFiles, errors, warnings, environment };
   469→  }

   471→  private async compile(params: {
   472→    sourcePath: string;
   473→    outputPath: string;
   474→    compiler?: string;
   475→    strict?: boolean;
   476→    sourceMap?: boolean;
   477→  }): Promise<{
   478→    success: boolean;
   479→    filesCompiled: number;
   480→    duration: number;
   481→    errors: CompileError[];
   482→    warnings: CompileError[];
   483→  }> {
   484→    const { sourcePath, outputPath, compiler = 'tsc', strict = true, sourceMap = true } = params;
   485→
   486→    if (!sourcePath || typeof sourcePath !== 'string') {
   487→      throw new Error('Source path is required');
   488→    }
   489→    if (!outputPath || typeof sourcePath !== 'string') {
   490→      throw new Error('Output path is required');
   491→    }
   492→
   493→    const validCompilers = ['tsc', 'swc', 'esbuild'];
   494→    if (!validCompilers.includes(compiler)) {
   495→      throw new Error(`Invalid compiler: ${compiler}. Valid: ${validCompilers.join(', ')}`);
   496→    }
   497→
   498→    const compileStart = Date.now();
   499→
   500→    // Simulate compilation
   501→    const result = this.simulateCompilation(sourcePath);
   502→
   503→    // Apply strict mode
   504→    if (strict) {
   505→      const strictWarnings = this.simulateStrictModeCheck(sourcePath);
   506→      result.warnings.push(...strictWarnings);
   507→    }
   508→
   509→    // Estimate files compiled
   510→    const filesCompiled = 10 + Math.floor(Math.random() * 40);
   511→
   512→    const duration = Date.now() - compileStart;
   513→    const success = result.errors.length === 0;
   514→
   515→    this.logger.log(
   516→      `Compile ${success ? 'succeeded' : 'failed'}: ${compiler}, ${filesCompiled} file(s), ${duration}ms`,
   517→    );
   518→
   519→    return {
   520→      success,
   521→      filesCompiled,
   522→      duration,
   523→      errors: result.errors,
   524→      warnings: result.warnings,
   525→    };
   526→  }

   528→  private async bundle(params: {
   529→    entryPoint: string;
   530→    outputPath: string;
   531→    bundler?: string;
   532→    minify?: boolean;
   533→    treeShaking?: boolean;
   534→    format?: string;
   535→  }): Promise<{
   536→    success: boolean;
   537→    bundleSize: number;
   538→    duration: number;
   539→    modules: number;
   540→    chunks: BuildChunk[];
   541→  }> {
   542→    const {
   543→      entryPoint,
   544→      outputPath,
   545→      bundler = 'esbuild',
   546→      minify = true,
   547→      treeShaking = true,
   548→      format = 'esm',
   549→    } = params;
   550→
   551→    if (!entryPoint || typeof entryPoint !== 'string') {
   552→      throw new Error('Entry point is required');
   553→    }
   554→    if (!outputPath || typeof outputPath !== 'string') {
   555→      throw new Error('Output path is required');
   556→    }
   557→
   558→    const validBundlers = ['webpack', 'esbuild', 'rollup'];
   559→    if (!validBundlers.includes(bundler)) {
   560→      throw new Error(`Invalid bundler: ${bundler}. Valid: ${validBundlers.join(', ')}`);
   561→    }
   562→
   563→    const validFormats = ['esm', 'cjs', 'iife'];
   564→    if (!validFormats.includes(format)) {
   565→      throw new Error(`Invalid format: ${format}. Valid: ${validFormats.join(', ')}`);
   566→    }
   567→
   568→    const bundleStart = Date.now();
   569→
   570→    // Simulate bundling
   571→    const modules = 20 + Math.floor(Math.random() * 100);
   572→
   573→    // Calculate simulated bundle size
   574→    let baseSize = modules * 2048; // ~2KB per module
   575→    if (minify) baseSize = Math.round(baseSize * 0.6);
   576→    if (treeShaking) baseSize = Math.round(baseSize * 0.75);
   577→
   578→    const bundleSize = baseSize;
   579→
   580→    // Generate chunks
   581→    const chunks: BuildChunk[] = [
   582→      {
   583→        name: 'main',
   584→        size: Math.round(bundleSize * 0.7),
   585→        files: [outputPath],
   586→        isEntry: true,
   587→      },
   588→    ];
   589→
   590→    // Add vendor chunk if using webpack
   591→    if (bundler === 'webpack') {
   592→      chunks.push({
   593→        name: 'vendor',
   594→        size: Math.round(bundleSize * 0.3),
   595→        files: [outputPath.replace(/\.\w+$/, '.vendor$&')],
   596→        isEntry: false,
   597→      });
   598→    }
   599→
   600→    // Add CSS chunk for browser targets
   601→    if (format === 'iife') {
   602→      chunks.push({
   603→        name: 'styles',
   604→        size: Math.round(bundleSize * 0.05),
   605→        files: [outputPath.replace(/\.\w+$/, '.css')],
   606→        isEntry: false,
   607→      });
   608→    }
   609→
   610→    const duration = Date.now() - bundleStart;
   611→
   612→    this.logger.log(
   613→      `Bundle ${bundler}: ${bundleSize} bytes, ${modules} modules, ${chunks.length} chunk(s), ${duration}ms`,
   614→    );
   615→
   616→    return {
   617→      success: true,
   618→      bundleSize,
   619→      duration,
   620→      modules,
   621→      chunks,
   622→    };
   623→  }

   628→    environment?: string;
   629→  }): Promise<{
   630→    cleaned: string[];

   633→  }> {

   635→      projectPath,
   636→      cleanTargets = ['dist', 'build', '.cache', '.tmp'],
   637→      environment = 'production',
   638→    } = params;
   639→
   640→    if (!projectPath || typeof projectPath !== 'string') {
   641→      throw new Error('Project path is required');
   642→    }
   643→
   644→    const totalStart = Date.now();
   645→
   646→    // Step 1: Clean targets
   647→    const cleaned: string[] = [];
   648→    for (const target of cleanTargets) {
   649→      cleaned.push(`${projectPath}/${target}`);
   650→      this.logger.log(`Cleaned: ${projectPath}/${target}`);

   655→      projectPath,
   656→      environment,
   657→      verbose: true,
   658→    });
   659→
   660→    const totalDuration = Date.now() - totalStart;

   662→    this.logger.log(
   663→      `Clean build: ${cleaned.length} dir(s) cleaned, build ${buildResult.success ? 'succeeded' : 'failed'}, ${totalDuration}ms total`,
   664→    );
   665→
   666→    return {
   667→      cleaned,
   668→      buildResult,
   669→      totalDuration,
   670→    };
   671→  }

   673→  private async configureBuild(params: {
   674→    projectPath: string;
   675→    buildTool: string;
   676→    options?: Record<string, any>;
   677→    overwrite?: boolean;
   678→  }): Promise<{
   679→    configPath: string;
   680→    configContent: string;
   681→    created: boolean;
   682→    warnings: string[];
   683→  }> {
   684→    const { projectPath, buildTool, options = {}, overwrite = false } = params;
   685→
   686→    if (!projectPath || typeof projectPath !== 'string') {
   687→      throw new Error('Project path is required');
   688→    }
   689→
   690→    const validBuildTools = ['tsc', 'webpack', 'esbuild', 'rollup', 'nest-cli'];
   691→    if (!validBuildTools.includes(buildTool)) {
   692→      throw new Error(`Invalid build tool: ${buildTool}. Valid: ${validBuildTools.join(', ')}`);
   693→    }
   694→
   695→    const warnings: string[] = [];
   696→    let configPath: string;
   697→    let configContent: string;
   698→
   699→    switch (buildTool) {
   700→      case 'tsc':
   701→        configPath = `${projectPath}/tsconfig.json`;
   702→        configContent = this.generateTsConfig(options);
   703→        break;
   704→      case 'webpack':
   705→        configPath = `${projectPath}/webpack.config.js`;
   706→        configContent = this.generateWebpackConfig(options);
   707→        break;
   708→      case 'esbuild':
   709→        configPath = `${projectPath}/esbuild.config.js`;
   710→        configContent = this.generateEsbuildConfig(options);
   711→        break;
   712→      case 'rollup':
   713→        configPath = `${projectPath}/rollup.config.js`;
   714→        configContent = this.generateRollupConfig(options);
   715→        break;
   716→      case 'nest-cli':
   717→        configPath = `${projectPath}/nest-cli.json`;
   718→        configContent = this.generateNestCliConfig(options);
   719→        break;
   720→      default:
   721→        throw new Error(`Unsupported build tool: ${buildTool}`);
   722→    }
   723→
   724→    if (!overwrite) {
   725→      warnings.push(
   726→        'If the config file already exists, it will not be overwritten. Set overwrite: true to replace.',
   727→      );
   728→    }
   729→
   730→    this.logger.log(`Configured build: ${buildTool}, config at ${configPath}`);
   731→
   732→    return { configPath, configContent, created: true, warnings };
   733→  }
