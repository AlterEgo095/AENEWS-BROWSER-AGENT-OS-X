   258→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   259→    const startTime = Date.now();
   260→    const { action, ...params } = input.payload;
   261→
   262→    if (!action) {
   263→      return this.createAgentOutput(
   264→        input.taskId,
   265→        false,
   266→        null,
   267→        'Missing required parameter: action',
   268→        startTime,
   269→      );
   270→    }
   271→
   272→    const supportedActions = [
   273→      'generateDocs',
   274→      'generateApiDocs',
   275→      'generateReadme',
   276→      'updateChangelog',
   277→      'generateTypeDocs',
   278→    ];
   279→
   280→    if (!supportedActions.includes(action)) {
   281→      return this.createAgentOutput(
   282→        input.taskId,
   283→        false,
   284→        null,
   285→        `Unknown documentation action: ${action}. Supported: ${supportedActions.join(', ')}`,
   286→        startTime,
   287→      );
   288→    }
   289→
   290→    try {
   291→      const tool = this.getTool(action);
   292→      if (!tool) {
   293→        return this.createAgentOutput(
   294→          input.taskId,
   295→          false,
   296→          null,
   297→          `Tool not found: ${action}`,
   298→          startTime,
   299→        );
   300→      }
   301→
   302→      const result = await tool.execute(params);
   303→
   304→      this.docHistory.push({
   305→        action,
   306→        timestamp: new Date(),
   307→        outputSize: JSON.stringify(result).length,
   308→      });
   309→
   310→      await this.storeInWorkingMemory(
   311→        `docs:last:${action}`,
   312→        { params, result, timestamp: new Date() },
   313→        300000,
   314→      );
   315→
   316→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   317→    } catch (error) {
   318→      const msg = (error as Error).message;
   319→      this.logger.error(`Documentation execution failed for ${action}: ${msg}`);
   320→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   321→    }
   322→  }
   323→
   324→  protected async onDestroy(): Promise<void> {
   325→    this.docHistory = [];
   326→    this.logger.log('Documentation agent destroyed, history cleared');
   327→  }

   331→  private async generateDocs(params: {
   332→    code: string;
   333→    language: string;
   334→    style?: string;
   335→    includeExamples?: boolean;
   336→  }): Promise<{
   337→    documentedCode: string;
   338→    docBlocks: DocBlock[];
   339→    coverage: number;
   340→  }> {
   341→    const { code, language, style = 'jsdoc', includeExamples = true } = params;
   342→
   343→    if (!code || typeof code !== 'string') {
   344→      throw new Error('Source code is required');
   345→    }
   346→    if (!language || typeof language !== 'string') {
   347→      throw new Error('Programming language is required');
   348→    }
   349→
   350→    const lines = code.split('\n');
   351→    const docBlocks: DocBlock[] = [];
   352→    const resultLines: string[] = [];
   353→
   354→    let totalDocTargets = 0;
   355→    let documentedTargets = 0;
   356→
   357→    for (let i = 0; i < lines.length; i++) {
   358→      const line = lines[i];
   359→
   360→      // Check if this line already has documentation
   361→      const prevLine = i > 0 ? lines[i - 1].trim() : '';
   362→      const hasExistingDoc =
   363→        prevLine.endsWith('*/') || prevLine.startsWith('"""') || prevLine.includes("'''");
   364→
   365→      // Detect functions
   366→      const funcMatch = line.match(
   367→        /(?:export\s+)?(?:async\s+)?function\s+(\w+)\s*\(([^)]*)\)(?:\s*:\s*(\w+))?/,
   368→      );
   369→      if (funcMatch) {
   370→        totalDocTargets++;
   371→        const doc = this.generateFunctionDoc(
   372→          funcMatch[1],
   373→          funcMatch[2],
   374→          funcMatch[3],
   375→          style,
   376→          language,
   377→          includeExamples,
   378→        );
   379→
   380→        if (!hasExistingDoc) {
   381→          resultLines.push(doc);
   382→          documentedTargets++;
   383→        }
   384→
   385→        docBlocks.push({
   386→          target: funcMatch[1],
   387→          type: 'function',
   388→          line: i + 1,
   389→          doc: doc,
   390→        });
   391→      }
   392→
   393→      // Detect classes
   394→      const classMatch = line.match(/(?:export\s+)?(?:abstract\s+)?class\s+(\w+)/);
   395→      if (classMatch) {
   396→        totalDocTargets++;
   397→        const doc = this.generateClassDoc(classMatch[1], style);
   398→
   399→        if (!hasExistingDoc) {
   400→          resultLines.push(doc);
   401→          documentedTargets++;
   402→        }
   403→
   404→        docBlocks.push({
   405→          target: classMatch[1],
   406→          type: 'class',
   407→          line: i + 1,
   408→          doc,
   409→        });
   410→      }
   411→
   412→      // Detect methods
   413→      const methodMatch = line.match(
   414→        /(?:public|private|protected)?\s*(?:async\s+)?(\w+)\s*\(([^)]*)\)(?:\s*:\s*(\w+))?\s*{/,
   415→      );
   416→      if (
   417→        methodMatch &&
   418→        !funcMatch &&
   419→        !['if', 'for', 'while', 'switch', 'catch'].includes(methodMatch[1])
   420→      ) {
   421→        totalDocTargets++;
   422→        const doc = this.generateMethodDoc(
   423→          methodMatch[1],
   424→          methodMatch[2],
   425→          methodMatch[3],
   426→          style,
   427→          language,
   428→        );
   429→
   430→        if (!hasExistingDoc) {
   431→          resultLines.push(doc);
   432→          documentedTargets++;
   433→        }
   434→
   435→        docBlocks.push({
   436→          target: methodMatch[1],
   437→          type: 'method',
   438→          line: i + 1,
   439→          doc,
   440→        });
   441→      }
   442→
   443→      // Detect interfaces
   444→      const interfaceMatch = line.match(/(?:export\s+)?interface\s+(\w+)/);
   445→      if (interfaceMatch) {
   446→        totalDocTargets++;
   447→        const doc = this.generateInterfaceDoc(interfaceMatch[1], style);
   448→
   449→        if (!hasExistingDoc) {
   450→          resultLines.push(doc);
   451→          documentedTargets++;
   452→        }
   453→
   454→        docBlocks.push({
   455→          target: interfaceMatch[1],
   456→          type: 'interface',
   457→          line: i + 1,
   458→          doc,
   459→        });
   460→      }
   461→
   462→      resultLines.push(line);
   463→    }
   464→
   465→    const coverage =
   466→      totalDocTargets > 0 ? Math.round((documentedTargets / totalDocTargets) * 100) : 100;
   467→    const documentedCode = resultLines.join('\n');
   468→
   469→    this.logger.log(
   470→      `Generated documentation: ${docBlocks.length} doc block(s), ${coverage}% coverage`,
   471→    );
   472→
   473→    return { documentedCode, docBlocks, coverage };
   474→  }

   476→  private async generateApiDocs(params: {
   477→    code: string;
   478→    language: string;
   479→    format?: string;
   480→    version?: string;
   481→  }): Promise<{
   482→    documentation: string;
   483→    endpoints: ApiEndpoint[];
   484→    format: string;
   485→  }> {
   486→    const { code, language, format = 'openapi', version = '1.0.0' } = params;
   487→
   488→    if (!code || typeof code !== 'string') {
   489→      throw new Error('Source code is required');
   490→    }
   491→
   492→    // Extract API endpoints from code
   493→    const endpoints = this.extractApiEndpoints(code, language);
   494→
   495→    if (endpoints.length === 0) {
   496→      throw new Error('No API endpoints found in the provided code');
   497→    }
   498→
   499→    // Generate documentation in requested format
   500→    let documentation: string;
   501→
   502→    switch (format) {
   503→      case 'openapi':
   504→        documentation = this.renderOpenApiDocs(endpoints, version);
   505→        break;
   506→      case 'markdown':
   507→        documentation = this.renderMarkdownApiDocs(endpoints, version);
   508→        break;
   509→      case 'html':
   510→        documentation = this.renderHtmlApiDocs(endpoints, version);
   511→        break;
   512→      default:
   513→        documentation = this.renderOpenApiDocs(endpoints, version);
   514→    }
   515→
   516→    this.logger.log(`Generated API docs: ${endpoints.length} endpoint(s), format=${format}`);
   517→
   518→    return { documentation, endpoints, format };
   519→  }

   521→  private async generateReadme(params: {
   522→    projectName: string;
   523→    description: string;
   524→    packageJson?: Record<string, any>;
   525→    features?: string[];
   526→    installationSteps?: string[];
   527→  }): Promise<{
   528→    content: string;
   529→    sections: string[];
   530→  }> {
   531→    const { projectName, description, packageJson, features = [], installationSteps = [] } = params;
   532→
   533→    if (!projectName || typeof projectName !== 'string') {
   534→      throw new Error('Project name is required');
   535→    }
   536→    if (!description || typeof description !== 'string') {
   537→      throw new Error('Project description is required');
   538→    }
   539→
   540→    const sections: string[] = [];
   541→    let content = '';
   542→
   543→    // Title and description
   544→    content += `# ${projectName}\n\n`;
   545→    content += `${description}\n\n`;
   546→    sections.push('header');
   547→
   548→    // Badges
   549→    if (packageJson) {
   550→      content += this.generateBadges(packageJson);
   551→      sections.push('badges');
   552→    }
   553→
   554→    // Features
   555→    if (features.length > 0) {
   556→      content += `## Features\n\n`;
   557→      for (const feature of features) {
   558→        content += `- ${feature}\n`;
   559→      }
   560→      content += '\n';
   561→      sections.push('features');
   562→    }
   563→
   564→    // Installation
   565→    content += `## Installation\n\n`;
   566→    if (installationSteps.length > 0) {
   567→      for (const step of installationSteps) {
   568→        content += `${step}\n`;
   569→      }
   570→    } else if (packageJson?.name) {
   571→      content += '```bash\n';
   572→      content += `npm install ${packageJson.name}\n`;
   573→      content += '```\n\n';
   574→      content += 'or\n\n';
   575→      content += '```bash\n';
   576→      content += `yarn add ${packageJson.name}\n`;
   577→      content += '```\n';
   578→    } else {
   579→      content += '```bash\n';
   580→      content += 'npm install\n';
   581→      content += '```\n';
   582→    }
   583→    content += '\n';
   584→    sections.push('installation');
   585→
   586→    // Usage
   587→    content += `## Usage\n\n`;
   588→    content += '```typescript\n';
   589→    content += `import { ${this.toPascalCase(projectName)} } from '${packageJson?.name || projectName}';\n\n`;
   590→    content += `const instance = new ${this.toPascalCase(projectName)}();\n`;
   591→    content += '// Configure and use the module\n';
   592→    content += '```\n\n';
   593→    sections.push('usage');
   594→
   595→    // Configuration
   596→    if (packageJson?.config || features.some((f) => f.toLowerCase().includes('config'))) {
   597→      content += `## Configuration\n\n`;
   598→      content += '| Option | Type | Default | Description |\n';
   599→      content += '|--------|------|---------|-------------|\n';
   600→      content += "| `option1` | `string` | `'default'` | Description for option1 |\n";
   601→      content += '| `option2` | `boolean` | `false` | Description for option2 |\n\n';
   602→      sections.push('configuration');
   603→    }
   604→
   605→    // API Reference
   606→    content += `## API Reference\n\n`;
   607→    content += `See [API Documentation](./docs/api.md) for detailed API reference.\n\n`;
   608→    sections.push('api');
   609→
   610→    // Contributing
   611→    content += `## Contributing\n\n`;
   612→    content += `1. Fork the repository\n`;
   613→    content += `2. Create your feature branch (\`git checkout -b feature/amazing-feature\`)\n`;
   614→    content += `3. Commit your changes (\`git commit -m 'Add amazing feature'\`)\n`;
   615→    content += `4. Push to the branch (\`git push origin feature/amazing-feature\`)\n`;
   616→    content += `5. Open a Pull Request\n\n`;
   617→    sections.push('contributing');
   618→
   619→    // License
   620→    const license = packageJson?.license || 'MIT';
   621→    content += `## License\n\n`;
   622→    content += `This project is licensed under the ${license} License.\n`;
   623→    sections.push('license');
   624→
   625→    this.logger.log(`Generated README: ${sections.length} section(s)`);
   626→
   627→    return { content, sections };
   628→  }

   630→  private async updateChangelog(params: {
   631→    currentChangelog?: string;
   632→    version: string;
   633→    changes: ChangelogEntry[];
   634→    date?: string;
   635→  }): Promise<{
   636→    content: string;
   637→    version: string;
   638→    entriesAdded: number;
   639→  }> {
   640→    const { currentChangelog, version, changes, date } = params;
   641→
   642→    if (!version || typeof version !== 'string') {
   643→      throw new Error('Version is required');
   644→    }
   645→    if (!changes || !Array.isArray(changes) || changes.length === 0) {
   646→      throw new Error('At least one change entry is required');
   647→    }
   648→
   649→    // Validate version format
   650→    if (!/^\d+\.\d+\.\d+/.test(version)) {
   651→      throw new Error('Version must follow semver format (e.g., 1.0.0)');
   652→    }
   653→
   654→    const releaseDate = date || new Date().toISOString().split('T')[0];
   655→
   656→    // Group changes by type
   657→    const grouped = this.groupChangesByType(changes);
   658→
   659→    // Build new version entry
   660→    let newEntry = `## [${version}] - ${releaseDate}\n\n`;
   661→
   662→    const typeLabels: Record<string, string> = {
   663→      added: '### Added',
   664→      changed: '### Changed',
   665→      deprecated: '### Deprecated',
   666→      removed: '### Removed',
   667→      fixed: '### Fixed',
   668→      security: '### Security',
   669→    };
   670→
   671→    for (const [type, entries] of Object.entries(grouped)) {
   672→      const label = typeLabels[type] || `### ${type}`;
   673→      newEntry += `${label}\n\n`;
   674→      for (const entry of entries) {
   675→        const scope = entry.scope ? `**${entry.scope}:** ` : '';
   676→        newEntry += `- ${scope}${entry.description}\n`;
   677→      }
   678→      newEntry += '\n';
   679→    }
   680→
   681→    // Prepend to existing changelog or create new one
   682→    let content: string;
   683→
   684→    if (currentChangelog) {
   685→      // Insert after the header
   686→      const headerEnd = currentChangelog.indexOf('\n\n');
   687→      if (headerEnd !== -1) {
   688→        content =
   689→          currentChangelog.substring(0, headerEnd + 2) +
   690→          newEntry +
   691→          currentChangelog.substring(headerEnd + 2);
   692→      } else {
   693→        content = currentChangelog + '\n\n' + newEntry;
   694→      }
   695→    } else {
   696→      content = `# Changelog\n\nAll notable changes to this project will be documented in this file.\n\nThe format is based on [Keep a Changelog](https://keepachangelog.com/en/1.1.0/),\nand this project adheres to [Semantic Versioning](https://semver.org/spec/v2.0.0.html).\n\n${newEntry}`;
   697→    }
   698→
   699→    this.logger.log(`Updated changelog: version ${version}, ${changes.length} entr(ies)`);
   700→
   701→    return { content, version, entriesAdded: changes.length };
   702→  }

   888→  private extractApiEndpoints(code: string, language: string): ApiEndpoint[] {
   889→    const endpoints: ApiEndpoint[] = [];
   890→
   891→    // NestJS decorators
   892→    const nestjsRegex =
   893→      /@(Get|Post|Put|Delete|Patch)\(['"]([^'"]+)['"]\)(?:\s*\n\s*@HttpCode\(\d+\))?(?:\s*\n\s*@\w+)*\s*\n\s*(?:async\s+)?(\w+)\s*\(([^)]*)\)/g;
   894→    let match: RegExpExecArray | null;
   895→
   896→    while ((match = nestjsRegex.exec(code)) !== null) {
   897→      const method = match[1].toUpperCase();
   898→      const path = match[2];
   899→      const handlerName = match[3];
   900→      const params = this.parseApiParams(match[4], path);
   901→
   902→      endpoints.push({
   903→        method,
   904→        path,
   905→        description: `${handlerName} handler`,
   906→        parameters: params,
   907→        responses: this.inferResponses(method),
   908→        tags: this.extractControllerTag(code, match.index),
   909→      });
   910→    }
   911→
   912→    // Express-style routes
   913→    const expressRegex = /(?:router|app)\.(get|post|put|delete|patch)\(['"]([^'"]+)['"]/g;
   914→    while ((match = expressRegex.exec(code)) !== null) {
   915→      endpoints.push({
   916→        method: match[1].toUpperCase(),
   917→        path: match[2],
   918→        description: `${match[1].toUpperCase()} ${match[2]}`,
   919→        parameters: [],
   920→        responses: this.inferResponses(match[1].toUpperCase()),
   921→        tags: [],
   922→      });
   923→    }
   924→
   925→    return endpoints;
   926→  }

   928→  private parseApiParams(
   929→    paramsStr: string,
   930→    path: string,
   931→  ): Array<{ name: string; type: string; required: boolean; description: string }> {
   932→    const params: Array<{ name: string; type: string; required: boolean; description: string }> =
   933→      [];
   934→
   935→    // Path parameters
   936→    const pathParams = path.match(/:(\w+)/g);
   937→    if (pathParams) {
   938→      for (const param of pathParams) {
   939→        const name = param.replace(':', '');
   940→        params.push({
   941→          name,
   942→          type: 'string',
   943→          required: true,
   944→          description: `Path parameter: ${name}`,
   945→        });
   946→      }
   947→    }
   948→
   949→    // Method parameters
   950→    const methodParams = paramsStr.split(',').filter((p) => p.trim().length > 0);
   951→    for (const param of methodParams) {
   952→      const trimmed = param.trim();
   953→      if (trimmed.includes('@Body()') || trimmed.includes('@Req()') || trimmed.includes('@Res()')) {
   954→        continue;
   955→      }
   956→      const nameMatch = trimmed.match(/(\w+)$/);
   957→      if (nameMatch) {
   958→        params.push({
   959→          name: nameMatch[1],
   960→          type: 'string',
   961→          required: !trimmed.includes('?'),
   962→          description: `Parameter: ${nameMatch[1]}`,
   963→        });
   964→      }
   965→    }
   966→
   967→    return params;
   968→  }

   970→  private inferResponses(
   971→    method: string,
   972→  ): Array<{ code: number; description: string; type?: string }> {
   973→    const responses: Array<{ code: number; description: string; type?: string }> = [
   974→      { code: 200, description: 'Successful response' },
   975→    ];

   978→      responses.unshift({ code: 201, description: 'Resource created successfully' });
   979→    }
   980→
   981→    responses.push({ code: 400, description: 'Bad request' });
   982→    responses.push({ code: 401, description: 'Unauthorized' });
   983→    responses.push({ code: 500, description: 'Internal server error' });
   984→
   985→    return responses;
   986→  }

   988→  private extractControllerTag(code: string, position: number): string[] {
   989→    const beforePosition = code.substring(0, position);
   990→    const controllerMatch = beforePosition.match(/@Controller\(['"]?([^'")\]]+)/g);
   991→    if (controllerMatch) {
   992→      const lastController = controllerMatch[controllerMatch.length - 1];
   993→      const tag = lastController
   994→        .replace("@Controller('", '')
   995→        .replace('@Controller("', '')
   996→        .replace('@Controller(', '');
   997→      return [tag];
   998→    }
   999→    return ['default'];
  1000→  }

  1121→    }
  1122→
  1123→    // Extract type aliases
  1124→    const typeRegex = /(?:export\s+)?type\s+(\w+)\s*(?:<[^>]+>)?\s*=\s*([^;]+);/g;

  1129→

  1137→
  1138→    // Extract enums

  1167→        name,
  1168→        kind: 'class',
  1169→        description: `${name} class`,

  1206→  private renderMarkdownTypeDocs(types: TypeDocumentation[]): string {

  1208→
  1209→    for (const type of types) {
  1210→      md += `## ${type.name}\n\n`;
  1211→      md += `**Kind:** ${type.kind} ${type.exported ? '(exported)' : '(internal)'}\n\n`;
  1212→      md += `${type.description}\n\n`;
  1213→
  1214→      if (type.properties && type.properties.length > 0) {
  1215→        md += `### Properties\n\n`;

  1312→  private parseParamsList(paramsStr: string): Array<{ name: string; type: string }> {
  1313→    if (!paramsStr.trim()) return [];
  1314→    return paramsStr
  1315→      .split(',')
  1316→      .map((param) => {
  1317→        const trimmed = param.trim();
  1318→        const parts = trimmed.split(':');
  1319→        const name = parts[0].replace(/\?.*$/, '').trim();
  1320→        const type = parts[1] ? parts[1].replace(/\s*=\s*.*$/, '').trim() : 'any';
  1321→        return { name, type };
  1322→      })
  1323→      .filter((p) => p.name.length > 0);
  1324→  }

  1326→  private parsePythonParamsList(paramsStr: string): Array<{ name: string; type: string }> {

  1329→      .split(',')
  1330→      .map((param) => {

  1333→        const parts = trimmed.split(':');

  1338→      .filter((p) => p.name.length > 0 && p.type !== 'self');
