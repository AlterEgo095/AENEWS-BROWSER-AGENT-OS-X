   394→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   395→    const startTime = Date.now();
   396→    const { action, ...params } = input.payload;
   397→
   398→    if (!action) {
   399→      return this.createAgentOutput(
   400→        input.taskId,
   401→        false,
   402→        null,
   403→        'Missing required parameter: action',
   404→        startTime,
   405→      );
   406→    }
   407→
   408→    const supportedActions = [
   409→      'createPresentation',
   410→      'addSlide',
   411→      'addContent',
   412→      'applyTheme',
   413→      'exportPresentation',
   414→      'addTransition',
   415→    ];
   416→
   417→    if (!supportedActions.includes(action)) {
   418→      return this.createAgentOutput(
   419→        input.taskId,
   420→        false,
   421→        null,
   422→        `Unknown presentation action: ${action}. Supported: ${supportedActions.join(', ')}`,
   423→        startTime,
   424→      );
   425→    }
   426→
   427→    try {
   428→      const tool = this.getTool(action);
   429→      if (!tool) {
   430→        return this.createAgentOutput(
   431→          input.taskId,
   432→          false,
   433→          null,
   434→          `Tool not found: ${action}`,
   435→          startTime,
   436→        );
   437→      }
   438→
   439→      const result = await tool.execute(params);
   440→
   441→      await this.storeInWorkingMemory(
   442→        `presentation:last:${action}`,
   443→        { params, result, timestamp: new Date() },
   444→        300000,
   445→      );
   446→
   447→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   448→    } catch (error) {
   449→      const msg = (error as Error).message;
   450→      this.logger.error(`Presentation execution failed for ${action}: ${msg}`);
   451→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   452→    }
   453→  }

   466→  private async createPresentation(params: {
   467→    title: string;
   468→    author?: string;
   469→    theme?: string;
   470→    slides?: Array<{ layout: string; title?: string; subtitle?: string }>;
   471→    format?: string;
   472→  }): Promise<{
   473→    presentationId: string;
   474→    title: string;
   475→    slideCount: number;
   476→    createdAt: string;
   477→  }> {
   478→    const {
   479→      title,
   480→      author = 'agent@aenews.system',
   481→      theme = 'professional',
   482→      slides = [],
   483→      format = 'pptx',
   484→    } = params;
   485→
   486→    if (!title || typeof title !== 'string') {
   487→      throw new Error('A valid presentation title is required');
   488→    }
   489→
   490→    const validFormats = ['pptx', 'pdf', 'odp'];
   491→    if (!validFormats.includes(format)) {
   492→      throw new Error(`Invalid format: ${format}. Supported: ${validFormats.join(', ')}`);
   493→    }
   494→
   495→    const presentationId = this.generatePresentationId();
   496→    const themeDef = this.themes.get(theme) || this.themes.get('professional')!;
   497→
   498→    // Create title slide
   499→    const presentationSlides: Slide[] = [];
   500→    const titleSlide = this.createSlideObject('title', title, author, '');
   501→    presentationSlides.push(titleSlide);
   502→
   503→    // Add additional slides
   504→    for (const slideDef of slides) {
   505→      const slide = this.createSlideObject(
   506→        slideDef.layout as SlideLayout,
   507→        slideDef.title || '',
   508→        slideDef.subtitle || '',
   509→        '',
   510→      );
   511→      presentationSlides.push(slide);
   512→    }
   513→
   514→    const presentation: Presentation = {
   515→      id: presentationId,
   516→      title,
   517→      author,
   518→      format,
   519→      theme: themeDef,
   520→      slides: presentationSlides,
   521→      createdAt: new Date(),
   522→      updatedAt: new Date(),
   523→    };
   524→
   525→    this.presentations.set(presentationId, presentation);
   526→
   527→    this.logger.log(
   528→      `Created presentation: ${presentationId}, title="${title}", slides=${presentationSlides.length}`,
   529→    );
   530→
   531→    return {
   532→      presentationId,
   533→      title,
   534→      slideCount: presentationSlides.length,
   535→      createdAt: presentation.createdAt.toISOString(),
   536→    };
   537→  }

   539→  private async addSlide(params: {
   540→    presentationId: string;
   541→    layout: string;
   542→    position?: number;
   543→    title?: string;
   544→    subtitle?: string;
   545→    notes?: string;
   546→  }): Promise<{
   547→    presentationId: string;
   548→    slideId: string;
   549→    position: number;
   550→    status: string;
   551→  }> {
   552→    const { presentationId, layout, position, title = '', subtitle = '', notes = '' } = params;
   553→
   554→    if (!presentationId || typeof presentationId !== 'string') {
   555→      throw new Error('A valid presentationId is required');
   556→    }
   557→
   558→    const validLayouts = [
   559→      'title',
   560→      'titleAndContent',
   561→      'twoContent',
   562→      'blank',
   563→      'sectionHeader',
   564→      'comparison',
   565→    ];
   566→    if (!validLayouts.includes(layout)) {
   567→      throw new Error(`Invalid layout: ${layout}. Supported: ${validLayouts.join(', ')}`);
   568→    }
   569→
   570→    const presentation = this.presentations.get(presentationId);
   571→    if (!presentation) {
   572→      throw new Error(`Presentation not found: ${presentationId}`);
   573→    }
   574→
   575→    const slide = this.createSlideObject(layout as SlideLayout, title, subtitle, notes);
   576→    const insertPosition =
   577→      position !== undefined
   578→        ? Math.min(position - 1, presentation.slides.length)
   579→        : presentation.slides.length;
   580→
   581→    presentation.slides.splice(insertPosition, 0, slide);
   582→    presentation.updatedAt = new Date();
   583→
   584→    this.logger.log(
   585→      `Added slide: ${slide.id}, layout=${layout}, position=${insertPosition + 1} to presentation: ${presentationId}`,
   586→    );
   587→
   588→    return {
   589→      presentationId,
   590→      slideId: slide.id,
   591→      position: insertPosition + 1,
   592→      status: 'added',
   593→    };
   594→  }

   596→  private async addContent(params: {
   597→    presentationId: string;
   598→    slideId: string;
   599→    contentType: string;
   600→    content: any;
   601→    position?: ContentPosition;
   602→    style?: ContentStyle;
   603→  }): Promise<{
   604→    presentationId: string;
   605→    slideId: string;
   606→    contentId: string;
   607→    status: string;
   608→  }> {
   609→    const {
   610→      presentationId,
   611→      slideId,
   612→      contentType,
   613→      content,
   614→      position = { x: 0, y: 0, width: 800, height: 400 },
   615→      style = {},
   616→    } = params;
   617→
   618→    if (!presentationId || typeof presentationId !== 'string') {
   619→      throw new Error('A valid presentationId is required');
   620→    }
   621→    if (!slideId || typeof slideId !== 'string') {
   622→      throw new Error('A valid slideId is required');
   623→    }
   624→
   625→    const validContentTypes = ['text', 'image', 'table', 'chart', 'shape', 'bulletList'];
   626→    if (!validContentTypes.includes(contentType)) {
   627→      throw new Error(
   628→        `Invalid contentType: ${contentType}. Supported: ${validContentTypes.join(', ')}`,
   629→      );
   630→    }
   631→
   632→    if (!content) {
   633→      throw new Error('Content data is required');
   634→    }
   635→
   636→    const presentation = this.presentations.get(presentationId);
   637→    if (!presentation) {
   638→      throw new Error(`Presentation not found: ${presentationId}`);
   639→    }
   640→
   641→    const slide = presentation.slides.find((s) => s.id === slideId);
   642→    if (!slide) {
   643→      throw new Error(`Slide not found: ${slideId} in presentation ${presentationId}`);
   644→    }
   645→
   646→    const contentId = `content-${++this.contentCounter}`;
   647→    const slideContent: SlideContent = {
   648→      id: contentId,
   649→      type: contentType as SlideContent['type'],
   650→      data: this.processContentData(contentType, content),
   651→      position,
   652→      style: {
   653→        fontSize: 18,
   654→        fontColor: presentation.theme.fontColor,
   655→        fontFamily: presentation.theme.fontFamily,
   656→        ...style,
   657→      },
   658→    };
   659→
   660→    slide.content.push(slideContent);
   661→    presentation.updatedAt = new Date();
   662→
   663→    this.logger.log(`Added ${contentType} content to slide: ${slideId}, contentId=${contentId}`);
   664→
   665→    return {
   666→      presentationId,
   667→      slideId,
   668→      contentId,
   669→      status: 'added',
   670→    };
   671→  }

   673→  private async applyTheme(params: {
   674→    presentationId: string;
   675→    themeId: string;
   676→    slideId?: string;
   677→    overrideColors?: Record<string, string>;
   678→  }): Promise<{
   679→    presentationId: string;
   680→    themeId: string;
   681→    appliedToSlides: number;
   682→    status: string;
   683→  }> {
   684→    const { presentationId, themeId, slideId, overrideColors = {} } = params;
   685→
   686→    if (!presentationId || typeof presentationId !== 'string') {
   687→      throw new Error('A valid presentationId is required');
   688→    }
   689→    if (!themeId || typeof themeId !== 'string') {
   690→      throw new Error('A valid themeId is required');
   691→    }
   692→
   693→    const theme = this.themes.get(themeId);
   694→    if (!theme) {
   695→      const available = Array.from(this.themes.keys()).join(', ');
   696→      throw new Error(`Theme not found: ${themeId}. Available: ${available}`);
   697→    }
   698→
   699→    const presentation = this.presentations.get(presentationId);
   700→    if (!presentation) {
   701→      throw new Error(`Presentation not found: ${presentationId}`);
   702→    }
   703→
   704→    // Apply theme with optional color overrides
   705→    const appliedTheme: ThemeDefinition = {
   706→      ...theme,
   707→      ...overrideColors,
   708→    };
   709→
   710→    presentation.theme = appliedTheme;
   711→
   712→    let appliedToSlides = 0;
   713→
   714→    if (slideId) {
   715→      // Apply to specific slide
   716→      const slide = presentation.slides.find((s) => s.id === slideId);
   717→      if (!slide) {
   718→        throw new Error(`Slide not found: ${slideId}`);
   719→      }
   720→      slide.backgroundColor = appliedTheme.backgroundColor;
   721→
   722→      // Update content styles
   723→      for (const content of slide.content) {
   724→        content.style.fontColor = appliedTheme.fontColor;
   725→        content.style.fontFamily = appliedTheme.fontFamily;
   726→      }
   727→
   728→      appliedToSlides = 1;
   729→    } else {
   730→      // Apply to all slides
   731→      for (const slide of presentation.slides) {
   732→        slide.backgroundColor = appliedTheme.backgroundColor;
   733→
   734→        for (const content of slide.content) {
   735→          content.style.fontColor = appliedTheme.fontColor;
   736→          content.style.fontFamily = appliedTheme.fontFamily;
   737→        }
   738→      }
   739→      appliedToSlides = presentation.slides.length;
   740→    }
   741→
   742→    presentation.updatedAt = new Date();
   743→
   744→    this.logger.log(
   745→      `Applied theme: ${themeId} to ${appliedToSlides} slide(s) in presentation: ${presentationId}`,
   746→    );
   747→
   748→    return {
   749→      presentationId,
   750→      themeId,
   751→      appliedToSlides,
   752→      status: 'applied',
   753→    };
   754→  }

   756→  private async exportPresentation(params: {
   757→    presentationId: string;
   758→    format: string;
   759→    slideRange?: { start: number; end: number };
   760→    quality?: string;
   761→  }): Promise<{
   762→    presentationId: string;
   763→    format: string;
   764→    slideCount: number;
   765→    exportSize: string;
   766→    status: string;
   767→  }> {
   768→    const { presentationId, format, slideRange, quality = 'high' } = params;
   769→
   770→    if (!presentationId || typeof presentationId !== 'string') {
   771→      throw new Error('A valid presentationId is required');
   772→    }
   773→
   774→    const validFormats = ['pptx', 'pdf', 'odp', 'images'];
   775→    if (!validFormats.includes(format)) {
   776→      throw new Error(`Invalid export format: ${format}. Supported: ${validFormats.join(', ')}`);
   777→    }
   778→
   779→    const presentation = this.presentations.get(presentationId);
   780→    if (!presentation) {
   781→      throw new Error(`Presentation not found: ${presentationId}`);
   782→    }
   783→
   784→    // Determine slide range
   785→    let slides = presentation.slides;
   786→    if (slideRange) {
   787→      const start = Math.max(0, slideRange.start - 1);
   788→      const end = Math.min(slides.length, slideRange.end);
   789→      slides = slides.slice(start, end);
   790→    }
   791→
   792→    // Estimate export size based on quality and slide count
   793→    const baseSizeKB = quality === 'high' ? 500 : quality === 'medium' ? 250 : 100;
   794→    const estimatedSizeKB = slides.length * baseSizeKB;
   795→    const exportSize =
   796→      estimatedSizeKB > 1024
   797→        ? `${(estimatedSizeKB / 1024).toFixed(1)} MB`
   798→        : `${estimatedSizeKB} KB`;
   799→
   800→    this.logger.log(
   801→      `Exported presentation: ${presentationId}, format=${format}, slides=${slides.length}, size=${exportSize}`,
   802→    );
   803→
   804→    return {
   805→      presentationId,
   806→      format,
   807→      slideCount: slides.length,
   808→      exportSize,
   809→      status: 'exported',
   810→    };
   811→  }

   813→  private async addTransition(params: {
   814→    presentationId: string;
   815→    slideId: string;
   816→    type: string;
   817→    duration?: number;
   818→    advanceAfter?: number;
   819→  }): Promise<{
   820→    presentationId: string;
   821→    slideId: string;
   822→    transitionType: string;
   823→    status: string;
   824→  }> {
   825→    const { presentationId, slideId, type, duration = 1000, advanceAfter = 0 } = params;
   826→
   827→    if (!presentationId || typeof presentationId !== 'string') {
   828→      throw new Error('A valid presentationId is required');
   829→    }
   830→    if (!slideId || typeof slideId !== 'string') {
   831→      throw new Error('A valid slideId is required');
   832→    }
   833→
   834→    const validTransitionTypes = [
   835→      'fade',
   836→      'slide',
   837→      'push',
   838→      'wipe',
   839→      'split',
   840→      'reveal',
   841→      'dissolve',
   842→      'none',
   843→    ];
   844→    if (!validTransitionTypes.includes(type)) {
   845→      throw new Error(
   846→        `Invalid transition type: ${type}. Supported: ${validTransitionTypes.join(', ')}`,
   847→      );
   848→    }
   849→
   850→    const presentation = this.presentations.get(presentationId);
   851→    if (!presentation) {
   852→      throw new Error(`Presentation not found: ${presentationId}`);
   853→    }
   854→
   855→    const slide = presentation.slides.find((s) => s.id === slideId);
   856→    if (!slide) {
   857→      throw new Error(`Slide not found: ${slideId} in presentation ${presentationId}`);
   858→    }
   859→
   860→    slide.transition = {
   861→      type,
   862→      duration,
   863→      advanceAfter,
   864→    };
   865→
   866→    presentation.updatedAt = new Date();
   867→
   868→    this.logger.log(`Added transition: ${type} to slide: ${slideId}, duration=${duration}ms`);
   869→
   870→    return {
   871→      presentationId,
   872→      slideId,
   873→      transitionType: type,
   874→      status: 'added',
   875→    };
   876→  }

   971→  private processContentData(contentType: string, content: any): any {
   972→    switch (contentType) {
   973→      case 'text':
   974→        if (typeof content === 'string') {
   975→          return { text: content };
   976→        }
   977→        return content;
   978→      case 'image':
   979→        return {
   980→          src: content.src || content.url || '',
   981→          alt: content.alt || '',
   982→          width: content.width || 400,
   983→          height: content.height || 300,
   984→        };
   985→      case 'table':
   986→        return {
   987→          headers: content.headers || [],
   988→          rows: content.rows || [],
   989→          colCount: (content.headers || []).length,
   990→          rowCount: (content.rows || []).length,
   991→        };
   992→      case 'chart':
   993→        return {
   994→          chartType: content.chartType || 'bar',
   995→          title: content.title || '',
   996→          dataRange: content.dataRange || '',
   997→          xLabel: content.xLabel || '',
   998→          yLabel: content.yLabel || '',
   999→        };
  1000→      case 'shape':
  1001→        return {
  1002→          shapeType: content.shapeType || 'rectangle',
  1003→          fill: content.fill || '#4A90D9',
  1004→          border: content.border || 'none',
  1005→        };
  1006→      case 'bulletList':
  1007→        return {
  1008→          items: Array.isArray(content) ? content : content.items || [],
  1009→          bulletStyle: content.bulletStyle || 'disc',
  1010→        };
  1011→      default:
  1012→        return content;
  1013→    }
  1014→  }
