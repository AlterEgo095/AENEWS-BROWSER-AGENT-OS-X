   218→  // ─── Lifecycle: Execute ──────────────────────────────────────────
   219→
   220→  async execute(input: AgentInput): Promise<AgentOutput> {
   221→    // 1. Check status
   222→    if (this.status !== AgentStatus.RUNNING && this.status !== AgentStatus.IDLE) {
   223→      throw new AgentError(
   224→        `Agent cannot execute in ${this.status} state`,
   225→        AgentErrorCode.NOT_RUNNING,
   226→        this.config.id,
   227→        input.taskId,
   228→      );
   229→    }
   230→
   231→    // 2. Check circuit breaker
   232→    if (this.circuitBreaker.state === 'open') {
   233→      throw new AgentError(
   234→        'Circuit breaker is open; agent is not accepting tasks',
   235→        AgentErrorCode.CIRCUIT_BREAKER_OPEN,
   236→        this.config.id,
   237→        input.taskId,
   238→      );
   239→    }
   240→
   241→    // 3. Check max concurrent tasks
   242→    if (this.currentTasks.size >= this.config.maxConcurrentTasks) {
   243→      throw new AgentError(
   244→        `Max concurrent tasks (${this.config.maxConcurrentTasks}) reached`,
   245→        AgentErrorCode.MAX_CONCURRENT_TASKS,
   246→        this.config.id,
   247→        input.taskId,
   248→      );
   249→    }
   250→
   251→    // 4. Check permissions
   252→    await this.checkPermission(PermissionAction.EXECUTE, PermissionResource.TASK);
   253→
   254→    // 5. Validate input
   255→    const isValid = await this.onValidateInput(input);
   256→    if (!isValid) {
   257→      throw new AgentError(
   258→        'Invalid input provided',
   259→        AgentErrorCode.INVALID_INPUT,
   260→        this.config.id,
   261→        input.taskId,
   262→      );
   263→    }
   264→
   265→    // 6. Set up execution context
   266→    this.correlationId = input.context?.correlationId || uuidv4();
   267→    this.currentTasks.add(input.taskId);
   268→    this.lastActivity = new Date();
   269→
   270→    // 7. Record start time
   271→    const startTime = Date.now();
   272→    const startMemory = process.memoryUsage().heapUsed;
   273→
   274→    this.executeLifecyclePhase(LifecyclePhase.PRE_EXECUTE, { taskId: input.taskId });
   275→    this.emitEvent(AgentEventType.TASK_STARTED, {
   276→      taskId: input.taskId,
   277→      agentId: this.config.id,
   278→      priority: input.priority || TaskPriority.NORMAL,
   279→    });
   280→
   281→    try {
   282→      // Auto-start if idle
   283→      if (this.status === AgentStatus.IDLE) {
   284→        await this.start();
   285→      }
   286→
   287→      // 8. Execute with retry logic (exponential backoff)
   288→      let retryCount = 0;
   289→      let lastError: Error | null = null;
   290→
   291→      const output = await this.executeWithRetry<AgentOutput>(async () => {
   292→        try {
   293→          // Execute with timeout
   294→          const result = await this.executeWithTimeout(input);
   295→          return result;
   296→        } catch (execError) {
   297→          retryCount++;
   298→          lastError = execError as Error;
   299→
   300→          if (retryCount > this.config.retryPolicy.maxRetries) {
   301→            throw execError;
   302→          }
   303→
   304→          const backoffMs = this.calculateBackoff(retryCount);
   305→          this.logger.warn(
   306→            `Task ${input.taskId} failed (attempt ${retryCount}/${this.config.retryPolicy.maxRetries}), ` +
   307→            `retrying in ${backoffMs}ms: ${lastError.message}`,
   308→          );
   309→          await this.sleep(backoffMs);
   310→
   311→          // Re-throw so executeWithRetry can handle it
   312→          throw execError;
   313→        }
   314→      });
   315→
   316→      // 9. Collect metrics and emit events on success
   317→      this.currentTasks.delete(input.taskId);
   318→      this.completedTaskCount++;
   319→      this.lastActivity = new Date();
   320→
   321→      this.recordCircuitBreakerSuccess();
   322→
   323→      // Store result in memory
   324→      await this.storeExecutionResult(input, output);
   325→
   326→      // Lifecycle and event hooks
   327→      this.executeLifecyclePhase(LifecyclePhase.POST_EXECUTE, { taskId: input.taskId });
   328→      this.emitEvent(AgentEventType.TASK_COMPLETED, {
   329→        taskId: input.taskId,
   330→        agentId: this.config.id,
   331→        success: true,
   332→        executionTimeMs: Date.now() - startTime,
   333→        result: output.result,
   334→      });
   335→
   336→      await this.onSuccess(input, output);
   337→
   338→      // 10. Return output
   339→      return output;
   340→    } catch (error) {
   341→      this.currentTasks.delete(input.taskId);
   342→      this.failedTaskCount++;
   343→      this.recordCircuitBreakerFailure();
   344→      this.lastActivity = new Date();
   345→
   346→      const agentError = error as AgentError;
   347→      this.logger.error(
   348→        `Task ${input.taskId} execution error: ${agentError.message}`,
   349→        agentError.stack,
   350→      );
   351→
   352→      this.emitEvent(AgentEventType.AGENT_ERROR, {
   353→        errorCode: agentError.code || AgentErrorCode.EXECUTION_FAILED,
   354→        errorMessage: agentError.message,
   355→        taskId: input.taskId,
   356→        recoverable: true,
   357→      } as AgentErrorPayload);
   358→
   359→      this.emitEvent(AgentEventType.TASK_FAILED, {
   360→        taskId: input.taskId,
   361→        agentId: this.config.id,
   362→        success: false,
   363→        executionTimeMs: Date.now() - startTime,
   364→        error: agentError.message,
   365→      });
   366→
   367→      await this.onFailure(input, agentError);
   368→
   369→      // 10. Return output (with error info)
   370→      return {
   371→        taskId: input.taskId,
   372→        success: false,
   373→        result: null,
   374→        error: agentError.message,
   375→        metrics: this.collectMetrics(startTime, startMemory),
   376→        timestamp: new Date(),
   377→      };
   378→    }
   379→  }

   413→  // ─── Lifecycle: Stop ─────────────────────────────────────────────
   414→
   415→  async stop(): Promise<void> {
   416→    this.executeLifecyclePhase(LifecyclePhase.PRE_STOP);
   417→
   418→    // Clear all active timeouts
   419→    for (const [taskId, timeout] of this.activeTimeouts) {
   420→      clearTimeout(timeout);
   421→      this.activeTimeouts.delete(taskId);
   422→    }
   423→
   424→    // Wait for current tasks to finish (with a grace period)
   425→    const gracePeriodMs = 5000;
   426→    const startTime = Date.now();
   427→    while (this.currentTasks.size > 0 && Date.now() - startTime < gracePeriodMs) {
   428→      await this.sleep(100);
   429→    }
   430→
   431→    // Force-clear remaining tasks
   432→    if (this.currentTasks.size > 0) {
   433→      this.logger.warn(
   434→        `Force-stopping with ${this.currentTasks.size} active tasks`,
   435→      );
   436→      this.currentTasks.clear();
   437→    }
   438→
   439→    this.stopHealthMonitoring();
   440→    this.stopMetricsCollection();
   441→
   442→    await this.transitionTo(AgentStatus.STOPPED);
   443→    this.executeLifecyclePhase(LifecyclePhase.POST_STOP);
   444→
   445→    this.logger.log(`Agent stopped: ${this.config.id}`);
   446→    this.emitEvent(AgentEventType.AGENT_STOPPED, {
   447→      agentId: this.config.id,
   448→      completedTasks: this.completedTaskCount,
   449→      failedTasks: this.failedTaskCount,
   450→    });
   451→  }

   472→    this.emitEvent(AgentEventType.AGENT_DESTROYED, {
   473→      agentId: this.config.id,
   474→      cluster: this.config.cluster,

   586→  // ─── Permission Checking ─────────────────────────────────────────
   587→
   588→  /**
   589→   * Check if the agent has a specific permission.
   590→   * Overloaded: supports both enum-based and string-based checks.
   591→   */
   592→  protected async checkPermission(
   593→    action: PermissionAction | string,
   594→    resource: PermissionResource | string,
   595→  ): Promise<boolean> {
   596→    const permissionString = `${action}:${resource}`;
   597→
   598→    const hasPermission = this.config.permissions.includes(permissionString) ||
   599→      this.config.permissions.includes('*');
   600→
   601→    if (!hasPermission) {
   602→      // Also check via the permission evaluator if available
   603→      if (this.permissionEvaluator && typeof this.permissionEvaluator.hasPermission === 'function') {
   604→        const evaluated = await this.permissionEvaluator.hasPermission(
   605→          this.config.id,
   606→          action as PermissionAction,
   607→          resource as PermissionResource,
   608→        );
   609→        if (evaluated) return true;
   610→      }
   611→
   612→      throw new AgentError(
   613→        `Permission denied: ${permissionString}`,
   614→        AgentErrorCode.PERMISSION_DENIED,
   615→        this.config.id,
   616→      );
   617→    }
   618→
   619→    return true;
   620→  }

   622→  /**
   623→   * Check permission by resource and action strings.
   624→   * Returns boolean without throwing.
   625→   */
   626→  hasPermissionForResource(resource: string, action: string): boolean {
   627→    return this.config.permissions.includes(`${action}:${resource}`) ||
   628→      this.config.permissions.includes('*');
   629→  }

   631→  // ─── Timeout Handling ────────────────────────────────────────────
   632→
   633→  /**
   634→   * Execute agent task with a timeout.
   635→   */
   636→  protected async executeWithTimeout(input: AgentInput): Promise<AgentOutput> {
   637→    const timeoutMs = input.context?.timeout || this.config.timeout;
   638→
   639→    return new Promise<AgentOutput>((resolve, reject) => {
   640→      const timeoutId = setTimeout(() => {
   641→        this.activeTimeouts.delete(input.taskId);
   642→        reject(
   643→          new AgentError(
   644→            `Task timed out after ${timeoutMs}ms`,
   645→            AgentErrorCode.TIMEOUT,
   646→            this.config.id,
   647→            input.taskId,
   648→          ),
   649→        );
   650→      }, timeoutMs);
   651→
   652→      this.activeTimeouts.set(input.taskId, timeoutId);
   653→
   654→      this.onExecute(input)
   655→        .then((output) => {
   656→          clearTimeout(timeoutId);
   657→          this.activeTimeouts.delete(input.taskId);
   658→          resolve(output);
   659→        })
   660→        .catch((error) => {
   661→          clearTimeout(timeoutId);
   662→          this.activeTimeouts.delete(input.taskId);
   663→          reject(error);
   664→        });
   665→    });
   666→  }

   668→  /**
   669→   * Generic timeout wrapper for any promise.
   670→   * Rejects with AgentError if the promise does not resolve within the specified time.
   671→   */
   672→  protected withTimeout<T>(promise: Promise<T>, ms: number): Promise<T> {
   673→    return new Promise<T>((resolve, reject) => {
   674→      const timeoutId = setTimeout(() => {
   675→        reject(
   676→          new AgentError(
   677→            `Operation timed out after ${ms}ms`,
   678→            AgentErrorCode.TIMEOUT,
   679→            this.config.id,
   680→          ),
   681→        );
   682→      }, ms);
   683→
   684→      promise
   685→        .then((result) => {
   686→          clearTimeout(timeoutId);
   687→          resolve(result);
   688→        })
   689→        .catch((error) => {
   690→          clearTimeout(timeoutId);
   691→          reject(error);
   692→        });
   693→    });
   694→  }

   696→  // ─── Retry Logic ─────────────────────────────────────────────────
   697→
   698→  /**
   699→   * Calculate exponential backoff delay for a given retry attempt.
   700→   */
   701→  protected calculateBackoff(retryCount: number): number {
   702→    const { backoffMs, exponentialBackoff } = this.config.retryPolicy;
   703→    if (exponentialBackoff) {
   704→      return backoffMs * Math.pow(2, retryCount - 1) + Math.random() * 100;
   705→    }
   706→    return backoffMs;
   707→  }
   708→
   709→  /**
   710→   * Execute a function with retry logic and exponential backoff.
   711→   * Retries up to maxRetries times on failure.
   712→   */
   713→  protected async executeWithRetry<T>(fn: () => Promise<T>): Promise<T> {
   714→    const { maxRetries, backoffMs, exponentialBackoff } = this.config.retryPolicy;
   715→    let lastError: Error | null = null;
   716→
   717→    for (let attempt = 0; attempt <= maxRetries; attempt++) {
   718→      try {
   719→        return await fn();
   720→      } catch (error) {
   721→        lastError = error as Error;
   722→
   723→        if (attempt >= maxRetries) {
   724→          this.logger.error(
   725→            `All ${maxRetries} retries exhausted: ${lastError.message}`,
   726→          );
   727→          throw new AgentError(
   728→            `Retry exhausted after ${maxRetries} attempts: ${lastError.message}`,
   729→            AgentErrorCode.RETRY_EXHAUSTED,
   730→            this.config.id,
   731→            undefined,
   732→            lastError,
   733→          );
   734→        }
   735→
   736→        const delay = exponentialBackoff
   737→          ? backoffMs * Math.pow(2, attempt) + Math.random() * 100
   738→          : backoffMs;
   739→
   740→        this.logger.warn(
   741→          `Attempt ${attempt + 1}/${maxRetries} failed, retrying in ${Math.round(delay)}ms: ${lastError.message}`,
   742→        );
   743→
   744→        await this.sleep(delay);
   745→      }
   746→    }
   747→
   748→    // This should never be reached, but TypeScript needs it
   749→    throw lastError!;
   750→  }

   780→    try {
   781→      const customHealthy = await this.onHealthCheck();
   782→      const isHealthy = customHealthy && this.status !== AgentStatus.ERROR;
   783→
   784→      const previousHealth = this.healthState.isHealthy;
   785→      this.healthState = {
   786→        isHealthy,
   787→        lastHealthCheck: new Date(),

   790→      };
   791→
   792→      if (!isHealthy && this.healthState.consecutiveFailures >= 3) {

   801→      }

   803→      return isHealthy;
   804→    } catch (error) {
   805→      this.healthState.consecutiveFailures++;
   806→      this.healthState.isHealthy = false;
   807→      this.healthState.lastHealthCheck = new Date();

   842→  protected async transitionTo(newStatus: AgentStatus): Promise<void> {
   843→    const previousStatus = this.status;
   844→
   845→    if (!this.isValidTransition(previousStatus, newStatus)) {
   846→      this.logger.warn(
   847→        `Invalid state transition: ${previousStatus} → ${newStatus}`,
   848→      );
   849→      // Allow some transitions for recovery scenarios
   850→      if (newStatus !== AgentStatus.ERROR && newStatus !== AgentStatus.STOPPED) {
   851→        throw new AgentError(
   852→          `Invalid state transition: ${previousStatus} → ${newStatus}`,
   853→          AgentErrorCode.EXECUTION_FAILED,
   854→          this.config.id,
   855→        );
   856→      }
   857→    }
   858→
   859→    this.status = newStatus;
   860→    this.lastActivity = new Date();
   861→
   862→    this.emitEvent(AgentEventType.AGENT_STATUS_CHANGED, {
   863→      previousStatus,
   864→      newStatus,
   865→      reason: `Transition from ${previousStatus} to ${newStatus}`,
   866→    } as AgentStatusChangedPayload);
   867→
   868→    this.logger.log(`State transition: ${previousStatus} → ${newStatus}`);
   869→  }

  1008→  protected recordCircuitBreakerSuccess(): void {
  1009→    if (this.circuitBreaker.state === 'half_open') {
  1010→      this.circuitBreaker.halfOpenSuccesses++;
  1011→      if (this.circuitBreaker.halfOpenSuccesses >= this.circuitBreaker.successThreshold) {
  1012→        this.circuitBreaker.state = 'closed';
  1013→        this.circuitBreaker.failureCount = 0;
  1014→        this.circuitBreaker.halfOpenSuccesses = 0;
  1015→        this.logger.log('Circuit breaker closed');
  1016→        this.emitEvent(AgentEventType.CIRCUIT_BREAKER_CLOSED, {
  1017→          agentId: this.config.id,
  1018→          state: 'closed',
  1019→          failureCount: 0,
  1020→          lastFailureTime: this.circuitBreaker.lastFailureTime || new Date(),
  1021→        });
  1022→      }
  1023→    } else if (this.circuitBreaker.state === 'closed') {
  1024→      this.circuitBreaker.failureCount = 0;
  1025→    }
  1026→  }
  1027→
  1028→  protected recordCircuitBreakerFailure(): void {
  1029→    this.circuitBreaker.failureCount++;
  1030→    this.circuitBreaker.lastFailureTime = new Date();
  1031→
  1032→    if (this.circuitBreaker.state === 'half_open') {
  1033→      this.circuitBreaker.state = 'open';
  1034→      this.logger.warn('Circuit breaker opened (failure during half-open)');
  1035→    } else if (this.circuitBreaker.failureCount >= this.circuitBreaker.failureThreshold) {
  1036→      this.circuitBreaker.state = 'open';
  1037→      this.logger.error(
  1038→        `Circuit breaker opened after ${this.circuitBreaker.failureCount} failures`,
  1039→      );
  1040→      this.emitEvent(AgentEventType.CIRCUIT_BREAKER_OPENED, {
  1041→        agentId: this.config.id,
  1042→        state: 'open',
  1043→        failureCount: this.circuitBreaker.failureCount,
  1044→        lastFailureTime: this.circuitBreaker.lastFailureTime,
  1045→      });
  1046→
  1047→      // Schedule transition to half-open
  1048→      setTimeout(() => {
  1049→        if (this.circuitBreaker.state === 'open') {
  1050→          this.circuitBreaker.state = 'half_open';
  1051→          this.circuitBreaker.halfOpenSuccesses = 0;
  1052→          this.logger.log('Circuit breaker transitioned to half-open');
  1053→        }
  1054→      }, this.circuitBreaker.resetTimeoutMs);
  1055→    }
  1056→  }

  1081→  protected async executeLifecyclePhase(
  1082→    phase: LifecyclePhase,
  1083→    extraContext?: Record<string, any>,
  1084→  ): Promise<void> {
  1085→    this.lifecyclePhase = phase;
  1086→    const hooks = this.lifecycleHooks.get(phase) || [];
  1087→
  1088→    const context: LifecycleContext = {
  1089→      agentId: this.config.id,
  1090→      phase,
  1091→      previousStatus: this.status,
  1092→      newStatus: this.status,
  1093→      timestamp: new Date(),
  1094→      correlationId: this.correlationId,
  1095→      ...extraContext,
  1096→    };
  1097→
  1098→    for (const hook of hooks) {
  1099→      try {
  1100→        const result = await hook(context);
  1101→        if (!result.shouldProceed) {
  1102→          this.logger.warn(
  1103→            `Lifecycle hook for ${phase} blocked execution: ${result.error?.message || 'No reason provided'}`,
  1104→          );
  1105→          break;
  1106→        }
  1107→      } catch (error) {
  1108→        this.logger.error(
  1109→          `Lifecycle hook error for ${phase}: ${(error as Error).message}`,
  1110→        );
  1111→      }
  1112→    }
  1113→  }
