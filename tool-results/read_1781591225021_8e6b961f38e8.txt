   109→function StatCard({ label, value, sub, icon: Icon, color = 'text-primary' }: {
   110→  label: string; value: string | number; sub?: string; icon: any; color?: string;

   124→function TrendBadge({ trend }: { trend: 'improving' | 'degrading' | 'stable' | 'up' | 'down' | 'flat' }) {
   125→  const isUp = trend === 'improving' || trend === 'up';
   126→  const isDown = trend === 'degrading' || trend === 'down';
   127→  return (
   128→    <span className={cn(
   129→      'inline-flex items-center gap-1 rounded-full px-2 py-0.5 text-[10px] font-semibold',
   130→      isUp && 'bg-emerald-500/15 text-emerald-400',
   131→      isDown && 'bg-red-500/15 text-red-400',
   132→      !isUp && !isDown && 'bg-yellow-500/15 text-yellow-400',
   133→    )}>
   134→      {isUp && <ArrowUpRight className="h-3 w-3" />}
   135→      {isDown && <ArrowDownRight className="h-3 w-3" />}
   136→      {!isUp && !isDown && <Minus className="h-3 w-3" />}
   137→      {trend}
   138→    </span>
   139→  );
   140→}
   141→
   142→function ImpactBadge({ impact }: { impact: 'positive' | 'negative' | 'neutral' }) {
   143→  return (
   144→    <span className={cn(
   145→      'inline-flex items-center gap-1 rounded-full px-2 py-0.5 text-[10px] font-semibold',
   146→      impact === 'positive' && 'bg-emerald-500/15 text-emerald-400',
   147→      impact === 'negative' && 'bg-red-500/15 text-red-400',
   148→      impact === 'neutral' && 'bg-yellow-500/15 text-yellow-400',
   149→    )}>
   150→      {impact === 'positive' && <CheckCircle2 className="h-3 w-3" />}
   151→      {impact === 'negative' && <XCircle className="h-3 w-3" />}
   152→      {impact === 'neutral' && <Minus className="h-3 w-3" />}
   153→      {impact}
   154→    </span>
   155→  );
   156→}

   158→function PriorityBadge({ priority }: { priority: string }) {
   159→  const colors: Record<string, string> = {
   160→    critical: 'bg-red-500/15 text-red-400',
   161→    high: 'bg-orange-500/15 text-orange-400',
   162→    medium: 'bg-yellow-500/15 text-yellow-400',
   163→    low: 'bg-blue-500/15 text-blue-400',
   164→  };
   165→  return (
   166→    <span className={cn('inline-flex rounded-full px-2 py-0.5 text-[10px] font-semibold', colors[priority] || colors.low)}>
   167→      {priority}
   168→    </span>
   169→  );
   170→}
