   184→/**

   187→function generateAuthHeader(role: UserRole = UserRole.SUPER_ADMIN): string {
   188→  const jwtService = new JwtService({ secret: 'test-jwt-secret' });
   189→  const token = jwtService.sign({
   190→    sub: 'user-123',
   191→    email: 'admin@aenews.io',
   192→    role,
   193→    tenantId: 'tenant-123',
   194→  });
   195→  return `Bearer ${token}`;
   196→}
