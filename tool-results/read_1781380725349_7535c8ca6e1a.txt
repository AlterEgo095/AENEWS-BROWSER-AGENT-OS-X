   250→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   251→    const startTime = Date.now();
   252→    const { action, ...params } = input.payload;
   253→
   254→    if (!action) {
   255→      return this.createAgentOutput(
   256→        input.taskId,
   257→        false,
   258→        null,
   259→        'Missing required parameter: action',
   260→        startTime,
   261→      );
   262→    }
   263→
   264→    const supportedActions = [
   265→      'reviewCode',
   266→      'checkBestPractices',
   267→      'findBugs',
   268→      'checkSecurity',
   269→      'analyzeComplexity',
   270→    ];
   271→
   272→    if (!supportedActions.includes(action)) {
   273→      return this.createAgentOutput(
   274→        input.taskId,
   275→        false,
   276→        null,
   277→        `Unknown code review action: ${action}. Supported: ${supportedActions.join(', ')}`,
   278→        startTime,
   279→      );
   280→    }
   281→
   282→    try {
   283→      const tool = this.getTool(action);
   284→      if (!tool) {
   285→        return this.createAgentOutput(
   286→          input.taskId,
   287→          false,
   288→          null,
   289→          `Tool not found: ${action}`,
   290→          startTime,
   291→        );
   292→      }
   293→
   294→      const result = await tool.execute(params);
   295→
   296→      this.reviewHistory.push({
   297→        action,
   298→        language: params.language || 'unknown',
   299→        timestamp: new Date(),
   300→        issueCount:
   301→          result.issues?.length || result.bugs?.length || result.vulnerabilities?.length || 0,
   302→      });
   303→
   304→      // Store the review result for future reference
   305→      await this.storeInWorkingMemory(
   306→        `review:last:${action}`,
   307→        { params, result, timestamp: new Date() },
   308→        300000,
   309→      );
   310→
   311→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   312→    } catch (error) {
   313→      const msg = (error as Error).message;
   314→      this.logger.error(`CodeReview execution failed for ${action}: ${msg}`);
   315→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   316→    }
   317→  }
   318→

   326→  private async reviewCode(params: {
   327→    code: string;
   328→    language: string;
   329→    filePath?: string;
   330→    severityThreshold?: string;
   331→  }): Promise<{
   332→    issues: ReviewIssue[];
   333→    score: number;
   334→    summary: string;
   335→    filePath?: string;
   336→  }> {
   337→    const { code, language, filePath, severityThreshold = 'low' } = params;
   338→
   339→    if (!code || typeof code !== 'string') {
   340→      throw new Error('Source code is required for review');
   341→    }
   342→    if (!language || typeof language !== 'string') {
   343→      throw new Error('Programming language is required');
   344→    }
   345→
   346→    const lines = code.split('\n');
   347→    const issues: ReviewIssue[] = [];
   348→
   349→    // Style and formatting checks
   350→    this.checkFormatting(lines, issues, language);
   351→
   352→    // Naming conventions
   353→    this.checkNamingConventions(lines, issues, language);
   354→
   355→    // Code structure checks
   356→    this.checkCodeStructure(lines, issues, language);
   357→
   358→    // Error handling checks
   359→    this.checkErrorHandling(lines, issues, language);
   360→
   361→    // Performance anti-patterns
   362→    this.checkPerformanceAntiPatterns(lines, issues, language);
   363→
   364→    // Filter by severity threshold
   365→    const severityOrder: Record<string, number> = { low: 0, medium: 1, high: 2, critical: 3 };
   366→    const minSeverity = severityOrder[severityThreshold] || 0;
   367→    const filteredIssues = issues.filter(
   368→      (issue) => (severityOrder[issue.severity] || 0) >= minSeverity,
   369→    );
   370→
   371→    // Calculate review score (0-100)
   372→    const score = this.calculateReviewScore(lines.length, filteredIssues);
   373→
   374→    // Generate summary
   375→    const criticalCount = filteredIssues.filter((i) => i.severity === 'critical').length;
   376→    const highCount = filteredIssues.filter((i) => i.severity === 'high').length;
   377→    const mediumCount = filteredIssues.filter((i) => i.severity === 'medium').length;
   378→    const lowCount = filteredIssues.filter((i) => i.severity === 'low').length;
   379→
   380→    const summary =
   381→      `Code review complete: ${filteredIssues.length} issue(s) found ` +
   382→      `(${criticalCount} critical, ${highCount} high, ${mediumCount} medium, ${lowCount} low). ` +
   383→      `Score: ${score}/100.`;
   384→
   385→    this.logger.log(`Code review: ${filteredIssues.length} issues, score=${score}`);
   386→
   387→    return { issues: filteredIssues, score, summary, filePath };
   388→  }

   390→  private async checkBestPractices(params: {
   391→    code: string;
   392→    language: string;
   393→    framework?: string;
   394→  }): Promise<{
   395→    practices: Array<{ name: string; followed: boolean; details: string; suggestion?: string }>;
   396→    adherence: number;
   397→    suggestions: string[];
   398→  }> {
   399→    const { code, language, framework } = params;
   400→
   401→    if (!code || typeof code !== 'string') {
   402→      throw new Error('Source code is required');
   403→    }
   404→    if (!language || typeof language !== 'string') {
   405→      throw new Error('Programming language is required');
   406→    }
   407→
   408→    const lines = code.split('\n');
   409→    const practices: Array<{
   410→      name: string;
   411→      followed: boolean;
   412→      details: string;
   413→      suggestion?: string;
   414→    }> = [];
   415→    const suggestions: string[] = [];
   416→
   417→    // Language-agnostic best practices
   418→    practices.push({
   419→      name: 'Single Responsibility Principle',
   420→      followed: this.checkSingleResponsibility(lines),
   421→      details: 'Each function/class should have one reason to change',
   422→      suggestion: 'Consider splitting large functions into smaller, focused ones',
   423→    });
   424→
   425→    practices.push({
   426→      name: "DRY (Don't Repeat Yourself)",
   427→      followed: this.checkDRY(lines),
   428→      details: 'Avoid code duplication',
   429→      suggestion: 'Extract repeated code into reusable functions or utilities',
   430→    });
   431→
   432→    practices.push({
   433→      name: 'Proper Error Handling',
   434→      followed: this.checkProperErrorHandling(lines, language),
   435→      details: 'All error-prone operations should have proper error handling',
   436→      suggestion: 'Add try-catch blocks around operations that can fail',
   437→    });
   438→
   439→    practices.push({
   440→      name: 'Consistent Naming',
   441→      followed: this.checkConsistentNaming(lines, language),
   442→      details: 'Naming conventions should be consistent throughout the codebase',
   443→      suggestion: 'Follow language-specific naming conventions (camelCase, snake_case, etc.)',
   444→    });
   445→
   446→    practices.push({
   447→      name: 'No Magic Numbers',
   448→      followed: this.checkMagicNumbers(lines),
   449→      details: 'Avoid unexplained numeric literals',
   450→      suggestion: 'Extract magic numbers into named constants',
   451→    });
   452→
   453→    // Language-specific practices
   454→    if (language === 'typescript' || language === 'javascript') {
   455→      practices.push({
   456→        name: 'Use Strict Mode',
   457→        followed:
   458→          code.includes('"use strict"') ||
   459→          code.includes("'use strict'") ||
   460→          language === 'typescript',
   461→        details: 'Enable strict mode for better error detection',
   462→        suggestion: 'Add "use strict" directive or use TypeScript strict mode',
   463→      });
   464→
   465→      practices.push({
   466→        name: 'Prefer Const Over Let',
   467→        followed: this.checkConstOverLet(code),
   468→        details: 'Use const for variables that are not reassigned',
   469→        suggestion: 'Replace let with const where the variable is not reassigned',
   470→      });
   471→
   472→      if (framework === 'nestjs') {
   473→        practices.push({
   474→          name: 'Dependency Injection',
   475→          followed: code.includes('constructor(') && code.includes('@Injectable'),
   476→          details: 'Use NestJS dependency injection pattern',
   477→          suggestion: 'Inject dependencies through constructor parameters',
   478→        });
   479→      }
   480→    }
   481→
   482→    if (language === 'python') {
   483→      practices.push({
   484→        name: 'Type Hints',
   485→        followed: code.includes('def ') && code.includes(': ') && code.includes('-> '),
   486→        details: 'Use Python type hints for better code documentation',
   487→        suggestion: 'Add type hints to function signatures',
   488→      });
   489→    }
   490→
   491→    const followedCount = practices.filter((p) => p.followed).length;
   492→    const adherence = Math.round((followedCount / practices.length) * 100);
   493→
   494→    for (const practice of practices) {
   495→      if (!practice.followed && practice.suggestion) {
   496→        suggestions.push(practice.suggestion);
   497→      }
   498→    }
   499→
   500→    this.logger.log(
   501→      `Best practices check: ${followedCount}/${practices.length} followed, ${adherence}% adherence`,
   502→    );
   503→
   504→    return { practices, adherence, suggestions };
   505→  }

   507→  private async findBugs(params: {
   508→    code: string;
   509→    language: string;
   510→    includeWarnings?: boolean;
   511→  }): Promise<{
   512→    bugs: BugFinding[];
   513→    warnings: BugFinding[];
   514→    riskLevel: string;
   515→  }> {
   516→    const { code, language, includeWarnings = true } = params;
   517→
   518→    if (!code || typeof code !== 'string') {
   519→      throw new Error('Source code is required');
   520→    }
   521→    if (!language || typeof language !== 'string') {
   522→      throw new Error('Programming language is required');
   523→    }
   524→
   525→    const lines = code.split('\n');
   526→    const bugs: BugFinding[] = [];
   527→    const warnings: BugFinding[] = [];
   528→
   529→    // Bug detection patterns
   530→    this.detectOffByOneErrors(lines, bugs, language);
   531→    this.detectNullReferenceRisks(lines, bugs, language);
   532→    this.detectResourceLeaks(lines, bugs, language);
   533→    this.detectRaceConditions(lines, bugs, language);
   534→    this.detectUnhandledPromises(lines, bugs, language);
   535→    this.detectIncorrectComparisons(lines, bugs, language);
   536→
   537→    // Warning-level patterns
   538→    if (includeWarnings) {
   539→      this.detectUnusedVariables(lines, warnings, language);
   540→      this.detectUnreachableCode(lines, warnings, language);
   541→      this.detectImplicitTypeCoercion(lines, warnings, language);
   542→    }
   543→
   544→    // Calculate overall risk level
   545→    const criticalBugs = bugs.filter((b) => b.severity === 'critical').length;
   546→    const highBugs = bugs.filter((b) => b.severity === 'high').length;
   547→    let riskLevel = 'low';
   548→    if (criticalBugs > 0) riskLevel = 'critical';
   549→    else if (highBugs > 2) riskLevel = 'high';
   550→    else if (highBugs > 0 || bugs.length > 5) riskLevel = 'medium';
   551→
   552→    this.logger.log(
   553→      `Bug detection: ${bugs.length} bug(s), ${warnings.length} warning(s), risk=${riskLevel}`,
   554→    );
   555→
   556→    return { bugs, warnings, riskLevel };
   557→  }

   605→  private async analyzeComplexity(params: {
   606→    code: string;
   607→    language: string;
   608→    thresholds?: Record<string, number>;
   609→  }): Promise<{
   610→    cyclomaticComplexity: number;
   611→    cognitiveComplexity: number;
   612→    nestingDepth: number;
   613→    maintainabilityIndex: number;
   614→    hotspots: ComplexityHotspot[];
   615→  }> {
   616→    const { code, language, thresholds } = params;
   617→
   618→    if (!code || typeof code !== 'string') {
   619→      throw new Error('Source code is required');
   620→    }
   621→    if (!language || typeof language !== 'string') {
   622→      throw new Error('Programming language is required');
   623→    }
   624→
   625→    const defaultThresholds = {
   626→      maxCyclomatic: 10,
   627→      maxCognitive: 15,
   628→      maxNesting: 4,
   629→      minMaintainability: 20,
   630→    };
   631→
   632→    const effectiveThresholds = { ...defaultThresholds, ...thresholds };
   633→
   634→    const lines = code.split('\n');
   635→
   636→    // Calculate overall metrics

   638→    const cognitiveComplexity = this.calculateCognitiveComplexity(lines, language);
   639→    const nestingDepth = this.calculateMaxNesting(lines, language);
   640→    const maintainabilityIndex = this.calculateMaintainabilityIndex(
   641→      lines.length,
   642→      cyclomaticComplexity,
   643→    );
   644→
   645→    // Find complexity hotspots (functions with high complexity)
   646→    const hotspots = this.findComplexityHotspots(lines, language, effectiveThresholds);
   647→
   648→    this.logger.log(
   649→      `Complexity analysis: cyclomatic=${cyclomaticComplexity}, cognitive=${cognitiveComplexity}, ` +
   650→        `nesting=${nestingDepth}, maintainability=${maintainabilityIndex}`,

   653→    return {

   756→  private checkCodeStructure(lines: string[], issues: ReviewIssue[], language: string): void {
   757→    // Check for functions that are too long
   758→    let functionStart = -1;
   759→    let functionBraceCount = 0;
   760→    let functionName = '';
   761→
   762→    for (let i = 0; i < lines.length; i++) {

   769→        ),
   770→      );
   771→      if (funcMatch && functionStart === -1) {
   772→        functionName = funcMatch[1] || funcMatch[2];
   773→        functionStart = i;
   774→        functionBraceCount = 0;
   775→      }
   776→
   777→      // Track brace depth
   778→      if (functionStart !== -1) {
   779→        functionBraceCount += (line.match(/{/g) || []).length;
   780→        functionBraceCount -= (line.match(/}/g) || []).length;
   781→
   782→        if (functionBraceCount <= 0 && functionStart !== -1) {
   783→          const functionLength = i - functionStart + 1;
   784→          if (functionLength > 50) {
   785→            issues.push({
   786→              severity: 'medium',
   787→              category: 'structure',
   788→              message: `Function "${functionName}" is too long (${functionLength} lines)`,
   789→              line: functionStart + 1,
   790→              ruleId: 'max-function-length',
   791→              suggestion: 'Break the function into smaller, focused functions',
   792→            });
   793→          }
   794→          functionStart = -1;
   795→        }
   796→      }
   797→    }
   798→
   799→    // Check for too many parameters
   800→    for (let i = 0; i < lines.length; i++) {
   801→      const line = lines[i];
   802→      const paramMatch = line.match(
   803→        /(?:function\s+\w+|(?:const|let)\s+\w+\s*=\s*(?:async\s+)?)\(([^)]*)\)/,
   804→      );
   805→      if (paramMatch) {
   806→        const params = paramMatch[1].split(',').filter((p) => p.trim().length > 0);
   807→        if (params.length > 5) {
   808→          issues.push({
   809→            severity: 'medium',
   810→            category: 'structure',
   811→            message: `Function has too many parameters (${params.length})`,
   812→            line: i + 1,
   813→            ruleId: 'max-params',
   814→            suggestion: 'Consider using an options object pattern instead of many parameters',
   815→          });
   816→        }
   817→      }
   818→    }
   819→  }

   823→  private checkErrorHandling(lines: string[], issues: ReviewIssue[], language: string): void {

   838→          });

   840→      }
   841→
   842→      // Catch with only console.log
   843→      if (line.match(/catch/) && i + 1 < lines.length) {
   844→        for (let j = i + 1; j < Math.min(i + 5, lines.length); j++) {

   854→            break;
   855→          }
   856→          if (lines[j].includes('}')) break;
   857→        }
   858→      }
   859→
   860→      // Unhandled promise (missing await or .catch)
   861→      if (
   862→        (language === 'typescript' || language === 'javascript') &&
   863→        line.includes('.then(') &&

   871→            message: 'Promise chain missing .catch() handler',

   881→  // ─── Performance Anti-Pattern Checks ──────────────────────────
   882→
   883→  private checkPerformanceAntiPatterns(
   884→    lines: string[],
   885→    issues: ReviewIssue[],
   886→    language: string,
   887→  ): void {
   888→    for (let i = 0; i < lines.length; i++) {
   889→      const line = lines[i];
   890→
   891→      // Array mutation in loop
   892→      if (line.match(/\.push\(|\.splice\(/) && i > 0) {
   893→        for (let j = Math.max(0, i - 3); j < i; j++) {
   894→          if (lines[j].match(/for\s*\(|\.forEach\(|\.map\(/)) {
   895→            issues.push({
   896→              severity: 'medium',
   897→              category: 'performance',
   898→              message: 'Array mutation inside loop can be inefficient',
   899→              line: i + 1,
   900→              ruleId: 'no-mutation-in-loop',
   901→              suggestion: 'Use functional approaches like map/filter/reduce or collect items first',
   902→            });
   903→            break;
   904→          }
   905→        }
   906→      }
   907→
   908→      // Synchronous file operations
   909→      if (line.includes('readFileSync') || line.includes('writeFileSync')) {
   910→        issues.push({
   911→          severity: 'medium',
   912→          category: 'performance',
   913→          message: 'Synchronous file operation blocks the event loop',
   914→          line: i + 1,
   915→          ruleId: 'no-sync-file-ops',
   916→          suggestion: 'Use async file operations (readFile, writeFile) instead',
   917→        });
   918→      }
   919→
   920→      // Nested loops (O(n²) complexity indicator)
   921→      let indentLevel = 0;
   922→      const match = line.match(/^(\s*)/);
   923→      if (match) indentLevel = match[1].length;
   924→
   925→      if (line.match(/for\s*\(|while\s*\(|\.forEach\(/)) {
   926→        for (let j = i + 1; j < Math.min(i + 10, lines.length); j++) {
   927→          const innerMatch = lines[j].match(/^(\s*)/);
   928→          if (
   929→            innerMatch &&
   930→            innerMatch[1].length > indentLevel &&
   931→            lines[j].match(/for\s*\(|while\s*\(|\.forEach\(/)
   932→          ) {
   933→            issues.push({
   934→              severity: 'medium',
   935→              category: 'performance',
   936→              message: 'Nested loop detected — potential O(n²) complexity',
   937→              line: j + 1,
   938→              ruleId: 'no-nested-loops',
   939→              suggestion: 'Consider optimizing with hash maps, sorting, or breaking early',
   940→            });
   941→            break;
   942→          }
   943→        }
   944→      }
   945→    }
   946→  }

  1501→  private calculateCyclomaticComplexity(lines: string[], language: string): number {
  1502→    let complexity = 1; // Base complexity
  1503→
  1504→    const branchKeywords = [
  1505→      /\bif\b/,
  1506→      /\belse\s+if\b/,
  1507→      /\bfor\b/,
  1508→      /\bwhile\b/,
  1509→      /\bcase\b/,
  1510→      /\bcatch\b/,
  1511→      /\?.*:/,
  1512→      /\&\&/,
  1513→      /\|\|/,
  1514→    ];
  1515→
  1516→    for (const line of lines) {
  1517→      for (const keyword of branchKeywords) {
  1518→        const matches = line.match(new RegExp(keyword.source, 'g'));
  1519→        if (matches) {
  1520→          complexity += matches.length;
  1521→        }
  1522→      }
  1523→    }
  1524→
  1525→    return complexity;
  1526→  }
  1527→
  1528→  private calculateCognitiveComplexity(lines: string[], language: string): number {
  1529→    let complexity = 0;
  1530→    let nestingLevel = 0;
  1531→
  1532→    for (const line of lines) {
  1533→      const trimmed = line.trim();
  1534→
  1535→      // Increase nesting
  1536→      if (trimmed.includes('{')) nestingLevel += (trimmed.match(/{/g) || []).length;
  1537→
  1538→      // Cognitive complexity adds nesting level for each branch
  1539→      if (
  1540→        trimmed.match(/\bif\b/) ||
  1541→        trimmed.match(/\belse\b/) ||
  1542→        trimmed.match(/\bfor\b/) ||
  1543→        trimmed.match(/\bwhile\b/) ||
  1544→        trimmed.match(/\bcatch\b/) ||
  1545→        trimmed.match(/\bswitch\b/)
  1546→      ) {
  1547→        complexity += 1 + Math.max(0, nestingLevel - 1);
  1548→      }
  1549→
  1550→      // Logical operators
  1551→      complexity += (trimmed.match(/&&/g) || []).length;
  1552→      complexity += (trimmed.match(/\|\|/g) || []).length;
  1553→
  1554→      // Decrease nesting
  1555→      if (trimmed.includes('}')) nestingLevel -= (trimmed.match(/}/g) || []).length;
  1556→      nestingLevel = Math.max(0, nestingLevel);
  1557→    }
  1558→
  1559→    return complexity;
  1560→  }
  1561→
  1562→  private calculateMaxNesting(lines: string[], language: string): number {
  1563→    let maxNesting = 0;
  1564→    let currentNesting = 0;
  1565→
  1566→    for (const line of lines) {
  1567→      currentNesting += (line.match(/{/g) || []).length;
  1568→      currentNesting -= (line.match(/}/g) || []).length;
  1569→      currentNesting = Math.max(0, currentNesting);
  1570→      maxNesting = Math.max(maxNesting, currentNesting);
  1571→    }
  1572→
  1573→    return maxNesting;
  1574→  }
  1575→
  1576→  private calculateMaintainabilityIndex(linesOfCode: number, cyclomaticComplexity: number): number {
  1577→    // Simplified Maintainability Index (Microsoft formula approximation)
  1578→    const avgLineLength = 40; // Assumed average
  1579→    const halsteadVolume = linesOfCode * avgLineLength;
  1580→    const mi = Math.max(
  1581→      0,
  1582→      171 -
  1583→        5.2 * Math.log(halsteadVolume + 1) -
  1584→        0.23 * cyclomaticComplexity -
  1585→        16.2 * Math.log(linesOfCode + 1),
  1586→    );
  1587→    return Math.round(Math.min(100, Math.max(0, mi)));
  1588→  }

  1590→  private findComplexityHotspots(
  1591→    lines: string[],
  1592→    language: string,
  1593→    thresholds: Record<string, number>,
  1594→  ): ComplexityHotspot[] {
  1595→    const hotspots: ComplexityHotspot[] = [];
  1596→    let funcStart = -1;
  1597→    let funcName = '';
  1598→    let braceCount = 0;
  1599→
  1600→    for (let i = 0; i < lines.length; i++) {
  1601→      const line = lines[i];
  1602→
  1603→      const funcMatch = line.match(
  1604→        /(?:function\s+(\w+)|(?:const|let)\s+(\w+)\s*=\s*(?:async\s+)?(?:function|\())/,
  1605→      );
  1606→      if (funcMatch && funcStart === -1) {
  1607→        funcName = funcMatch[1] || funcMatch[2];
  1608→        funcStart = i;
  1609→        braceCount = 0;
  1610→      }
  1611→
  1612→      if (funcStart !== -1) {
  1613→        braceCount += (line.match(/{/g) || []).length;
  1614→        braceCount -= (line.match(/}/g) || []).length;
  1615→
  1616→        if (braceCount <= 0) {
  1617→          const funcLines = lines.slice(funcStart, i + 1);
  1618→          const cyclomatic = this.calculateCyclomaticComplexity(funcLines, language);
  1619→          const cognitive = this.calculateCognitiveComplexity(funcLines, language);
  1620→          const nesting = this.calculateMaxNesting(funcLines, language);
  1621→
  1622→          if (
  1623→            cyclomatic > thresholds.maxCyclomatic ||
  1624→            cognitive > thresholds.maxCognitive ||
  1625→            nesting > thresholds.maxNesting
  1626→          ) {
  1627→            hotspots.push({
  1628→              functionName: funcName,
  1629→              startLine: funcStart + 1,
  1630→              endLine: i + 1,
  1631→              cyclomatic,
  1632→              cognitive,
  1633→              nesting,
  1634→              suggestion: `Function "${funcName}" has high complexity (cyclomatic=${cyclomatic}, cognitive=${cognitive}, nesting=${nesting}). Consider refactoring into smaller functions.`,
  1635→            });
  1636→          }
  1637→
  1638→          funcStart = -1;
  1639→        }
  1640→      }
  1641→    }
  1642→
  1643→    return hotspots;
  1644→  }
