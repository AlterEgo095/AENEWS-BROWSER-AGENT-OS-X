   492→  private async generateWithLLM(config: Record<string, any>, startTime: number): Promise<AgentResult> {
   493→    const topic = config.topic || config.title;
   494→    const slideCount = config.slideCount || 8;
   495→    const language = config.language || 'fr';
   496→    const style = config.style || 'professional';
   497→    const theme = config.theme || 'corporate';
   498→
   499→    if (!topic) {
   500→      return { success: false, error: 'Topic/title is required for LLM-generated presentations' };
   501→    }
   502→
   503→    this.logger.log(`Generating LLM-powered presentation on "${topic}" (${slideCount} slides, ${language})`);
   504→    this.emitEvent(AgentEventType.AGENT_STARTED, { action: 'generate-with-llm', topic });
   505→
   506→    const llmResult = await this.executeWithLLM(
   507→      `You are a professional presentation designer. Create a ${style} presentation about the given topic.
   508→The presentation should have exactly ${slideCount} slides.
   509→Respond in ${language === 'fr' ? 'French' : language === 'en' ? 'English' : language}.
   510→Return a JSON object with this exact structure:
   511→{
   512→  "title": "Presentation Title",
   513→  "subtitle": "Optional subtitle",
   514→  "author": "AENEWS Agent OS X",
   515→  "slides": [
   516→    {
   517→      "layout": "content",
   518→      "title": "Slide Title",
   519→      "bullets": ["Point 1", "Point 2", "Point 3"],
   520→      "notes": "Speaker notes for this slide"
   521→    }
   522→  ]
   523→}
   524→The first slide should be a title slide (layout "cover" with title and subtitle).
   525→The last slide should be a conclusion/thank you slide (layout "closing").
   526→Each content slide should have 3-5 bullet points.
   527→Use varied layouts: "cover", "content", "section", "stat-block", "two-column", "comparison", "timeline", "quote", "closing".`,
   528→      `Create a ${style} presentation about: ${topic}`,
   529→      { responseFormat: 'json', temperature: 0.7, maxTokens: 4096 },
   530→    );
   531→
   532→    let slideData: any;
   533→    if (llmResult) {
   534→      slideData = this.safeJsonParse(llmResult);
   535→    }

   537→    if (!slideData || !slideData.slides) {
   538→      this.logger.warn('LLM did not return valid slide data — using fallback structure');
   539→      slideData = this.buildPremiumFallback(topic, language);
   540→    }
   541→
   542→    return this.buildPremiumPptx(slideData, theme, startTime);
   543→  }

   545→  /* ── PREMIUM PPTX BUILDER ──────────────────────────────────────── */

   547→  private async buildPremiumPptx(slideData: any, themeName: string, startTime: number): Promise<AgentResult> {
   548→    const pptx = new PptxGenJS();
   549→    const theme = PREMIUM_THEMES[themeName] || PREMIUM_THEMES.corporate;

   603→  private inferLayout(index: number, total: number): SlideLayout {
   604→    if (index === 0) return 'cover';
   605→    if (index === total - 1) return 'closing';
   606→    return 'content';

   609→  private renderSlide(
   610→    pptx: PptxGenJS,
   611→    layout: string,
   612→    def: any,
   613→    theme: PremiumTheme,
   614→    index: number,
   615→    total: number,
   616→  ): void {
   617→    switch (layout) {
   618→      case 'cover':
   619→        this.renderCoverSlide(pptx, def, theme);
   620→        break;
   621→      case 'section':
   622→        this.renderSectionSlide(pptx, def, theme);
   623→        break;
   624→      case 'content':
   625→        this.renderContentSlide(pptx, def, theme, index);
   626→        break;
   627→      case 'two-column':
   628→        this.renderTwoColumnSlide(pptx, def, theme, index);
   629→        break;
   630→      case 'stat-block':
   631→        this.renderStatBlockSlide(pptx, def, theme, index);
   632→        break;
   633→      case 'quote':
   634→        this.renderQuoteSlide(pptx, def, theme);
   635→        break;
   636→      case 'comparison':
   637→        this.renderComparisonSlide(pptx, def, theme, index);
   638→        break;
   639→      case 'timeline':
   640→        this.renderTimelineSlide(pptx, def, theme, index);
   641→        break;
   642→      case 'icon-grid':
   643→        this.renderIconGridSlide(pptx, def, theme, index);
   644→        break;
   645→      case 'image-text':
   646→        this.renderImageTextSlide(pptx, def, theme, index);
   647→        break;
   648→      case 'closing':
   649→        this.renderClosingSlide(pptx, def, theme);
   650→        break;
   651→      default:
   652→        this.renderContentSlide(pptx, def, theme, index);
   653→    }
   654→  }
