   298→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   299→    const startTime = Date.now();
   300→    const { action, ...params } = input.payload;
   301→
   302→    if (!action) {
   303→      return this.createAgentOutput(
   304→        input.taskId,
   305→        false,
   306→        null,
   307→        'Missing required parameter: action',
   308→        startTime,
   309→      );
   310→    }
   311→
   312→    const supportedActions = [
   313→      'orchestrateTask',
   314→      'assignAgent',
   315→      'monitorProgress',
   316→      'rebalanceWorkload',
   317→      'generateOrchestrationPlan',
   318→      'evaluateOutcome',
   319→    ];
   320→
   321→    if (!supportedActions.includes(action)) {
   322→      return this.createAgentOutput(
   323→        input.taskId,
   324→        false,
   325→        null,
   326→        `Unknown orchestration action: ${action}. Supported: ${supportedActions.join(', ')}`,
   327→        startTime,
   328→      );
   329→    }
   330→
   331→    try {
   332→      const tool = this.getTool(action);
   333→      if (!tool) {
   334→        return this.createAgentOutput(
   335→          input.taskId,
   336→          false,
   337→          null,
   338→          `Tool not found: ${action}`,
   339→          startTime,
   340→        );
   341→      }
   342→
   343→      const result = await tool.execute(params);
   344→
   345→      await this.storeInWorkingMemory(
   346→        `orchestrator:last:${action}`,
   347→        { params, result, timestamp: new Date() },
   348→        300000,
   349→      );
   350→
   351→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   352→    } catch (error) {
   353→      const msg = (error as Error).message;
   354→      this.logger.error(`MetaOrchestrator execution failed for ${action}: ${msg}`);
   355→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   356→    }
   357→  }

   367→  private async orchestrateTask(params: {
   368→    taskDescription: string;
   369→    priority?: string;
   370→    constraints?: Record<string, any>;
   371→  }): Promise<{
   372→    orchestrationId: string;
   373→    steps: Array<{ order: number; agentId: string; description: string }>;
   374→    assignedAgents: string[];
   375→    estimatedDurationMs: number;
   376→  }> {
   377→    const { taskDescription, priority = 'normal', constraints = {} } = params;
   378→
   379→    if (!taskDescription || typeof taskDescription !== 'string') {
   380→      throw new Error('Valid taskDescription string is required for orchestration');
   381→    }
   382→
   383→    const orchestrationId = this.generateId();
   384→
   385→    // Decompose the task into logical steps
   386→    const steps = this.decomposeTask(taskDescription);
   387→    const assignedAgents = steps.map((s) => s.agentId);
   388→    const estimatedDurationMs = steps.reduce((sum, s) => sum + s.estimatedDurationMs, 0);
   389→
   390→    const record: OrchestrationRecord = {
   391→      id: orchestrationId,
   392→      taskDescription,
   393→      status: 'planning',
   394→      steps: steps.map((s, i) => ({
   395→        id: `${orchestrationId}-step-${i}`,
   396→        order: i,
   397→        agentId: s.agentId,
   398→        description: s.description,
   399→        status: 'pending' as const,
   400→      })),
   401→      createdAt: new Date(),
   402→      updatedAt: new Date(),
   403→      estimatedDurationMs,
   404→    };
   405→
   406→    this.orchestrations.set(orchestrationId, record);
   407→
   408→    this.logger.log(
   409→      `Orchestration created: id=${orchestrationId}, steps=${steps.length}, agents=${assignedAgents.length}`,
   410→    );
   411→
   412→    return {
   413→      orchestrationId,
   414→      steps: steps.map((s) => ({ order: s.order, agentId: s.agentId, description: s.description })),
   415→      assignedAgents: [...new Set(assignedAgents)],
   416→      estimatedDurationMs,
   417→    };
   418→  }

   461→  private async monitorProgress(params: {
   462→    orchestrationId: string;
   463→    includeDetails?: boolean;
   464→  }): Promise<{
   465→    orchestrationId: string;
   466→    overallProgress: number;
   467→    stepStatuses: Array<{ stepId: string; order: number; status: string; progress: number }>;
   468→    bottlenecks: string[];
   469→  }> {
   470→    const { orchestrationId, includeDetails = false } = params;
   471→
   472→    if (!orchestrationId || typeof orchestrationId !== 'string') {
   473→      throw new Error('Valid orchestrationId string is required');
   474→    }
   475→
   476→    const record = this.orchestrations.get(orchestrationId);
   477→    if (!record) {
   478→      throw new Error(`Orchestration not found: ${orchestrationId}`);
   479→    }
   480→
   481→    // Simulate progress based on time elapsed
   482→    const elapsedMs = Date.now() - record.createdAt.getTime();
   483→    const totalSteps = record.steps.length;
   484→    const completedSteps = record.steps.filter((s) => s.status === 'completed').length;
   485→
   486→    // Simulate some steps progressing
   487→    const progressPerStep = totalSteps > 0 ? elapsedMs / record.estimatedDurationMs : 0;
   488→    const stepStatuses = record.steps.map((step) => {
   489→      const stepProgress = Math.min(
   490→        100,
   491→        Math.round(((progressPerStep * 100) / totalSteps) * (step.order + 1)),
   492→      );
   493→      return {
   494→        stepId: step.id,
   495→        order: step.order,
   496→        status: step.status,
   497→        progress: stepProgress,
   498→      };
   499→    });
   500→
   501→    const overallProgress =
   502→      totalSteps > 0
   503→        ? Math.min(100, Math.round((completedSteps / totalSteps) * 100 + progressPerStep * 10))
   504→        : 0;
   505→
   506→    const bottlenecks = this.identifyBottlenecks(record);
   507→
   508→    this.logger.log(
   509→      `Progress monitored: id=${orchestrationId}, progress=${overallProgress}%, bottlenecks=${bottlenecks.length}`,
   510→    );
   511→
   512→    return {
   513→      orchestrationId,
   514→      overallProgress,
   515→      stepStatuses,
   516→      bottlenecks,
   517→    };
   518→  }

   520→  private async rebalanceWorkload(params: { strategy?: string; agentIds?: string[] }): Promise<{
   521→    rebalanced: boolean;
   522→    redistributions: Array<{ from: string; to: string; taskCount: number }>;
   523→    loadBefore: Record<string, number>;
   524→    loadAfter: Record<string, number>;
   525→  }> {
   526→    const { strategy = 'even', agentIds = [] } = params;
   527→
   528→    const targetAgents =
   529→      agentIds.length > 0
   530→        ? agentIds
   531→        : ['meta-planner', 'meta-critic', 'meta-repair', 'meta-learning', 'meta-task-router'];
   532→
   533→    // Simulate current workload
   534→    const loadBefore: Record<string, number> = {};
   535→    for (const agentId of targetAgents) {
   536→      loadBefore[agentId] = Math.floor(Math.random() * 8) + 1;
   537→    }
   538→
   539→    const totalLoad = Object.values(loadBefore).reduce((sum, l) => sum + l, 0);
   540→    const targetLoad = Math.round(totalLoad / targetAgents.length);
   541→
   542→    const loadAfter: Record<string, number> = {};
   543→    const redistributions: Array<{ from: string; to: string; taskCount: number }> = [];
   544→
   545→    if (strategy === 'even') {
   546→      // Distribute load evenly
   547→      let remaining = totalLoad;
   548→      for (let i = 0; i < targetAgents.length; i++) {
   549→        const isLast = i === targetAgents.length - 1;
   550→        loadAfter[targetAgents[i]] = isLast ? remaining : targetLoad;
   551→        remaining -= loadAfter[targetAgents[i]];
   552→      }
   553→
   554→      // Calculate redistributions
   555→      for (const agentId of targetAgents) {
   556→        const diff = loadBefore[agentId] - (loadAfter[agentId] || 0);
   557→        if (diff > 0) {
   558→          const underloaded = targetAgents.find(
   559→            (a) => (loadBefore[a] || 0) < (loadAfter[a] || 0) && a !== agentId,
   560→          );
   561→          if (underloaded) {
   562→            redistributions.push({ from: agentId, to: underloaded, taskCount: diff });
   563→          }
   564→        }
   565→      }
   566→    } else if (strategy === 'priority-based') {
   567→      // High-priority agents get fewer tasks
   568→      for (const agentId of targetAgents) {
   569→        const isHighPriority = agentId.includes('judge') || agentId.includes('governance');
   570→        loadAfter[agentId] = isHighPriority
   571→          ? Math.max(1, Math.floor(targetLoad * 0.6))
   572→          : Math.ceil(targetLoad * 1.2);
   573→      }
   574→    } else {
   575→      // Capability-based: assign based on agent type
   576→      for (const agentId of targetAgents) {
   577→        const isSpecialized = agentId.includes('repair') || agentId.includes('critic');
   578→        loadAfter[agentId] = isSpecialized ? Math.max(1, Math.floor(targetLoad * 0.8)) : targetLoad;
   579→      }
   580→    }
   581→
   582→    this.logger.log(
   583→      `Workload rebalanced: strategy=${strategy}, agents=${targetAgents.length}, redistributions=${redistributions.length}`,
   584→    );
   585→
   586→    return { rebalanced: true, redistributions, loadBefore, loadAfter };
   587→  }

   591→    availableAgents?: string[];

   594→  }): Promise<{
   595→    planId: string;
   596→    phases: Array<{

   598→      steps: Array<{ agentId: string; action: string; description: string }>;
   599→      estimatedDurationMs: number;
   600→    }>;
   601→    dependencies: Array<{ from: string; to: string; type: string }>;
   602→    estimatedDurationMs: number;
   603→    riskAssessment: { level: string; factors: string[]; mitigations: string[] };
   604→  }> {
   605→    const {
   606→      workflowDescription,
   607→      availableAgents = [
   608→        'meta-planner',
   609→        'meta-critic',
   610→        'meta-repair',

   612→        'meta-learning',

   614→        'meta-knowledge-synthesis',

   618→    } = params;
   619→
   620→    if (!workflowDescription || typeof workflowDescription !== 'string') {
   621→      throw new Error('Valid workflowDescription string is required');

   826→  private decomposeTask(description: string): Array<{
   827→    order: number;
   828→    agentId: string;
   829→    description: string;
   830→    estimatedDurationMs: number;
   831→  }> {
   832→    const words = description.split(/\s+/);
   833→    const complexity = Math.min(words.length / 10, 5);
   834→
   835→    const steps = [
   836→      {
   837→        order: 0,
   838→        agentId: 'meta-planner',
   839→        description: `Plan approach for: ${description.substring(0, 100)}`,
   840→        estimatedDurationMs: 5000 + complexity * 2000,
   841→      },
   842→      {
   843→        order: 1,
   844→        agentId: 'meta-task-router',
   845→        description: `Route subtasks for: ${description.substring(0, 100)}`,
   846→        estimatedDurationMs: 3000 + complexity * 1000,
   847→      },
   848→      {
   849→        order: 2,
   850→        agentId: 'meta-critic',
   851→        description: `Evaluate intermediate results for: ${description.substring(0, 80)}`,
   852→        estimatedDurationMs: 4000 + complexity * 1500,
   853→      },
   854→      {
   855→        order: 3,
   856→        agentId: 'meta-judge',
   857→        description: `Final decision on: ${description.substring(0, 80)}`,
   858→        estimatedDurationMs: 3000 + complexity * 1000,
   859→      },
   860→    ];
   861→
   862→    if (complexity > 2) {
   863→      steps.splice(2, 0, {
   864→        order: 2,
   865→        agentId: 'meta-repair',
   866→        description: `Repair any issues found during execution`,
   867→        estimatedDurationMs: 5000 + complexity * 2000,
   868→      });
   869→      // Re-number
   870→      steps.forEach((s, i) => {
   871→        s.order = i;
   872→      });
   873→    }
   874→
   875→    return steps;
   876→  }

   878→  private identifyBottlenecks(record: OrchestrationRecord): string[] {
   879→    const bottlenecks: string[] = [];
   880→    const runningSteps = record.steps.filter((s) => s.status === 'running');
   881→    const pendingSteps = record.steps.filter((s) => s.status === 'pending');
   882→
   883→    if (runningSteps.length > 3) {
   884→      bottlenecks.push('Too many concurrent steps may be causing resource contention');
   885→    }
   886→
   887→    const longRunningSteps = runningSteps.filter((s) => {
   888→      if (!s.assignedAt) return false;
   889→      return Date.now() - s.assignedAt.getTime() > 60000;
   890→    });
   891→
   892→    for (const step of longRunningSteps) {
   893→      bottlenecks.push(
   894→        `Step ${step.order} (${step.agentId}) has been running for an extended period`,
   895→      );
   896→    }
   897→
   898→    if (pendingSteps.length > runningSteps.length && runningSteps.length > 0) {
   899→      bottlenecks.push('Pending steps outnumber running steps; consider parallelization');
   900→    }
   901→
   902→    return bottlenecks;
   903→  }

   905→  private assessCompletionScore(record: OrchestrationRecord): number {
   906→    const totalSteps = record.steps.length;
   907→    if (totalSteps === 0) return 50;
   908→    const completedSteps = record.steps.filter((s) => s.status === 'completed').length;
   909→    return Math.round((completedSteps / totalSteps) * 100);
   910→  }

   912→  private extractLessons(
   913→    criteriaResults: Array<{ criterion: string; score: number; passed: boolean }>,
   914→    _record?: OrchestrationRecord,
   915→  ): string[] {
   916→    const lessons: string[] = [];
   917→
   918→    const failedCriteria = criteriaResults.filter((c) => !c.passed || c.score < 60);
   919→    if (failedCriteria.length > 0) {
   920→      lessons.push(
   921→        `Areas needing improvement: ${failedCriteria.map((c) => c.criterion).join(', ')}`,
   922→      );
   923→    }
   924→
   925→    const highScores = criteriaResults.filter((c) => c.score >= 85);
   926→    if (highScores.length > 0) {
   927→      lessons.push(`Strengths to maintain: ${highScores.map((c) => c.criterion).join(', ')}`);
   928→    }
   929→
   930→    lessons.push('Regular monitoring during execution helps identify issues early');
   931→    lessons.push('Clear objective definitions improve outcome evaluation accuracy');
   932→
   933→    return lessons;
   934→  }

   936→  private generateRecommendations(
   937→    criteriaResults: Array<{ criterion: string; score: number }>,
   938→    overallScore: number,
   939→  ): string[] {
   940→    const recommendations: string[] = [];
   941→
   942→    if (overallScore < 60) {
   943→      recommendations.push('Consider revising the orchestration strategy for better outcomes');
   944→    }
   945→
   946→    const weakCriteria = criteriaResults.filter((c) => c.score < 70);
   947→    for (const criterion of weakCriteria) {
   948→      recommendations.push(
   949→        `Improve ${criterion.criterion} by adjusting agent selection and task allocation`,
   950→      );
   951→    }
   952→
   953→    if (overallScore >= 80) {
   954→      recommendations.push(
   955→        'Current orchestration approach is effective; consider using it as a template',
   956→      );
   957→    }
   958→
   959→    recommendations.push('Document successful patterns for reuse in future orchestrations');
   960→
   961→    return recommendations;
   962→  }
