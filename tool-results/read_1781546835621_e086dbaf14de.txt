   271→            success: true,
   272→            data: {
   273→              action,
   274→              scenario,
   275→              model: model as
   276→                | 'monte_carlo'
   277→                | 'discrete_event'
   278→                | 'agent_based'
   279→                | 'system_dynamics'
   280→                | 'stochastic',

   286→              }>,

   288→              iterations,

   308→                  ? ([] as Array<{
   309→                      pathId: number;
   310→                      probability: number;

   368→          const includeDuals = config.includeDuals || false;

   372→            return {
   373→              success: false,

   379→          this.logger.log(
   380→            `Optimizing: ${direction} "${objective}" (method: ${method}, solver: ${solver})`,

   382→
   383→          return {
   384→            success: true,
   385→            data: {
   386→              action,
   387→              objective,
   388→              direction: direction as 'maximize' | 'minimize',
   389→              variables: variables as Array<{
   390→                name: string;
   391→                lowerBound: number;
   392→                upperBound: number;
   393→                type: 'continuous' | 'integer' | 'binary';
   394→                currentValue: number;
   395→              }>,
   396→              constraints: constraints as Array<{
   397→                name: string;
   398→                expression: string;
   399→                operator: 'lte' | 'gte' | 'eq';
   400→                rhs: number;
   401→                slack: number;
   402→              }>,
   403→              method: method as 'linear' | 'nonlinear' | 'integer' | 'quadratic' | 'stochastic' | 'multi_objective',
   404→              solver,

   406→              maxIterations,
   407→              includeRelaxation,
   408→              includeDuals,
   409→              includePareto,
   410→              optimization: {
   411→                status: 'optimal' as 'optimal' | 'suboptimal' | 'infeasible' | 'unbounded' | 'time_limit',

   418→              sensitivity: {

   440→              duals: includeDuals

   446→              paretoFront: includePareto

   448→                    solutions: [] as Array<{
   449→                      objectives: Record<string, number>;
   450→                      variables: Record<string, number>;

   452→                    idealPoint: {} as Record<string, number>,

   454→                  }
   455→                : undefined,
   456→              recommendations: [] as Array<{

   460→              }>,
   461→              status: 'optimization_complete',

   464→            metadata: { duration: Date.now() - startTime },

   466→        }

   471→          const historicalData = config.historicalData || [];
   472→          const horizon = config.horizon || 12;
   473→          const frequency = config.frequency || 'monthly';
