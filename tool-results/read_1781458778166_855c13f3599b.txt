    27→// ─── Status Badge ────────────────────────────────────────────────
    28→function StatusBadge({ status }: { status: string }) {
    29→  const config: Record<string, { color: string; dot: string; pulse: boolean }> = {
    30→    idle: { color: 'bg-slate-500/20 text-slate-400 border-slate-500/30', dot: 'bg-slate-400', pulse: false },
    31→    thinking: { color: 'bg-violet-500/20 text-violet-400 border-violet-500/30', dot: 'bg-violet-400', pulse: true },
    32→    executing: { color: 'bg-amber-500/20 text-amber-400 border-amber-500/30', dot: 'bg-amber-400', pulse: true },
    33→    waiting: { color: 'bg-slate-500/20 text-slate-400 border-slate-500/30', dot: 'bg-slate-400', pulse: false },
    34→    collaborating: { color: 'bg-pink-500/20 text-pink-400 border-pink-500/30', dot: 'bg-pink-400', pulse: true },
    35→    completed: { color: 'bg-emerald-500/20 text-emerald-400 border-emerald-500/30', dot: 'bg-emerald-400', pulse: false },
    36→    error: { color: 'bg-red-500/20 text-red-400 border-red-500/30', dot: 'bg-red-400', pulse: false },
    37→  };
    38→  const c = config[status] || config.idle;
    39→
    40→  return (
    41→    <span className={cn('inline-flex items-center gap-1.5 rounded-full border px-2 py-0.5 text-[10px] font-semibold uppercase', c.color)}>
    42→      <div className={cn('h-1.5 w-1.5 rounded-full', c.dot, c.pulse && 'animate-pulse')} />
    43→      {status}
    44→    </span>
    45→  );
    46→}

   108→// ─── Agent Card (Left Panel) ─────────────────────────────────────
   109→function AgentCard({ session, isSelected, onClick }: { session: AgentSession; isSelected: boolean; onClick: () => void }) {
   110→  const isActive = session.status !== 'idle' && session.status !== 'completed';
   111→  const clusterEmojis: Record<string, string> = {
   112→    browser: '🌐', computer: '💻', coding: '🔧', office: '📋',
   113→    marketing: '📢', business: '💼', infrastructure: '🏗️',
   114→    security: '🔒', 'meta-intelligence': '🧠', 'llm-intelligence': '🤖',
   115→    'intelligent-orchestration': '🎛️', watchdog: '👁️',
   116→    'self-evolution': '🧬', certification: '✅',
   117→  };
   118→
   119→  return (
   120→    <button
   121→      onClick={onClick}
   122→      className={cn(
   123→        'w-full text-left rounded-xl border p-3 transition-all duration-200',
   124→        isSelected
   125→          ? 'border-primary/50 bg-primary/10 shadow-lg shadow-primary/5'
   126→          : 'border-border bg-card hover:border-primary/30',
   127→        isActive && !isSelected && 'border-amber-500/30 bg-amber-500/5'
   128→      )}
   129→    >
   130→      <div className="flex items-center justify-between">
   131→        <div className="flex items-center gap-2 min-w-0">
   132→          <span className="text-base">{clusterEmojis[session.cluster] || '🤖'}</span>
   133→          <div className="min-w-0">
   134→            <p className="text-xs font-semibold text-foreground truncate">{session.agentName}</p>
   135→            <p className="text-[10px] text-muted-foreground capitalize">{session.cluster.replace(/-/g, ' ')}</p>
   136→          </div>
   137→        </div>
   138→        <StatusBadge status={session.status} />
   139→      </div>
   140→
   141→      {/* Progress bar for active agents */}
   142→      {isActive && session.progress > 0 && (
   143→        <div className="mt-2">
   144→          <div className="h-1 rounded-full bg-border overflow-hidden">
   145→            <div
   146→              className={cn(
   147→                'h-full rounded-full transition-all duration-500',
   148→                session.status === 'thinking' ? 'bg-violet-400' :
   149→                session.status === 'executing' ? 'bg-amber-400' :
   150→                session.status === 'collaborating' ? 'bg-pink-400' : 'bg-primary'
   151→              )}
   152→              style={{ width: `${session.progress}%` }}
   153→            />
   154→          </div>
   155→        </div>
   156→      )}
   157→
   158→      {/* Current step */}
   159→      {session.currentStep && (
   160→        <div className="mt-1.5 flex items-center gap-1.5">
   161→          {session.status === 'thinking' ? (
   162→            <Brain className="h-3 w-3 text-violet-400 animate-pulse" />
   163→          ) : session.status === 'executing' ? (
   164→            <Loader2 className="h-3 w-3 text-amber-400 animate-spin" />
   165→          ) : (
   166→            <Activity className="h-3 w-3 text-pink-400" />
   167→          )}
   168→          <p className="text-[10px] text-muted-foreground truncate">{session.currentStep}</p>
   169→        </div>
   170→      )}
   171→
   172→      {/* Metrics */}
   173→      {(session.tokensUsed || session.toolCalls || session.llmCalls) && (
   174→        <div className="mt-2 flex items-center gap-3 text-[10px] text-muted-foreground">
   175→          {session.tokensUsed && <span><Hash className="inline h-3 w-3 mr-0.5" />{session.tokensUsed} tokens</span>}
   176→          {session.llmCalls && <span><Brain className="inline h-3 w-3 mr-0.5" />{session.llmCalls} LLM</span>}
   177→          {session.toolCalls && <span><Wrench className="inline h-3 w-3 mr-0.5" />{session.toolCalls} tools</span>}
   178→        </div>
   179→      )}
   180→    </button>
   181→  );
   182→}
