   276→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   277→    const startTime = Date.now();
   278→    // Bridge: use LLM for quality evaluation, issue identification, and improvement suggestions
   279→    if (this.bridge) {
   280→      try {
   281→        const llmResult = await this.bridge.callLLM({
   282→          systemPrompt: `You are the ${this.config.name} agent in the Meta-Intelligence cluster. Analyze the following task and provide detailed quality evaluation, issue identification, and improvement suggestions.`,
   283→          userPrompt: JSON.stringify(input.payload),
   284→          temperature: 0.3,
   285→          maxTokens: 2048,
   286→        });
   287→
   288→        const analysis = llmResult.content;
   289→
   290→        return this.createAgentOutput(
   291→          input.taskId,
   292→          true,
   293→          { analysis, costUsd: llmResult.costUsd, tokensUsed: llmResult.tokenCount },
   294→          undefined,
   295→          startTime,
   296→        );
   297→      } catch (error) {
   298→        this.logger.warn(`Bridge LLM failed, fallback: ${(error as Error).message}`);
   299→      }
   300→    }
   301→
   302→const { action, ...params } = input.payload;
   303→
   304→    if (!action) {
   305→      return this.createAgentOutput(
   306→        input.taskId,
   307→        false,
   308→        null,
   309→        'Missing required parameter: action',
   310→        startTime,
   311→      );
   312→    }
   313→
   314→    const supportedActions = [
   315→      'evaluateOutput',
   316→      'scoreQuality',
   317→      'identifyIssues',
   318→      'suggestImprovements',
   319→      'compareOutputs',
   320→      'validateConsistency',
   321→    ];
   322→
   323→    if (!supportedActions.includes(action)) {
   324→      return this.createAgentOutput(
   325→        input.taskId,
   326→        false,
   327→        null,
   328→        `Unknown critic action: ${action}. Supported: ${supportedActions.join(', ')}`,
   329→        startTime,
   330→      );
   331→    }
   332→
   333→    try {
   334→      const tool = this.getTool(action);
   335→      if (!tool) {
   336→        return this.createAgentOutput(
   337→          input.taskId,
   338→          false,
   339→          null,
   340→          `Tool not found: ${action}`,
   341→          startTime,
   342→        );
   343→      }
   344→
   345→      const result = await tool.execute(params);
   346→      await this.storeInWorkingMemory(
   347→        `critic:last:${action}`,
   348→        { params, result, timestamp: new Date() },
   349→        300000,
   350→      );
   351→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   352→    } catch (error) {
   353→      const msg = (error as Error).message;
   354→      this.logger.error(`MetaCritic execution failed for ${action}: ${msg}`);
   355→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   356→    }
   357→  }

   366→  private async evaluateOutput(params: {
   367→    output: any;
   368→    criteria?: string[];
   369→    context?: Record<string, any>;
   370→  }): Promise<{
   371→    overallScore: number;
   372→    criteriaScores: Record<string, number>;
   373→    passed: boolean;
   374→    summary: string;
   375→  }> {
   376→    const {
   377→      output,
   378→      criteria = ['completeness', 'accuracy', 'relevance', 'clarity'],
   379→      context = {},
   380→    } = params;
   381→
   382→    if (output === null || output === undefined) {
   383→      throw new Error('Output cannot be null or undefined for evaluation');
   384→    }
   385→
   386→    const criteriaScores: Record<string, number> = {};
   387→
   388→    for (const criterion of criteria) {
   389→      criteriaScores[criterion] = this.evaluateCriterion(output, criterion, context);
   390→    }
   391→
   392→    const overallScore = Math.round(
   393→      Object.values(criteriaScores).reduce((sum, s) => sum + s, 0) / criteria.length,
   394→    );
   395→
   396→    const passed = overallScore >= 60;
   397→
   398→    const summary = this.generateEvaluationSummary(criteriaScores, overallScore, passed);
   399→
   400→    const outputId = this.generateId();
   401→    this.evaluationHistory.push({ outputId, score: overallScore, timestamp: new Date() });
   402→
   403→    this.logger.log(
   404→      `Output evaluated: score=${overallScore}, passed=${passed}, criteria=${criteria.length}`,
   405→    );
   406→
   407→    return { overallScore, criteriaScores, passed, summary };
   408→  }

   410→  private async scoreQuality(params: { content: any; dimensions?: string[] }): Promise<{
   411→    scores: Record<string, number>;
   412→    overallScore: number;
   413→    grade: string;
   414→  }> {
   415→    const {
   416→      content,
   417→      dimensions = ['correctness', 'completeness', 'consistency', 'efficiency', 'readability'],
   418→    } = params;
   419→
   420→    if (content === null || content === undefined) {
   421→      throw new Error('Content cannot be null or undefined for quality scoring');
   422→    }
   423→
   424→    const scores: Record<string, number> = {};
   425→
   426→    for (const dim of dimensions) {
   427→      scores[dim] = this.scoreDimension(content, dim);
   428→    }
   429→
   430→    const overallScore = Math.round(
   431→      Object.values(scores).reduce((sum, s) => sum + s, 0) / dimensions.length,
   432→    );
   433→
   434→    const grade = this.scoreToGrade(overallScore);
   435→
   436→    this.logger.log(
   437→      `Quality scored: overall=${overallScore}, grade=${grade}, dimensions=${dimensions.length}`,
   438→    );
   439→
   440→    return { scores, overallScore, grade };
   441→  }

   443→  private async identifyIssues(params: {
   444→    output: any;
   445→    expectedFormat?: Record<string, any>;
   446→    severity?: string;
   447→  }): Promise<{
   448→    issues: Issue[];
   449→    issueCount: number;
   450→    criticalCount: number;
   451→  }> {
   452→    const { output, expectedFormat, severity = 'all' } = params;
   453→
   454→    if (output === null || output === undefined) {
   455→      throw new Error('Output cannot be null or undefined for issue identification');
   456→    }
   457→
   458→    const issues: Issue[] = [];
   459→
   460→    // Check for null/undefined fields
   461→    if (typeof output === 'object') {
   462→      for (const [key, value] of Object.entries(output)) {
   463→        if (value === null || value === undefined) {
   464→          issues.push({
   465→            id: this.generateId(),
   466→            severity: 'warning',
   467→            category: 'missing-field',
   468→            message: `Field "${key}" is null or undefined`,
   469→            location: key,
   470→            recommendation: `Provide a value for field "${key}"`,
   471→          });
   472→        }
   473→      }
   474→    }
   475→
   476→    // Check against expected format
   477→    if (expectedFormat) {
   478→      for (const [key, type] of Object.entries(expectedFormat)) {
   479→        if (!(key in (output || {}))) {
   480→          issues.push({
   481→            id: this.generateId(),
   482→            severity: 'critical',
   483→            category: 'missing-field',
   484→            message: `Required field "${key}" is missing`,
   485→            location: key,
   486→            recommendation: `Add the required field "${key}" of type ${type}`,
   487→          });
   488→        } else if (typeof output[key] !== type) {
   489→          issues.push({
   490→            id: this.generateId(),
   491→            severity: 'warning',
   492→            category: 'type-mismatch',
   493→            message: `Field "${key}" expected type ${type}, got ${typeof output[key]}`,
   494→            location: key,
   495→            recommendation: `Convert field "${key}" to type ${type}`,
   496→          });
   497→        }
   498→      }
   499→    }
   500→
   501→    // Check for empty strings
   502→    if (typeof output === 'string' && output.trim().length === 0) {
   503→      issues.push({
   504→        id: this.generateId(),
   505→        severity: 'critical',
   506→        category: 'empty-output',
   507→        message: 'Output is an empty string',
   508→        recommendation: 'Provide non-empty content',
   509→      });
   510→    }
   511→
   512→    // Check for error indicators
   513→    if (typeof output === 'object' && output.error) {
   514→      issues.push({
   515→        id: this.generateId(),
   516→        severity: 'critical',
   517→        category: 'error-indicator',
   518→        message: `Output contains error: ${output.error}`,
   519→        location: 'error',
   520→        recommendation: 'Investigate and resolve the error before using this output',
   521→      });
   522→    }
   523→
   524→    // Filter by severity
   525→    const filteredIssues =
   526→      severity === 'all'
   527→        ? issues
   528→        : issues.filter((i) => {
   529→            if (severity === 'critical') return i.severity === 'critical';
   530→            if (severity === 'warning')
   531→              return i.severity === 'critical' || i.severity === 'warning';
   532→            return true;
   533→          });
   534→
   535→    const criticalCount = filteredIssues.filter((i) => i.severity === 'critical').length;
   536→
   537→    this.logger.log(`Issues identified: total=${filteredIssues.length}, critical=${criticalCount}`);
   538→
   539→    return { issues: filteredIssues, issueCount: filteredIssues.length, criticalCount };
   540→  }

   542→  private async suggestImprovements(params: {
   543→    output: any;
   544→    goals?: string[];
   545→    constraints?: Record<string, any>;
   546→  }): Promise<{
   547→    suggestions: Suggestion[];
   548→    priorityOrder: string[];
   549→    estimatedImpact: string;
   550→  }> {
   551→    const { output, goals = ['quality', 'completeness'], constraints = {} } = params;
   552→
   553→    if (output === null || output === undefined) {
   554→      throw new Error('Output cannot be null or undefined for improvement suggestions');
   555→    }
   556→
   557→    const suggestions: Suggestion[] = [];
   558→
   559→    // Analyze output and generate suggestions
   560→    if (typeof output === 'string') {
   561→      if (output.length < 100) {
   562→        suggestions.push({
   563→          id: this.generateId(),
   564→          category: 'content-depth',
   565→          description: 'Output is brief; consider adding more detail and context',
   566→          priority: 'medium',
   567→          estimatedImpact: 'moderate',
   568→          implementation: 'Expand on key points with supporting evidence or examples',
   569→        });
   570→      }
   571→      if (!output.includes('.')) {
   572→        suggestions.push({
   573→          id: this.generateId(),
   574→          category: 'completeness',
   575→          description: 'Output appears to be incomplete (no sentence termination)',
   576→          priority: 'high',
   577→          estimatedImpact: 'significant',
   578→          implementation: 'Complete the thought with proper sentence structure',
   579→        });
   580→      }
   581→    }
   582→
   583→    if (typeof output === 'object') {
   584→      const keys = Object.keys(output || {});
   585→      if (keys.length < 3) {
   586→        suggestions.push({
   587→          id: this.generateId(),
   588→          category: 'completeness',
   589→          description: 'Output object has few fields; consider adding more structured data',
   590→          priority: 'medium',
   591→          estimatedImpact: 'moderate',
   592→          implementation: 'Add additional relevant fields to provide comprehensive information',
   593→        });
   594→      }
   595→
   596→      if (!output.success && output.success !== undefined) {
   597→        suggestions.push({
   598→          id: this.generateId(),
   599→          category: 'error-handling',
   600→          description: 'Output indicates failure; improve error handling and recovery',
   601→          priority: 'high',
   602→          estimatedImpact: 'significant',
   603→          implementation: 'Add fallback logic and graceful degradation paths',
   604→        });
   605→      }
   606→    }
   607→
   608→    // Goal-based suggestions
   609→    if (goals.includes('quality')) {
   610→      suggestions.push({
   611→        id: this.generateId(),
   612→        category: 'quality',
   613→        description: 'Add validation and verification steps to ensure output quality',
   614→        priority: 'medium',
   615→        estimatedImpact: 'moderate',
   616→        implementation: 'Implement schema validation and cross-check mechanisms',
   617→      });
   618→    }
   619→
   620→    if (goals.includes('completeness')) {
   621→      suggestions.push({
   622→        id: this.generateId(),
   623→        category: 'completeness',
   624→        description: 'Ensure all required fields and edge cases are covered',
   625→        priority: 'medium',
   626→        estimatedImpact: 'moderate',
   627→        implementation: 'Add checklist verification for all required output elements',
   628→      });
   629→    }
   630→
   631→    // Sort by priority
   632→    const priorityOrder = suggestions
   633→      .sort((a, b) => {
   634→        const priorityVal = { high: 3, medium: 2, low: 1 };
   635→        return (priorityVal[b.priority] || 0) - (priorityVal[a.priority] || 0);
   636→      })
   637→      .map((s) => s.id);
   638→
   639→    const estimatedImpact = suggestions.some((s) => s.priority === 'high')
   640→      ? 'significant'
   641→      : suggestions.length > 0
   642→        ? 'moderate'
   643→        : 'minimal';
   644→
   645→    this.logger.log(
   646→      `Improvements suggested: count=${suggestions.length}, impact=${estimatedImpact}`,
   647→    );
   648→
   649→    return { suggestions, priorityOrder, estimatedImpact };
   650→  }

   656→    rankings: Array<{ id: string; score: number; rank: number }>;
   657→    bestOutput: string;
   658→    comparison: Record<string, Record<string, number>>;
   659→  }> {
   660→    const { outputs, criteria = ['quality', 'completeness', 'accuracy', 'efficiency'] } = params;
   661→
   662→    if (!outputs || !Array.isArray(outputs) || outputs.length < 2) {
   663→      throw new Error('At least two outputs are required for comparison');
   664→    }
   665→
   666→    const comparison: Record<string, Record<string, number>> = {};
   667→    const scored: Array<{ id: string; score: number }> = [];
   668→
   669→    for (const output of outputs) {
   670→      const criteriaScores: Record<string, number> = {};
   671→
   672→      for (const criterion of criteria) {
   673→        criteriaScores[criterion] = this.evaluateCriterion(output.content, criterion, {});
   674→      }
   675→
   676→      comparison[output.id] = criteriaScores;
   677→      const avgScore =
   678→        Object.values(criteriaScores).reduce((sum, s) => sum + s, 0) / criteria.length;
   679→      scored.push({ id: output.id, score: Math.round(avgScore) });
   680→    }
   681→
   682→    scored.sort((a, b) => b.score - a.score);
   683→
   684→    const rankings = scored.map((s, i) => ({
   685→      id: s.id,
   686→      score: s.score,
   687→      rank: i + 1,
   688→    }));
   689→
   690→    const bestOutput = rankings[0].id;
   691→
   692→    this.logger.log(`Outputs compared: count=${outputs.length}, best=${bestOutput}`);
   693→
   694→    return { rankings, bestOutput, comparison };
   695→  }

   697→  private async validateConsistency(params: {
   698→    outputs: any[];
   699→    rules?: Array<{ field: string; type: string }>;
   700→  }): Promise<{
   701→    consistent: boolean;
   702→    inconsistencies: Array<{
   703→      outputIndex: number;
   704→      field: string;
   705→      issue: string;
   706→      expected: string;
   707→      actual: string;
   708→    }>;
   709→    consistencyScore: number;
   710→  }> {
   711→    const { outputs, rules = [] } = params;
   712→
   713→    if (!outputs || !Array.isArray(outputs) || outputs.length === 0) {
   714→      throw new Error('Non-empty outputs array is required for consistency validation');
   715→    }
   716→
   717→    const inconsistencies: Array<{
   718→      outputIndex: number;
   719→      field: string;
   720→      issue: string;
   721→      expected: string;
   722→      actual: string;
   723→    }> = [];
   724→
   725→    // Validate against rules
   726→    for (let i = 0; i < outputs.length; i++) {
   727→      const output = outputs[i];
   728→
   729→      for (const rule of rules) {
   730→        const value = output?.[rule.field];
   731→        if (rule.type === 'required' && (value === null || value === undefined)) {
   732→          inconsistencies.push({
   733→            outputIndex: i,
   734→            field: rule.field,
   735→            issue: 'Missing required field',
   736→            expected: rule.type,
   737→            actual: 'undefined',
   738→          });
   739→        } else if (rule.type === 'string' && typeof value !== 'string') {
   740→          inconsistencies.push({
   741→            outputIndex: i,
   742→            field: rule.field,
   743→            issue: 'Type mismatch',
   744→            expected: 'string',
   745→            actual: typeof value,
   746→          });
   747→        } else if (rule.type === 'number' && typeof value !== 'number') {
   748→          inconsistencies.push({
   749→            outputIndex: i,
   750→            field: rule.field,
   751→            issue: 'Type mismatch',
   752→            expected: 'number',
   753→            actual: typeof value,
   754→          });
   755→        }
   756→      }
   757→    }
   758→
   759→    // Check cross-output consistency
   760→    if (outputs.length > 1) {
   761→      const firstKeys = typeof outputs[0] === 'object' ? Object.keys(outputs[0] || {}).sort() : [];
   762→      for (let i = 1; i < outputs.length; i++) {
   763→        const currentKeys =
   764→          typeof outputs[i] === 'object' ? Object.keys(outputs[i] || {}).sort() : [];
   765→        if (JSON.stringify(firstKeys) !== JSON.stringify(currentKeys)) {
   766→          inconsistencies.push({
   767→            outputIndex: i,
   768→            field: 'structure',
   769→            issue: 'Structural inconsistency with first output',
   770→            expected: firstKeys.join(','),
   771→            actual: currentKeys.join(','),
   772→          });
   773→        }
   774→      }
   775→    }
   776→
   777→    const consistent = inconsistencies.length === 0;
   778→    const consistencyScore =
   779→      outputs.length > 0
   780→        ? Math.round(Math.max(0, 100 - (inconsistencies.length / outputs.length) * 50))
   781→        : 100;
   782→
   783→    this.logger.log(
   784→      `Consistency validated: outputs=${outputs.length}, consistent=${consistent}, score=${consistencyScore}`,
   785→    );
   786→
   787→    return { consistent, inconsistencies, consistencyScore };
   788→  }

   792→  private evaluateCriterion(output: any, criterion: string, context: Record<string, any>): number {
   793→    switch (criterion) {
   794→      case 'completeness': {
   795→        if (typeof output === 'object' && output !== null) {
   796→          const keys = Object.keys(output);
   797→          const nullValues = keys.filter((k) => output[k] === null || output[k] === undefined);
   798→          return keys.length > 0
   799→            ? Math.round(((keys.length - nullValues.length) / keys.length) * 100)
   800→            : 50;
   801→        }
   802→        return typeof output === 'string' && output.length > 0 ? 80 : 40;
   803→      }
   804→      case 'accuracy': {
   805→        if (output?.error) return 30;
   806→        if (output?.success === true) return 90;
   807→        if (output?.success === false) return 40;
   808→        return 65 + Math.floor(Math.random() * 20);
   809→      }
   810→      case 'relevance': {
   811→        const contextKeys = Object.keys(context);
   812→        if (contextKeys.length === 0) return 70 + Math.floor(Math.random() * 15);
   813→        return 60 + Math.floor(Math.random() * 25);
   814→      }
   815→      case 'clarity': {
   816→        if (typeof output === 'string') {
   817→          const avgWordLength =
   818→            output.split(/\s+/).reduce((sum, w) => sum + w.length, 0) /
   819→            (output.split(/\s+/).length || 1);
   820→          return avgWordLength < 8 ? 85 : avgWordLength < 12 ? 70 : 55;
   821→        }
   822→        return 75;
   823→      }
   824→      case 'correctness': {
   825→        if (output?.error) return 30;
   826→        return 70 + Math.floor(Math.random() * 20);
   827→      }
   828→      case 'consistency': {
   829→        return 65 + Math.floor(Math.random() * 25);
   830→      }
   831→      case 'efficiency': {
   832→        return 60 + Math.floor(Math.random() * 30);
   833→      }
   834→      case 'readability': {
   835→        if (typeof output === 'string') {
   836→          const sentences = output.split(/[.!?]+/).filter((s) => s.trim().length > 0);
   837→          const avgLen =
   838→            sentences.length > 0
   839→              ? sentences.reduce((sum, s) => sum + s.split(/\s+/).length, 0) / sentences.length
   840→              : 0;
   841→          return avgLen > 0 && avgLen < 20 ? 85 : avgLen < 30 ? 70 : 55;
   842→        }
   843→        return 75;
   844→      }
   845→      case 'quality': {
   846→        return 65 + Math.floor(Math.random() * 25);
   847→      }
   848→      default:
   849→        return 60 + Math.floor(Math.random() * 20);
   850→    }
   851→  }

   853→  private scoreDimension(content: any, dimension: string): number {
   854→    switch (dimension) {
   855→      case 'correctness':
   856→        return content?.success === true
   857→          ? 90
   858→          : content?.error
   859→            ? 35
   860→            : 65 + Math.floor(Math.random() * 20);
   861→      case 'completeness':
   862→        if (typeof content === 'object' && content !== null) {
   863→          const keys = Object.keys(content);
   864→          const filled = keys.filter((k) => content[k] !== null && content[k] !== undefined);
   865→          return keys.length > 0 ? Math.round((filled.length / keys.length) * 100) : 50;
   866→        }
   867→        return 70;
   868→      case 'consistency':
   869→        return 70 + Math.floor(Math.random() * 20);
   870→      case 'efficiency':
   871→        return 60 + Math.floor(Math.random() * 30);
   872→      case 'readability':
   873→        return 65 + Math.floor(Math.random() * 25);
   874→      default:
   875→        return 60 + Math.floor(Math.random() * 20);
   876→    }
   877→  }

   879→  private scoreToGrade(score: number): string {
   880→    if (score >= 90) return 'A';
   881→    if (score >= 80) return 'B';
   882→    if (score >= 70) return 'C';
   883→    if (score >= 60) return 'D';
   884→    return 'F';
   885→  }

   887→  private generateEvaluationSummary(
   888→    criteriaScores: Record<string, number>,
   889→    overallScore: number,
   890→    passed: boolean,
   891→  ): string {
   892→    const strengths = Object.entries(criteriaScores)
   893→      .filter(([, score]) => score >= 80)
   894→      .map(([criterion]) => criterion);
   895→    const weaknesses = Object.entries(criteriaScores)
   896→      .filter(([, score]) => score < 60)
   897→      .map(([criterion]) => criterion);
   898→
   899→    let summary = `Overall score: ${overallScore}/100. ${passed ? 'PASSED' : 'FAILED'}.`;
   900→
   901→    if (strengths.length > 0) {
   902→      summary += ` Strengths: ${strengths.join(', ')}.`;
   903→    }
   904→    if (weaknesses.length > 0) {
   905→      summary += ` Areas for improvement: ${weaknesses.join(', ')}.`;
   906→    }
   907→
   908→    return summary;
   909→  }
