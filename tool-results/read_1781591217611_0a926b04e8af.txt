    26→  title, value, subtitle, icon: Icon, trend, trendValue, color = 'text-primary',
    27→}: {
    28→  title: string; value: string | number; subtitle?: string;
    29→  icon: React.ElementType; trend?: 'up' | 'down' | 'neutral'; trendValue?: string;

    31→}) {

    33→    <div className="rounded-xl border border-border bg-card p-5 transition-all duration-200 hover:border-primary/30 hover:shadow-lg hover:shadow-primary/5">
    34→      <div className="flex items-start justify-between">

    60→// ─── Circular Progress ───────────────────────────────────────────
    61→function CircularProgress({ value, size = 80, strokeWidth = 6, color = '#3b82f6', label }: {
    62→  value: number; size?: number; strokeWidth?: number; color?: string; label: string;
    63→}) {
    64→  const radius = (size - strokeWidth) / 2;
    65→  const circumference = 2 * Math.PI * radius;
    66→  const offset = circumference - (value / 100) * circumference;
    67→
    68→  return (
    69→    <div className="flex flex-col items-center gap-1">
    70→      <svg width={size} height={size} className="-rotate-90">
    71→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke="#334155" strokeWidth={strokeWidth} />
    72→        <circle cx={size / 2} cy={size / 2} r={radius} fill="none" stroke={color} strokeWidth={strokeWidth}
    73→          strokeDasharray={circumference} strokeDashoffset={offset} strokeLinecap="round"
    74→          className="transition-all duration-1000 ease-out" />
    75→      </svg>
    76→      <p className="text-[10px] font-medium text-muted-foreground">{label}</p>
    77→    </div>
    78→  );
    79→}

   189→// MAIN DASHBOARD PAGE

   191→export default function DashboardPage() {

   200→  const { connected, subscribe, unsubscribe } = useWebSocket();
   201→
   202→  // ─── Fetch data ────────────────────────────────────────────────
   203→  const fetchData = async () => {
   204→    try {
   205→      const results = await Promise.allSettled([
   206→        api.getAgentStats(),
   207→        api.getHealth(),
   208→        api.getAgents({ limit: 100 }),
   209→        api.getEvents({ limit: 20 }),
   210→        api.getMissions({ limit: 10 }),
   211→      ]);
   212→
   213→      if (results[0].status === 'fulfilled') setClusterStats(results[0].value);
   214→
   215→      if (results[1].status === 'fulfilled') setHealth(results[1].value);
   216→
   217→      if (results[2].status === 'fulfilled') setAgents(results[2].value.data || []);
   218→
   219→      if (results[3].status === 'fulfilled') setRecentEvents(results[3].value.data || []);
   220→
   221→      if (results[4].status === 'fulfilled') setRecentMissions(results[4].value.data || []);
   222→    } catch {
   223→    } finally {
   224→      setLoading(false);
   225→      setLastRefresh(new Date());
   226→    }
   227→  };
   228→
   229→  useEffect(() => { fetchData(); }, []); // eslint-disable-line react-hooks/set-state-in-effect
   230→
   231→  // ─── WebSocket updates ─────────────────────────────────────────
   232→  useEffect(() => {
   233→    const handler = () => { fetchData(); };
   234→    subscribe('*', handler);
   235→    return () => { unsubscribe('*', handler); };
   236→  }, [subscribe, unsubscribe]);
   237→
   238→  // ─── Computed values ───────────────────────────────────────────
   239→  const totalAgents = agents.length;
   240→  const activeAgents = agents.filter((a) => a.status === 'running').length;
   241→  const errorAgents = agents.filter((a) => a.status === 'error').length;
   242→  const idleAgents = agents.filter((a) => a.status === 'idle').length;
   243→  const activeMissions = recentMissions.filter(
   244→    (m) => ![MS.DRAFT, MS.COMPLETED, MS.CANCELLED, MS.ARCHIVED, MS.FAILED].includes(m.state)
   245→  ).length;
   246→  const completedMissions = recentMissions.filter((m) => m.state === MS.COMPLETED).length;
   247→  const healthyServices = health ? Object.values(health.info || {}).filter((s: { status: string }) => s.status === 'up').length : 0;
   248→  const totalServices = health ? Object.keys(health.info || {}).length : 0;
   249→  const uptimePct = totalServices > 0 ? Math.round((healthyServices / totalServices) * 100) : 0;
   250→
   251→  // ─── Chart data ────────────────────────────────────────────────
   252→  const agentStatusData = useMemo(() => [
   253→    { name: 'Running', value: activeAgents, color: '#22c55e' },
   254→    { name: 'Idle', value: idleAgents, color: '#94a3b8' },
   255→    { name: 'Error', value: errorAgents, color: '#ef4444' },
   256→    { name: 'Stopped', value: agents.filter(a => a.status === 'stopped').length, color: '#6b7280' },

   260→  // Until one is available, we show only the current snapshot as a flat line

   262→  const timeSeriesData = useMemo(() => {
   263→    const base = new Date();
   264→    base.setMinutes(0, 0, 0);
   265→    return Array.from({ length: 24 }, (_, i) => {
   266→      const hour = new Date(base.getTime() - (23 - i) * 3600000);
   267→      return {
   268→        time: hour.toLocaleTimeString('en', { hour: '2-digit', minute: '2-digit' }),
   269→        agents: i === 23 ? activeAgents : 0,
   270→        tasks: 0,
   271→        events: 0,
   272→      };
   273→    });
   274→  }, [activeAgents]);
   275→
   276→  // NOTE: Performance time-series requires the /performance/overview API.
   277→  // Showing current snapshot until real historical data is available from the backend.
   278→  useEffect(() => {
   279→    async function fetchPerf() {
   280→      try {
   281→        const res = await fetch('/api/v1/performance/overview', { credentials: 'include' });
   282→        if (res.ok) {
   283→          const json = await res.json();
   284→          const report = json.data?.profiling || json.profiling;
   285→          if (report) {
   286→            const base = new Date();
   287→            base.setMinutes(base.getMinutes() - base.getMinutes() % 5, 0, 0);
   288→            const memUtil = parseInt(report.memory?.heapUtilization || '0');
   289→            const cpuUtil = parseFloat(report.cpu?.utilizationPercent || '0');
   290→            const evLag = report.eventLoop?.currentLagMs || 0;
   291→            setPerfData(Array.from({ length: 12 }, (_, i) => {
   292→              const slot = new Date(base.getTime() - (11 - i) * 300000);
   293→              return {
   294→                time: slot.toLocaleTimeString('en', { hour: '2-digit', minute: '2-digit' }),
   295→                cpu: i === 11 ? cpuUtil : 0,
   296→                memory: i === 11 ? memUtil : 0,
   297→                eventLoop: i === 11 ? evLag : 0,
   298→              };
   299→            }));
   300→            return;
   301→          }
   302→        }
   303→      } catch { /* performance endpoint unavailable */ }
   304→      // Fallback: empty chart slots
   305→      const base = new Date();
   306→      base.setMinutes(base.getMinutes() - base.getMinutes() % 5, 0, 0);
   307→      setPerfData(Array.from({ length: 12 }, (_, i) => {
   308→        const slot = new Date(base.getTime() - (11 - i) * 300000);
   309→        return { time: slot.toLocaleTimeString('en', { hour: '2-digit', minute: '2-digit' }), cpu: 0, memory: 0, eventLoop: 0 };
   310→      }));
   311→    }
   312→    fetchPerf();
   313→  }, []);
   314→
   315→  const performanceData = useMemo(() => perfData, [perfData]);
   316→
   317→  // ─── Loading ───────────────────────────────────────────────────
   318→  if (loading) {
   319→    return (
   320→      <div className="space-y-6">
   321→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
   322→          {[1, 2, 3, 4].map((i) => (
   323→            <div key={i} className="h-32 animate-shimmer rounded-xl" />
   324→          ))}
   325→        </div>
   326→        <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
   327→          {[1, 2, 3].map((i) => (
   328→            <div key={i} className="h-64 animate-shimmer rounded-xl" />
   329→          ))}
   330→        </div>
   331→      </div>
   332→    );
   333→  }
   334→
   335→  // ═══════════════════════════════════════════════════════════════
   336→  return (
   337→    <div className="space-y-6 animate-slide-in">
   338→      {/* Top Stats Row */}
   339→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4 xl:grid-cols-6">
   340→        <StatCard title="Total Agents" value={totalAgents} subtitle="Across all clusters" icon={Bot} color="text-primary" />
   341→        <StatCard title="Active" value={activeAgents} subtitle="Currently running" icon={Activity} color="text-emerald-400" trend="up" trendValue={`${Math.round(activeAgents/Math.max(totalAgents,1)*100)}% active`} />
   342→        <StatCard title="Active Missions" value={activeMissions} subtitle="In progress" icon={Rocket} color="text-orange-400" />
   343→        <StatCard title="Services UP" value={`${healthyServices}/${totalServices}`} subtitle="System health" icon={CheckCircle2} color="text-cyan-400" />
