    68→    68→/**
    69→    69→ * LLMService — facade that selects and manages LLM providers.
    70→    70→ *
    71→    71→ * Features:
    72→    72→ * - Chat with a specific provider or the configured default
    73→    73→ * - Automatic fallback to secondary provider when primary fails
    74→    74→ * - Per-provider usage metrics tracking
    75→    75→ * - Circuit breaker integration — each provider call is wrapped in a circuit breaker
    76→    76→ * - Dead host cooldown — failed hosts are temporarily removed from rotation
    77→    77→ * - Response caching — identical requests are served from cache
    78→    78→ * - Streaming support — async generator for real-time token delivery
    79→    79→ * - Graceful degradation — never throws when LLM is unavailable
    80→    80→ *
    81→    81→ * Circuit Breaker Integration:
    82→    82→ * - Circuit key: `llm:{providerName}`
    83→    83→ * - On OPEN: automatically tries fallback provider
    84→    84→ * - On HALF_OPEN: allows a probe request through to test recovery
    85→    85→ *
    86→    86→ * Dead Host Cooldown Integration:
    87→    87→ * - Tracks consecutive failures per provider host
    88→    88→ * - After 2 consecutive failures, the host enters a 20s cooldown
    89→    89→ * - Cooldown is automatically cleared on expiry or on success
    90→    90→ *
    91→    91→ * Cache Integration:
    92→    92→ * - SHA256-based cache key derived from (model, messages hash, temperature, maxTokens)
    93→    93→ * - TTL-based expiration (default 5 minutes)
    94→    94→ * - LRU eviction when max cache size is reached
    95→    95→ * - Cache is bypassed for streaming requests
    96→    96→ */
    97→    97→@Injectable()
    98→    98→export class LLMService {
    99→    99→  private readonly logger = new Logger(LLMService.name);

   102→   102→  private defaultProviderName: string;
   103→   103→  private fallbackEnabled: boolean;
   104→   104→  private secondaryProviderName: string;
   105→   105→  private readonly cacheEnabled: boolean;

   126→   126→        failedRequests: 0,

   128→   128→        totalCompletionTokens: 0,

   155→   155→        `fallback: ${this.fallbackEnabled ? this.secondaryProviderName : 'disabled'}, ` +
   156→   156→        `circuit breaker: ${this.circuitBreakerService ? 'enabled' : 'disabled'}, ` +
   157→   157→        `dead host cooldown: ${this.deadHostCooldownService ? 'enabled' : 'disabled'}, ` +
   158→   158→        `cache: ${this.cacheService && this.cacheEnabled ? 'enabled' : 'disabled'}`,
   159→   159→    );

   162→   162→  /**
   163→   163→   * Send a chat request using a specific provider (or the default).
   164→   164→   * If the primary provider fails and fallback is enabled, tries the secondary.
   165→   165→   * Each provider call is wrapped in a circuit breaker (when available).
   166→   166→   *
   167→   167→   * Cache: If caching is enabled, identical requests will be served from cache
   168→   168→   * unless options.skipCache is true.
   169→   169→   */
   170→   170→  async chat(
   171→   171→    messages: LLMMessage[],
   172→   172→    options?: LLMOptions & { skipCache?: boolean },
   173→   173→    providerName?: string,
   174→   174→  ): Promise<LLMResponse> {
   175→   175→    const targetName = providerName || this.defaultProviderName;
   176→   176→    const provider = this.getProvider(targetName);
   177→   177→
   178→   178→    if (!provider) {
   179→   179→      throw new Error(`LLM provider not found: ${targetName}`);
   180→   180→    }
   181→   181→
   182→   182→    // Check dead host cooldown
   183→   183→    if (this.deadHostCooldownService && !this.deadHostCooldownService.isAvailable(targetName)) {
   184→   184→      this.logger.warn(
   185→   185→        `Provider "${targetName}" is in cooldown — trying fallback or alternative`,
   186→   186→      );
   187→   187→      if (this.fallbackEnabled && targetName !== this.secondaryProviderName) {
   188→   188→        return this.chatWithFallback(messages, options, targetName);
   189→   189→      }
   190→   190→      throw new Error(
   191→   191→        `LLM provider "${targetName}" is in cooldown — try again later`,
   192→   192→      );
   193→   193→    }
   194→   194→
   195→   195→    if (!provider.isAvailable()) {
   196→   196→      // Mark as dead host
   197→   197→      if (this.deadHostCooldownService) {
   198→   198→        this.deadHostCooldownService.markFailed(targetName, 'Provider unavailable — API key not configured');
   199→   199→      }
   200→   200→
   201→   201→      // Try fallback if enabled
   202→   202→      if (this.fallbackEnabled) {
   203→   203→        this.logger.warn(
   204→   204→          `Provider "${targetName}" unavailable, trying fallback "${this.secondaryProviderName}"`,
   205→   205→        );
   206→   206→        return this.chatWithFallback(messages, options, targetName);
   207→   207→      }
   208→   208→      throw new Error(
   209→   209→        `LLM provider "${targetName}" is not available — API key not configured`,
   210→   210→      );
   211→   211→    }
   212→   212→
   213→   213→    // Check cache before making the request
   214→   214→    if (this.cacheService && this.cacheEnabled && !options?.skipCache) {
   215→   215→      const model = options?.model || this.getDefaultModel(targetName);
   216→   216→      const cacheKey = this.cacheService.buildKey(
   217→   217→        model,
   218→   218→        messages,
   219→   219→        options?.temperature,
   220→   220→        options?.maxTokens,
   221→   221→      );
   222→   222→      const cached = this.cacheService.get<LLMResponse>(cacheKey);
   223→   223→      if (cached) {
   224→   224→        this.logger.debug(`Cache hit for provider "${targetName}" — returning cached response`);
   225→   225→        return cached;
   226→   226→      }
   227→   227→    }
   228→   228→
   229→   229→    // Wrap the provider call with circuit breaker if available
   230→   230→    if (this.circuitBreakerService) {
   231→   231→      const circuitKey = `${CIRCUIT_KEY_PREFIX.LLM}:${targetName}`;
   232→   232→
   233→   233→      const response = await this.circuitBreakerService.execute<LLMResponse>(
   234→   234→        circuitKey,
   235→   235→        () => this.executeProviderChat(provider, targetName, messages, options),
   236→   236→        // Fallback: try secondary provider when circuit is OPEN
   237→   237→        this.fallbackEnabled && targetName !== this.secondaryProviderName
   238→   238→          ? () => this.chatWithFallback(messages, options, targetName)
   239→   239→          : undefined,
   240→   240→      );
   241→   241→
   242→   242→      // Store in cache on success
   243→   243→      this.storeInCache(targetName, messages, options, response);
   244→   244→      return response;
   245→   245→    }
   246→   246→
   247→   247→    // No circuit breaker — direct execution
   248→   248→    const response = await this.executeProviderChat(provider, targetName, messages, options);
   249→   249→    this.storeInCache(targetName, messages, options, response);
   250→   250→    return response;
   251→   251→  }

   269→   269→  /**
   270→   270→   * Streaming chat — returns an async generator that yields chunks of the response.
   271→   271→   *
   272→   272→   * Note: Streaming bypasses the cache since partial responses aren't cacheable.
   273→   273→   * If a provider doesn't support streaming, the full response is yielded as a
   274→   274→   * single chunk.
   275→   275→   */
   276→   276→  async *chatStream(
   277→   277→    messages: LLMMessage[],
   278→   278→    options?: LLMOptions,
   279→   279→    providerName?: string,
   280→   280→  ): AsyncGenerator<LLMStreamChunk, void, undefined> {
   281→   281→    const targetName = providerName || this.defaultProviderName;
   282→   282→    const provider = this.getProvider(targetName);
   283→   283→
   284→   284→    if (!provider || !provider.isAvailable()) {
   285→   285→      // Try fallback
   286→   286→      if (this.fallbackEnabled && targetName !== this.secondaryProviderName) {
   287→   287→        const fallbackProvider = this.getProvider(this.secondaryProviderName);
   288→   288→        if (fallbackProvider?.isAvailable()) {
   289→   289→          yield* this.streamFromProvider(fallbackProvider, this.secondaryProviderName, messages, options);
   290→   290→          return;
   291→   291→        }
   292→   292→      }
   293→   293→      throw new Error(`LLM provider "${targetName}" is not available for streaming`);
   294→   294→    }
   295→   295→
   296→   296→    // Check dead host cooldown
   297→   297→    if (this.deadHostCooldownService && !this.deadHostCooldownService.isAvailable(targetName)) {
   298→   298→      throw new Error(`LLM provider "${targetName}" is in cooldown — try again later`);
   299→   299→    }
   300→   300→
   301→   301→    yield* this.streamFromProvider(provider, targetName, messages, options);
   302→   302→  }

   318→   318→  /**
   319→   319→   * Lists all registered providers with their availability status and metrics.

   321→   321→  listProviders(): ProviderInfo[] {
   322→   322→    const result: ProviderInfo[] = [];
   323→   323→    for (const [name, provider] of this.providers) {
   324→   324→      const info: ProviderInfo = {
   325→   325→        name,
   326→   326→        available: provider.isAvailable(),
   327→   327→        metrics: this.metrics.get(name) || this.emptyMetrics(),
   328→   328→      };
   329→   329→
   330→   330→      // Add circuit breaker state if available
   331→   331→      if (this.circuitBreakerService) {
   332→   332→        const circuitState = this.circuitBreakerService.getState(`${CIRCUIT_KEY_PREFIX.LLM}:${name}`);
   333→   333→        info.circuitState = circuitState.state;
   334→   334→      }
   335→   335→
   336→   336→      // Add cooldown state if available
   337→   337→      if (this.deadHostCooldownService) {
   338→   338→        info.inCooldown = !this.deadHostCooldownService.isAvailable(name);
   339→   339→      }
   340→   340→
   341→   341→      result.push(info);
   342→   342→    }
   343→   343→    return result;
   344→   344→  }

   346→   346→  /**
   347→   347→   * Check if any LLM provider is available.
   348→   348→   * Considers circuit breaker state and dead host cooldown.
   349→   349→   */
   350→   350→  isAnyAvailable(): boolean {
   351→   351→    for (const [name, provider] of this.providers) {
   352→   352→      if (!provider.isAvailable()) continue;
   353→   353→
   354→   354→      // If circuit breaker is available, check if the circuit is OPEN
   355→   355→      if (this.circuitBreakerService) {
   356→   356→        const circuitKey = `${CIRCUIT_KEY_PREFIX.LLM}:${name}`;
   357→   357→        if (this.circuitBreakerService.isOpen(circuitKey)) {
   358→   358→          this.logger.debug(`Provider "${name}" is available but circuit is OPEN — skipping`);
   359→   359→          continue;
   360→   360→        }
   361→   361→      }
   362→   362→
   363→   363→      // If dead host cooldown is available, check if the host is in cooldown
   364→   364→      if (this.deadHostCooldownService && !this.deadHostCooldownService.isAvailable(name)) {
   365→   365→        this.logger.debug(`Provider "${name}" is available but in cooldown — skipping`);
   366→   366→        continue;
   367→   367→      }
   368→   368→
   369→   369→      return true;
   370→   370→    }
   371→   371→    return false;
   372→   372→  }

   398→   398→
   399→   399→  /**
   400→   400→   * Switch the default LLM provider at runtime.
   401→   401→   * Invalidates the cache and resets circuit breakers for a clean switch.
   402→   402→   *
   403→   403→   * @param providerName - The name of the provider to switch to ('zai', 'openai', 'anthropic')
   404→   404→   * @throws Error if the provider name is not registered
   405→   405→   */
   406→   406→  switchProvider(providerName: string): void {
   407→   407→    const validProviders = Array.from(this.providers.keys());
   408→   408→    if (!validProviders.includes(providerName)) {
   409→   409→      throw new Error(
   410→   410→        `Invalid provider "${providerName}". Available providers: ${validProviders.join(', ')}`,
   411→   411→      );
   412→   412→    }
   413→   413→
   414→   414→    const previousProvider = this.defaultProviderName;
   415→   415→    this.defaultProviderName = providerName;
   416→   416→
   417→   417→    // Invalidate cache to avoid stale responses from the previous provider
   418→   418→    this.invalidateCache();
   419→   419→
   420→   420→    // Reset circuit breaker for the new provider so it gets a fresh start
   421→   421→    if (this.circuitBreakerService) {
   422→   422→      const circuitKey = `${CIRCUIT_KEY_PREFIX.LLM}:${providerName}`;
   423→   423→      this.circuitBreakerService.reset(circuitKey);
   424→   424→    }
   425→   425→
   426→   426→    // Clear dead host cooldown for the new provider
   427→   427→    if (this.deadHostCooldownService) {
   428→   428→      this.deadHostCooldownService.resetHost(providerName);
   429→   429→    }
   430→   430→
   431→   431→    this.logger.log(
   432→   432→      `Provider switched: "${previousProvider}" → "${providerName}"` +
   433→   433→        ` (available: ${this.providers.get(providerName)?.isAvailable() ?? false})`,
   434→   434→    );
   435→   435→  }
   436→   436→
   437→   437→  /**
   438→   438→   * Update the fallback configuration at runtime.
   439→   439→   *
   440→   440→   * @param enabled - Whether fallback is enabled
   441→   441→   * @param secondaryProvider - The fallback provider name
   442→   442→   */
   443→   443→  setFallbackConfig(enabled: boolean, secondaryProvider?: string): void {
   444→   444→    if (secondaryProvider) {
   445→   445→      const validProviders = Array.from(this.providers.keys());
   446→   446→      if (!validProviders.includes(secondaryProvider)) {
   447→   447→        throw new Error(
   448→   448→          `Invalid secondary provider "${secondaryProvider}". Available providers: ${validProviders.join(', ')}`,
   449→   449→        );
   450→   450→      }
   451→   451→      this.secondaryProviderName = secondaryProvider;
   452→   452→    }
   453→   453→    this.fallbackEnabled = enabled;
   454→   454→
   455→   455→    this.logger.log(
   456→   456→      `Fallback config updated: enabled=${enabled}, secondary="${this.secondaryProviderName}"`,
   457→   457→    );
   458→   458→  }
   459→   459→
   460→   460→  /**
   461→   461→   * Apply a full LLM configuration at runtime.
   462→   462→   * Used by the admin API to persist and apply provider changes.
   463→   463→   *

   466→   466→  applyConfig(config: LLMConfig): void {

   472→   472→    );
   473→   473→  }

   487→   487→  // ─── Private Methods ─────────────────────────────────────────
   488→   488→
   489→   489→  /**
   490→   490→   * Execute a provider chat call with metrics tracking and dead host management.
   491→   491→   */
   492→   492→  private async executeProviderChat(
   493→   493→    provider: ILLMProvider,
   494→   494→    providerName: string,
   495→   495→    messages: LLMMessage[],
   496→   496→    options: LLMOptions | undefined,
   497→   497→  ): Promise<LLMResponse> {
   498→   498→    try {
   499→   499→      const response = await provider.chat(messages, options);
   500→   500→      this.recordSuccess(providerName, response);
   501→   501→
   502→   502→      // Mark host as healthy
   503→   503→      if (this.deadHostCooldownService) {
   504→   504→        this.deadHostCooldownService.markSuccess(providerName);
   505→   505→      }
   506→   506→
   507→   507→      return response;
   508→   508→    } catch (error: any) {
   509→   509→      this.recordFailure(providerName, error.message);
   510→   510→
   511→   511→      // Mark host as failed for cooldown tracking
   512→   512→      if (this.deadHostCooldownService) {
   513→   513→        this.deadHostCooldownService.markFailed(providerName, error.message);
   514→   514→      }
   515→   515→
   516→   516→      // Try fallback on failure (when NOT already in circuit breaker context)
   517→   517→      if (!this.circuitBreakerService && this.fallbackEnabled && providerName !== this.secondaryProviderName) {
   518→   518→        this.logger.warn(
   519→   519→          `Provider "${providerName}" failed, trying fallback "${this.secondaryProviderName}": ${error.message}`,
   520→   520→        );
   521→   521→        return this.chatWithFallback(messages, options, providerName);
   522→   522→      }
   523→   523→
   524→   524→      throw error;
   525→   525→    }
   526→   526→  }
   527→   527→
   528→   528→  /**
   529→   529→   * Attempt chat with the fallback provider.
   530→   530→   * Also wrapped in a circuit breaker when available.
   531→   531→   */
   532→   532→  private async chatWithFallback(
   533→   533→    messages: LLMMessage[],
   534→   534→    options: (LLMOptions & { skipCache?: boolean }) | undefined,
   535→   535→    failedProvider: string,
   536→   536→  ): Promise<LLMResponse> {
   537→   537→    const fallbackProvider = this.providers.get(this.secondaryProviderName);
   538→   538→
   539→   539→    if (!fallbackProvider) {
   540→   540→      throw new Error(
   541→   541→        `Fallback provider "${this.secondaryProviderName}" not found`,
   542→   542→      );
   543→   543→    }
   544→   544→
   545→   545→    // Check dead host cooldown for fallback
   546→   546→    if (this.deadHostCooldownService && !this.deadHostCooldownService.isAvailable(this.secondaryProviderName)) {
   547→   547→      throw new Error(
   548→   548→        `Both primary ("${failedProvider}") and fallback ("${this.secondaryProviderName}") are in cooldown`,
   549→   549→      );
   550→   550→    }
   551→   551→
   552→   552→    if (!fallbackProvider.isAvailable()) {
   553→   553→      if (this.deadHostCooldownService) {
   554→   554→        this.deadHostCooldownService.markFailed(this.secondaryProviderName, 'Fallback unavailable');
   555→   555→      }
   556→   556→      throw new Error(
   557→   557→        `Both primary ("${failedProvider}") and fallback ("${this.secondaryProviderName}") providers are unavailable`,
   558→   558→      );
   559→   559→    }
   560→   560→
   561→   561→    // Wrap fallback in its own circuit breaker
   562→   562→    if (this.circuitBreakerService) {
   563→   563→      const fallbackCircuitKey = `${CIRCUIT_KEY_PREFIX.LLM}:${this.secondaryProviderName}`;
   564→   564→
   565→   565→      try {
   566→   566→        const response = await this.circuitBreakerService.execute<LLMResponse>(
   567→   567→          fallbackCircuitKey,
   568→   568→          () => {
   569→   569→            const resp = fallbackProvider.chat(messages, options);
   570→   570→            return resp.then((r) => {
   571→   571→              this.recordSuccess(this.secondaryProviderName, r);
   572→   572→              if (this.deadHostCooldownService) {
   573→   573→                this.deadHostCooldownService.markSuccess(this.secondaryProviderName);
   574→   574→              }
   575→   575→              return r;
   576→   576→            });
   577→   577→          },
   578→   578→        );
   579→   579→
   580→   580→        // Store fallback result in cache
   581→   581→        this.storeInCache(this.secondaryProviderName, messages, options, response);
   582→   582→        return response;
   583→   583→      } catch (error: any) {
   584→   584→        if (error instanceof CircuitBreakerOpenError) {
   585→   585→          throw new Error(
   586→   586→            `Both primary ("${failedProvider}") and fallback ("${this.secondaryProviderName}") circuits are OPEN`,
   587→   587→          );
   588→   588→        }
   589→   589→        this.recordFailure(this.secondaryProviderName, error.message);
   590→   590→        if (this.deadHostCooldownService) {
   591→   591→          this.deadHostCooldownService.markFailed(this.secondaryProviderName, error.message);
   592→   592→        }
   593→   593→        throw error;
   594→   594→      }
   595→   595→    }
   596→   596→
   597→   597→    // No circuit breaker — direct fallback execution
   598→   598→    try {
   599→   599→      const response = await fallbackProvider.chat(messages, options);
   600→   600→      this.recordSuccess(this.secondaryProviderName, response);
   601→   601→      if (this.deadHostCooldownService) {
   602→   602→        this.deadHostCooldownService.markSuccess(this.secondaryProviderName);
   603→   603→      }
   604→   604→      this.storeInCache(this.secondaryProviderName, messages, options, response);
   605→   605→      return response;
   606→   606→    } catch (error: any) {
   607→   607→      this.recordFailure(this.secondaryProviderName, error.message);
   608→   608→      if (this.deadHostCooldownService) {
   609→   609→        this.deadHostCooldownService.markFailed(this.secondaryProviderName, error.message);
   610→   610→      }
   611→   611→      throw error;
   612→   612→    }
   613→   613→  }

   615→   615→  /**
   616→   616→   * Stream from a provider, falling back to non-streaming if streaming isn't supported.
   617→   617→   */
   618→   618→  private async *streamFromProvider(
   619→   619→    provider: ILLMProvider,
   620→   620→    providerName: string,
   621→   621→    messages: LLMMessage[],
   622→   622→    options: LLMOptions | undefined,
   623→   623→  ): AsyncGenerator<LLMStreamChunk, void, undefined> {
   624→   624→    try {
   625→   625→      // Check if the provider supports streaming (OpenAI SDK has streaming)
   626→   626→      if (provider instanceof OpenAIProvider && (provider as any).client) {
   627→   627→        const client = (provider as any).client;
   628→   628→        const model = options?.model || this.configService.get<string>('llm.openai.model', 'gpt-4o');
   629→   629→        const maxTokens = options?.maxTokens || this.configService.get<number>('llm.openai.maxTokens', 4096);
   630→   630→        const temperature = options?.temperature ?? 0.7;
   631→   631→
   632→   632→        const openaiMessages = messages.map((m) => ({
   633→   633→          role: m.role,
   634→   634→          content: m.content,
   635→   635→        }));
   636→   636→
   637→   637→        const stream = await client.chat.completions.create({
   638→   638→          model,
   639→   639→          messages: openaiMessages,
   640→   640→          max_tokens: maxTokens,
   641→   641→          temperature,
   642→   642→          stream: true,
   643→   643→        });
   644→   644→
   645→   645→        let fullContent = '';
   646→   646→        for await (const chunk of stream) {
   647→   647→          const content = chunk.choices[0]?.delta?.content || '';
   648→   648→          if (content) {
   649→   649→            fullContent += content;
   650→   650→            yield {
   651→   651→              content,
   652→   652→              model: chunk.model || model,
   653→   653→              finishReason: null,
   654→   654→            };
   655→   655→          }
   656→   656→          if (chunk.choices[0]?.finish_reason) {
   657→   657→            yield {
   658→   658→              content: '',
   659→   659→              model: chunk.model || model,
   660→   660→              finishReason: chunk.choices[0].finish_reason,
   661→   661→              usage: chunk.usage
   662→   662→                ? {
   663→   663→                    promptTokens: chunk.usage.prompt_tokens ?? 0,
   664→   664→                    completionTokens: chunk.usage.completion_tokens ?? 0,
   665→   665→                    totalTokens: chunk.usage.total_tokens ?? 0,
   666→   666→                  }
   667→   667→                : undefined,
   668→   668→            };
   669→   669→          }
   670→   670→        }
   671→   671→
   672→   672→        // Record success for the full streaming response
   673→   673→        this.recordSuccess(providerName, {
   674→   674→          content: fullContent,
   675→   675→          model,
   676→   676→          usage: { promptTokens: 0, completionTokens: 0, totalTokens: 0 },
   677→   677→          finishReason: 'stop',
   678→   678→        });
   679→   679→
   680→   680→        if (this.deadHostCooldownService) {
   681→   681→          this.deadHostCooldownService.markSuccess(providerName);
   682→   682→        }
   683→   683→        return;
   684→   684→      }
   685→   685→
   686→   686→      if (provider instanceof AnthropicProvider && (provider as any).client) {
   687→   687→        const client = (provider as any).client;
   688→   688→        const model = options?.model || this.configService.get<string>('llm.anthropic.model', 'claude-sonnet-4-20250514');
   689→   689→        const maxTokens = options?.maxTokens || this.configService.get<number>('llm.anthropic.maxTokens', 4096);
   690→   690→        const temperature = options?.temperature ?? 0.7;
   691→   691→
   692→   692→        let systemPrompt: string | undefined;
   693→   693→        const anthropicMessages: any[] = [];
   694→   694→        for (const msg of messages) {
   695→   695→          if (msg.role === 'system') {
   696→   696→            systemPrompt = msg.content;
   697→   697→          } else {
   698→   698→            anthropicMessages.push({ role: msg.role, content: msg.content });
   699→   699→          }
   700→   700→        }
   701→   701→
   702→   702→        const stream = client.messages.stream({
   703→   703→          model,
   704→   704→          messages: anthropicMessages,
   705→   705→          max_tokens: maxTokens,
   706→   706→          temperature,
   707→   707→          ...(systemPrompt ? { system: systemPrompt } : {}),
   708→   708→        });
   709→   709→
   710→   710→        let fullContent = '';
   711→   711→        for await (const event of stream) {
   712→   712→          if (event.type === 'content_block_delta' && event.delta?.type === 'text_delta') {
   713→   713→            const content = event.delta.text || '';
   714→   714→            fullContent += content;
   715→   715→            yield {
   716→   716→              content,
   717→   717→              model,
   718→   718→              finishReason: null,
   719→   719→            };
   720→   720→          }
   721→   721→          if (event.type === 'message_stop') {
   722→   722→            yield {
   723→   723→              content: '',
   724→   724→              model,
   725→   725→              finishReason: 'end_turn',
   726→   726→            };
   727→   727→          }
   728→   728→        }
   729→   729→
   730→   730→        this.recordSuccess(providerName, {
   731→   731→          content: fullContent,
   732→   732→          model,
   733→   733→          usage: { promptTokens: 0, completionTokens: 0, totalTokens: 0 },
   734→   734→          finishReason: 'end_turn',
   735→   735→        });
   736→   736→
   737→   737→        if (this.deadHostCooldownService) {
   738→   738→          this.deadHostCooldownService.markSuccess(providerName);
   739→   739→        }
   740→   740→        return;
   741→   741→      }
   742→   742→
   743→   743→      // Fallback: provider doesn't support streaming — make a regular call and yield as single chunk
   744→   744→      const response = await provider.chat(messages, options);
   745→   745→      this.recordSuccess(providerName, response);
   746→   746→      if (this.deadHostCooldownService) {
   747→   747→        this.deadHostCooldownService.markSuccess(providerName);
   748→   748→      }
   749→   749→
   750→   750→      yield {
   751→   751→        content: response.content,
   752→   752→        model: response.model,
   753→   753→        finishReason: response.finishReason,
   754→   754→        usage: response.usage,
   755→   755→      };
   756→   756→    } catch (error: any) {
   757→   757→      this.recordFailure(providerName, error.message);
   758→   758→      if (this.deadHostCooldownService) {
   759→   759→        this.deadHostCooldownService.markFailed(providerName, error.message);
   760→   760→      }
   761→   761→      throw error;
   762→   762→    }
   763→   763→  }

   765→   765→  /**
   766→   766→   * Store a response in cache if caching is enabled.
   767→   767→   */
   768→   768→  private storeInCache(
   769→   769→    providerName: string,
   770→   770→    messages: LLMMessage[],
   771→   771→    options: (LLMOptions & { skipCache?: boolean }) | undefined,
   772→   772→    response: LLMResponse,
   773→   773→  ): void {
   774→   774→    if (!this.cacheService || !this.cacheEnabled || options?.skipCache) return;
   775→   775→
   776→   776→    try {
   777→   777→      const model = options?.model || this.getDefaultModel(providerName);
   778→   778→      const cacheKey = this.cacheService.buildKey(
   779→   779→        model,
   780→   780→        messages,
   781→   781→        options?.temperature,
   782→   782→        options?.maxTokens,
   783→   783→      );
   784→   784→      this.cacheService.set(cacheKey, response);
   785→   785→    } catch (err: any) {
   786→   786→      this.logger.debug(`Failed to store response in cache: ${err.message}`);
   787→   787→    }
   788→   788→  }

   790→   790→  /**
   791→   791→   * Get the default model name for a given provider.
   792→   792→   */
   793→   793→  private getDefaultModel(providerName: string): string {
   794→   794→    if (providerName === 'openai') {
   795→   795→      return this.configService.get<string>('llm.openai.model', 'gpt-4o');
   796→   796→    }
   797→   797→    if (providerName === 'anthropic') {
   798→   798→      return this.configService.get<string>('llm.anthropic.model', 'claude-sonnet-4-20250514');
   799→   799→    }
   800→   800→    return 'unknown';
   801→   801→  }

   803→   803→  private recordSuccess(providerName: string, response: LLMResponse): void {
   804→   804→    const metrics = this.metrics.get(providerName) || this.emptyMetrics();
   805→   805→    metrics.totalRequests++;
   806→   806→    metrics.successfulRequests++;
   807→   807→    metrics.totalPromptTokens += response.usage.promptTokens;
   808→   808→    metrics.totalCompletionTokens += response.usage.completionTokens;
   809→   809→    metrics.totalTokens += response.usage.totalTokens;
   810→   810→    metrics.lastRequestAt = new Date();
   811→   811→    metrics.lastError = null;
   812→   812→    this.metrics.set(providerName, metrics);
   813→   813→  }
   814→   814→
   815→   815→  private recordFailure(providerName: string, errorMessage: string): void {
   816→   816→    const metrics = this.metrics.get(providerName) || this.emptyMetrics();
   817→   817→    metrics.totalRequests++;
   818→   818→    metrics.failedRequests++;
   819→   819→    metrics.lastRequestAt = new Date();
   820→   820→    metrics.lastError = errorMessage;
   821→   821→    this.metrics.set(providerName, metrics);
   822→   822→  }

   829→   829→      totalPromptTokens: 0,
   830→   830→      totalCompletionTokens: 0,
   831→   831→      totalTokens: 0,
   832→   832→      lastRequestAt: null,
   833→   833→      lastError: null,
   834→   834→    };
   835→   835→  }
