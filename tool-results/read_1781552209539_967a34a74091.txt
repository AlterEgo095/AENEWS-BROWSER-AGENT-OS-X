   102→async function apiFetch<T>(url: string, options?: RequestInit): Promise<T> {
   103→  const res = await fetch(url, options)
   104→  if (!res.ok) {
   105→    const err = await res.json().catch(() => ({ error: 'Request failed' }))
   106→    throw new Error(err.error || 'Request failed')
   107→  }
   108→  return res.json()
   109→}

   111→// ─── Animated Counter ────────────────────────────────────────────────────────
   112→
   113→function AnimatedCounter({ value, duration = 800 }: { value: number; duration?: number }) {
   114→  const [display, setDisplay] = useState(0)
   115→
   116→  useEffect(() => {
   117→    const start = display
   118→    const end = value
   119→    const diff = end - start
   120→    if (diff === 0) return
   121→
   122→    const startTime = performance.now()
   123→    let rafId: number
   124→
   125→    function animate(currentTime: number) {
   126→      const elapsed = currentTime - startTime
   127→      const progress = Math.min(elapsed / duration, 1)
   128→      const eased = 1 - Math.pow(1 - progress, 3)
   129→      setDisplay(Math.round(start + diff * eased))
   130→      if (progress < 1) {
   131→        rafId = requestAnimationFrame(animate)
   132→      }
   133→    }
   134→
   135→    rafId = requestAnimationFrame(animate)
   136→    return () => cancelAnimationFrame(rafId)
   137→    // eslint-disable-next-line react-hooks/exhaustive-deps
   138→  }, [value, duration])
   139→
   140→  return <span>{display.toLocaleString()}</span>
   141→}

   145→function getTransactionIcon(type: string) {
   146→  switch (type) {
   147→    case 'purchase':
   148→      return <CreditCard className="size-4 text-emerald-400" />
   149→    case 'admin_add':
   150→      return <ArrowUpCircle className="size-4 text-emerald-400" />
   151→    case 'admin_deduct':
   152→      return <ArrowDownCircle className="size-4 text-red-400" />
   153→    case 'bonus':
   154→      return <Sparkles className="size-4 text-amber-400" />
   155→    case 'usage':
   156→      return <Minus className="size-4 text-red-400" />
   157→    default:
   158→      return <Coins className="size-4 text-slate-400" />
   159→  }
   160→}
