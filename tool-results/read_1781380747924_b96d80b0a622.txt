   337→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   338→    const startTime = Date.now();
   339→    const { action, ...params } = input.payload;
   340→
   341→    if (!action) {
   342→      return this.createAgentOutput(
   343→        input.taskId,
   344→        false,
   345→        null,
   346→        'Missing required parameter: action',
   347→        startTime,
   348→      );
   349→    }
   350→
   351→    const supportedActions = [
   352→      'readFile',
   353→      'writeFile',
   354→      'createDirectory',
   355→      'deleteFile',
   356→      'moveFile',
   357→      'copyFile',
   358→      'listDirectory',
   359→      'getFileInfo',
   360→    ];
   361→
   362→    if (!supportedActions.includes(action)) {
   363→      return this.createAgentOutput(
   364→        input.taskId,
   365→        false,
   366→        null,
   367→        `Unknown filesystem action: ${action}. Supported: ${supportedActions.join(', ')}`,
   368→        startTime,
   369→      );
   370→    }
   371→
   372→    try {
   373→      const tool = this.getTool(action);
   374→      if (!tool) {
   375→        return this.createAgentOutput(
   376→          input.taskId,
   377→          false,
   378→          null,
   379→          `Tool not found: ${action}`,
   380→          startTime,
   381→        );
   382→      }
   383→
   384→      const result = await tool.execute(params);
   385→
   386→      this.operationLog.push({
   387→        operation: action,
   388→        path: params.path || params.source || '',
   389→        timestamp: new Date(),
   390→        success: true,
   391→      });
   392→
   393→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   394→    } catch (error) {
   395→      const msg = (error as Error).message;
   396→      this.logger.error(`FileSystem execution failed for ${action}: ${msg}`);
   397→
   398→      this.operationLog.push({
   399→        operation: action,
   400→        path: params.path || params.source || '',
   401→        timestamp: new Date(),
   402→        success: false,
   403→      });
   404→
   405→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   406→    }
   407→  }

   413→  }

   417→  private async readFile(
   418→    path: string,
   419→    encoding: string = 'utf-8',
   420→  ): Promise<{ content: string; size: number; encoding: string; lastModified: string }> {
   421→    this.validatePath(path);
   422→
   423→    const node = this.resolveNode(path);
   424→    if (!node) {
   425→      throw new Error(`File not found: ${path}`);
   426→    }
   427→    if (node.type === 'directory') {
   428→      throw new Error(`Path is a directory, not a file: ${path}`);
   429→    }
   430→
   431→    this.logger.log(`Read file: ${path} (${node.size} bytes)`);
   432→    return {
   433→      content: node.content || '',
   434→      size: node.size,
   435→      encoding,
   436→      lastModified: node.modified.toISOString(),
   437→    };
   438→  }

   440→  private async writeFile(
   441→    path: string,
   442→    content: string,
   443→    encoding: string = 'utf-8',
   444→    overwrite: boolean = true,
   445→  ): Promise<{ path: string; bytesWritten: number; created: boolean }> {
   446→    this.validatePath(path);
   447→
   448→    const parentPath = this.getParentPath(path);
   449→    const fileName = this.getBaseName(path);
   450→
   451→    const parentNode = this.resolveNode(parentPath);
   452→    if (!parentNode) {
   453→      throw new Error(`Parent directory not found: ${parentPath}`);
   454→    }
   455→    if (parentNode.type !== 'directory') {
   456→      throw new Error(`Parent path is not a directory: ${parentPath}`);
   457→    }
   458→
   459→    const existingNode = parentNode.children.get(fileName);
   460→    if (existingNode && existingNode.type === 'directory') {
   461→      throw new Error(`Cannot write file: a directory exists at ${path}`);
   462→    }
   463→    if (existingNode && !overwrite) {
   464→      throw new Error(`File already exists and overwrite is false: ${path}`);
   465→    }
   466→
   467→    const isCreated = !existingNode;
   468→    const byteSize = Buffer.byteLength(content, encoding as BufferEncoding);
   469→
   470→    const fileNode: VirtualNode = {
   471→      name: fileName,
   472→      type: 'file',
   473→      content,
   474→      children: new Map(),
   475→      created: existingNode?.created || new Date(),
   476→      modified: new Date(),
   477→      permissions: 'rw-r--r--',
   478→      size: byteSize,
   479→    };
   480→
   481→    parentNode.children.set(fileName, fileNode);
   482→    parentNode.modified = new Date();
   483→
   484→    this.logger.log(`Wrote file: ${path} (${byteSize} bytes, created: ${isCreated})`);
   485→    return { path, bytesWritten: byteSize, created: isCreated };
   486→  }

   488→  private async createDirectory(
   489→    path: string,
   490→    recursive: boolean = true,
   491→  ): Promise<{ path: string; created: boolean }> {
   492→    this.validatePath(path);
   493→
   494→    const segments = this.splitPath(path);
   495→
   496→    if (recursive) {
   497→      let current = this.root;
   498→      for (const segment of segments) {
   499→        if (!current.children.has(segment)) {
   500→          current.children.set(segment, this.createDirectoryNode(segment));
   501→          current.modified = new Date();
   502→        }
   503→        const next = current.children.get(segment)!;
   504→        if (next.type !== 'directory') {
   505→          throw new Error(`Path component is not a directory: ${segment}`);
   506→        }
   507→        current = next;
   508→      }
   509→      this.logger.log(`Created directory (recursive): ${path}`);
   510→      return { path, created: true };
   511→    } else {
   512→      // Non-recursive: parent must exist
   513→      const parentPath = this.getParentPath(path);
   514→      const dirName = this.getBaseName(path);
   515→      const parentNode = this.resolveNode(parentPath);
   516→
   517→      if (!parentNode) {
   518→        throw new Error(
   519→          `Parent directory not found: ${parentPath}. Use recursive: true to create parents.`,
   520→        );
   521→      }
   522→      if (parentNode.type !== 'directory') {
   523→        throw new Error(`Parent path is not a directory: ${parentPath}`);
   524→      }
   525→      if (parentNode.children.has(dirName)) {
   526→        throw new Error(`Directory already exists: ${path}`);
   527→      }
   528→
   529→      parentNode.children.set(dirName, this.createDirectoryNode(dirName));
   530→      parentNode.modified = new Date();
   531→
   532→      this.logger.log(`Created directory: ${path}`);
   533→      return { path, created: true };
   534→    }
   535→  }

   537→  private async deleteFile(
   538→    path: string,
   539→    recursive: boolean = false,
   540→    force: boolean = false,
   541→  ): Promise<{ path: string; deleted: boolean; type: string }> {
   542→    this.validatePath(path);
   543→
   544→    if (path === '/') {
   545→      throw new Error('Cannot delete root directory');
   546→    }
   547→
   548→    const node = this.resolveNode(path);
   549→    if (!node) {
   550→      throw new Error(`Path not found: ${path}`);
   551→    }
   552→
   553→    const parentPath = this.getParentPath(path);
   554→    const baseName = this.getBaseName(path);
   555→    const parentNode = this.resolveNode(parentPath);
   556→
   557→    if (node.type === 'directory' && node.children.size > 0) {
   558→      if (!recursive && !force) {
   559→        throw new Error(`Directory not empty: ${path}. Use recursive: true or force: true.`);
   560→      }
   561→    }
   562→
   563→    parentNode!.children.delete(baseName);
   564→    parentNode!.modified = new Date();
   565→
   566→    this.logger.log(`Deleted ${node.type}: ${path}`);
   567→    return { path, deleted: true, type: node.type };
   568→  }

   570→  private async moveFile(
   571→    source: string,
   572→    destination: string,
   573→    overwrite: boolean = false,
   574→  ): Promise<{ source: string; destination: string; moved: boolean }> {
   575→    this.validatePath(source);
   576→    this.validatePath(destination);
   577→
   578→    if (source === '/') {
   579→      throw new Error('Cannot move root directory');
   580→    }
   581→
   582→    const sourceNode = this.resolveNode(source);
   583→    if (!sourceNode) {
   584→      throw new Error(`Source not found: ${source}`);
   585→    }
   586→
   587→    const sourceParentPath = this.getParentPath(source);
   588→    const sourceBaseName = this.getBaseName(source);
   589→    const sourceParent = this.resolveNode(sourceParentPath);
   590→
   591→    const destParentPath = this.getParentPath(destination);
   592→    const destBaseName = this.getBaseName(destination);
   593→    const destParent = this.resolveNode(destParentPath);
   594→
   595→    if (!destParent) {
   596→      throw new Error(`Destination parent directory not found: ${destParentPath}`);
   597→    }
   598→    if (destParent.type !== 'directory') {
   599→      throw new Error(`Destination parent is not a directory: ${destParentPath}`);
   600→    }
   601→
   602→    const existingDest = destParent.children.get(destBaseName);
   603→    if (existingDest && !overwrite) {
   604→      throw new Error(`Destination already exists and overwrite is false: ${destination}`);
   605→    }
   606→
   607→    // Perform the move
   608→    const movedNode = { ...sourceNode, name: destBaseName, modified: new Date() };
   609→    destParent.children.set(destBaseName, movedNode);
   610→    sourceParent!.children.delete(sourceBaseName);
   611→    destParent.modified = new Date();
   612→    sourceParent!.modified = new Date();
   613→
   614→    this.logger.log(`Moved: ${source} → ${destination}`);
   615→    return { source, destination, moved: true };
   616→  }

   618→  private async copyFile(
   619→    source: string,
   620→    destination: string,
   621→    overwrite: boolean = false,
   622→    recursive: boolean = false,
   623→  ): Promise<{ source: string; destination: string; copied: boolean; bytesCopied: number }> {
   624→    this.validatePath(source);
   625→    this.validatePath(destination);
   626→
   627→    const sourceNode = this.resolveNode(source);
   628→    if (!sourceNode) {
   629→      throw new Error(`Source not found: ${source}`);
   630→    }
   631→
   632→    const destParentPath = this.getParentPath(destination);
   633→    const destBaseName = this.getBaseName(destination);
   634→    const destParent = this.resolveNode(destParentPath);
   635→
   636→    if (!destParent) {
   637→      throw new Error(`Destination parent directory not found: ${destParentPath}`);
   638→    }
   639→
   640→    const existingDest = destParent.children.get(destBaseName);
   641→    if (existingDest && !overwrite) {
   642→      throw new Error(`Destination already exists and overwrite is false: ${destination}`);
   643→    }
   644→
   645→    const copiedNode = this.deepCloneNode(sourceNode, destBaseName);
   646→
   647→    destParent.children.set(destBaseName, copiedNode);
   648→    destParent.modified = new Date();
   649→
   650→    const bytesCopied = this.calculateNodeSize(copiedNode);
   651→
   652→    this.logger.log(`Copied: ${source} → ${destination} (${bytesCopied} bytes)`);
   653→    return { source, destination, copied: true, bytesCopied };
   654→  }

   656→  private async listDirectory(
   657→    path: string,
   658→    recursive: boolean = false,
   659→    includeHidden: boolean = false,
   660→    pattern?: string,
   661→  ): Promise<{
   662→    path: string;
   663→    entries: Array<{ name: string; type: string; size: number; modified: string }>;
   664→    totalFiles: number;
   665→    totalDirectories: number;
   666→  }> {
   667→    this.validatePath(path);
   668→
   669→    const node = this.resolveNode(path);
   670→    if (!node) {
   671→      throw new Error(`Directory not found: ${path}`);
   672→    }
   673→    if (node.type !== 'directory') {
   674→      throw new Error(`Path is not a directory: ${path}`);
   675→    }
   676→
   677→    const entries: Array<{ name: string; type: string; size: number; modified: string }> = [];
   678→    let totalFiles = 0;
   679→    let totalDirectories = 0;
   680→
   681→    const collectEntries = (currentNode: VirtualNode, currentPath: string) => {
   682→      for (const [childName, childNode] of currentNode.children) {
   683→        // Skip hidden files unless requested
   684→        if (childName.startsWith('.') && !includeHidden) continue;
   685→
   686→        // Apply glob pattern filter
   687→        if (pattern && !this.matchGlob(childName, pattern)) continue;
   688→
   689→        const childPath = currentPath === '/' ? `/${childName}` : `${currentPath}/${childName}`;
   690→        entries.push({
   691→          name: childName,
   692→          type: childNode.type,
   693→          size: childNode.size,
   694→          modified: childNode.modified.toISOString(),
   695→        });
   696→
   697→        if (childNode.type === 'file') {
   698→          totalFiles++;
   699→        } else {
   700→          totalDirectories++;
   701→          if (recursive) {
   702→            collectEntries(childNode, childPath);
   703→          }
   704→        }
   705→      }
   706→    };
   707→
   708→    collectEntries(node, path);
   709→
   710→    this.logger.log(`Listed directory: ${path} (${totalFiles} files, ${totalDirectories} dirs)`);
   711→    return { path, entries, totalFiles, totalDirectories };
   712→  }

   744→  private createDirectoryNode(name: string): VirtualNode {
   745→    return {
   746→      name,
   747→      type: 'directory',
   748→      children: new Map(),
   749→      created: new Date(),
   750→      modified: new Date(),
   751→      permissions: 'rwxr-xr-x',
   752→      size: 4096,
   753→    };
   754→  }
   755→
   756→  private validatePath(path: string): void {
   757→    if (!path || typeof path !== 'string') {
   758→      throw new Error('Path must be a non-empty string');
   759→    }
   760→    if (!path.startsWith('/')) {
   761→      throw new Error('Path must be absolute (start with /)');
   762→    }
   763→    // Prevent path traversal
   764→    const normalized = path.replace(/\/+/g, '/');
   765→    if (normalized.includes('..')) {
   766→      throw new Error('Path traversal (..) is not allowed');
   767→    }
   768→  }
   769→
   770→  private resolveNode(path: string): VirtualNode | null {
   771→    if (path === '/') return this.root;
   772→
   773→    const segments = this.splitPath(path);
   774→    let current: VirtualNode = this.root;
   775→
   776→    for (const segment of segments) {
   777→      if (!segment) continue;
   778→      if (current.type !== 'directory') return null;
   779→      const child = current.children.get(segment);
   780→      if (!child) return null;
   781→      current = child;
   782→    }
   783→
   784→    return current;
   785→  }
   786→
   787→  private splitPath(path: string): string[] {
   788→    return path.split('/').filter((s) => s.length > 0);
   789→  }
   790→
   791→  private getParentPath(path: string): string {
   792→    const segments = this.splitPath(path);
   793→    if (segments.length <= 1) return '/';
   794→    return '/' + segments.slice(0, -1).join('/');
   795→  }
   796→
   797→  private getBaseName(path: string): string {
   798→    const segments = this.splitPath(path);
   799→    return segments[segments.length - 1] || '/';
   800→  }

   802→  private deepCloneNode(node: VirtualNode, newName: string): VirtualNode {
   803→    const clone: VirtualNode = {
   804→      name: newName,
   805→      type: node.type,
   806→      content: node.content,
   807→      children: new Map(),
   808→      created: new Date(),
   809→      modified: new Date(),
   810→      permissions: node.permissions,
   811→      size: node.size,
   812→    };
   813→
   814→    for (const [key, child] of node.children) {
   815→      clone.children.set(key, this.deepCloneNode(child, child.name));
   816→    }
   817→
   818→    return clone;
   819→  }

   821→  private calculateNodeSize(node: VirtualNode): number {
   822→    let size = node.size;
   823→    for (const [, child] of node.children) {
   824→      size += this.calculateNodeSize(child);
   825→    }
   826→    return size;
   827→  }

   829→  private matchGlob(name: string, pattern: string): boolean {
   830→    // Simple glob matching: supports * and ?
   831→    const regexStr = pattern
   832→      .replace(/[.+^${}()|[\]\\]/g, '\\$&')
   833→      .replace(/\*/g, '.*')
   834→      .replace(/\?/g, '.');
   835→    try {
   836→      const regex = new RegExp(`^${regexStr}$`);
   837→      return regex.test(name);
   838→    } catch {
   839→      return name.includes(pattern);
   840→    }
   841→  }
