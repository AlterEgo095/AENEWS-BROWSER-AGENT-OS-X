   385→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   386→    const startTime = Date.now();
   387→    const { action, ...params } = input.payload;
   388→
   389→    if (!action) {
   390→      return this.createAgentOutput(
   391→        input.taskId,
   392→        false,
   393→        null,
   394→        'Missing required parameter: action',
   395→        startTime,
   396→      );
   397→    }
   398→
   399→    const supportedActions = [
   400→      'compose',
   401→      'send',
   402→      'read',
   403→      'reply',
   404→      'forward',
   405→      'search',
   406→      'deleteEmail',
   407→      'organizeInFolder',
   408→    ];
   409→
   410→    if (!supportedActions.includes(action)) {
   411→      return this.createAgentOutput(
   412→        input.taskId,
   413→        false,
   414→        null,
   415→        `Unknown email action: ${action}. Supported: ${supportedActions.join(', ')}`,
   416→        startTime,
   417→      );
   418→    }
   419→
   420→    try {
   421→      const tool = this.getTool(action);
   422→      if (!tool) {
   423→        return this.createAgentOutput(
   424→          input.taskId,
   425→          false,
   426→          null,
   427→          `Tool not found: ${action}`,
   428→          startTime,
   429→        );
   430→      }
   431→
   432→      const result = await tool.execute(params);
   433→
   434→      // Store result in working memory for quick access
   435→      await this.storeInWorkingMemory(
   436→        `email:last:${action}`,
   437→        { params, result, timestamp: new Date() },
   438→        300000,
   439→      );
   440→
   441→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   442→    } catch (error) {
   443→      const msg = (error as Error).message;
   444→      this.logger.error(`Email execution failed for ${action}: ${msg}`);
   445→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   446→    }
   447→  }

   458→  private async compose(params: {
   459→    to: string[];
   460→    cc?: string[];
   461→    bcc?: string[];
   462→    subject: string;
   463→    body: string;
   464→    attachments?: EmailAttachment[];
   465→    priority?: 'low' | 'normal' | 'high';
   466→  }): Promise<{
   467→    emailId: string;
   468→    status: string;
   469→    composedAt: string;
   470→  }> {
   471→    const { to, cc = [], bcc = [], subject, body, attachments = [], priority = 'normal' } = params;
   472→
   473→    if (!to || !Array.isArray(to) || to.length === 0) {
   474→      throw new Error('At least one recipient is required');
   475→    }
   476→    if (!subject || typeof subject !== 'string') {
   477→      throw new Error('A valid subject is required');
   478→    }
   479→    if (!body || typeof body !== 'string') {
   480→      throw new Error('A valid body is required');
   481→    }
   482→
   483→    // Validate email addresses
   484→    for (const addr of [...to, ...cc, ...bcc]) {
   485→      if (!this.isValidEmail(addr)) {
   486→        throw new Error(`Invalid email address: ${addr}`);
   487→      }
   488→    }
   489→
   490→    const emailId = this.generateEmailId();
   491→    const email: EmailMessage = {
   492→      id: emailId,
   493→      from: 'agent@aenews.system',
   494→      to,
   495→      cc,
   496→      bcc,
   497→      subject,
   498→      body,
   499→      attachments,
   500→      folder: 'drafts',
   501→      labels: [],
   502→      isRead: true,
   503→      isImportant: false,
   504→      priority,
   505→      status: 'draft',
   506→      createdAt: new Date(),
   507→    };
   508→
   509→    this.emails.set(emailId, email);
   510→
   511→    // Add to drafts folder
   512→    this.addEmailToFolder(emailId, 'drafts');
   513→
   514→    this.logger.log(
   515→      `Composed email: ${emailId}, to=${to.join(',')}, subject="${subject.substring(0, 50)}"`,
   516→    );
   517→
   518→    return {
   519→      emailId,
   520→      status: 'draft',
   521→      composedAt: new Date().toISOString(),
   522→    };
   523→  }

   525→  private async send(params: { emailId: string; scheduleAt?: string }): Promise<{
   526→    emailId: string;
   527→    sentAt: string;
   528→    status: string;
   529→  }> {
   530→    const { emailId, scheduleAt } = params;
   531→
   532→    if (!emailId || typeof emailId !== 'string') {
   533→      throw new Error('A valid emailId is required');
   534→    }
   535→
   536→    const email = this.emails.get(emailId);
   537→    if (!email) {
   538→      throw new Error(`Email not found: ${emailId}`);
   539→    }
   540→
   541→    if (email.status === 'sent') {
   542→      throw new Error(`Email ${emailId} has already been sent`);
   543→    }
   544→
   545→    if (email.status === 'trashed' || email.status === 'deleted') {
   546→      throw new Error(`Cannot send email in ${email.status} status`);
   547→    }
   548→
   549→    // If scheduled, store the schedule but don't send now
   550→    if (scheduleAt) {
   551→      const scheduledTime = new Date(scheduleAt);
   552→      if (isNaN(scheduledTime.getTime())) {
   553→        throw new Error('Invalid scheduleAt timestamp');
   554→      }
   555→      this.logger.log(`Email ${emailId} scheduled for ${scheduleAt}`);
   556→    }
   557→
   558→    // Update email status
   559→    email.status = 'sent';
   560→    email.sentAt = new Date();
   561→
   562→    // Move from drafts to sent
   563→    this.removeEmailFromFolder(emailId, 'drafts');
   564→    this.addEmailToFolder(emailId, 'sent');
   565→
   566→    this.logger.log(`Sent email: ${emailId}, to=${email.to.join(',')}`);
   567→
   568→    return {
   569→      emailId,
   570→      sentAt: email.sentAt.toISOString(),
   571→      status: 'sent',
   572→    };
   573→  }

   575→  private async read(params: { emailId: string; markAsRead?: boolean }): Promise<{
   576→    email: EmailMessage;
   577→    readAt: string;
   578→  }> {
   579→    const { emailId, markAsRead = true } = params;
   580→
   581→    if (!emailId || typeof emailId !== 'string') {
   582→      throw new Error('A valid emailId is required');
   583→    }
   584→
   585→    const email = this.emails.get(emailId);
   586→    if (!email) {
   587→      throw new Error(`Email not found: ${emailId}`);
   588→    }
   589→
   590→    if (email.status === 'deleted') {
   591→      throw new Error(`Email ${emailId} has been permanently deleted`);
   592→    }
   593→
   594→    const readAt = new Date().toISOString();
   595→
   596→    if (markAsRead && !email.isRead) {
   597→      email.isRead = true;
   598→      email.readAt = new Date();
   599→    }
   600→
   601→    this.logger.log(`Read email: ${emailId}, markAsRead=${markAsRead}`);
   602→
   603→    return {
   604→      email: { ...email },
   605→      readAt,
   606→    };
   607→  }

   609→  private async reply(params: {
   610→    emailId: string;
   611→    body: string;
   612→    replyAll?: boolean;
   613→    attachments?: EmailAttachment[];
   614→  }): Promise<{
   615→    replyId: string;
   616→    status: string;
   617→  }> {
   618→    const { emailId, body, replyAll = false, attachments = [] } = params;
   619→
   620→    if (!emailId || typeof emailId !== 'string') {
   621→      throw new Error('A valid emailId is required');
   622→    }
   623→    if (!body || typeof body !== 'string') {
   624→      throw new Error('A valid body is required for the reply');
   625→    }
   626→
   627→    const originalEmail = this.emails.get(emailId);
   628→    if (!originalEmail) {
   629→      throw new Error(`Email not found: ${emailId}`);
   630→    }
   631→
   632→    // Build recipient list
   633→    const to = replyAll
   634→      ? [originalEmail.from, ...originalEmail.to.filter((addr) => addr !== 'agent@aenews.system')]
   635→      : [originalEmail.from];
   636→
   637→    const cc = replyAll ? originalEmail.cc : [];
   638→    const subject = originalEmail.subject.startsWith('Re:')
   639→      ? originalEmail.subject
   640→      : `Re: ${originalEmail.subject}`;
   641→
   642→    const replyId = this.generateEmailId();
   643→    const reply: EmailMessage = {
   644→      id: replyId,
   645→      from: 'agent@aenews.system',
   646→      to,
   647→      cc,
   648→      bcc: [],
   649→      subject,
   650→      body,
   651→      attachments,
   652→      folder: 'sent',
   653→      labels: [],
   654→      isRead: true,
   655→      isImportant: false,
   656→      priority: originalEmail.priority,
   657→      status: 'sent',
   658→      replyToId: emailId,
   659→      threadId: originalEmail.threadId || emailId,
   660→      createdAt: new Date(),
   661→      sentAt: new Date(),
   662→    };
   663→
   664→    this.emails.set(replyId, reply);
   665→    this.addEmailToFolder(replyId, 'sent');
   666→
   667→    this.logger.log(`Replied to email: ${emailId}, replyId=${replyId}, replyAll=${replyAll}`);
   668→
   669→    return {
   670→      replyId,
   671→      status: 'sent',

   675→  private async forward(params: { emailId: string; to: string[]; message?: string }): Promise<{
   676→    forwardedId: string;
   677→    status: string;
   678→  }> {
   679→    const { emailId, to, message = '' } = params;
   680→
   681→    if (!emailId || typeof emailId !== 'string') {
   682→      throw new Error('A valid emailId is required');
   683→    }
   684→    if (!to || !Array.isArray(to) || to.length === 0) {
   685→      throw new Error('At least one recipient is required for forwarding');
   686→    }
   687→
   688→    for (const addr of to) {
   689→      if (!this.isValidEmail(addr)) {
   690→        throw new Error(`Invalid email address: ${addr}`);
   691→      }
   692→    }
   693→
   694→    const originalEmail = this.emails.get(emailId);
   695→    if (!originalEmail) {
   696→      throw new Error(`Email not found: ${emailId}`);
   697→    }
   698→
   699→    const subject = originalEmail.subject.startsWith('Fwd:')
   700→      ? originalEmail.subject
   701→      : `Fwd: ${originalEmail.subject}`;
   702→
   703→    const forwardedBody = message
   704→      ? `${message}\n\n---------- Forwarded message ----------\nFrom: ${originalEmail.from}\nDate: ${originalEmail.createdAt.toISOString()}\nSubject: ${originalEmail.subject}\n\n${originalEmail.body}`
   705→      : `---------- Forwarded message ----------\nFrom: ${originalEmail.from}\nDate: ${originalEmail.createdAt.toISOString()}\nSubject: ${originalEmail.subject}\n\n${originalEmail.body}`;
   706→
   707→    const forwardedId = this.generateEmailId();
   708→    const forwarded: EmailMessage = {
   709→      id: forwardedId,
   710→      from: 'agent@aenews.system',
   711→      to,
   712→      cc: [],
   713→      bcc: [],
   714→      subject,
   715→      body: forwardedBody,
   716→      attachments: [...originalEmail.attachments],
   717→      folder: 'sent',
   718→      labels: [],
   719→      isRead: true,
   720→      isImportant: false,
   721→      priority: originalEmail.priority,
   722→      status: 'sent',
   723→      forwardedFromId: emailId,
   724→      threadId: originalEmail.threadId || emailId,
   725→      createdAt: new Date(),
   726→      sentAt: new Date(),
   727→    };
   728→
   729→    this.emails.set(forwardedId, forwarded);
   730→    this.addEmailToFolder(forwardedId, 'sent');
   731→
   732→    this.logger.log(`Forwarded email: ${emailId}, forwardedId=${forwardedId}, to=${to.join(',')}`);
   733→
   734→    return {
   735→      forwardedId,
   736→      status: 'sent',
   737→    };
   738→  }

   740→  private async search(params: {
   741→    query: string;
   742→    from?: string;
   743→    to?: string;
   744→    subject?: string;
   745→    folder?: string;
   746→    dateFrom?: string;
   747→    dateTo?: string;
   748→    limit?: number;
   749→    offset?: number;
   750→  }): Promise<{
   751→    results: EmailMessage[];
   752→    total: number;
   753→    hasMore: boolean;
   754→  }> {
   755→    const { query, from, to, subject, folder, dateFrom, dateTo, limit = 20, offset = 0 } = params;
   756→
   757→    if (!query || typeof query !== 'string') {
   758→      throw new Error('A valid search query is required');
   759→    }
   760→
   761→    const queryLower = query.toLowerCase();
   762→
   763→    let results = Array.from(this.emails.values()).filter((email) => {
   764→      // Skip deleted emails
   765→      if (email.status === 'deleted') return false;
   766→
   767→      // Text matching on subject and body
   768→      const matchesQuery =
   769→        email.subject.toLowerCase().includes(queryLower) ||
   770→        email.body.toLowerCase().includes(queryLower) ||
   771→        email.from.toLowerCase().includes(queryLower);
   772→
   773→      // Filter by sender
   774→      const matchesFrom = !from || email.from.toLowerCase().includes(from.toLowerCase());
   775→
   776→      // Filter by recipient
   777→      const matchesTo = !to || email.to.some((r) => r.toLowerCase().includes(to!.toLowerCase()));
   778→
   779→      // Filter by subject
   780→      const matchesSubject =
   781→        !subject || email.subject.toLowerCase().includes(subject.toLowerCase());
   782→
   783→      // Filter by folder
   784→      const matchesFolder = !folder || email.folder === folder;
   785→
   786→      // Filter by date range
   787→      let matchesDate = true;
   788→      if (dateFrom) {
   789→        const fromDate = new Date(dateFrom);
   790→        if (!isNaN(fromDate.getTime())) {
   791→          matchesDate = matchesDate && email.createdAt >= fromDate;
   792→        }
   793→      }
   794→      if (dateTo) {
   795→        const toDate = new Date(dateTo);
   796→        if (!isNaN(toDate.getTime())) {
   797→          matchesDate = matchesDate && email.createdAt <= toDate;
   798→        }
   799→      }
   800→
   801→      return (
   802→        matchesQuery && matchesFrom && matchesTo && matchesSubject && matchesFolder && matchesDate
   803→      );
   804→    });
   805→
   806→    // Sort by date, newest first
   807→    results.sort((a, b) => b.createdAt.getTime() - a.createdAt.getTime());
   808→
   809→    const total = results.length;
   810→    const hasMore = offset + limit < total;
   811→    results = results.slice(offset, offset + limit);
   812→
   813→    this.logger.log(`Email search: query="${query}", results=${total}, returned=${results.length}`);
   814→
   815→    return { results, total, hasMore };
   816→  }

   818→  private async deleteEmail(params: { emailId: string; permanent?: boolean }): Promise<{
   819→    emailId: string;
   820→    status: string;
   821→  }> {
   822→    const { emailId, permanent = false } = params;
   823→
   824→    if (!emailId || typeof emailId !== 'string') {
   825→      throw new Error('A valid emailId is required');
   826→    }
   827→
   828→    const email = this.emails.get(emailId);
   829→    if (!email) {
   830→      throw new Error(`Email not found: ${emailId}`);
   831→    }
   832→
   833→    if (permanent) {
   834→      // Permanently delete
   835→      this.removeEmailFromFolder(emailId, email.folder);
   836→      email.status = 'deleted';
   837→      email.folder = 'deleted';
   838→      this.emails.delete(emailId);
   839→
   840→      this.logger.log(`Permanently deleted email: ${emailId}`);
   841→      return { emailId, status: 'deleted' };
   842→    }
   843→
   844→    // Move to trash
   845→    this.removeEmailFromFolder(emailId, email.folder);
   846→    email.status = 'trashed';
   847→    email.folder = 'trash';
   848→    this.addEmailToFolder(emailId, 'trash');
   849→
   850→    this.logger.log(`Moved email to trash: ${emailId}`);
   851→    return { emailId, status: 'trashed' };
   852→  }

   854→  private async organizeInFolder(params: {
   855→    emailId: string;
   856→    folder?: string;
   857→    labels?: string[];
   858→    markAsRead?: boolean;
   859→    markAsImportant?: boolean;
   860→  }): Promise<{
   861→    emailId: string;
   862→    folder: string;
   863→    labels: string[];
   864→    status: string;
   865→  }> {
   866→    const { emailId, folder, labels = [], markAsRead, markAsImportant } = params;
   867→
   868→    if (!emailId || typeof emailId !== 'string') {
   869→      throw new Error('A valid emailId is required');
   870→    }
   871→
   872→    const email = this.emails.get(emailId);
   873→    if (!email) {
   874→      throw new Error(`Email not found: ${emailId}`);
   875→    }
   876→
   877→    // Move to folder if specified
   878→    if (folder) {
   879→      this.removeEmailFromFolder(emailId, email.folder);
   880→      email.folder = folder;
   881→      this.addEmailToFolder(emailId, folder);
   882→
   883→      // Create folder if it doesn't exist
   884→      if (!this.folders.has(folder)) {
   885→        this.folders.set(folder, {
   886→          name: folder,
   887→          emailIds: [emailId],
   888→          isSystem: false,
   889→        });
   890→      }
   891→    }
   892→
   893→    // Apply labels
   894→    for (const label of labels) {
   895→      if (!email.labels.includes(label)) {
   896→        email.labels.push(label);
   897→      }
   898→    }
   899→
   900→    // Mark as read
   901→    if (markAsRead !== undefined) {
   902→      email.isRead = markAsRead;
   903→      if (markAsRead) {
   904→        email.readAt = new Date();
   905→      }
   906→    }
   907→
   908→    // Mark as important
   909→    if (markAsImportant !== undefined) {
   910→      email.isImportant = markAsImportant;
   911→    }
   912→
   913→    this.logger.log(
   914→      `Organized email: ${emailId}, folder=${folder || email.folder}, labels=[${labels.join(',')}]`,
   915→    );
   916→
   917→    return {
   918→      emailId,
   919→      folder: email.folder,
   920→      labels: [...email.labels],
   921→      status: 'organized',
   922→    };
   923→  }
