     1→/**
     2→ * AENEWS Agent OS X - Auto Recovery Service
     3→ *
     4→ * When an agent fails, this service orchestrates the full recovery pipeline:
     5→ *   Health Check → Detect Failure → Restart → Restore Memory → Resume Task
     6→ *
     7→ * No human intervention required — unless auto-recovery is exhausted, in which
     8→ * case the service escalates to a human operator.
     9→ *
    10→ * Recovery strategies range from a simple restart all the way to quarantine,
    11→ * with policy-driven configuration per failure type.
    12→ */
    13→
    14→import { Injectable, Logger, OnModuleInit } from '@nestjs/common';
    15→
    16→// ─── Type Definitions ──────────────────────────────────────────────
    17→
    18→export enum FailureType {
    19→  CRASH = 'crash',
    20→  TIMEOUT = 'timeout',
    21→  OOM = 'oom',
    22→  CIRCUIT_BREAKER_OPEN = 'circuit_breaker_open',
    23→  HEALTH_CHECK_FAILED = 'health_check_failed',
    24→  UNHANDLED_EXCEPTION = 'unhandled_exception',
    25→  DEADLOCK = 'deadlock',
    26→  DEPENDENCY_FAILURE = 'dependency_failure',
    27→}
    28→
    29→export enum RecoveryStrategy {
    30→  RESTART = 'restart', // Simple restart
    31→  RESTORE_MEMORY_RESUME = 'restore_memory_resume', // Restart + restore memory + resume task
    32→  FAILOVER = 'failover', // Switch to backup agent
    33→  SCALE_OUT = 'scale_out', // Create new agent instance
    34→  DEGRADE = 'degrade', // Continue with reduced functionality
    35→  QUARANTINE = 'quarantine', // Isolate the failing agent
    36→}
    37→
    38→export enum RecoveryStatus {
    39→  DETECTED = 'detected',
    40→  ANALYZING = 'analyzing',
    41→  RECOVERING = 'recovering',
    42→  SUCCEEDED = 'succeeded',
    43→  FAILED = 'failed',
    44→  ESCALATED = 'escalated',
    45→}

    56→  maxAttempts: number;
    57→  lastAttemptAt: Date | null;
    58→  result: string | null;
    59→  taskId: string | null;
    60→  memorySnapshotId: string | null;
    61→  escalationReason: string | null;
    62→  metadata: Record<string, any>;
    63→}

   215→  // ─── Event helpers ────────────────────────────────────────────────
   216→
   217→  /**
   218→   * Register a listener for a recovery event.
   219→   * Returns an unsubscribe function.
   220→   */
   221→  on(event: string, listener: (payload: RecoveryEventPayload) => void): () => void {
   222→    if (!this.eventListeners.has(event)) {
   223→      this.eventListeners.set(event, []);
   224→    }
   225→    this.eventListeners.get(event)!.push(listener);
   226→    return () => {
   227→      const listeners = this.eventListeners.get(event);
   228→      if (listeners) {
   229→        const idx = listeners.indexOf(listener);
   230→        if (idx >= 0) listeners.splice(idx, 1);
   231→      }
   232→    };
   233→  }
   234→
   235→  private emitEvent(event: string, payload: RecoveryEventPayload): void {
   236→    const listeners = this.eventListeners.get(event);
   237→    if (listeners) {
   238→      for (const listener of listeners) {
   239→        try {
   240→          listener(payload);
   241→        } catch (err) {
   242→          this.logger.warn(`Event listener error on "${event}": ${err}`);
   243→        }
   244→      }
   245→    }
   246→  }

   341→  // ─── 2. detectFailure ─────────────────────────────────────────────
   342→
   343→  /**
   344→   * Called when a failure is detected. Creates an AutoRecoveryAction,
   345→   * determines the strategy from the matching policy, and begins the
   346→   * recovery process immediately.
   347→   *
   348→   * @returns The created AutoRecoveryAction
   349→   */
   350→  detectFailure(
   351→    agentId: string,
   352→    failureType: FailureType,
   353→    context?: RecoveryContext,
   354→  ): AutoRecoveryAction {
   355→    const policy = this.policies.get(failureType);
   356→    if (!policy) {
   357→      this.logger.error(
   358→        `No recovery policy found for failure type "${failureType}". Using default RESTART strategy.`,
   359→      );
   360→    }
   361→
   362→    const strategy = policy?.strategy ?? RecoveryStrategy.RESTART;
   363→    const maxAttempts = policy?.maxAttempts ?? 2;
   364→
   365→    // Build a human-readable description
   366→    const failureDescription = this.buildFailureDescription(agentId, failureType, context);
   367→
   368→    const actionId = this.generateActionId();
   369→
   370→    const action: AutoRecoveryAction = {
   371→      id: actionId,
   372→      agentId,
   373→      failureType,
   374→      failureDescription,
   375→      detectionTime: new Date(),
   376→      recoveryStrategy: strategy,
   377→      status: RecoveryStatus.DETECTED,
   378→      attempts: 0,
   379→      maxAttempts,
   380→      lastAttemptAt: null,
   381→      result: null,
   382→      taskId: context?.taskId ?? null,
   383→      memorySnapshotId: null,
   384→      escalationReason: null,
   385→      metadata: {
   386→        errorMessage: context?.errorMessage ?? null,
   387→        stackTrace: context?.stackTrace ?? null,
   388→        policySnapshot: policy ? { ...policy } : null,
   389→      },
   390→    };
   391→
   392→    this.actions.set(actionId, action);
   393→
   394→    this.logger.warn(
   395→      `Failure detected: agent="${agentId}" type=${failureType} strategy=${strategy} action=${actionId}`,
   396→    );
   397→
   398→    this.emitEvent(RECOVERY_STARTED, {
   399→      actionId,
   400→      agentId,
   401→      failureType,
   402→      strategy,
   403→      timestamp: Date.now(),
   404→    });
   405→
   406→    // Begin recovery immediately
   407→    this.executeRecovery(actionId);
   408→
   409→    return { ...action };
   410→  }
   411→
   412→  // ─── 3. executeRecovery ───────────────────────────────────────────
   413→
   414→  /**
   415→   * THE CORE METHOD. Execute the recovery strategy for the given action.
   416→   *
   417→   * Strategies:
   418→   *   RESTART               — Stop agent, reinitialize, start
   419→   *   RESTORE_MEMORY_RESUME — Snapshot memory, restart agent, restore memory, resume task
   420→   *   FAILOVER              — Find alternative agent with same capabilities, transfer task
   421→   *   SCALE_OUT             — Create new agent instance
   422→   *   DEGRADE               — Mark agent as degraded, reduce capabilities
   423→   *   QUARANTINE            — Isolate agent, prevent it from receiving new tasks
   424→   */
   425→  async executeRecovery(actionId: string): Promise<void> {
   426→    const action = this.actions.get(actionId);
   427→    if (!action) {
   428→      this.logger.error(`Cannot execute recovery — action "${actionId}" not found`);
   429→      return;
   430→    }
   431→
   432→    // Guard: if already succeeded, do not re-execute
   433→    if (action.status === RecoveryStatus.SUCCEEDED) {
   434→      this.logger.warn(`Action "${actionId}" already succeeded — skipping re-execution`);
   435→      return;
   436→    }
   437→
   438→    // Guard: if escalated, do not re-execute
   439→    if (action.status === RecoveryStatus.ESCALATED) {
   440→      this.logger.warn(`Action "${actionId}" is escalated — requires human intervention`);
   441→      return;
   442→    }
   443→
   444→    const policy = this.policies.get(action.failureType);
   445→    const cooldownMs = policy?.cooldownMs ?? 5_000;
   446→
   447→    // If this is a retry, apply cooldown
   448→    if (action.attempts > 0 && action.lastAttemptAt) {
   449→      const elapsed = Date.now() - action.lastAttemptAt.getTime();
   450→      if (elapsed < cooldownMs) {
   451→        const waitMs = cooldownMs - elapsed;
   452→        this.logger.log(
   453→          `Cooldown in effect for action "${actionId}" — waiting ${waitMs}ms before retry`,
   454→        );
   455→        await this.sleep(waitMs);
   456→      }
   457→    }
   458→
   459→    // Transition to ANALYZING
   460→    action.status = RecoveryStatus.ANALYZING;
   461→    this.logger.log(
   462→      `Analyzing failure for action "${actionId}": agent="${action.agentId}" type=${action.failureType}`,
   463→    );
   464→
   465→    // Transition to RECOVERING
   466→    action.status = RecoveryStatus.RECOVERING;
   467→    action.attempts++;
   468→    action.lastAttemptAt = new Date();
   469→
   470→    const recoveryStart = Date.now();
   471→
   472→    try {
   473→      switch (action.recoveryStrategy) {
   474→        case RecoveryStrategy.RESTART:
   475→          await this.executeRestart(action);
   476→          break;
   477→
   478→        case RecoveryStrategy.RESTORE_MEMORY_RESUME:
   479→          await this.executeRestoreMemoryResume(action);
   480→          break;
   481→
   482→        case RecoveryStrategy.FAILOVER:
   483→          await this.executeFailover(action);
   484→          break;
   485→
   486→        case RecoveryStrategy.SCALE_OUT:
   487→          await this.executeScaleOut(action);
   488→          break;
   489→
   490→        case RecoveryStrategy.DEGRADE:
   491→          await this.executeDegrade(action);
   492→          break;
   493→
   494→        case RecoveryStrategy.QUARANTINE:
   495→          await this.executeQuarantine(action);
   496→          break;
   497→
   498→        default:
   499→          throw new Error(`Unknown recovery strategy: ${action.recoveryStrategy}`);
   500→      }
   501→
   502→      // Record recovery duration
   503→      const duration = Date.now() - recoveryStart;
   504→      this.recoveryDurations.push(duration);
   505→      if (this.recoveryDurations.length > MAX_HISTORY_SIZE) {
   506→        this.recoveryDurations.splice(0, this.recoveryDurations.length - MAX_HISTORY_SIZE);
   507→      }
   508→
   509→      action.result = `Recovery succeeded via ${action.recoveryStrategy} in ${duration}ms`;
   510→      this.handleRecoverySuccess(actionId);
   511→    } catch (error) {
   512→      const errMsg = error instanceof Error ? error.message : String(error);
   513→      this.logger.error(
   514→        `Recovery attempt ${action.attempts}/${action.maxAttempts} failed for action "${actionId}": ${errMsg}`,
   515→      );
   516→
   517→      action.result = `Attempt ${action.attempts} failed: ${errMsg}`;
   518→      this.handleRecoveryFailure(actionId);
   519→    }
   520→  }
   521→
   522→  // ─── 4. handleRecoverySuccess ─────────────────────────────────────
   523→
   524→  /**
   525→   * Process a successful recovery. Updates the action status to SUCCEEDED,
   526→   * records the action in history, clears any transient failure state for
   527→   * the agent, and emits the RECOVERY_SUCCEEDED event.
   528→   */
   529→  handleRecoverySuccess(actionId: string): void {
   530→    const action = this.actions.get(actionId);
   531→    if (!action) {
   532→      this.logger.error(`Cannot handle recovery success — action "${actionId}" not found`);
   533→      return;
   534→    }
   535→
   536→    action.status = RecoveryStatus.SUCCEEDED;
   537→
   538→    // Clear degraded / quarantined state if the agent was recovered
   539→    if (action.recoveryStrategy !== RecoveryStrategy.QUARANTINE) {
   540→      this.degradedAgents.delete(action.agentId);
   541→    }
   542→
   543→    // Move to history
   544→    this.addToHistory(action);
   545→    this.actions.delete(actionId);
   546→
   547→    this.logger.log(
   548→      `Recovery succeeded: action="${actionId}" agent="${action.agentId}" strategy=${action.recoveryStrategy} attempts=${action.attempts}`,
   549→    );
   550→
   551→    this.emitEvent(RECOVERY_SUCCEEDED, {
   552→      actionId,
   553→      agentId: action.agentId,
   554→      failureType: action.failureType,
   555→      strategy: action.recoveryStrategy,
   556→      timestamp: Date.now(),
   557→      attempts: action.attempts,
   558→    });
   559→  }
   560→
   561→  // ─── 5. handleRecoveryFailure ─────────────────────────────────────
   562→
   563→  /**
   564→   * Process a failed recovery attempt. If the action has remaining
   565→   * attempts, schedule a retry after cooldown. If all attempts are
   566→   * exhausted, escalate to human.
   567→   */
   568→  handleRecoveryFailure(actionId: string): void {
   569→    const action = this.actions.get(actionId);
   570→    if (!action) {
   571→      this.logger.error(`Cannot handle recovery failure — action "${actionId}" not found`);
   572→      return;
   573→    }
   574→
   575→    const policy = this.policies.get(action.failureType);
   576→
   577→    if (action.attempts < action.maxAttempts) {
   578→      // Still have retries remaining — schedule another attempt
   579→      const cooldownMs = policy?.cooldownMs ?? 5_000;
   580→
   581→      this.logger.warn(
   582→        `Recovery attempt ${action.attempts}/${action.maxAttempts} failed for agent "${action.agentId}". ` +
   583→          `Retrying after ${cooldownMs}ms cooldown...`,
   584→      );
   585→
   586→      action.status = RecoveryStatus.RECOVERING;
   587→
   588→      // Schedule retry (fire-and-forget — in production use a scheduler)
   589→      setTimeout(() => {
   590→        this.executeRecovery(actionId).catch((err) => {
   591→          this.logger.error(`Scheduled retry failed for action "${actionId}": ${err}`);
   592→        });
   593→      }, cooldownMs);
   594→    } else {
   595→      // All attempts exhausted — mark as failed and escalate
   596→      action.status = RecoveryStatus.FAILED;
   597→
   598→      this.logger.error(
   599→        `All ${action.maxAttempts} recovery attempts exhausted for agent "${action.agentId}" ` +
   600→          `(failure: ${action.failureType}). Escalating to human.`,
   601→      );
   602→
   603→      // Check if we should escalate before the escalation threshold
   604→      const escalationAfter = policy?.escalationAfterAttempts ?? action.maxAttempts;
   605→      if (action.attempts >= escalationAfter) {
   606→        this.escalate(actionId);
   607→      }
   608→    }
   609→  }
   610→
   611→  // ─── 6. escalate ──────────────────────────────────────────────────
   612→
   613→  /**
   614→   * Escalate to a human when auto-recovery fails. Creates an escalation
   615→   * record, marks the action as ESCALATED, and emits the
   616→   * RECOVERY_ESCALATED event.
   617→   */
   618→  escalate(actionId: string): EscalationRecord {
   619→    const action = this.actions.get(actionId);
   620→    if (!action) {
   621→      throw new Error(`Cannot escalate — action "${actionId}" not found`);
   622→    }
   623→
   624→    action.status = RecoveryStatus.ESCALATED;
   625→
   626→    const escalationReason =
   627→      `Auto-recovery failed after ${action.attempts} attempts. ` +
   628→      `Failure type: ${action.failureType}. Strategy: ${action.recoveryStrategy}. ` +
   629→      `Last result: ${action.result ?? 'unknown'}`;
   630→
   631→    action.escalationReason = escalationReason;
   632→
   633→    const escalation: EscalationRecord = {
   634→      id: this.generateEscalationId(),
   635→      actionId,
   636→      agentId: action.agentId,
   637→      failureType: action.failureType,
   638→      reason: escalationReason,
   639→      timestamp: new Date(),
   640→      acknowledged: false,
   641→      acknowledgedBy: null,
   642→      acknowledgedAt: null,
   643→    };
   644→
   645→    this.escalations.push(escalation);
   646→
   647→    // Bound escalation records
   648→    if (this.escalations.length > MAX_ESCALATION_RECORDS) {
   649→      this.escalations.splice(0, this.escalations.length - MAX_ESCALATION_RECORDS);
   650→    }
   651→
   652→    // Move action to history
   653→    this.addToHistory(action);
   654→    this.actions.delete(actionId);
   655→
   656→    this.logger.error(
   657→      `ESCALATION: action="${actionId}" agent="${action.agentId}" — ${escalationReason}`,
   658→    );
   659→
   660→    this.emitEvent(RECOVERY_ESCALATED, {
   661→      actionId,
   662→      agentId: action.agentId,
   663→      failureType: action.failureType,
   664→      strategy: action.recoveryStrategy,
   665→      timestamp: Date.now(),
   666→      escalationId: escalation.id,
   667→      reason: escalationReason,
   668→    });
   669→
   670→    return { ...escalation };
   671→  }

   726→  // ─── 11. updatePolicy ─────────────────────────────────────────────
   727→
   728→  /**
   729→   * Update a recovery policy for a given failure type.
   730→   * Validates the policy before applying.
   731→   */
   732→  updatePolicy(failureType: FailureType, policy: Partial<RecoveryPolicy>): RecoveryPolicy {
   733→    const existing = this.policies.get(failureType);
   734→
   735→    if (!existing) {
   736→      // Creating a new policy for this failure type
   737→      if (!policy.strategy) {
   738→        throw new Error(`New policy for "${failureType}" must specify a strategy`);
   739→      }
   740→
   741→      const newPolicy: RecoveryPolicy = {
   742→        failureType,
   743→        strategy: policy.strategy,
   744→        maxAttempts: policy.maxAttempts ?? 3,
   745→        cooldownMs: policy.cooldownMs ?? 5_000,
   746→        escalationAfterAttempts: policy.escalationAfterAttempts ?? 3,
   747→        autoRestart: policy.autoRestart ?? true,
   748→        preserveMemory: policy.preserveMemory ?? false,
   749→      };
   750→
   751→      this.policies.set(failureType, newPolicy);
   752→      this.logger.log(`Created new recovery policy for failure type "${failureType}"`);
   753→
   754→      return { ...newPolicy };
   755→    }
   756→
   757→    // Merge updates into the existing policy
   758→    const updated: RecoveryPolicy = {
   759→      ...existing,
   760→      ...policy,
   761→      failureType, // failureType is the identity key — never change it
   762→    };
   763→
   764→    // Validate
   765→    if (!Object.values(RecoveryStrategy).includes(updated.strategy)) {
   766→      throw new Error(`Invalid recovery strategy: ${updated.strategy}`);
   767→    }
   768→    if (updated.maxAttempts < 1) {
   769→      throw new Error('maxAttempts must be >= 1');
   770→    }
   771→    if (updated.cooldownMs < 0) {
   772→      throw new Error('cooldownMs must be >= 0');
   773→    }
   774→    if (updated.escalationAfterAttempts < 1) {
   775→      throw new Error('escalationAfterAttempts must be >= 1');
   776→    }
   777→
   778→    this.policies.set(failureType, updated);
   779→    this.logger.log(
   780→      `Updated recovery policy for failure type "${failureType}": strategy=${updated.strategy} maxAttempts=${updated.maxAttempts}`,
   781→    );
   782→
   783→    return { ...updated };
   784→  }

   786→  // ─── 12. getRecoveryStats ─────────────────────────────────────────
   787→
   788→  /**
   789→   * Compute and return recovery statistics:
   790→   *   - total recoveries
   791→   *   - by failure type, strategy, and status
   792→   *   - success rate
   793→   *   - average recovery time
   794→   *   - total escalations
   795→   *   - currently active count
   796→   */
   797→  getRecoveryStats(): RecoveryStats {
   798→    const allActions = [...this.history, ...this.actions.values()];
   799→
   800→    const byFailureType: Record<string, number> = {};
   801→    const byStrategy: Record<string, number> = {};
   802→    const byStatus: Record<string, number> = {};
   803→
   804→    let succeeded = 0;
   805→    const total = allActions.length;
   806→
   807→    for (const action of allActions) {
   808→      byFailureType[action.failureType] = (byFailureType[action.failureType] ?? 0) + 1;
   809→      byStrategy[action.recoveryStrategy] = (byStrategy[action.recoveryStrategy] ?? 0) + 1;
   810→      byStatus[action.status] = (byStatus[action.status] ?? 0) + 1;
   811→
   812→      if (action.status === RecoveryStatus.SUCCEEDED) {
   813→        succeeded++;
   814→      }
   815→    }
   816→
   817→    const successRate = total > 0 ? succeeded / total : 0;
   818→
   819→    const averageRecoveryTimeMs =
   820→      this.recoveryDurations.length > 0
   821→        ? this.recoveryDurations.reduce((sum, d) => sum + d, 0) / this.recoveryDurations.length
   822→        : 0;
   823→
   824→    return {
   825→      totalRecoveries: total,
   826→      byFailureType,
   827→      byStrategy,
   828→      byStatus,
   829→      successRate,
   830→      averageRecoveryTimeMs,
   831→      totalEscalations: this.escalations.length,
   832→      activeRecoveryCount: this.actions.size,
   833→    };
   834→  }

   836→  // ─── 13. performHealthChecks ──────────────────────────────────────
   837→
   838→  /**
   839→   * Proactively check agent health. Detect issues before they become
   840→   * failures. If an agent is unhealthy, preemptively start recovery.
   841→   *
   842→   * @returns Array of health statuses for all known agents
   843→   */
   844→  performHealthChecks(): AgentHealthStatus[] {
   845→    const results: AgentHealthStatus[] = [];
   846→
   847→    for (const [agentId, runningState] of this.agentRunningState.entries()) {
   848→      const status = this.checkAgentHealth(agentId, runningState);
   849→      this.healthStatuses.set(agentId, status);
   850→      results.push(status);
   851→
   852→      // If unhealthy and not already being recovered, preemptively start recovery
   853→      if (!status.healthy) {
   854→        const hasActiveRecovery = [...this.actions.values()].some((a) => a.agentId === agentId);
   855→
   856→        if (!hasActiveRecovery) {
   857→          this.logger.warn(
   858→            `Preemptive recovery triggered: agent="${agentId}" is unhealthy — ${status.issues.join(', ')}`,
   859→          );
   860→
   861→          this.emitEvent(HEALTH_CHECK_FAILED_EVENT, {
   862→            actionId: '',
   863→            agentId,
   864→            failureType: FailureType.HEALTH_CHECK_FAILED,
   865→            strategy: RecoveryStrategy.RESTART,
   866→            timestamp: Date.now(),
   867→            issues: status.issues,
   868→          });
   869→
   870→          this.detectFailure(agentId, FailureType.HEALTH_CHECK_FAILED, {
   871→            agentId,
   872→            failureType: FailureType.HEALTH_CHECK_FAILED,
   873→            errorMessage: `Health check failed: ${status.issues.join('; ')}`,
   874→          });
   875→        }
   876→      }
   877→    }
   878→
   879→    return results;
   880→  }

   882→  // ─── 14. snapshotAgentMemory ──────────────────────────────────────
   883→
   884→  /**
   885→   * Take a snapshot of agent memory for recovery purposes.
   886→   * Returns the snapshot ID. Snapshots are bounded per agent
   887→   * (oldest are pruned when the limit is exceeded).
   888→   */
   889→  snapshotAgentMemory(agentId: string): string {
   890→    const snapshotId = this.generateSnapshotId();
   891→
   892→    // Retrieve the agent's current memory state
   893→    // In production this would call the agent's memory export API
   894→    const memoryState = this.agentRunningState.get(agentId) ?? {};
   895→
   896→    // Find the current task if any
   897→    const activeAction = [...this.actions.values()].find((a) => a.agentId === agentId);
   898→
   899→    const snapshot: MemorySnapshot = {
   900→      id: snapshotId,
   901→      agentId,
   902→      timestamp: new Date(),
   903→      state: memoryState,
   904→      taskId: activeAction?.taskId ?? null,
   905→      metadata: {
   906→        agentRunningState: this.agentRunningState.get(agentId) ?? 'unknown',
   907→        isDegraded: this.degradedAgents.has(agentId),
   908→        isQuarantined: this.quarantinedAgents.has(agentId),
   909→      },
   910→    };
   911→
   912→    this.snapshots.set(snapshotId, snapshot);
   913→
   914→    // Track per-agent snapshots
   915→    if (!this.agentSnapshots.has(agentId)) {
   916→      this.agentSnapshots.set(agentId, []);
   917→    }
   918→    const agentSnapshotList = this.agentSnapshots.get(agentId)!;
   919→    agentSnapshotList.push(snapshotId);
   920→
   921→    // Prune oldest snapshots if over the per-agent limit
   922→    while (agentSnapshotList.length > MAX_SNAPSHOTS_PER_AGENT) {
   923→      const oldestId = agentSnapshotList.shift()!;
   924→      this.snapshots.delete(oldestId);
   925→      this.logger.debug(`Pruned old snapshot "${oldestId}" for agent "${agentId}"`);
   926→    }
   927→
   928→    this.logger.log(`Memory snapshot taken: snapshot="${snapshotId}" agent="${agentId}"`);
   929→
   930→    this.emitEvent(MEMORY_SNAPSHOT_TAKEN, {
   931→      actionId: '',
   932→      agentId,
   933→      failureType: FailureType.CRASH,
   934→      strategy: RecoveryStrategy.RESTORE_MEMORY_RESUME,
   935→      timestamp: Date.now(),
   936→      snapshotId,
   937→    });
   938→
   939→    return snapshotId;
   940→  }

   942→  // ─── 15. restoreAgentMemory ───────────────────────────────────────
   943→
   944→  /**
   945→   * Restore agent memory from a snapshot. Returns true if the snapshot
   946→   * was found and restoration succeeded.
   947→   */
   948→  restoreAgentMemory(agentId: string, snapshotId: string): boolean {
   949→    const snapshot = this.snapshots.get(snapshotId);
   950→    if (!snapshot) {
   951→      this.logger.error(`Cannot restore memory — snapshot "${snapshotId}" not found`);
   952→      return false;
   953→    }
   954→
   955→    if (snapshot.agentId !== agentId) {
   956→      this.logger.error(
   957→        `Cannot restore memory — snapshot "${snapshotId}" belongs to agent "${snapshot.agentId}", not "${agentId}"`,
   958→      );
   959→      return false;
   960→    }
   961→
   962→    // In production, this would call the agent's memory import API
   963→    // and validate the state is compatible with the agent's current version
   964→    this.agentRunningState.set(agentId, 'running');
   965→
   966→    this.logger.log(
   967→      `Memory restored: snapshot="${snapshotId}" agent="${agentId}" (snapshot age: ${Date.now() - snapshot.timestamp.getTime()}ms)`,
   968→    );
   969→
   970→    this.emitEvent(MEMORY_RESTORED, {
   971→      actionId: '',
   972→      agentId,
   973→      failureType: FailureType.CRASH,
   974→      strategy: RecoveryStrategy.RESTORE_MEMORY_RESUME,
   975→      timestamp: Date.now(),
   976→      snapshotId,
   977→    });
   978→
   979→    return true;
   980→  }

  1073→  /**
  1074→   * RESTART: Stop the agent, reinitialize it, and start it again.
  1075→   */
  1076→  private async executeRestart(action: AutoRecoveryAction): Promise<void> {

  1088→    await this.sleep(300); // Simulate initialization
  1089→
  1090→    // Start
  1091→    this.logger.log(`RESTART: Starting agent "${agentId}"...`);
  1092→    this.agentRunningState.set(agentId, 'running');
  1093→

  1100→  /**
  1101→   * RESTORE_MEMORY_RESUME: Take a memory snapshot, restart the agent,
  1102→   * restore its memory, and resume the interrupted task.
  1103→   */
  1104→  private async executeRestoreMemoryResume(action: AutoRecoveryAction): Promise<void> {
  1105→    const { agentId } = action;
  1106→
  1107→    // Step 1: Snapshot current memory before restart
  1108→    this.logger.log(`RESTORE_MEMORY_RESUME: Snapshotting memory for agent "${agentId}"...`);
  1109→    const snapshotId = this.snapshotAgentMemory(agentId);
  1110→    action.memorySnapshotId = snapshotId;
  1111→
  1112→    // Step 2: Restart the agent
  1113→    this.logger.log(`RESTORE_MEMORY_RESUME: Restarting agent "${agentId}"...`);
  1114→    this.agentRunningState.set(agentId, 'stopped');
  1115→    await this.sleep(500);
  1116→
  1117→    this.agentRunningState.set(agentId, 'initializing');
  1118→    await this.sleep(300);
  1119→
  1120→    this.agentRunningState.set(agentId, 'running');
  1121→
  1122→    // Step 3: Restore memory from snapshot
  1123→    this.logger.log(
  1124→      `RESTORE_MEMORY_RESUME: Restoring memory for agent "${agentId}" from snapshot "${snapshotId}"...`,
  1125→    );
  1126→    const restored = this.restoreAgentMemory(agentId, snapshotId);
  1127→    if (!restored) {
  1128→      throw new Error(`Failed to restore memory from snapshot "${snapshotId}"`);
  1129→    }
  1130→
  1131→    // Step 4: Resume interrupted task
  1132→    if (action.taskId) {
  1133→      this.logger.log(
  1134→        `RESTORE_MEMORY_RESUME: Resuming task "${action.taskId}" for agent "${agentId}"...`,
  1135→      );
  1136→      // In production, this would call the task scheduler to resume
  1137→      action.metadata.taskResumed = true;
  1138→      action.metadata.resumedAt = new Date().toISOString();
  1139→    } else {
  1140→      this.logger.log(`RESTORE_MEMORY_RESUME: No task to resume for agent "${agentId}"`);
  1141→    }
  1142→
  1143→    this.logger.log(`RESTORE_MEMORY_RESUME: Agent "${agentId}" recovered successfully`);
  1144→  }

  1146→  /**
  1147→   * FAILOVER: Find an alternative agent with the same capabilities
  1148→   * and transfer the task to it.
  1149→   */
  1150→  private async executeFailover(action: AutoRecoveryAction): Promise<void> {
  1151→    const { agentId } = action;
  1152→    const capabilities = this.agentCapabilities.get(agentId) ?? [];
  1153→
  1154→    this.logger.log(
  1155→      `FAILOVER: Looking for alternative agent for "${agentId}" with capabilities: [${capabilities.join(', ')}]`,
  1156→    );
  1157→
  1158→    // Find an agent with matching capabilities that is running and not degraded/quarantined
  1159→    let failoverTarget: string | null = null;
  1160→
  1161→    for (const [candidateId, candidateCaps] of this.agentCapabilities.entries()) {
  1162→      if (candidateId === agentId) continue;
  1163→      if (this.quarantinedAgents.has(candidateId)) continue;
  1164→      if (this.degradedAgents.has(candidateId)) continue;
  1165→      if (this.agentRunningState.get(candidateId) !== 'running') continue;
  1166→
  1167→      // Check if the candidate has all required capabilities
  1168→      const hasAllCapabilities = capabilities.every((cap) => candidateCaps.includes(cap));
  1169→
  1170→      if (hasAllCapabilities) {
  1171→        failoverTarget = candidateId;
  1172→        break;
  1173→      }
  1174→    }
  1175→
  1176→    if (!failoverTarget) {
  1177→      // If no exact match, try to find an agent with at least some matching capabilities
  1178→      for (const [candidateId, candidateCaps] of this.agentCapabilities.entries()) {
  1179→        if (candidateId === agentId) continue;
  1180→        if (this.quarantinedAgents.has(candidateId)) continue;
  1181→        if (this.agentRunningState.get(candidateId) !== 'running') continue;
  1182→
  1183→        const matchingCaps = capabilities.filter((cap) => candidateCaps.includes(cap));
  1184→
  1185→        if (matchingCaps.length > 0) {
  1186→          failoverTarget = candidateId;
  1187→          break;
  1188→        }
  1189→      }
  1190→    }
  1191→
  1192→    if (!failoverTarget) {
  1193→      throw new Error(
  1194→        `No suitable failover target found for agent "${agentId}" with capabilities: [${capabilities.join(', ')}]`,
  1195→      );
  1196→    }
  1197→
  1198→    this.logger.log(`FAILOVER: Transferring task from "${agentId}" to "${failoverTarget}"`);
  1199→
  1200→    // Mark the original agent as stopped
  1201→    this.agentRunningState.set(agentId, 'stopped');
  1202→
  1203→    // Transfer the task to the failover target
  1204→    if (action.taskId) {
  1205→      this.logger.log(
  1206→        `FAILOVER: Task "${action.taskId}" transferred from "${agentId}" to "${failoverTarget}"`,
  1207→      );
  1208→      action.metadata.failoverTarget = failoverTarget;
  1209→      action.metadata.taskTransferred = true;
  1210→      action.metadata.transferredAt = new Date().toISOString();
  1211→    }
  1212→
  1213→    // Take a memory snapshot of the original agent for the new agent
  1214→    // (so the failover target has context about what happened)
  1215→    const snapshotId = this.snapshotAgentMemory(agentId);
  1216→    action.memorySnapshotId = snapshotId;
  1217→
  1218→    this.logger.log(`FAILOVER: Agent "${agentId}" failed over to "${failoverTarget}" successfully`);
  1219→  }

  1221→  /**
  1222→   * SCALE_OUT: Create a new agent instance to handle the load
  1223→   * while the failing agent recovers.
  1224→   */
  1225→  private async executeScaleOut(action: AutoRecoveryAction): Promise<void> {
  1226→    const { agentId } = action;
  1227→    const capabilities = this.agentCapabilities.get(agentId) ?? [];
  1228→
  1229→    const newAgentId = `${agentId}-clone-${Date.now()}`;
  1230→
  1231→    this.logger.log(
  1232→      `SCALE_OUT: Creating new agent instance "${newAgentId}" (clone of "${agentId}")`,
  1233→    );
  1234→
  1235→    // Register the new agent with the same capabilities
  1236→    this.registerAgent(newAgentId, [...capabilities]);
  1237→
  1238→    // Simulate the time it takes to spin up a new instance
  1239→    await this.sleep(1000);
  1240→
  1241→    this.logger.log(
  1242→      `SCALE_OUT: New agent "${newAgentId}" is running. Original agent "${agentId}" remains stopped.`,
  1243→    );
  1244→
  1245→    // Keep the original agent stopped
  1246→    this.agentRunningState.set(agentId, 'stopped');
  1247→
  1248→    // Transfer the task
  1249→    if (action.taskId) {
  1250→      this.logger.log(
  1251→        `SCALE_OUT: Task "${action.taskId}" reassigned from "${agentId}" to "${newAgentId}"`,
  1252→      );
  1253→      action.metadata.scaledOutAgentId = newAgentId;
  1254→      action.metadata.taskTransferred = true;
  1255→      action.metadata.transferredAt = new Date().toISOString();
  1256→    }
  1257→  }

  1260→   * DEGRADE: Mark the agent as degraded, reducing its capabilities.
  1261→   * It continues running but with limited functionality.
  1262→   */
  1263→  private async executeDegrade(action: AutoRecoveryAction): Promise<void> {
  1264→    const { agentId } = action;
  1265→
  1266→    this.logger.log(`DEGRADE: Marking agent "${agentId}" as degraded — reducing capabilities`);
  1267→
  1268→    this.degradedAgents.add(agentId);
  1269→
  1270→    // Reduce capabilities — keep only essential ones
  1271→    const currentCaps = this.agentCapabilities.get(agentId) ?? [];
  1272→    if (currentCaps.length > 0) {
  1273→      // Keep at most half of the capabilities (the first ones, assumed most critical)
  1274→      const reducedCaps = currentCaps.slice(0, Math.max(1, Math.floor(currentCaps.length / 2)));
  1275→      this.agentCapabilities.set(agentId, reducedCaps);
  1276→
  1277→      this.logger.log(
  1278→        `DEGRADE: Agent "${agentId}" capabilities reduced from [${currentCaps.join(', ')}] to [${reducedCaps.join(', ')}]`,
  1279→      );
  1280→    }
  1281→
  1282→    // Agent remains running but in a degraded state
  1283→    action.metadata.degradedCapabilities = this.agentCapabilities.get(agentId);
  1284→
  1285→    this.logger.log(`DEGRADE: Agent "${agentId}" is now running in degraded mode`);
  1286→  }

  1289→   * QUARANTINE: Isolate the agent so it cannot receive new tasks
  1290→   * or interact with other agents. This is a safety measure for
  1291→   * agents that are exhibiting dangerous or erratic behaviour.
  1292→   */
  1293→  private async executeQuarantine(action: AutoRecoveryAction): Promise<void> {
  1294→    const { agentId } = action;
  1295→
  1296→    this.logger.warn(`QUARANTINE: Isolating agent "${agentId}" — preventing new task assignments`);
  1297→
  1298→    this.quarantinedAgents.add(agentId);
  1299→    this.degradedAgents.delete(agentId);
  1300→
  1301→    // Snapshot memory before quarantine for forensic analysis
  1302→    const snapshotId = this.snapshotAgentMemory(agentId);
  1303→    action.memorySnapshotId = snapshotId;
  1304→
  1305→    // The agent is NOT stopped — it's isolated but still running
  1306→    // so we can observe its behaviour and potentially recover it later
  1307→    action.metadata.quarantinedAt = new Date().toISOString();
  1308→    action.metadata.memorySnapshotForForensics = snapshotId;
  1309→
  1310→    this.logger.warn(
  1311→      `QUARANTINE: Agent "${agentId}" is now quarantined. Memory snapshot "${snapshotId}" saved for forensics.`,
  1312→    );
  1313→  }

  1320→  private buildFailureDescription(
  1321→    agentId: string,

  1323→    context?: RecoveryContext,
  1324→  ): string {
  1325→    const descriptions: Record<FailureType, string> = {

  1333→      [FailureType.DEPENDENCY_FAILURE]: `Agent "${agentId}" experienced a dependency failure`,

  1335→
  1336→    let description = descriptions[failureType] ?? `Agent "${agentId}" failed with ${failureType}`;

  1345→  /**
  1346→   * Perform a health check on a single agent.
  1347→   * In production this would call the agent's health endpoint.
  1348→   */
  1349→  private checkAgentHealth(agentId: string, runningState: string): AgentHealthStatus {
  1350→    const issues: string[] = [];
  1351→    let healthy = true;
  1352→
  1353→    // Check if agent is running
  1354→    if (runningState !== 'running') {
  1355→      issues.push(`Agent is not running (state: ${runningState})`);
  1356→      healthy = false;
  1357→    }
  1358→
  1359→    // Check if agent is degraded
  1360→    if (this.degradedAgents.has(agentId)) {
  1361→      issues.push('Agent is in degraded mode');
  1362→      // Degraded agents are still "healthy" — just limited
  1363→    }
  1364→
  1365→    // Check if agent is quarantined
  1366→    if (this.quarantinedAgents.has(agentId)) {
  1367→      issues.push('Agent is quarantined');
  1368→      healthy = false;
  1369→    }
  1370→
  1371→    // Simulate metric collection (in production, these would be real metrics)
  1372→    const metrics: AgentHealthStatus['metrics'] = {
  1373→      responseTimeMs: Math.random() * 500,
  1374→      errorRate: Math.random() * 0.1,
  1375→      memoryUsageMb: 128 + Math.random() * 512,
  1376→      cpuUsagePercent: Math.random() * 100,
  1377→      taskQueueDepth: Math.floor(Math.random() * 50),
  1378→    };
  1379→
  1380→    // Check for concerning metric thresholds
  1381→    if (metrics.errorRate > 0.05) {
  1382→      issues.push(`High error rate: ${(metrics.errorRate * 100).toFixed(1)}%`);
  1383→      healthy = false;
  1384→    }
  1385→    if (metrics.cpuUsagePercent > 90) {
  1386→      issues.push(`High CPU usage: ${metrics.cpuUsagePercent.toFixed(1)}%`);
  1387→      healthy = false;
  1388→    }
  1389→    if (metrics.memoryUsageMb > 512) {
  1390→      issues.push(`High memory usage: ${metrics.memoryUsageMb.toFixed(0)}MB`);
  1391→      healthy = false;
  1392→    }
  1393→
  1394→    return {
  1395→      agentId,
  1396→      healthy,
  1397→      lastChecked: new Date(),
  1398→      issues,
  1399→      metrics,
  1400→    };
  1401→  }
