   368→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   369→    const startTime = Date.now();
   370→    const { action, ...params } = input.payload;
   371→
   372→    if (!action) {
   373→      return this.createAgentOutput(
   374→        input.taskId,
   375→        false,
   376→        null,
   377→        'Missing required parameter: action',
   378→        startTime,
   379→      );
   380→    }
   381→
   382→    const supportedActions = [
   383→      'searchLogs',
   384→      'aggregateLogs',
   385→      'createLogAlert',
   386→      'exportLogs',
   387→      'analyzePatterns',
   388→      'setRetentionPolicy',
   389→    ];
   390→
   391→    if (!supportedActions.includes(action)) {
   392→      return this.createAgentOutput(
   393→        input.taskId,
   394→        false,
   395→        null,
   396→        `Unknown logging action: ${action}. Supported: ${supportedActions.join(', ')}`,
   397→        startTime,
   398→      );
   399→    }
   400→
   401→    try {
   402→      const tool = this.getTool(action);
   403→      if (!tool) {
   404→        return this.createAgentOutput(
   405→          input.taskId,
   406→          false,
   407→          null,
   408→          `Tool not found: ${action}`,
   409→          startTime,
   410→        );
   411→      }
   412→
   413→      const result = await tool.execute(params);
   414→
   415→      await this.storeInWorkingMemory(
   416→        `logging:last:${action}`,
   417→        { params, result, timestamp: new Date() },
   418→        300000,
   419→      );
   420→
   421→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   422→    } catch (error) {
   423→      const msg = (error as Error).message;
   424→      this.logger.error(`Logging execution failed for ${action}: ${msg}`);
   425→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   426→    }
   427→  }

   437→  private async searchLogs(params: {
   438→    query: string;
   439→    service?: string;
   440→    level?: string;
   441→    timeRange?: string;
   442→    limit?: number;
   443→    offset?: number;
   444→  }): Promise<{
   445→    query: string;
   446→    total: number;
   447→    logs: LogEntry[];
   448→    hasMore: boolean;
   449→    searchTimeMs: number;
   450→  }> {
   451→    const { query, service, level, timeRange = '1h', limit = 100, offset = 0 } = params;
   452→
   453→    if (!query || typeof query !== 'string') {
   454→      throw new Error('Search query is required');
   455→    }
   456→    if (limit < 1 || limit > 1000) {
   457→      throw new Error('Limit must be between 1 and 1000');
   458→    }
   459→
   460→    const searchStart = Date.now();
   461→
   462→    // Generate simulated log entries matching the query
   463→    const totalCount = Math.floor(Math.random() * 5000) + 50;
   464→    const logs: LogEntry[] = [];
   465→
   466→    const effectiveLimit = Math.min(limit, totalCount - offset);
   467→    for (let i = 0; i < effectiveLimit; i++) {
   468→      const logService =
   469→        service ||
   470→        this.simulatedServices[Math.floor(Math.random() * this.simulatedServices.length)];
   471→      const logLevel =
   472→        level || this.simulatedLevels[Math.floor(Math.random() * this.simulatedLevels.length)];
   473→      const message =
   474→        this.simulatedMessages[Math.floor(Math.random() * this.simulatedMessages.length)];
   475→
   476→      logs.push({
   477→        id: `log-${offset + i + 1}`,
   478→        timestamp: new Date(
   479→          Date.now() - Math.random() * this.timeRangeToMs(timeRange),
   480→        ).toISOString(),
   481→        service: logService,
   482→        level: logLevel,
   483→        message: query === '*' ? message : `${message} [matched: "${query}"]`,
   484→        host: `host-${Math.floor(Math.random() * 10) + 1}.dc1`,
   485→        traceId: Math.random() > 0.5 ? this.generateTraceId() : undefined,
   486→        metadata:
   487→          Math.random() > 0.7
   488→            ? {
   489→                requestId: this.generateTraceId(),
   490→                userId: `user-${Math.floor(Math.random() * 1000)}`,
   491→              }
   492→            : undefined,
   493→      });
   494→    }
   495→
   496→    const searchTimeMs = Date.now() - searchStart;
   497→    const hasMore = offset + effectiveLimit < totalCount;
   498→
   499→    this.logger.log(
   500→      `searchLogs: "${query}", ${totalCount} total, ${logs.length} returned, ${searchTimeMs}ms`,
   501→    );
   502→
   503→    return { query, total: totalCount, logs, hasMore, searchTimeMs };
   504→  }

   506→  private async aggregateLogs(params: {
   507→    service?: string;
   508→    field: string;
   509→    aggregation?: string;
   510→    timeRange?: string;
   511→    interval?: string;
   512→  }): Promise<{
   513→    field: string;
   514→    aggregation: string;
   515→    buckets: Array<{ key: string; docCount: number; percentage?: number }>;
   516→    totalDocuments: number;
   517→    timeRange: string;
   518→  }> {
   519→    const { service, field, aggregation = 'terms', timeRange = '24h', interval = '1h' } = params;
   520→
   521→    if (!field || typeof field !== 'string') {
   522→      throw new Error('Aggregation field is required');
   523→    }
   524→
   525→    // Generate aggregated data based on field type
   526→    const buckets: Array<{ key: string; docCount: number; percentage?: number }> = [];
   527→    const totalDocuments = Math.floor(Math.random() * 500000) + 10000;
   528→
   529→    if (field === 'level') {
   530→      const levels = [
   531→        { key: 'info', weight: 0.55 },
   532→        { key: 'debug', weight: 0.2 },
   533→        { key: 'warn', weight: 0.15 },
   534→        { key: 'error', weight: 0.08 },
   535→        { key: 'fatal', weight: 0.02 },
   536→      ];
   537→      for (const l of levels) {
   538→        const docCount = Math.round(totalDocuments * l.weight * (0.9 + Math.random() * 0.2));
   539→        buckets.push({ key: l.key, docCount, percentage: Math.round(l.weight * 10000) / 100 });
   540→      }
   541→    } else if (field === 'service') {
   542→      for (const svc of this.simulatedServices) {
   543→        const docCount = Math.round(
   544→          (totalDocuments / this.simulatedServices.length) * (0.8 + Math.random() * 0.4),
   545→        );
   546→        buckets.push({ key: svc, docCount });
   547→      }
   548→    } else if (field === 'host') {
   549→      for (let i = 1; i <= 10; i++) {
   550→        buckets.push({
   551→          key: `host-${i}.dc1`,
   552→          docCount: Math.round((totalDocuments / 10) * (0.7 + Math.random() * 0.6)),
   553→        });
   554→      }
   555→    } else {
   556→      // Time-based buckets
   557→      const intervals =
   558→        interval === '1m'
   559→          ? 60
   560→          : interval === '5m'
   561→            ? 288
   562→            : interval === '15m'
   563→              ? 96
   564→              : interval === '1h'
   565→                ? 24
   566→                : 30;
   567→      for (let i = 0; i < intervals; i++) {
   568→        const bucketTime = new Date(Date.now() - (intervals - i) * this.intervalToMs(interval));
   569→        buckets.push({
   570→          key: bucketTime.toISOString(),
   571→          docCount: Math.round((totalDocuments / intervals) * (0.5 + Math.random())),
   572→        });
   573→      }
   574→    }
   575→
   576→    this.logger.log(
   577→      `aggregateLogs: ${field} by ${aggregation}, ${buckets.length} buckets, ${totalDocuments} docs`,
   578→    );
   579→
   580→    return { field, aggregation, buckets, totalDocuments, timeRange };
   581→  }

   583→  private async createLogAlert(params: {
   584→    name: string;
   585→    query: string;
   586→    threshold: number;
   587→    timeWindow?: string;
   588→    severity?: string;
   589→    channels?: string[];
   590→  }): Promise<{
   591→    alertId: string;
   592→    name: string;
   593→    query: string;
   594→    threshold: number;
   595→    created: boolean;
   596→  }> {
   597→    const {
   598→      name,
   599→      query,
   600→      threshold,
   601→      timeWindow = '5m',
   602→      severity = 'warning',
   603→      channels = ['email', 'slack'],
   604→    } = params;
   605→
   606→    if (!name || typeof name !== 'string') {
   607→      throw new Error('Alert name is required');
   608→    }
   609→    if (!query || typeof query !== 'string') {
   610→      throw new Error('Query is required');
   611→    }
   612→    if (threshold < 1) {
   613→      throw new Error('Threshold must be at least 1');
   614→    }
   615→
   616→    this.logAlertCounter++;
   617→    const alertId = `log-alert-${this.logAlertCounter}-${Date.now()}`;
   618→
   619→    this.logAlerts.set(alertId, {
   620→      id: alertId,
   621→      name,
   622→      query,
   623→      threshold,
   624→      timeWindow,
   625→      severity,
   626→      channels,
   627→      enabled: true,
   628→      createdAt: new Date(),
   629→    });
   630→
   631→    this.logger.log(
   632→      `Created log alert: ${name} [${alertId}], threshold: ${threshold} in ${timeWindow}`,
   633→    );
   634→
   635→    return { alertId, name, query, threshold, created: true };
   636→  }

   655→      throw new Error('Query is required');
   656→    }

   658→    const validFormats = ['json', 'csv', 'ndjson'];
   659→    if (!validFormats.includes(format)) {
   660→      throw new Error(`Invalid format: ${format}. Valid: ${validFormats.join(', ')}`);

   663→    this.exportCounter++;

   671→        : fileSizeBytes > 1048576
   672→          ? `${(fileSizeBytes / 1048576).toFixed(1)} MB`
   673→          : `${(fileSizeBytes / 1024).toFixed(1)} KB`;
   674→
   675→    const servicePath = service ? `/${service}` : '/all';
   676→    const downloadUrl = `https://logs.example.com/exports${servicePath}/${exportId}.${format}`;
   677→
   678→    this.logger.log(
   679→      `Export logs: ${exportId}, ${format}, ${recordCount} records, ~${fileSizeEstimate}`,
   680→    );
   681→
   682→    return {
   683→      exportId,
   684→      format,
   685→      recordCount,
   686→      downloadUrl,
   687→      fileSizeEstimate,
   688→      expiresAt: new Date(Date.now() + 86400000).toISOString(),
   689→    };

   784→    service: string;

   786→    maxStorageGb?: number;
   787→    compressionEnabled?: boolean;
   788→    archiveAfterDays?: number;
   789→  }): Promise<{
   790→    policyId: string;
   791→    name: string;
   792→    service: string;
   793→    retentionDays: number;
   794→    created: boolean;
   795→    message: string;
   796→  }> {
   797→    const {
   798→      name,
   799→      service,
   800→      retentionDays,
   801→      maxStorageGb,
   802→      compressionEnabled = true,
   803→      archiveAfterDays,
   804→    } = params;
   805→
   806→    if (!name || typeof name !== 'string') {
   807→      throw new Error('Policy name is required');
   808→    }
   809→    if (!service || typeof service !== 'string') {
   810→      throw new Error('Service name is required (use "*" for all services)');
   811→    }
   812→    if (retentionDays < 1 || retentionDays > 3650) {
   813→      throw new Error('Retention days must be between 1 and 3650 (10 years)');
   814→    }
   815→    if (archiveAfterDays !== undefined && archiveAfterDays >= retentionDays) {
   816→      throw new Error('Archive days must be less than retention days');
   817→    }
   818→
   819→    this.retentionCounter++;

   821→

   823→      id: policyId,
   824→      name,
   825→      service,
   826→      retentionDays,
   827→      maxStorageGb,
   828→      compressionEnabled,
   829→      archiveAfterDays,
   830→      createdAt: new Date(),
   831→    });
   832→
   833→    this.logger.log(
   834→      `Set retention policy: ${name} [${policyId}], service=${service}, retention=${retentionDays} days`,
   835→    );

   837→    return {

   839→      name,
   840→      service,
   841→      retentionDays,
   842→      created: true,
   843→      message: `Retention policy "${name}" created for ${service === '*' ? 'all services' : service}: ${retentionDays} days retention${archiveAfterDays ? `, archive after ${archiveAfterDays} days` : ''}`,
   844→    };
   845→  }
