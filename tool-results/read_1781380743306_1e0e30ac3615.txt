   417→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   418→    const startTime = Date.now();
   419→    const { action, ...params } = input.payload;
   420→
   421→    if (!action) {
   422→      return this.createAgentOutput(
   423→        input.taskId,
   424→        false,
   425→        null,
   426→        'Missing required parameter: action',
   427→        startTime,
   428→      );
   429→    }
   430→
   431→    const supportedActions = [
   432→      'createSpreadsheet',
   433→      'updateCell',
   434→      'applyFormula',
   435→      'createChart',
   436→      'importData',
   437→      'exportData',
   438→      'pivotTable',
   439→    ];
   440→
   441→    if (!supportedActions.includes(action)) {
   442→      return this.createAgentOutput(
   443→        input.taskId,
   444→        false,
   445→        null,
   446→        `Unknown spreadsheet action: ${action}. Supported: ${supportedActions.join(', ')}`,
   447→        startTime,
   448→      );
   449→    }
   450→
   451→    try {
   452→      const tool = this.getTool(action);
   453→      if (!tool) {
   454→        return this.createAgentOutput(
   455→          input.taskId,
   456→          false,
   457→          null,
   458→          `Tool not found: ${action}`,
   459→          startTime,
   460→        );
   461→      }
   462→
   463→      const result = await tool.execute(params);
   464→
   465→      await this.storeInWorkingMemory(
   466→        `spreadsheet:last:${action}`,
   467→        { params, result, timestamp: new Date() },
   468→        300000,
   469→      );
   470→
   471→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   472→    } catch (error) {
   473→      const msg = (error as Error).message;
   474→      this.logger.error(`Spreadsheet execution failed for ${action}: ${msg}`);
   475→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   476→    }
   477→  }

   489→  private async createSpreadsheet(params: {
   490→    title: string;
   491→    sheets?: Array<{ name: string; data?: string[][] }>;
   492→    author?: string;
   493→    format?: string;
   494→  }): Promise<{
   495→    spreadsheetId: string;
   496→    title: string;
   497→    createdAt: string;
   498→  }> {
   499→    const { title, sheets = [], author = 'agent@aenews.system', format = 'xlsx' } = params;
   500→
   501→    if (!title || typeof title !== 'string') {
   502→      throw new Error('A valid spreadsheet title is required');
   503→    }
   504→
   505→    const validFormats = ['xlsx', 'csv', 'ods'];
   506→    if (!validFormats.includes(format)) {
   507→      throw new Error(`Invalid format: ${format}. Supported: ${validFormats.join(', ')}`);
   508→    }
   509→
   510→    const spreadsheetId = this.generateSpreadsheetId();
   511→    const sheetsMap = new Map<string, Sheet>();
   512→
   513→    // Create default sheet
   514→    const defaultSheet: Sheet = {
   515→      name: 'Sheet1',
   516→      cells: new Map(),
   517→      rowCount: 1000,
   518→      columnCount: 26,
   519→    };
   520→    sheetsMap.set('Sheet1', defaultSheet);
   521→
   522→    // Create additional sheets
   523→    for (const sheetDef of sheets) {
   524→      const sheet: Sheet = {
   525→        name: sheetDef.name,
   526→        cells: new Map(),
   527→        rowCount: 1000,
   528→        columnCount: 26,
   529→      };
   530→
   531→      // Populate initial data
   532→      if (sheetDef.data) {
   533→        for (let row = 0; row < sheetDef.data.length; row++) {
   534→          for (let col = 0; col < sheetDef.data[row].length; col++) {
   535→            const cellRef = this.columnToLetter(col + 1) + (row + 1);
   536→            sheet.cells.set(cellRef, {
   537→              value: sheetDef.data[row][col],
   538→            });
   539→          }
   540→        }
   541→      }
   542→
   543→      sheetsMap.set(sheetDef.name, sheet);
   544→    }
   545→
   546→    const spreadsheet: Spreadsheet = {
   547→      id: spreadsheetId,
   548→      title,
   549→      author,
   550→      format,
   551→      sheets: sheetsMap,
   552→      charts: [],
   553→      pivotTables: [],
   554→      createdAt: new Date(),
   555→      updatedAt: new Date(),
   556→    };
   557→
   558→    this.spreadsheets.set(spreadsheetId, spreadsheet);
   559→
   560→    this.logger.log(
   561→      `Created spreadsheet: ${spreadsheetId}, title="${title}", sheets=${sheetsMap.size}`,
   562→    );
   563→
   564→    return {
   565→      spreadsheetId,
   566→      title,
   567→      createdAt: spreadsheet.createdAt.toISOString(),
   568→    };
   569→  }

   571→  private async updateCell(params: {
   572→    spreadsheetId: string;
   573→    sheet?: string;
   574→    cell: string;
   575→    value?: string;
   576→    formula?: string;
   577→    format?: CellFormat;
   578→    rangeUpdates?: Array<{ cell: string; value?: string; formula?: string }>;
   579→  }): Promise<{
   580→    spreadsheetId: string;
   581→    updatedCells: number;
   582→    status: string;
   583→  }> {
   584→    const { spreadsheetId, sheet = 'Sheet1', cell, value, formula, format, rangeUpdates } = params;
   585→
   586→    if (!spreadsheetId || typeof spreadsheetId !== 'string') {
   587→      throw new Error('A valid spreadsheetId is required');
   588→    }
   589→
   590→    const spreadsheet = this.spreadsheets.get(spreadsheetId);
   591→    if (!spreadsheet) {
   592→      throw new Error(`Spreadsheet not found: ${spreadsheetId}`);
   593→    }
   594→
   595→    const targetSheet = spreadsheet.sheets.get(sheet);
   596→    if (!targetSheet) {
   597→      throw new Error(`Sheet not found: ${sheet}`);
   598→    }
   599→
   600→    let updatedCells = 0;
   601→
   602→    // Single cell update
   603→    if (cell && (value !== undefined || formula !== undefined || format !== undefined)) {
   604→      this.validateCellReference(cell);
   605→      const existingCell = targetSheet.cells.get(cell) || { value: null };
   606→      targetSheet.cells.set(cell, {
   607→        ...existingCell,
   608→        ...(value !== undefined ? { value } : {}),
   609→        ...(formula !== undefined ? { formula } : {}),
   610→        ...(format !== undefined ? { format } : {}),
   611→      });
   612→
   613→      // Compute formula if present
   614→      if (formula) {
   615→        const computed = this.evaluateFormula(formula, targetSheet);
   616→        targetSheet.cells.get(cell)!.computedValue = computed;
   617→      }
   618→
   619→      updatedCells++;
   620→    }
   621→
   622→    // Batch range updates
   623→    if (rangeUpdates && Array.isArray(rangeUpdates)) {
   624→      for (const update of rangeUpdates) {
   625→        this.validateCellReference(update.cell);
   626→        const existingCell = targetSheet.cells.get(update.cell) || { value: null };
   627→        targetSheet.cells.set(update.cell, {
   628→          ...existingCell,
   629→          ...(update.value !== undefined ? { value: update.value } : {}),
   630→          ...(update.formula !== undefined ? { formula: update.formula } : {}),
   631→        });
   632→
   633→        if (update.formula) {
   634→          const computed = this.evaluateFormula(update.formula, targetSheet);
   635→          targetSheet.cells.get(update.cell)!.computedValue = computed;
   636→        }
   637→
   638→        updatedCells++;
   639→      }
   640→    }
   641→
   642→    spreadsheet.updatedAt = new Date();
   643→
   644→    this.logger.log(
   645→      `Updated cells in spreadsheet: ${spreadsheetId}, sheet=${sheet}, count=${updatedCells}`,
   646→    );
   647→
   648→    return {
   649→      spreadsheetId,
   650→      updatedCells,
   651→      status: 'updated',
   652→    };
   653→  }

   655→  private async applyFormula(params: {
   656→    spreadsheetId: string;
   657→    sheet?: string;
   658→    cell: string;

  1166→  private evaluateFormula(formula: string, sheet: Sheet): string | number {
  1167→    // Simple formula evaluation for SUM, AVG, COUNT, MIN, MAX
  1168→    const upperFormula = formula.toUpperCase();
  1169→
  1170→    // Match function patterns like SUM(A1:A10), AVG(B1:B5)
  1171→    const funcMatch = upperFormula.match(
  1172→      /^(SUM|AVG|AVERAGE|COUNT|MIN|MAX)\(([A-Z]+\d+):([A-Z]+\d+)\)$/,
  1173→    );
  1174→    if (funcMatch) {
  1175→      const [, func, startRef, endRef] = funcMatch;
  1176→      const start = this.parseCellReference(startRef);
  1177→      const end = this.parseCellReference(endRef);
  1178→
  1179→      const values: number[] = [];
  1180→      for (let row = start.row; row <= end.row; row++) {
  1181→        for (let col = start.col; col <= end.col; col++) {
  1182→          const cellRef = this.columnToLetter(col) + row;
  1183→          const cellData = sheet.cells.get(cellRef);
  1184→          const val = cellData?.computedValue ?? cellData?.value;
  1185→          if (val !== null && val !== undefined && !isNaN(Number(val))) {
  1186→            values.push(Number(val));
  1187→          }
  1188→        }
  1189→      }
  1190→
  1191→      switch (func) {
  1192→        case 'SUM':
  1193→          return values.reduce((a, b) => a + b, 0);
  1194→        case 'AVG':
  1195→        case 'AVERAGE':
  1196→          return values.length > 0 ? values.reduce((a, b) => a + b, 0) / values.length : 0;
  1197→        case 'COUNT':
  1198→          return values.length;
  1199→        case 'MIN':
  1200→          return values.length > 0 ? Math.min(...values) : 0;
  1201→        case 'MAX':
  1202→          return values.length > 0 ? Math.max(...values) : 0;
  1203→      }
  1204→    }
  1205→
  1206→    // Simple arithmetic: try to evaluate as expression
  1207→    try {
  1208→      // Replace cell references with their values
  1209→      let expression = formula;
  1210→      const cellRefPattern = /([A-Z]+\d+)/g;
  1211→      let match: RegExpExecArray | null;
  1212→      while ((match = cellRefPattern.exec(formula)) !== null) {
  1213→        const cellData = sheet.cells.get(match[1]);
  1214→        const val = cellData?.computedValue ?? cellData?.value ?? 0;
  1215→        expression = expression.replace(match[1], String(val));
  1216→      }
  1217→
  1218→      // Simple safe evaluation (only numbers and basic operators)
  1219→      if (/^[\d\s+\-*/().]+$/.test(expression)) {
  1220→        const result = new Function(`return ${expression}`)();
  1221→        return typeof result === 'number' ? Math.round(result * 1000) / 1000 : result;
  1222→      }
  1223→    } catch {
  1224→      // Formula evaluation failed, return formula as-is
  1225→    }
  1226→
  1227→    return formula;
  1228→  }

  1321→  private convertTo2DArray(cells: Array<{ ref: string; value: any }>, range?: string): string[][] {
  1322→    if (cells.length === 0) return [];
  1323→
  1324→    let minRow = Infinity,
  1325→      maxRow = 0,
  1326→      minCol = Infinity,
  1327→      maxCol = 0;
  1328→
  1329→    for (const cell of cells) {
  1330→      const ref = this.parseCellReference(cell.ref);
  1331→      minRow = Math.min(minRow, ref.row);
  1332→      maxRow = Math.max(maxRow, ref.row);
  1333→      minCol = Math.min(minCol, ref.col);
  1334→      maxCol = Math.max(maxCol, ref.col);
  1335→    }
  1336→
  1337→    const numRows = maxRow - minRow + 1;
  1338→    const numCols = maxCol - minCol + 1;
  1339→    const grid: string[][] = Array.from({ length: numRows }, () => Array(numCols).fill(''));
  1340→
  1341→    for (const cell of cells) {
  1342→      const ref = this.parseCellReference(cell.ref);
  1343→      grid[ref.row - minRow][ref.col - minCol] = String(cell.value ?? '');
  1344→    }
  1345→
  1346→    return grid;
  1347→  }

  1349→  private aggregatePivotData(
  1350→    rows: Array<Map<string, any>>,
  1351→    rowFields: string[],
  1352→    columnFields: string[],
  1353→    valueFields: Array<{ field: string; aggregation: string }>,
  1354→  ): { rowCount: number; columnCount: number; data: Map<string, Map<string, number[]>> } {
  1355→    // Group data by row fields
  1356→    const groups = new Map<string, Array<Map<string, any>>>();
  1357→
  1358→    for (const row of rows) {
  1359→      const key =
  1360→        rowFields.length > 0
  1361→          ? rowFields.map((f) => String(row.get(f) ?? 'undefined')).join('|')
  1362→          : '__all__';
  1363→
  1364→      if (!groups.has(key)) {
  1365→        groups.set(key, []);
  1366→      }
  1367→      groups.get(key)!.push(row);
  1368→    }
  1369→
  1370→    // Aggregate values for each group
  1371→    const pivotData = new Map<string, Map<string, number[]>>();
  1372→
  1373→    for (const [groupKey, groupRows] of groups) {
  1374→      const aggregated = new Map<string, number[]>();
  1375→
  1376→      for (const vf of valueFields) {
  1377→        const vals = groupRows.map((r) => Number(r.get(vf.field))).filter((v) => !isNaN(v));
  1378→
  1379→        let result: number;
  1380→        switch (vf.aggregation.toLowerCase()) {
  1381→          case 'sum':
  1382→            result = vals.reduce((a, b) => a + b, 0);
  1383→            break;
  1384→          case 'avg':
  1385→          case 'average':
  1386→            result = vals.length > 0 ? vals.reduce((a, b) => a + b, 0) / vals.length : 0;
  1387→            break;
  1388→          case 'count':
  1389→            result = vals.length;
  1390→            break;
  1391→          case 'min':
  1392→            result = vals.length > 0 ? Math.min(...vals) : 0;
  1393→            break;
  1394→          case 'max':
  1395→            result = vals.length > 0 ? Math.max(...vals) : 0;
  1396→            break;
  1397→          default:
  1398→            result = 0;
  1399→        }
  1400→
  1401→        const valueKey = `${vf.field}_${vf.aggregation}`;
  1402→        aggregated.set(valueKey, [result]);
  1403→      }
  1404→
  1405→      pivotData.set(groupKey, aggregated);
  1406→    }
  1407→
  1408→    const rowCount = groups.size + 1; // +1 for header
  1409→    const columnCount = valueFields.length + rowFields.length;
  1410→
  1411→    return { rowCount, columnCount, data: pivotData };
  1412→  }
