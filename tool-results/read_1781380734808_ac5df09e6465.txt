   218→  // ─── Lifecycle: Execute ──────────────────────────────────────────
   219→
   220→  async execute(input: AgentInput): Promise<AgentOutput> {
   221→    // 1. Check status
   222→    if (this.status !== AgentStatus.RUNNING && this.status !== AgentStatus.IDLE) {
   223→      throw new AgentError(
   224→        `Agent cannot execute in ${this.status} state`,
   225→        AgentErrorCode.NOT_RUNNING,
   226→        this.config.id,
   227→        input.taskId,
   228→      );
   229→    }
   230→
   231→    // 2. Check circuit breaker
   232→    if (this.circuitBreaker.state === 'open') {
   233→      throw new AgentError(
   234→        'Circuit breaker is open; agent is not accepting tasks',
   235→        AgentErrorCode.CIRCUIT_BREAKER_OPEN,
   236→        this.config.id,
   237→        input.taskId,
   238→      );
   239→    }
   240→
   241→    // 3. Check max concurrent tasks
   242→    if (this.currentTasks.size >= this.config.maxConcurrentTasks) {
   243→      throw new AgentError(
   244→        `Max concurrent tasks (${this.config.maxConcurrentTasks}) reached`,
   245→        AgentErrorCode.MAX_CONCURRENT_TASKS,
   246→        this.config.id,
   247→        input.taskId,
   248→      );
   249→    }
   250→
   251→    // 4. Check permissions
   252→    await this.checkPermission(PermissionAction.EXECUTE, PermissionResource.TASK);
   253→
   254→    // 5. Validate input
   255→    const isValid = await this.onValidateInput(input);
   256→    if (!isValid) {
   257→      throw new AgentError(
   258→        'Invalid input provided',
   259→        AgentErrorCode.INVALID_INPUT,
   260→        this.config.id,
   261→        input.taskId,
   262→      );
   263→    }
   264→
   265→    // 6. Set up execution context
   266→    this.correlationId = input.context?.correlationId || uuidv4();
   267→    this.currentTasks.add(input.taskId);
   268→    this.lastActivity = new Date();
   269→
   270→    // 7. Record start time
   271→    const startTime = Date.now();
   272→    const startMemory = process.memoryUsage().heapUsed;
   273→
   274→    this.executeLifecyclePhase(LifecyclePhase.PRE_EXECUTE, { taskId: input.taskId });
   275→    this.emitEvent(AgentEventType.TASK_STARTED, {
   276→      taskId: input.taskId,
   277→      agentId: this.config.id,
   278→      priority: input.priority || TaskPriority.NORMAL,
   279→    });
   280→
   281→    try {
   282→      // Auto-start if idle
   283→      if (this.status === AgentStatus.IDLE) {
   284→        await this.start();
   285→      }
   286→
   287→      // 8. Execute with retry logic (exponential backoff)
   288→      let retryCount = 0;
   289→      let lastError: Error | null = null;
   290→
   291→      const output = await this.executeWithRetry<AgentOutput>(async () => {
   292→        try {
   293→          // Execute with timeout
   294→          const result = await this.executeWithTimeout(input);
   295→          return result;
   296→        } catch (execError) {
   297→          retryCount++;
   298→          lastError = execError as Error;
   299→
   300→          if (retryCount > this.config.retryPolicy.maxRetries) {
   301→            throw execError;
   302→          }
   303→
   304→          const backoffMs = this.calculateBackoff(retryCount);
   305→          this.logger.warn(
   306→            `Task ${input.taskId} failed (attempt ${retryCount}/${this.config.retryPolicy.maxRetries}), ` +
   307→              `retrying in ${backoffMs}ms: ${lastError.message}`,
   308→          );
   309→          await this.sleep(backoffMs);
   310→
   311→          // Re-throw so executeWithRetry can handle it
   312→          throw execError;
   313→        }
   314→      });
   315→
   316→      // 9. Collect metrics and emit events on success
   317→      this.currentTasks.delete(input.taskId);
   318→      this.completedTaskCount++;
   319→      this.lastActivity = new Date();
   320→
   321→      this.recordCircuitBreakerSuccess();
   322→
   323→      // Store result in memory
   324→      await this.storeExecutionResult(input, output);
   325→
   326→      // Lifecycle and event hooks
   327→      this.executeLifecyclePhase(LifecyclePhase.POST_EXECUTE, { taskId: input.taskId });
   328→      this.emitEvent(AgentEventType.TASK_COMPLETED, {
   329→        taskId: input.taskId,
   330→        agentId: this.config.id,
   331→        success: true,
   332→        executionTimeMs: Date.now() - startTime,
   333→        result: output.result,
   334→      });
   335→
   336→      await this.onSuccess(input, output);
   337→
   338→      // 10. Return output
   339→      return output;
   340→    } catch (error) {
   341→      this.currentTasks.delete(input.taskId);
   342→      this.failedTaskCount++;
   343→      this.recordCircuitBreakerFailure();
   344→      this.lastActivity = new Date();
   345→
   346→      const agentError = error as AgentError;
   347→      this.logger.error(
   348→        `Task ${input.taskId} execution error: ${agentError.message}`,
   349→        agentError.stack,
   350→      );
   351→
   352→      this.emitEvent(AgentEventType.AGENT_ERROR, {
   353→        errorCode: agentError.code || AgentErrorCode.EXECUTION_FAILED,
   354→        errorMessage: agentError.message,
   355→        taskId: input.taskId,
   356→        recoverable: true,
   357→      } as AgentErrorPayload);
   358→
   359→      this.emitEvent(AgentEventType.TASK_FAILED, {
   360→        taskId: input.taskId,
   361→        agentId: this.config.id,
   362→        success: false,
   363→        executionTimeMs: Date.now() - startTime,
   364→        error: agentError.message,
   365→      });
   366→
   367→      await this.onFailure(input, agentError);
   368→
   369→      // 10. Return output (with error info)
   370→      return {
   371→        taskId: input.taskId,
   372→        success: false,
   373→        result: null,
   374→        error: agentError.message,
   375→        metrics: this.collectMetrics(startTime, startMemory),
   376→        timestamp: new Date(),
   377→      };
   378→    }
   379→  }

   413→  // ─── Lifecycle: Stop ─────────────────────────────────────────────
   414→
   415→  async stop(): Promise<void> {
   416→    this.executeLifecyclePhase(LifecyclePhase.PRE_STOP);
   417→
   418→    // Clear all active timeouts
   419→    for (const [taskId, timeout] of this.activeTimeouts) {
   420→      clearTimeout(timeout);
   421→      this.activeTimeouts.delete(taskId);
   422→    }
   423→
   424→    // Wait for current tasks to finish (with a grace period)
   425→    const gracePeriodMs = 5000;
   426→    const startTime = Date.now();
   427→    while (this.currentTasks.size > 0 && Date.now() - startTime < gracePeriodMs) {
   428→      await this.sleep(100);
   429→    }
   430→
   431→    // Force-clear remaining tasks
   432→    if (this.currentTasks.size > 0) {
   433→      this.logger.warn(`Force-stopping with ${this.currentTasks.size} active tasks`);
   434→      this.currentTasks.clear();
   435→    }
   436→
   437→    this.stopHealthMonitoring();
   438→    this.stopMetricsCollection();
   439→
   440→    await this.transitionTo(AgentStatus.STOPPED);
   441→    this.executeLifecyclePhase(LifecyclePhase.POST_STOP);
   442→
   443→    this.logger.log(`Agent stopped: ${this.config.id}`);
   444→    this.emitEvent(AgentEventType.AGENT_STOPPED, {
   445→      agentId: this.config.id,
   446→      completedTasks: this.completedTaskCount,
   447→      failedTasks: this.failedTaskCount,
   448→    });
   449→  }

   584→  // ─── Permission Checking ─────────────────────────────────────────
   585→
   586→  /**
   587→   * Check if the agent has a specific permission.
   588→   * Overloaded: supports both enum-based and string-based checks.
   589→   */
   590→  protected async checkPermission(
   591→    action: PermissionAction | string,
   592→    resource: PermissionResource | string,
   593→  ): Promise<boolean> {
   594→    const permissionString = `${action}:${resource}`;
   595→
   596→    const hasPermission =
   597→      this.config.permissions.includes(permissionString) || this.config.permissions.includes('*');
   598→
   599→    if (!hasPermission) {
   600→      // Also check via the permission evaluator if available
   601→      if (
   602→        this.permissionEvaluator &&
   603→        typeof this.permissionEvaluator.hasPermission === 'function'
   604→      ) {
   605→        const evaluated = await this.permissionEvaluator.hasPermission(
   606→          this.config.id,
   607→          action as PermissionAction,
   608→          resource as PermissionResource,
   609→        );
   610→        if (evaluated) return true;
   611→      }
   612→
   613→      throw new AgentError(
   614→        `Permission denied: ${permissionString}`,
   615→        AgentErrorCode.PERMISSION_DENIED,
   616→        this.config.id,
   617→      );
   618→    }
   619→
   620→    return true;
   621→  }

   623→  /**
   624→   * Check permission by resource and action strings.
   625→   * Returns boolean without throwing.
   626→   */
   627→  hasPermissionForResource(resource: string, action: string): boolean {
   628→    return (
   629→      this.config.permissions.includes(`${action}:${resource}`) ||
   630→      this.config.permissions.includes('*')
   631→    );
   632→  }

   634→  // ─── Timeout Handling ────────────────────────────────────────────
   635→
   636→  /**
   637→   * Execute agent task with a timeout.
   638→   */
   639→  protected async executeWithTimeout(input: AgentInput): Promise<AgentOutput> {
   640→    const timeoutMs = input.context?.timeout || this.config.timeout;
   641→
   642→    return new Promise<AgentOutput>((resolve, reject) => {
   643→      const timeoutId = setTimeout(() => {
   644→        this.activeTimeouts.delete(input.taskId);
   645→        reject(
   646→          new AgentError(
   647→            `Task timed out after ${timeoutMs}ms`,
   648→            AgentErrorCode.TIMEOUT,
   649→            this.config.id,
   650→            input.taskId,
   651→          ),
   652→        );
   653→      }, timeoutMs);
   654→
   655→      this.activeTimeouts.set(input.taskId, timeoutId);
   656→
   657→      this.onExecute(input)
   658→        .then((output) => {
   659→          clearTimeout(timeoutId);
   660→          this.activeTimeouts.delete(input.taskId);
   661→          resolve(output);
   662→        })
   663→        .catch((error) => {
   664→          clearTimeout(timeoutId);
   665→          this.activeTimeouts.delete(input.taskId);
   666→          reject(error);
   667→        });
   668→    });
   669→  }

   671→  /**
   672→   * Generic timeout wrapper for any promise.
   673→   * Rejects with AgentError if the promise does not resolve within the specified time.
   674→   */
   675→  protected withTimeout<T>(promise: Promise<T>, ms: number): Promise<T> {
   676→    return new Promise<T>((resolve, reject) => {
   677→      const timeoutId = setTimeout(() => {
   678→        reject(
   679→          new AgentError(
   680→            `Operation timed out after ${ms}ms`,
   681→            AgentErrorCode.TIMEOUT,
   682→            this.config.id,
   683→          ),
   684→        );
   685→      }, ms);
   686→
   687→      promise
   688→        .then((result) => {
   689→          clearTimeout(timeoutId);
   690→          resolve(result);
   691→        })
   692→        .catch((error) => {
   693→          clearTimeout(timeoutId);
   694→          reject(error);
   695→        });
   696→    });
   697→  }

   699→  // ─── Retry Logic ─────────────────────────────────────────────────
   700→
   701→  /**
   702→   * Calculate exponential backoff delay for a given retry attempt.
   703→   */
   704→  protected calculateBackoff(retryCount: number): number {
   705→    const { backoffMs, exponentialBackoff } = this.config.retryPolicy;
   706→    if (exponentialBackoff) {
   707→      return backoffMs * Math.pow(2, retryCount - 1) + Math.random() * 100;
   708→    }
   709→    return backoffMs;
   710→  }
   711→
   712→  /**
   713→   * Execute a function with retry logic and exponential backoff.
   714→   * Retries up to maxRetries times on failure.
   715→   */
   716→  protected async executeWithRetry<T>(fn: () => Promise<T>): Promise<T> {
   717→    const { maxRetries, backoffMs, exponentialBackoff } = this.config.retryPolicy;
   718→    let lastError: Error | null = null;
   719→
   720→    for (let attempt = 0; attempt <= maxRetries; attempt++) {
   721→      try {
   722→        return await fn();
   723→      } catch (error) {
   724→        lastError = error as Error;
   725→
   726→        if (attempt >= maxRetries) {
   727→          this.logger.error(`All ${maxRetries} retries exhausted: ${lastError.message}`);
   728→          throw new AgentError(
   729→            `Retry exhausted after ${maxRetries} attempts: ${lastError.message}`,
   730→            AgentErrorCode.RETRY_EXHAUSTED,
   731→            this.config.id,
   732→            undefined,
   733→            lastError,
   734→          );
   735→        }
   736→
   737→        const delay = exponentialBackoff
   738→          ? backoffMs * Math.pow(2, attempt) + Math.random() * 100
   739→          : backoffMs;
   740→
   741→        this.logger.warn(
   742→          `Attempt ${attempt + 1}/${maxRetries} failed, retrying in ${Math.round(delay)}ms: ${lastError.message}`,
   743→        );
   744→
   745→        await this.sleep(delay);
   746→      }
   747→    }
   748→
   749→    // This should never be reached, but TypeScript needs it
   750→    throw lastError!;
   751→  }

   781→    try {
   782→      const customHealthy = await this.onHealthCheck();

   784→
   785→      const previousHealth = this.healthState.isHealthy;

   807→      this.healthState.isHealthy = false;
   808→      this.healthState.lastHealthCheck = new Date();

   843→  protected async transitionTo(newStatus: AgentStatus): Promise<void> {
   844→    const previousStatus = this.status;
   845→
   846→    if (!this.isValidTransition(previousStatus, newStatus)) {
   847→      this.logger.warn(`Invalid state transition: ${previousStatus} → ${newStatus}`);
   848→      // Allow some transitions for recovery scenarios
   849→      if (newStatus !== AgentStatus.ERROR && newStatus !== AgentStatus.STOPPED) {
   850→        throw new AgentError(

   852→          AgentErrorCode.EXECUTION_FAILED,

   854→        );

   856→    }
   857→
   858→    this.status = newStatus;
   859→    this.lastActivity = new Date();
   860→
   861→    this.emitEvent(AgentEventType.AGENT_STATUS_CHANGED, {
   862→      previousStatus,
   863→      newStatus,
   864→      reason: `Transition from ${previousStatus} to ${newStatus}`,
   865→    } as AgentStatusChangedPayload);
   866→
   867→    this.logger.log(`State transition: ${previousStatus} → ${newStatus}`);
   868→  }

  1013→  protected recordCircuitBreakerSuccess(): void {
  1014→    if (this.circuitBreaker.state === 'half_open') {
  1015→      this.circuitBreaker.halfOpenSuccesses++;
  1016→      if (this.circuitBreaker.halfOpenSuccesses >= this.circuitBreaker.successThreshold) {
  1017→        this.circuitBreaker.state = 'closed';
  1018→        this.circuitBreaker.failureCount = 0;
  1019→        this.circuitBreaker.halfOpenSuccesses = 0;
  1020→        this.logger.log('Circuit breaker closed');
  1021→        this.emitEvent(AgentEventType.CIRCUIT_BREAKER_CLOSED, {
  1022→          agentId: this.config.id,
  1023→          state: 'closed',
  1024→          failureCount: 0,
  1025→          lastFailureTime: this.circuitBreaker.lastFailureTime || new Date(),
  1026→        });
  1027→      }
  1028→    } else if (this.circuitBreaker.state === 'closed') {
  1029→      this.circuitBreaker.failureCount = 0;
  1030→    }
  1031→  }

  1033→  protected recordCircuitBreakerFailure(): void {
  1034→    this.circuitBreaker.failureCount++;
  1035→    this.circuitBreaker.lastFailureTime = new Date();
  1036→
  1037→    if (this.circuitBreaker.state === 'half_open') {
  1038→      this.circuitBreaker.state = 'open';
  1039→      this.logger.warn('Circuit breaker opened (failure during half-open)');
  1040→    } else if (this.circuitBreaker.failureCount >= this.circuitBreaker.failureThreshold) {
  1041→      this.circuitBreaker.state = 'open';
  1042→      this.logger.error(
  1043→        `Circuit breaker opened after ${this.circuitBreaker.failureCount} failures`,
  1044→      );
  1045→      this.emitEvent(AgentEventType.CIRCUIT_BREAKER_OPENED, {
  1046→        agentId: this.config.id,
  1047→        state: 'open',
  1048→        failureCount: this.circuitBreaker.failureCount,
  1049→        lastFailureTime: this.circuitBreaker.lastFailureTime,
  1050→      });
  1051→
  1052→      // Schedule transition to half-open
  1053→      setTimeout(() => {
  1054→        if (this.circuitBreaker.state === 'open') {
  1055→          this.circuitBreaker.state = 'half_open';
  1056→          this.circuitBreaker.halfOpenSuccesses = 0;
  1057→          this.logger.log('Circuit breaker transitioned to half-open');
  1058→        }

  1088→    extraContext?: Record<string, any>,
  1089→  ): Promise<void> {

  1094→      agentId: this.config.id,
  1095→      phase,
  1096→      previousStatus: this.status,
  1097→      newStatus: this.status,
  1098→      timestamp: new Date(),
  1099→      correlationId: this.correlationId,
  1100→      ...extraContext,
  1101→    };
  1102→

  1112→      } catch (error) {

  1115→    }
