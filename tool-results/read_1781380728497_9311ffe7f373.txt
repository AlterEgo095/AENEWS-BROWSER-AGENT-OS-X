   281→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   282→    const startTime = Date.now();
   283→    const { action, ...params } = input.payload;
   284→
   285→    if (!action) {
   286→      return this.createAgentOutput(
   287→        input.taskId,
   288→        false,
   289→        null,
   290→        'Missing required parameter: action',
   291→        startTime,
   292→      );
   293→    }
   294→
   295→    const supportedActions = [
   296→      'analyzeError',
   297→      'traceExecution',
   298→      'suggestFix',
   299→      'applyFix',
   300→      'validateFix',
   301→    ];
   302→
   303→    if (!supportedActions.includes(action)) {
   304→      return this.createAgentOutput(
   305→        input.taskId,
   306→        false,
   307→        null,
   308→        `Unknown debugging action: ${action}. Supported: ${supportedActions.join(', ')}`,
   309→        startTime,
   310→      );
   311→    }
   312→
   313→    try {
   314→      const tool = this.getTool(action);
   315→      if (!tool) {
   316→        return this.createAgentOutput(
   317→          input.taskId,
   318→          false,
   319→          null,
   320→          `Tool not found: ${action}`,
   321→          startTime,
   322→        );
   323→      }
   324→
   325→      const result = await tool.execute(params);
   326→
   327→      // Store debug session data
   328→      const sessionId = input.context?.sessionId || input.taskId;
   329→      if (!this.debugSessions.has(sessionId)) {
   330→        this.debugSessions.set(sessionId, {
   331→          error: params.error,
   332→          appliedFixes: [],
   333→          validated: false,
   334→        });
   335→      }
   336→      const session = this.debugSessions.get(sessionId)!;
   337→
   338→      switch (action) {
   339→        case 'analyzeError':
   340→          session.analysis = result as ErrorAnalysis;
   341→          break;
   342→        case 'suggestFix':
   343→          session.suggestions = result.suggestions;
   344→          break;
   345→        case 'applyFix':
   346→          session.appliedFixes.push(...(result.changes || []));
   347→          break;
   348→        case 'validateFix':
   349→          session.validated = result.valid;
   350→          break;
   351→      }
   352→
   353→      await this.storeInWorkingMemory(
   354→        `debugging:session:${sessionId}`,
   355→        { action, result, timestamp: new Date() },
   356→        600000,
   357→      );
   358→
   359→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   360→    } catch (error) {
   361→      const msg = (error as Error).message;
   362→      this.logger.error(`Debugging execution failed for ${action}: ${msg}`);
   363→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   364→    }
   365→  }
   366→

   374→  private async analyzeError(params: {
   375→    error: Record<string, any>;
   376→    code?: string;
   377→    language?: string;
   378→    context?: Record<string, any>;
   379→  }): Promise<ErrorAnalysis> {
   380→    const { error, code, language = 'typescript', context } = params;
   381→
   382→    if (!error) {
   383→      throw new Error('Error object is required for analysis');
   384→    }
   385→
   386→    const errorMessage = error.message || error.msg || error.toString?.() || 'Unknown error';
   387→    const errorStack = error.stack || '';
   388→    const errorType = error.name || error.type || this.classifyError(errorMessage);
   389→
   390→    // Determine root cause
   391→    const rootCause = this.determineRootCause(errorMessage, errorStack, code, language);
   392→
   393→    // Determine severity
   394→    const severity = this.determineSeverity(errorType, errorMessage, code);
   395→
   396→    // Identify affected code paths
   397→    const affectedPaths = this.extractAffectedPaths(errorStack, code);
   398→
   399→    // Generate analysis text
   400→    const analysis = this.generateAnalysisText(
   401→      errorType,
   402→      errorMessage,
   403→      rootCause,
   404→      severity,
   405→      affectedPaths,
   406→      language,
   407→    );
   408→
   409→    // Find related errors
   410→    const relatedErrors = this.findRelatedErrors(errorType, errorMessage, code, language);
   411→
   412→    this.logger.log(
   413→      `Error analyzed: type=${errorType}, severity=${severity}, rootCause=${rootCause.substring(0, 80)}`,
   414→    );
   415→
   416→    return {
   417→      errorType,
   418→      message: errorMessage,
   419→      rootCause,
   420→      severity,
   421→      affectedPaths,
   422→      analysis,
   423→      relatedErrors,
   424→    };
   425→  }

   427→  private async traceExecution(params: {
   428→    code: string;
   429→    entryPoint: string;
   430→    language: string;
   431→    inputs?: Record<string, any>;
   432→    maxDepth?: number;
   433→  }): Promise<{
   434→    traceSteps: TraceStep[];
   435→    errorPoint: { line: number; function: string; reason: string } | null;
   436→    variablesAtError: Record<string, any>;
   437→    executionFlow: string;
   438→  }> {
   439→    const { code, entryPoint, language, inputs = {}, maxDepth = 50 } = params;
   440→
   441→    if (!code || typeof code !== 'string') {
   442→      throw new Error('Source code is required for tracing');
   443→    }
   444→    if (!entryPoint || typeof entryPoint !== 'string') {
   445→      throw new Error('Entry point is required for tracing');
   446→    }
   447→
   448→    const lines = code.split('\n');
   449→    const traceSteps: TraceStep[] = [];
   450→    let stepCounter = 0;
   451→
   452→    // Find entry point in code
   453→    const entryLine = this.findFunctionStartLine(lines, entryPoint, language);
   454→    if (entryLine === -1) {
   455→      throw new Error(`Entry point "${entryPoint}" not found in code`);
   456→    }
   457→
   458→    // Simulate execution trace
   459→    let currentLine = entryLine;
   460→    let currentFunction = entryPoint;
   461→    const variables: Record<string, any> = { ...inputs };
   462→    let errorPoint: { line: number; function: string; reason: string } | null = null;
   463→
   464→    while (currentLine < lines.length && stepCounter < maxDepth) {
   465→      const line = lines[currentLine].trim();
   466→
   467→      // Skip empty lines and comments
   468→      if (!line || line.startsWith('//') || line.startsWith('/*') || line.startsWith('*')) {
   469→        currentLine++;
   470→        continue;
   471→      }
   472→
   473→      stepCounter++;
   474→      const traceStep: TraceStep = {
   475→        step: stepCounter,
   476→        location: `line ${currentLine + 1}`,
   477→        line: currentLine + 1,
   478→        function: currentFunction,
   479→        action: this.describeLineAction(line, language),
   480→        variables: { ...variables },
   481→        timestamp: Date.now(),
   482→      };
   483→
   484→      // Track variable assignments
   485→      this.trackVariableAssignment(line, variables, language);
   486→
   487→      // Check for function calls (enter new function)
   488→      const funcCall = this.extractFunctionCall(line, language);
   489→      if (funcCall) {
   490→        const funcStart = this.findFunctionStartLine(lines, funcCall, language);
   491→        if (funcStart !== -1) {
   492→          currentFunction = funcCall;
   493→        }
   494→      }
   495→
   496→      // Check for potential error points
   497→      if (this.isErrorPoint(line, language)) {
   498→        errorPoint = {
   499→          line: currentLine + 1,
   500→          function: currentFunction,
   501→          reason: this.identifyErrorReason(line, language),
   502→        };
   503→      }
   504→
   505→      // Detect function exit
   506→      if (line === '}' && currentFunction !== entryPoint) {
   507→        currentFunction = entryPoint;
   508→      }
   509→
   510→      traceSteps.push(traceStep);
   511→      currentLine++;
   512→    }
   513→
   514→    // Build execution flow description
   515→    const executionFlow = this.buildExecutionFlowDescription(traceSteps, entryPoint, errorPoint);
   516→
   517→    this.logger.log(
   518→      `Execution trace complete: ${traceSteps.length} step(s), error at ${errorPoint ? `line ${errorPoint.line}` : 'none'}`,
   519→    );
   520→
   521→    return {
   522→      traceSteps,
   523→      errorPoint,
   524→      variablesAtError: variables,
   525→      executionFlow,
   526→    };
   527→  }

   529→  private async suggestFix(params: {
   530→    error: Record<string, any>;
   531→    code: string;
   532→    language: string;
   533→    maxSuggestions?: number;
   534→  }): Promise<{
   535→    suggestions: FixSuggestion[];
   536→    confidence: number;
   537→    autoFixable: boolean;
   538→  }> {
   539→    const { error, code, language, maxSuggestions = 3 } = params;
   540→
   541→    if (!error) {
   542→      throw new Error('Error description is required');
   543→    }
   544→    if (!code || typeof code !== 'string') {
   545→      throw new Error('Source code is required');
   546→    }
   547→    if (!language || typeof language !== 'string') {
   548→      throw new Error('Programming language is required');
   549→    }
   550→
   551→    const errorMessage = error.message || error.msg || error.toString?.() || 'Unknown error';
   552→    const errorType = error.name || error.type || this.classifyError(errorMessage);
   553→
   554→    const suggestions: FixSuggestion[] = [];
   555→    const lines = code.split('\n');
   556→
   557→    // Type error fixes
   558→    if (errorType.toLowerCase().includes('type') || errorMessage.toLowerCase().includes('type')) {
   559→      suggestions.push(...this.suggestTypeErrorFixes(errorMessage, lines, language));
   560→    }
   561→
   562→    // Reference error fixes
   563→    if (
   564→      errorType.toLowerCase().includes('reference') ||
   565→      errorMessage.toLowerCase().includes('is not defined')
   566→    ) {
   567→      suggestions.push(...this.suggestReferenceErrorFixes(errorMessage, lines, language));
   568→    }
   569→
   570→    // Null/undefined error fixes
   571→    if (
   572→      errorMessage.toLowerCase().includes('null') ||
   573→      errorMessage.toLowerCase().includes('undefined') ||
   574→      errorMessage.toLowerCase().includes('cannot read propert')
   575→    ) {
   576→      suggestions.push(...this.suggestNullErrorFixes(errorMessage, lines, language));
   577→    }
   578→
   579→    // Syntax error fixes
   580→    if (
   581→      errorType.toLowerCase().includes('syntax') ||
   582→      errorMessage.toLowerCase().includes('unexpected')
   583→    ) {
   584→      suggestions.push(...this.suggestSyntaxErrorFixes(errorMessage, lines, language));
   585→    }
   586→
   587→    // Range error fixes
   588→    if (
   589→      errorType.toLowerCase().includes('range') ||
   590→      errorMessage.toLowerCase().includes('out of range')
   591→    ) {
   592→      suggestions.push(...this.suggestRangeErrorFixes(errorMessage, lines, language));
   593→    }
   594→
   595→    // Generic fixes
   596→    if (suggestions.length === 0) {
   597→      suggestions.push(...this.suggestGenericFixes(errorMessage, lines, language));
   598→    }
   599→
   600→    // Limit suggestions
   601→    const limitedSuggestions = suggestions.slice(0, maxSuggestions);
   602→
   603→    // Calculate overall confidence
   604→    const avgConfidence =
   605→      limitedSuggestions.length > 0
   606→        ? limitedSuggestions.reduce((sum, s) => sum + s.confidence, 0) / limitedSuggestions.length
   607→        : 0;
   608→
   609→    const autoFixable = limitedSuggestions.some((s) => s.autoFixable);
   610→
   611→    this.logger.log(
   612→      `Suggested ${limitedSuggestions.length} fix(es), confidence=${avgConfidence.toFixed(2)}, autoFixable=${autoFixable}`,
   613→    );
   614→
   615→    return { suggestions: limitedSuggestions, confidence: avgConfidence, autoFixable };
   616→  }

   618→  private async applyFix(params: {
   619→    code: string;
   620→    fix: Record<string, any>;
   621→    language: string;
   622→    dryRun?: boolean;
   623→  }): Promise<{
   624→    fixedCode: string;
   625→    applied: boolean;
   626→    changes: AppliedChange[];
   627→    dryRun: boolean;
   628→  }> {
   629→    const { code, fix, language, dryRun = true } = params;
   630→
   631→    if (!code || typeof code !== 'string') {
   632→      throw new Error('Source code is required');
   633→    }
   634→    if (!fix || typeof fix !== 'object') {
   635→      throw new Error('Fix object is required');
   636→    }
   637→
   638→    const lines = code.split('\n');
   639→    const changes: AppliedChange[] = [];
   640→    let fixedCode = code;
   641→
   642→    const fixType = fix.type || fix.fixType || 'replacement';
   643→    const targetLine = fix.line ? fix.line - 1 : this.findTargetLine(lines, fix, language);
   644→
   645→    if (targetLine >= 0 && targetLine < lines.length) {
   646→      switch (fixType) {
   647→        case 'replacement': {
   648→          const before = lines[targetLine];
   649→          let after = before;
   650→
   651→          if (fix.search && fix.replace) {
   652→            after = before.replace(new RegExp(this.escapeRegex(fix.search), 'g'), fix.replace);
   653→          } else if (fix.newLine) {
   654→            after = fix.newLine;
   655→          }
   656→
   657→          if (before !== after) {
   658→            lines[targetLine] = after;
   659→            changes.push({
   660→              type: 'replacement',
   661→              lineStart: targetLine + 1,
   662→              lineEnd: targetLine + 1,
   663→              before,
   664→              after,
   665→              description: fix.description || `Replaced line ${targetLine + 1}`,
   666→            });
   667→          }
   668→          break;
   669→        }
   670→        case 'insertion': {
   671→          const insertLine = fix.insertBefore ? targetLine : targetLine + 1;
   672→          const newLine = fix.newLine || fix.code || '';
   673→          lines.splice(insertLine, 0, newLine);
   674→          changes.push({
   675→            type: 'insertion',
   676→            lineStart: insertLine + 1,
   677→            lineEnd: insertLine + 1,
   678→            before: '',
   679→            after: newLine,
   680→            description: fix.description || `Inserted line after ${targetLine + 1}`,
   681→          });
   682→          break;
   683→        }
   684→        case 'deletion': {
   685→          const deleteCount = fix.deleteCount || 1;
   686→          const deleted = lines.splice(targetLine, deleteCount);
   687→          changes.push({
   688→            type: 'deletion',
   689→            lineStart: targetLine + 1,
   690→            lineEnd: targetLine + deleteCount,
   691→            before: deleted.join('\n'),
   692→            after: '',
   693→            description:
   694→              fix.description || `Deleted line(s) ${targetLine + 1}-${targetLine + deleteCount}`,
   695→          });
   696→          break;
   697→        }
   698→      }
   699→    }
   700→
   701→    fixedCode = lines.join('\n');
   702→
   703→    if (!dryRun) {
   704→      await this.storeInWorkingMemory(
   705→        'debugging:lastFix',
   706→        {
   707→          fix,
   708→          changes,
   709→          timestamp: new Date(),
   710→          applied: true,
   711→        },
   712→        300000,
   713→      );
   714→    }
   715→
   716→    this.logger.log(
   717→      `${dryRun ? 'Dry run: ' : ''}Applied fix: ${changes.length} change(s), type=${fixType}`,
   718→    );
   719→
   720→    return { fixedCode, applied: !dryRun && changes.length > 0, changes, dryRun };
   721→  }

   723→  private async validateFix(params: {
   724→    originalCode: string;
   725→    fixedCode: string;
   726→    error: Record<string, any>;
   727→    language: string;
   728→  }): Promise<{
   729→    valid: boolean;
   730→    residualIssues: ValidationError[];
   731→    regressionRisk: 'low' | 'medium' | 'high';
   732→    verificationSteps: string[];
   733→  }> {
   734→    const { originalCode, fixedCode, error, language } = params;
   735→
   736→    if (!originalCode || typeof originalCode !== 'string') {
   737→      throw new Error('Original code is required');
   738→    }
   739→    if (!fixedCode || typeof fixedCode !== 'string') {
   740→      throw new Error('Fixed code is required');
   741→    }
   742→    if (!error) {
   743→      throw new Error('Error object is required for validation');
   744→    }
   745→
   746→    const errorMessage = error.message || error.msg || error.toString?.() || 'Unknown error';
   747→    const errorType = error.name || error.type || '';
   748→    const residualIssues: ValidationError[] = [];
   749→    const verificationSteps: string[] = [];
   750→
   751→    // 1. Check that the fixed code is different from original
   752→    if (originalCode === fixedCode) {
   753→      residualIssues.push({
   754→        type: 'no-change',
   755→        message: 'Fixed code is identical to original code — no fix was applied',
   756→        severity: 'high',
   757→      });
   758→    }
   759→    verificationSteps.push('Verify that the fixed code differs from the original');
   760→
   761→    // 2. Check for syntax validity (basic structural check)
   762→    const syntaxCheck = this.performBasicSyntaxCheck(fixedCode, language);
   763→    if (!syntaxCheck.valid) {
   764→      residualIssues.push({
   765→        type: 'syntax',
   766→        message: `Syntax issue after fix: ${syntaxCheck.message}`,
   767→        severity: 'high',
   768→      });
   769→    }
   770→    verificationSteps.push('Verify syntax correctness of fixed code');
   771→
   772→    // 3. Check that the original error pattern is no longer present
   773→    const errorPatternPresent = this.checkErrorPatternExists(
   774→      fixedCode,
   775→      errorMessage,
   776→      errorType,
   777→      language,
   778→    );
   779→    if (errorPatternPresent) {
   780→      residualIssues.push({
   781→        type: 'error-pattern',
   782→        message: 'The original error pattern may still exist in the fixed code',
   783→        severity: 'medium',
   784→      });
   785→    }
   786→    verificationSteps.push('Verify that the original error pattern is resolved');
   787→
   788→    // 4. Check for new issues introduced by the fix
   789→    const newIssues = this.detectNewIssues(originalCode, fixedCode, language);
   790→    residualIssues.push(...newIssues);
   791→    verificationSteps.push('Check for regressions or new issues introduced by the fix');
   792→
   793→    // 5. Check balanced brackets/braces
   794→    const balanceCheck = this.checkBalancedDelimiters(fixedCode);
   795→    if (!balanceCheck.balanced) {
   796→      residualIssues.push({
   797→        type: 'structure',
   798→        message: `Unbalanced delimiters: ${balanceCheck.message}`,
   799→        severity: 'high',
   800→      });
   801→    }
   802→    verificationSteps.push('Verify balanced delimiters (braces, brackets, parentheses)');
   803→
   804→    // 6. Verify imports/dependencies are still valid
   805→    const importCheck = this.checkImportConsistency(originalCode, fixedCode, language);
   806→    if (!importCheck.consistent) {
   807→      residualIssues.push({
   808→        type: 'imports',
   809→        message: `Import inconsistency: ${importCheck.message}`,
   810→        severity: 'medium',
   811→      });
   812→    }
   813→    verificationSteps.push('Verify import statements are still valid after fix');
   814→
   815→    // Determine overall validity
   816→    const highSeverityIssues = residualIssues.filter((i) => i.severity === 'high');
   817→    const valid = highSeverityIssues.length === 0 && !errorPatternPresent;
   818→
   819→    // Determine regression risk
   820→    const totalIssues = residualIssues.length;
   821→    let regressionRisk: 'low' | 'medium' | 'high' = 'low';
   822→    if (highSeverityIssues.length > 0) regressionRisk = 'high';
   823→    else if (totalIssues > 2) regressionRisk = 'medium';
   824→
   825→    this.logger.log(
   826→      `Fix validation: valid=${valid}, ${residualIssues.length} residual issue(s), risk=${regressionRisk}`,
   827→    );
   828→
   829→    return { valid, residualIssues, regressionRisk, verificationSteps };
   830→  }

   834→  private classifyError(message: string): string {
   835→    const lower = message.toLowerCase();
   836→
   837→    if (lower.includes('typeerror') || lower.includes('type error')) return 'TypeError';
   838→    if (lower.includes('referenceerror') || lower.includes('is not defined'))

   841→    if (lower.includes('rangeerror') || lower.includes('out of range')) return 'RangeError';
   842→    if (lower.includes('null') || lower.includes('undefined')) return 'NullReferenceError';
   843→    if (lower.includes('timeout') || lower.includes('timed out')) return 'TimeoutError';
   844→    if (lower.includes('permission') || lower.includes('access denied')) return 'PermissionError';
   845→    if (lower.includes('not found') || lower.includes('enoent')) return 'NotFoundError';
   846→    if (lower.includes('connection') || lower.includes('econnrefused')) return 'ConnectionError';
   847→    if (lower.includes('memory') || lower.includes('heap')) return 'MemoryError';
   848→
   849→    return 'UnknownError';

   852→  private determineRootCause(
   853→    errorMessage: string,
   854→    stack: string,
   855→    code?: string,
   856→    language?: string,
   857→  ): string {
   858→    const lower = errorMessage.toLowerCase();
   859→
   860→    // Null/undefined access
   861→    if (lower.includes('cannot read propert') || lower.includes('cannot read properties')) {
   862→      const match = errorMessage.match(/cannot read (?:properties of|propert[yies])\s+(\w+)/i);
   863→      const prop = match ? match[1] : 'property';
   864→      return `Attempting to access "${prop}" on a null or undefined value. The object is likely not initialized or the access path contains a null/undefined intermediate.`;
   865→    }
   866→
   867→    // Not defined
   868→    if (lower.includes('is not defined')) {
   869→      const match = errorMessage.match(/(\w+)\s+is not defined/i);
   870→      const name = match ? match[1] : 'variable';
   871→      return `"${name}" is referenced but not declared or imported. Check for typos, missing imports, or scope issues.`;
   872→    }
   873→
   874→    // Type mismatch
   875→    if (lower.includes('is not a function')) {
   876→      const match = errorMessage.match(/(\w+)\s+is not a function/i);
   877→      const name = match ? match[1] : 'value';
   878→      return `"${name}" is being called as a function but is not of callable type. Verify the variable type and method availability.`;
   879→    }
   880→
   881→    // Cannot read properties of undefined
   882→    if (lower.includes('is not iterable')) {
   883→      return 'Attempting to iterate over a non-iterable value. Ensure the target is an array, string, or other iterable object.';
   884→    }
   885→
   886→    // Syntax errors
   887→    if (lower.includes('unexpected token') || lower.includes('unexpected end')) {
   888→      return 'Syntax error in the code. Check for missing brackets, parentheses, semicolons, or string delimiters.';
   889→    }
   890→
   891→    // Timeout
   892→    if (lower.includes('timeout') || lower.includes('timed out')) {
   893→      return 'An operation exceeded its time limit. This may be caused by an infinite loop, unresponsive external service, or insufficient timeout configuration.';
   894→    }
   895→
   896→    // Default
   897→    return `Error: ${errorMessage}. Analyze the stack trace and code context to identify the specific cause.`;
   898→  }

   900→  private determineSeverity(
   901→    errorType: string,
   902→    errorMessage: string,
   903→    code?: string,
   904→  ): 'low' | 'medium' | 'high' | 'critical' {
   905→    // Critical: unhandled exceptions, security errors
   906→    if (
   907→      errorType.includes('Security') ||
   908→      errorType.includes('Permission') ||
   909→      errorMessage.toLowerCase().includes('fatal')
   910→    ) {
   911→      return 'critical';
   912→    }
   913→
   914→    // High: runtime errors that crash the application
   915→    if (
   916→      errorType.includes('TypeError') ||
   917→      errorType.includes('ReferenceError') ||
   918→      errorType.includes('NullReference')
   919→    ) {
   920→      return 'high';
   921→    }
   922→
   923→    // Medium: recoverable errors
   924→    if (
   925→      errorType.includes('RangeError') ||
   926→      errorType.includes('ConnectionError') ||
   927→      errorType.includes('Timeout')
   928→    ) {
   929→      return 'medium';
   930→    }
   931→
   932→    // Low: syntax errors caught at compile time, warnings
   933→    if (errorType.includes('SyntaxError')) {
   934→      return 'low';
   935→    }
   936→
   937→    return 'medium';
   938→  }

  1493→  private checkBalancedDelimiters(code: string): { balanced: boolean; message: string } {
  1494→    const stack: string[] = [];
  1495→    const pairs: Record<string, string> = { '{': '}', '(': ')', '[': ']' };
  1496→
  1497→    for (let i = 0; i < code.length; i++) {
  1498→      const char = code[i];
  1499→      if (char in pairs) {
  1500→        stack.push(char);
  1501→      } else if (Object.values(pairs).includes(char)) {
  1502→        if (stack.length === 0) {
  1503→          return {
  1504→            balanced: false,
  1505→            message: `Unmatched closing delimiter '${char}' at position ${i}`,
  1506→          };
  1507→        }
  1508→        const last = stack.pop()!;
  1509→        if (pairs[last] !== char) {
  1510→          return {
  1511→            balanced: false,
  1512→            message: `Mismatched delimiters: '${last}' and '${char}' at position ${i}`,
  1513→          };
  1514→        }
  1515→      }
  1516→    }
  1517→
  1518→    if (stack.length > 0) {
  1519→      return { balanced: false, message: `Unclosed delimiter(s): ${stack.join(', ')}` };
  1520→    }
  1521→
  1522→    return { balanced: true, message: 'All delimiters are balanced' };
  1523→  }
