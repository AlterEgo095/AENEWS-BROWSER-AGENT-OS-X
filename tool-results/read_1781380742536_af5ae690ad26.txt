   319→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   320→    const startTime = Date.now();
   321→    const { action, ...params } = input.payload;
   322→
   323→    if (!action) {
   324→      return this.createAgentOutput(
   325→        input.taskId,
   326→        false,
   327→        null,
   328→        'Missing required parameter: action',
   329→        startTime,
   330→      );
   331→    }
   332→
   333→    const supportedActions = [
   334→      'createDocument',
   335→      'editDocument',
   336→      'convertFormat',
   337→      'extractText',
   338→      'mergeDocuments',
   339→      'applyTemplate',
   340→    ];
   341→
   342→    if (!supportedActions.includes(action)) {
   343→      return this.createAgentOutput(
   344→        input.taskId,
   345→        false,
   346→        null,
   347→        `Unknown document action: ${action}. Supported: ${supportedActions.join(', ')}`,
   348→        startTime,
   349→      );
   350→    }
   351→
   352→    try {
   353→      const tool = this.getTool(action);
   354→      if (!tool) {
   355→        return this.createAgentOutput(
   356→          input.taskId,
   357→          false,
   358→          null,
   359→          `Tool not found: ${action}`,
   360→          startTime,
   361→        );
   362→      }
   363→
   364→      const result = await tool.execute(params);
   365→
   366→      await this.storeInWorkingMemory(
   367→        `document:last:${action}`,
   368→        { params, result, timestamp: new Date() },
   369→        300000,
   370→      );
   371→
   372→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   373→    } catch (error) {
   374→      const msg = (error as Error).message;
   375→      this.logger.error(`Document execution failed for ${action}: ${msg}`);
   376→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   377→    }
   378→  }

   389→  private async createDocument(params: {
   390→    title: string;
   391→    format: string;
   392→    content?: string;
   393→    author?: string;
   394→    template?: string;
   395→    metadata?: Record<string, any>;
   396→  }): Promise<{
   397→    documentId: string;
   398→    format: string;
   399→    createdAt: string;
   400→  }> {
   401→    const {
   402→      title,
   403→      format,
   404→      content = '',
   405→      author = 'agent@aenews.system',
   406→      template,
   407→      metadata = {},
   408→    } = params;
   409→
   410→    if (!title || typeof title !== 'string') {
   411→      throw new Error('A valid document title is required');
   412→    }
   413→
   414→    const validFormats = ['docx', 'pdf', 'html', 'md', 'txt', 'rtf'];
   415→    if (!validFormats.includes(format)) {
   416→      throw new Error(`Invalid format: ${format}. Supported: ${validFormats.join(', ')}`);
   417→    }
   418→
   419→    let documentContent = content;
   420→
   421→    // Apply template if specified
   422→    if (template) {
   423→      const tmpl = this.templates.get(template);
   424→      if (tmpl) {
   425→        documentContent = tmpl.content;
   426→        if (metadata.variables) {
   427→          for (const [key, value] of Object.entries(metadata.variables)) {
   428→            documentContent = documentContent.replaceAll(`{{${key}}}`, String(value));
   429→          }
   430→        }
   431→      }
   432→    }
   433→
   434→    const documentId = this.generateDocumentId();
   435→    const document: Document = {
   436→      id: documentId,
   437→      title,
   438→      format,
   439→      content: documentContent,
   440→      author,
   441→      metadata,
   442→      createdAt: new Date(),
   443→      updatedAt: new Date(),
   444→      version: 1,
   445→      pageCount: this.estimatePageCount(documentContent),
   446→    };
   447→
   448→    this.documents.set(documentId, document);
   449→
   450→    this.logger.log(`Created document: ${documentId}, title="${title}", format=${format}`);
   451→
   452→    return {
   453→      documentId,
   454→      format,
   455→      createdAt: document.createdAt.toISOString(),
   456→    };
   457→  }

   459→  private async editDocument(params: { documentId: string; operations: EditOperation[] }): Promise<{
   460→    documentId: string;
   461→    appliedOperations: number;
   462→    status: string;
   463→  }> {
   464→    const { documentId, operations } = params;
   465→
   466→    if (!documentId || typeof documentId !== 'string') {
   467→      throw new Error('A valid documentId is required');
   468→    }
   469→    if (!operations || !Array.isArray(operations) || operations.length === 0) {
   470→      throw new Error('At least one edit operation is required');
   471→    }
   472→
   473→    const document = this.documents.get(documentId);
   474→    if (!document) {
   475→      throw new Error(`Document not found: ${documentId}`);
   476→    }
   477→
   478→    let content = document.content;
   479→    let appliedOperations = 0;
   480→
   481→    for (const op of operations) {
   482→      try {
   483→        switch (op.type) {
   484→          case 'insert': {
   485→            if (op.position === undefined || op.text === undefined) {
   486→              throw new Error('Insert operation requires position and text');
   487→            }
   488→            const pos = Math.min(op.position, content.length);
   489→            content = content.slice(0, pos) + op.text + content.slice(pos);
   490→            appliedOperations++;
   491→            break;
   492→          }
   493→          case 'replace': {
   494→            if (op.search === undefined || op.replacement === undefined) {
   495→              throw new Error('Replace operation requires search and replacement');
   496→            }
   497→            const occurrences = (content.match(new RegExp(this.escapeRegex(op.search), 'g')) || [])
   498→              .length;
   499→            content = content.replaceAll(op.search, op.replacement);
   500→            if (occurrences > 0) appliedOperations++;
   501→            break;
   502→          }
   503→          case 'delete': {
   504→            if (op.start === undefined || op.end === undefined) {
   505→              throw new Error('Delete operation requires start and end positions');
   506→            }
   507→            content = content.slice(0, op.start) + content.slice(op.end);
   508→            appliedOperations++;
   509→            break;
   510→          }
   511→          case 'append': {
   512→            if (op.text === undefined) {
   513→              throw new Error('Append operation requires text');
   514→            }
   515→            content += op.text;
   516→            appliedOperations++;
   517→            break;
   518→          }
   519→          default:
   520→            this.logger.warn(`Unknown edit operation type: ${op.type}`);
   521→        }
   522→      } catch (opError) {
   523→        this.logger.warn(`Edit operation failed: ${(opError as Error).message}, skipping`);
   524→      }
   525→    }
   526→
   527→    document.content = content;
   528→    document.updatedAt = new Date();
   529→    document.version++;
   530→    document.pageCount = this.estimatePageCount(content);
   531→
   532→    this.logger.log(
   533→      `Edited document: ${documentId}, applied=${appliedOperations}/${operations.length}`,
   534→    );
   535→
   536→    return {
   537→      documentId,
   538→      appliedOperations,
   539→      status: 'updated',
   540→    };
   541→  }

   543→  private async convertFormat(params: {
   544→    documentId: string;
   545→    targetFormat: string;
   546→    options?: Record<string, any>;
   547→  }): Promise<{
   548→    documentId: string;
   549→    convertedDocumentId: string;
   550→    sourceFormat: string;
   551→    targetFormat: string;
   552→    status: string;
   553→  }> {
   554→    const { documentId, targetFormat, options = {} } = params;
   555→
   556→    if (!documentId || typeof documentId !== 'string') {
   557→      throw new Error('A valid documentId is required');
   558→    }
   559→
   560→    const validFormats = ['docx', 'pdf', 'html', 'md', 'txt', 'rtf'];
   561→    if (!validFormats.includes(targetFormat)) {
   562→      throw new Error(
   563→        `Invalid target format: ${targetFormat}. Supported: ${validFormats.join(', ')}`,
   564→      );
   565→    }
   566→
   567→    const sourceDocument = this.documents.get(documentId);
   568→    if (!sourceDocument) {
   569→      throw new Error(`Document not found: ${documentId}`);
   570→    }
   571→
   572→    if (sourceDocument.format === targetFormat) {
   573→      throw new Error(`Document is already in ${targetFormat} format`);
   574→    }
   575→
   576→    // Convert content based on source and target format
   577→    const convertedContent = this.performConversion(
   578→      sourceDocument.content,
   579→      sourceDocument.format,
   580→      targetFormat,
   581→      options,
   582→    );
   583→
   584→    const convertedDocumentId = this.generateDocumentId();
   585→    const convertedDocument: Document = {
   586→      id: convertedDocumentId,
   587→      title: sourceDocument.title,
   588→      format: targetFormat,
   589→      content: convertedContent,
   590→      author: sourceDocument.author,
   591→      metadata: {
   592→        ...sourceDocument.metadata,
   593→        convertedFrom: documentId,
   594→        sourceFormat: sourceDocument.format,
   595→        convertedAt: new Date().toISOString(),
   596→      },
   597→      createdAt: new Date(),
   598→      updatedAt: new Date(),
   599→      version: 1,
   600→      pageCount: this.estimatePageCount(convertedContent),
   601→    };
   602→
   603→    this.documents.set(convertedDocumentId, convertedDocument);
   604→
   605→    this.logger.log(
   606→      `Converted document: ${documentId} (${sourceDocument.format}) -> ${convertedDocumentId} (${targetFormat})`,
   607→    );
   608→
   609→    return {
   610→      documentId,
   611→      convertedDocumentId,
   612→      sourceFormat: sourceDocument.format,
   613→      targetFormat,
   614→      status: 'converted',
   615→    };
   616→  }

   618→  private async extractText(params: {
   619→    documentId: string;
   620→    includeMetadata?: boolean;
   621→    pageRange?: { start: number; end: number };
   622→  }): Promise<{
   623→    documentId: string;
   624→    text: string;
   625→    pageCount: number;
   626→    wordCount: number;
   627→  }> {
   628→    const { documentId, includeMetadata = false, pageRange } = params;
   629→
   630→    if (!documentId || typeof documentId !== 'string') {
   631→      throw new Error('A valid documentId is required');
   632→    }
   633→
   634→    const document = this.documents.get(documentId);
   635→    if (!document) {
   636→      throw new Error(`Document not found: ${documentId}`);
   637→    }
   638→
   639→    // Strip format-specific markup to get plain text
   640→    let text = this.stripMarkup(document.content, document.format);
   641→
   642→    // Apply page range if specified
   643→    if (pageRange) {
   644→      const lines = text.split('\n');
   645→      const linesPerPage = 50; // Approximate lines per page
   646→      const startLine = pageRange.start * linesPerPage;
   647→      const endLine = pageRange.end * linesPerPage;
   648→      text = lines.slice(startLine, endLine).join('\n');
   649→    }
   650→
   651→    const wordCount = text.split(/\s+/).filter((w) => w.length > 0).length;
   652→
   653→    const result: any = {
   654→      documentId,
   655→      text,
   656→      pageCount: document.pageCount,
   657→      wordCount,
   658→    };
   659→
   660→    if (includeMetadata) {
   661→      result.metadata = {
   662→        title: document.title,
   663→        author: document.author,
   664→        format: document.format,
   665→        createdAt: document.createdAt,
   666→        updatedAt: document.updatedAt,
   667→        version: document.version,
   668→      };
   669→    }
   670→
   671→    this.logger.log(`Extracted text from document: ${documentId}, words=${wordCount}`);
   672→
   673→    return result;
   674→  }

   676→  private async mergeDocuments(params: {
   677→    documentIds: string[];
   678→    outputTitle?: string;
   679→    outputFormat?: string;
   680→    separator?: string;
   681→  }): Promise<{
   682→    mergedDocumentId: string;
   683→    sourceDocumentCount: number;
   684→    status: string;
   685→  }> {
   686→    const { documentIds, outputTitle, outputFormat, separator = '\n\n---\n\n' } = params;
   687→
   688→    if (!documentIds || !Array.isArray(documentIds) || documentIds.length < 2) {
   689→      throw new Error('At least two document IDs are required for merging');
   690→    }
   691→
   692→    const sourceDocs: Document[] = [];
   693→    for (const id of documentIds) {
   694→      const doc = this.documents.get(id);
   695→      if (!doc) {
   696→        throw new Error(`Document not found: ${id}`);
   697→      }
   698→      sourceDocs.push(doc);
   699→    }
   700→
   701→    // Determine output format (use first document's format if not specified)
   702→    const format = outputFormat || sourceDocs[0].format;
   703→    const validFormats = ['docx', 'pdf', 'html', 'md', 'txt', 'rtf'];
   704→    if (!validFormats.includes(format)) {
   705→      throw new Error(`Invalid output format: ${format}`);
   706→    }
   707→
   708→    // Merge contents
   709→    const mergedContent = sourceDocs
   710→      .map((doc) => this.stripMarkup(doc.content, doc.format))
   711→      .join(separator);
   712→
   713→    const title = outputTitle || `Merged: ${sourceDocs.map((d) => d.title).join(' + ')}`;
   714→
   715→    const mergedDocumentId = this.generateDocumentId();
   716→    const mergedDocument: Document = {
   717→      id: mergedDocumentId,
   718→      title,
   719→      format,
   720→      content: this.applyFormatMarkup(mergedContent, format),
   721→      author: 'agent@aenews.system',
   722→      metadata: {
   723→        mergedFrom: documentIds,
   724→        mergedAt: new Date().toISOString(),
   725→      },
   726→      createdAt: new Date(),
   727→      updatedAt: new Date(),
   728→      version: 1,
   729→      pageCount: this.estimatePageCount(mergedContent),
   730→    };
   731→
   732→    this.documents.set(mergedDocumentId, mergedDocument);
   733→
   734→    this.logger.log(`Merged ${documentIds.length} documents into: ${mergedDocumentId}`);
   735→
   736→    return {
   737→      mergedDocumentId,
   738→      sourceDocumentCount: documentIds.length,
   739→      status: 'merged',
   740→    };
   741→  }

   743→  private async applyTemplate(params: {
   744→    documentId?: string;
   745→    templateId: string;
   746→    variables?: Record<string, string>;
   747→  }): Promise<{
   748→    documentId: string;
   749→    templateId: string;
   750→    substitutedVars: string[];
   751→    status: string;
   752→  }> {
   753→    const { documentId, templateId, variables = {} } = params;
   754→
   755→    if (!templateId || typeof templateId !== 'string') {
   756→      throw new Error('A valid templateId is required');
   757→    }
   758→
   759→    const template = this.templates.get(templateId);
   760→    if (!template) {
   761→      const available = Array.from(this.templates.keys()).join(', ');
   762→      throw new Error(`Template not found: ${templateId}. Available: ${available}`);
   763→    }
   764→
   765→    let content = template.content;
   766→    const substitutedVars: string[] = [];
   767→
   768→    // Substitute variables
   769→    for (const [key, value] of Object.entries(variables)) {
   770→      const placeholder = `{{${key}}}`;
   771→      if (content.includes(placeholder)) {
   772→        content = content.replaceAll(placeholder, value);
   773→        substitutedVars.push(key);
   774→      }
   775→    }
   776→
   777→    // Check for unsubstituted variables
   778→    const remainingPlaceholders = content.match(/\{\{(\w+)\}\}/g);
   779→    if (remainingPlaceholders) {
   780→      const missingVars = remainingPlaceholders.map((p) => p.replace(/[{}]/g, ''));
   781→      this.logger.warn(`Unsubstituted template variables: ${missingVars.join(', ')}`);
   782→    }
   783→
   784→    let targetDocumentId = documentId;
   785→
   786→    if (targetDocumentId) {
   787→      // Apply to existing document
   788→      const document = this.documents.get(targetDocumentId);
   789→      if (!document) {
   790→        throw new Error(`Document not found: ${targetDocumentId}`);
   791→      }
   792→      document.content = content;
   793→      document.updatedAt = new Date();
   794→      document.version++;
   795→      document.pageCount = this.estimatePageCount(content);
   796→    } else {
   797→      // Create new document from template
   798→      targetDocumentId = this.generateDocumentId();
   799→      const newDoc: Document = {
   800→        id: targetDocumentId,
   801→        title: `Document from template: ${template.name}`,
   802→        format: template.format,
   803→        content,
   804→        author: 'agent@aenews.system',
   805→        metadata: {
   806→          templateId,
   807→          substitutedVars,
   808→        },
   809→        createdAt: new Date(),
   810→        updatedAt: new Date(),
   811→        version: 1,
   812→        pageCount: this.estimatePageCount(content),
   813→      };
   814→      this.documents.set(targetDocumentId, newDoc);
   815→    }
   816→
   817→    this.logger.log(
   818→      `Applied template: ${templateId}, doc=${targetDocumentId}, substituted=${substitutedVars.length} var(s)`,
   819→    );
   820→
   821→    return {
   822→      documentId: targetDocumentId,
   823→      templateId,
   824→      substitutedVars,
   825→      status: 'applied',
   826→    };
   827→  }

   898→  private generateDocumentId(): string {
   899→    this.documentCounter++;
   900→    return `doc-${Date.now()}-${this.documentCounter}`;

   903→  private estimatePageCount(content: string): number {
   904→    if (!content) return 0;
   905→    const words = content.split(/\s+/).filter((w) => w.length > 0).length;
   906→    return Math.max(1, Math.ceil(words / 250)); // ~250 words per page
   907→  }

   909→  private stripMarkup(content: string, format: string): string {
   910→    switch (format) {
   911→      case 'html': {
   912→        // Remove HTML tags
   913→        return content
   914→          .replace(/<[^>]*>/g, '')
   915→          .replace(/&nbsp;/g, ' ')
   916→          .replace(/&amp;/g, '&')
   917→          .replace(/&lt;/g, '<')
   918→          .replace(/&gt;/g, '>')
   919→          .replace(/&quot;/g, '"')
   920→          .trim();
   921→      }
   922→      case 'md': {
   923→        // Remove markdown formatting
   924→        return content
   925→          .replace(/^#{1,6}\s+/gm, '')
   926→          .replace(/\*\*(.*?)\*\*/g, '$1')
   927→          .replace(/\*(.*?)\*/g, '$1')
   928→          .replace(/\[(.*?)\]\(.*?\)/g, '$1')
   929→          .replace(/^[-*]\s+/gm, '- ')
   930→          .trim();
   931→      }
   932→      case 'rtf': {
   933→        // Remove RTF control words
   934→        return content
   935→          .replace(/\\[a-z]+\d*\s?/gi, '')
   936→          .replace(/[{}]/g, '')
   937→          .trim();
   938→      }
   939→      case 'docx':
   940→      case 'pdf':
   941→      case 'txt':
   942→      default:
   943→        return content;
   944→    }
   945→  }
   946→
   947→  private applyFormatMarkup(text: string, format: string): string {
   948→    switch (format) {
   949→      case 'html':
   950→        return text
   951→          .split('\n')
   952→          .map((line) => (line.trim() ? `<p>${line}</p>` : ''))
   953→          .join('\n');
   954→      case 'md':
   955→        return text;
   956→      case 'txt':
   957→        return text;
   958→      default:
   959→        return text;
   960→    }
   961→  }
   962→
   963→  private performConversion(
   964→    content: string,
   965→    sourceFormat: string,
   966→    targetFormat: string,
   967→    options: Record<string, any>,
   968→  ): string {
   969→    // First, strip source format markup to plain text
   970→    const plainText = this.stripMarkup(content, sourceFormat);
   971→    // Then, apply target format markup
   972→    return this.applyFormatMarkup(plainText, targetFormat);
   973→  }

   975→  private escapeRegex(str: string): string {
   976→    return str.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
   977→  }
