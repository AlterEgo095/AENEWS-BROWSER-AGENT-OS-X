   370→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   371→    const startTime = Date.now();
   372→    const { action, ...params } = input.payload;
   373→
   374→    if (!action) {
   375→      return this.createAgentOutput(
   376→        input.taskId,
   377→        false,
   378→        null,
   379→        'Missing required parameter: action',
   380→        startTime,
   381→      );
   382→    }
   383→
   384→    const supportedActions = [
   385→      'createEvent',
   386→      'updateEvent',
   387→      'deleteEvent',
   388→      'findFreeSlots',
   389→      'getSchedule',
   390→      'sendInvitation',
   391→      'setReminder',
   392→    ];
   393→
   394→    if (!supportedActions.includes(action)) {
   395→      return this.createAgentOutput(
   396→        input.taskId,
   397→        false,
   398→        null,
   399→        `Unknown calendar action: ${action}. Supported: ${supportedActions.join(', ')}`,
   400→        startTime,
   401→      );
   402→    }
   403→
   404→    try {
   405→      const tool = this.getTool(action);
   406→      if (!tool) {
   407→        return this.createAgentOutput(
   408→          input.taskId,
   409→          false,
   410→          null,
   411→          `Tool not found: ${action}`,
   412→          startTime,
   413→        );
   414→      }
   415→
   416→      const result = await tool.execute(params);
   417→
   418→      await this.storeInWorkingMemory(
   419→        `calendar:last:${action}`,
   420→        { params, result, timestamp: new Date() },
   421→        300000,
   422→      );
   423→
   424→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   425→    } catch (error) {
   426→      const msg = (error as Error).message;
   427→      this.logger.error(`Calendar execution failed for ${action}: ${msg}`);
   428→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   429→    }
   430→  }

   441→  private async createEvent(params: {
   442→    title: string;
   443→    description?: string;
   444→    startTime: string;
   445→    endTime: string;
   446→    location?: string;
   447→    attendees?: string[];
   448→    isRecurring?: boolean;
   449→    recurrenceRule?: string;
   450→    color?: string;
   451→  }): Promise<{
   452→    eventId: string;
   453→    status: string;
   454→    createdAt: string;
   455→  }> {
   456→    const {
   457→      title,
   458→      description = '',
   459→      startTime,
   460→      endTime,
   461→      location = '',
   462→      attendees = [],
   463→      isRecurring = false,
   464→      recurrenceRule,
   465→      color = 'default',
   466→    } = params;
   467→
   468→    if (!title || typeof title !== 'string') {
   469→      throw new Error('A valid event title is required');
   470→    }
   471→    if (!startTime || !endTime) {
   472→      throw new Error('Both startTime and endTime are required');
   473→    }
   474→
   475→    const start = new Date(startTime);
   476→    const end = new Date(endTime);
   477→
   478→    if (isNaN(start.getTime())) {
   479→      throw new Error('Invalid startTime: must be a valid ISO date string');
   480→    }
   481→    if (isNaN(end.getTime())) {
   482→      throw new Error('Invalid endTime: must be a valid ISO date string');
   483→    }
   484→    if (end <= start) {
   485→      throw new Error('endTime must be after startTime');
   486→    }
   487→
   488→    // Check for conflicts
   489→    const conflicts = this.findConflicts(start, end);
   490→    if (conflicts.length > 0) {
   491→      this.logger.warn(`Event "${title}" conflicts with ${conflicts.length} existing event(s)`);
   492→    }
   493→
   494→    const eventId = this.generateEventId();
   495→    const eventAttendees: EventAttendee[] = attendees.map((email) => ({
   496→      email,
   497→      responseStatus: 'needsAction' as const,
   498→    }));
   499→
   500→    const event: CalendarEvent = {
   501→      id: eventId,
   502→      title,
   503→      description,
   504→      startTime: start,
   505→      endTime: end,
   506→      location,
   507→      organizer: 'agent@aenews.system',
   508→      attendees: eventAttendees,
   509→      status: 'confirmed',
   510→      isRecurring,
   511→      recurrenceRule,
   512→      reminders: [],
   513→      color,
   514→      createdAt: new Date(),
   515→      updatedAt: new Date(),
   516→    };
   517→
   518→    this.events.set(eventId, event);
   519→
   520→    this.logger.log(
   521→      `Created event: ${eventId}, title="${title}", ${start.toISOString()} - ${end.toISOString()}`,
   522→    );
   523→
   524→    return {
   525→      eventId,
   526→      status: 'confirmed',
   527→      createdAt: event.createdAt.toISOString(),
   528→    };
   529→  }

   531→  private async updateEvent(params: {
   532→    eventId: string;
   533→    title?: string;
   534→    description?: string;
   535→    startTime?: string;
   536→    endTime?: string;
   537→    location?: string;
   538→    attendees?: string[];
   539→  }): Promise<{
   540→    eventId: string;
   541→    updatedFields: string[];
   542→    status: string;
   543→  }> {
   544→    const { eventId, title, description, startTime, endTime, location, attendees } = params;
   545→
   546→    if (!eventId || typeof eventId !== 'string') {
   547→      throw new Error('A valid eventId is required');
   548→    }
   549→
   550→    const event = this.events.get(eventId);
   551→    if (!event) {
   552→      throw new Error(`Event not found: ${eventId}`);
   553→    }
   554→
   555→    if (event.status === 'cancelled') {
   556→      throw new Error(`Cannot update a cancelled event: ${eventId}`);
   557→    }
   558→
   559→    const updatedFields: string[] = [];
   560→
   561→    if (title !== undefined) {
   562→      event.title = title;
   563→      updatedFields.push('title');
   564→    }
   565→    if (description !== undefined) {
   566→      event.description = description;
   567→      updatedFields.push('description');
   568→    }
   569→    if (startTime !== undefined) {
   570→      const start = new Date(startTime);
   571→      if (isNaN(start.getTime())) {
   572→        throw new Error('Invalid startTime');
   573→      }
   574→      event.startTime = start;
   575→      updatedFields.push('startTime');
   576→    }
   577→    if (endTime !== undefined) {
   578→      const end = new Date(endTime);
   579→      if (isNaN(end.getTime())) {
   580→        throw new Error('Invalid endTime');
   581→      }
   582→      event.endTime = end;
   583→      updatedFields.push('endTime');
   584→    }
   585→    if (location !== undefined) {
   586→      event.location = location;
   587→      updatedFields.push('location');
   588→    }
   589→    if (attendees !== undefined) {
   590→      event.attendees = attendees.map((email) => ({
   591→        email,
   592→        responseStatus: 'needsAction' as const,
   593→      }));
   594→      updatedFields.push('attendees');
   595→    }
   596→
   597→    // Validate time range after update
   598→    if (event.endTime <= event.startTime) {
   599→      throw new Error('endTime must be after startTime');
   600→    }
   601→
   602→    event.updatedAt = new Date();
   603→
   604→    this.logger.log(`Updated event: ${eventId}, fields=[${updatedFields.join(',')}]`);
   605→
   606→    return {
   607→      eventId,
   608→      updatedFields,
   609→      status: event.status,
   610→    };
   611→  }

   614→    eventId: string;

   616→    cancellationReason?: string;
   617→  }): Promise<{
   618→    eventId: string;

   620→    notifiedAttendees: number;
   621→  }> {

   623→
   624→    if (!eventId || typeof eventId !== 'string') {
   625→      throw new Error('A valid eventId is required');
   626→    }
   627→
   628→    const event = this.events.get(eventId);
   629→    if (!event) {

   644→    );

   646→    return {
   647→      eventId,
   648→      status: 'cancelled',
   649→      notifiedAttendees,
   650→    };
   651→  }

   653→  private async findFreeSlots(params: {
   654→    dateFrom: string;
   655→    dateTo: string;
   656→    durationMinutes: number;
   657→    workingHoursOnly?: boolean;
   658→    attendees?: string[];
   659→  }): Promise<{
   660→    freeSlots: FreeSlot[];
   661→    totalSlots: number;
   662→  }> {
   663→    const { dateFrom, dateTo, durationMinutes, workingHoursOnly = true, attendees = [] } = params;
   664→
   665→    if (!dateFrom || !dateTo) {
   666→      throw new Error('Both dateFrom and dateTo are required');
   667→    }
   668→    if (!durationMinutes || durationMinutes <= 0) {
   669→      throw new Error('durationMinutes must be a positive number');
   670→    }
   671→
   672→    const rangeStart = new Date(dateFrom);
   673→    const rangeEnd = new Date(dateTo);
   674→
   675→    if (isNaN(rangeStart.getTime()) || isNaN(rangeEnd.getTime())) {
   676→      throw new Error('Invalid date range: must be valid ISO date strings');
   677→    }
   678→    if (rangeEnd <= rangeStart) {
   679→      throw new Error('dateTo must be after dateFrom');
   680→    }
   681→
   682→    // Define working hours (9 AM - 5 PM by default)
   683→    const workStartHour = 9;
   684→    const workEndHour = 17;
   685→
   686→    // Get all confirmed events in the range
   687→    const busyEvents = Array.from(this.events.values()).filter(
   688→      (event) =>
   689→        event.status === 'confirmed' && event.startTime < rangeEnd && event.endTime > rangeStart,
   690→    );
   691→
   692→    // Generate free slots by iterating through each day
   693→    const freeSlots: FreeSlot[] = [];
   694→    const currentDay = new Date(rangeStart);
   695→
   696→    while (currentDay < rangeEnd) {
   697→      const dayStart = new Date(currentDay);
   698→      const dayEnd = new Date(currentDay);
   699→
   700→      if (workingHoursOnly) {
   701→        dayStart.setHours(workStartHour, 0, 0, 0);
   702→        dayEnd.setHours(workEndHour, 0, 0, 0);
   703→      } else {
   704→        dayStart.setHours(0, 0, 0, 0);
   705→        dayEnd.setHours(23, 59, 59, 999);
   706→      }
   707→
   708→      // Adjust for range boundaries
   709→      const slotStart = dayStart < rangeStart ? rangeStart : dayStart;
   710→      const slotEnd = dayEnd > rangeEnd ? rangeEnd : dayEnd;
   711→
   712→      if (slotStart >= slotEnd) {
   713→        currentDay.setDate(currentDay.getDate() + 1);
   714→        continue;
   715→      }
   716→
   717→      // Skip weekends if working hours only
   718→      if (workingHoursOnly && (currentDay.getDay() === 0 || currentDay.getDay() === 6)) {
   719→        currentDay.setDate(currentDay.getDate() + 1);
   720→        continue;
   721→      }
   722→
   723→      // Get busy intervals for this day
   724→      const dayBusy = busyEvents
   725→        .filter((e) => e.startTime < slotEnd && e.endTime > slotStart)
   726→        .map((e) => ({
   727→          start: e.startTime < slotStart ? slotStart : e.startTime,
   728→          end: e.endTime > slotEnd ? slotEnd : e.endTime,
   729→        }))
   730→        .sort((a, b) => a.start.getTime() - b.start.getTime());
   731→
   732→      // Merge overlapping busy intervals
   733→      const mergedBusy = this.mergeIntervals(dayBusy);
   734→
   735→      // Find free slots in between busy intervals
   736→      let cursor = new Date(slotStart);
   737→
   738→      for (const busy of mergedBusy) {
   739→        const freeDuration = (busy.start.getTime() - cursor.getTime()) / 60000;
   740→        if (freeDuration >= durationMinutes) {
   741→          freeSlots.push({
   742→            startTime: new Date(cursor),
   743→            endTime: new Date(busy.start),
   744→            durationMinutes: Math.floor(freeDuration),
   745→          });
   746→        }
   747→        cursor = new Date(busy.end);
   748→      }
   749→
   750→      // Check remaining time after last busy interval
   751→      const remainingDuration = (slotEnd.getTime() - cursor.getTime()) / 60000;
   752→      if (remainingDuration >= durationMinutes) {
   753→        freeSlots.push({
   754→          startTime: new Date(cursor),
   755→          endTime: new Date(slotEnd),
   756→          durationMinutes: Math.floor(remainingDuration),
   757→        });
   758→      }
   759→
   760→      currentDay.setDate(currentDay.getDate() + 1);
   761→    }
   762→
   763→    this.logger.log(
   764→      `Found ${freeSlots.length} free slot(s) for ${durationMinutes}min between ${dateFrom} and ${dateTo}`,
   765→    );
   766→
   767→    return { freeSlots, totalSlots: freeSlots.length };
   768→  }

   770→  private async getSchedule(params: {
   771→    dateFrom: string;
   772→    dateTo: string;
   773→    includeCancelled?: boolean;
   774→  }): Promise<{
   775→    events: CalendarEvent[];
   776→    total: number;
   777→  }> {
   778→    const { dateFrom, dateTo, includeCancelled = false } = params;
   779→
   780→    if (!dateFrom || !dateTo) {
   781→      throw new Error('Both dateFrom and dateTo are required');
   782→    }
   783→
   784→    const rangeStart = new Date(dateFrom);
   785→    const rangeEnd = new Date(dateTo);
   786→
   787→    if (isNaN(rangeStart.getTime()) || isNaN(rangeEnd.getTime())) {
   788→      throw new Error('Invalid date range');
   789→    }
   790→
   791→    let events = Array.from(this.events.values()).filter(
   792→      (event) => event.startTime < rangeEnd && event.endTime > rangeStart,
   793→    );
   794→
   795→    if (!includeCancelled) {
   796→      events = events.filter((e) => e.status !== 'cancelled');
   797→    }
   798→
   799→    // Sort by start time
   800→    events.sort((a, b) => a.startTime.getTime() - b.startTime.getTime());
   801→
   802→    this.logger.log(
   803→      `Retrieved schedule: ${events.length} event(s) between ${dateFrom} and ${dateTo}`,
   804→    );
   805→
   806→    return { events, total: events.length };
   807→  }

   809→  private async sendInvitation(params: {
   810→    eventId: string;
   811→    attendees: string[];
   812→    message?: string;
   813→  }): Promise<{
   814→    eventId: string;
   815→    invitedAttendees: string[];
   816→    status: string;
   817→  }> {
   818→    const { eventId, attendees, message } = params;
   819→
   820→    if (!eventId || typeof eventId !== 'string') {
   821→      throw new Error('A valid eventId is required');
   822→    }
   823→    if (!attendees || !Array.isArray(attendees) || attendees.length === 0) {
   824→      throw new Error('At least one attendee email is required');
   825→    }
   826→
   827→    const event = this.events.get(eventId);
   828→    if (!event) {
   829→      throw new Error(`Event not found: ${eventId}`);
   830→    }
   831→
   832→    if (event.status === 'cancelled') {
   833→      throw new Error(`Cannot send invitations for a cancelled event: ${eventId}`);
   834→    }
   835→
   836→    // Add new attendees
   837→    const invitedAttendees: string[] = [];
   838→    for (const email of attendees) {
   839→      const existing = event.attendees.find((a) => a.email === email);
   840→      if (!existing) {
   841→        event.attendees.push({
   842→          email,
   843→          responseStatus: 'needsAction',
   844→        });
   845→        invitedAttendees.push(email);
   846→      } else {
   847→        this.logger.warn(`Attendee ${email} already invited to event ${eventId}`);
   848→      }
   849→    }
   850→
   851→    event.updatedAt = new Date();
   852→
   853→    this.logger.log(
   854→      `Sent invitation for event: ${eventId}, to=[${invitedAttendees.join(',')}], message=${message ? 'included' : 'none'}`,
   855→    );
   856→
   857→    return {
   858→      eventId,
   859→      invitedAttendees,
   860→      status: 'sent',
   861→    };
   862→  }

   864→  private async setReminder(params: {
   865→    eventId: string;
   866→    minutesBefore: number;
   867→    method?: 'email' | 'popup' | 'sms';
   868→  }): Promise<{
   869→    eventId: string;
   870→    reminderId: string;
   871→    triggerAt: string;
   872→    status: string;
   873→  }> {
   874→    const { eventId, minutesBefore, method = 'popup' } = params;
   875→
   876→    if (!eventId || typeof eventId !== 'string') {
   877→      throw new Error('A valid eventId is required');
   878→    }
   879→    if (!minutesBefore || minutesBefore <= 0) {
   880→      throw new Error('minutesBefore must be a positive number');
   881→    }
   882→
   883→    const event = this.events.get(eventId);
   884→    if (!event) {
   885→      throw new Error(`Event not found: ${eventId}`);
   886→    }
   887→
   888→    if (event.status === 'cancelled') {
   889→      throw new Error(`Cannot set reminder for a cancelled event: ${eventId}`);
   890→    }
   891→
   892→    const reminderId = `reminder-${++this.reminderCounter}`;
   893→    const triggerAt = new Date(event.startTime.getTime() - minutesBefore * 60000);
   894→
   895→    const reminder: EventReminder = {
   896→      id: reminderId,
   897→      minutesBefore,
   898→      method,
   899→      triggered: false,
   900→    };
   901→
   902→    event.reminders.push(reminder);
   903→    event.updatedAt = new Date();
   904→
   905→    this.logger.log(
   906→      `Set reminder for event: ${eventId}, ${minutesBefore}min before, method=${method}`,
   907→    );
   908→
   909→    return {
   910→      eventId,
   911→      reminderId,
   912→      triggerAt: triggerAt.toISOString(),
   913→      status: 'set',
   914→    };
   915→  }

   924→  private findConflicts(start: Date, end: Date): CalendarEvent[] {
   925→    return Array.from(this.events.values()).filter(
   926→      (event) => event.status === 'confirmed' && event.startTime < end && event.endTime > start,
   927→    );
   928→  }

   930→  private mergeIntervals(
   931→    intervals: Array<{ start: Date; end: Date }>,
   932→  ): Array<{ start: Date; end: Date }> {
   933→    if (intervals.length === 0) return [];
   934→
   935→    const merged: Array<{ start: Date; end: Date }> = [
   936→      { start: new Date(intervals[0].start), end: new Date(intervals[0].end) },
   937→    ];
   938→
   939→    for (let i = 1; i < intervals.length; i++) {
   940→      const last = merged[merged.length - 1];
   941→      if (intervals[i].start <= last.end) {
   942→        last.end = new Date(Math.max(last.end.getTime(), intervals[i].end.getTime()));
   943→      } else {
   944→        merged.push({
   945→          start: new Date(intervals[i].start),
   946→          end: new Date(intervals[i].end),
   947→        });
   948→      }
   949→    }
   950→
   951→    return merged;
   952→  }
