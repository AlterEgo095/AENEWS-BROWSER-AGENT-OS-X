   259→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   260→    const startTime = Date.now();
   261→    const { action, ...params } = input.payload;
   262→
   263→    if (!action) {
   264→      return this.createAgentOutput(
   265→        input.taskId,
   266→        false,
   267→        null,
   268→        'Missing required parameter: action',
   269→        startTime,
   270→      );
   271→    }
   272→
   273→    const supportedActions = [
   274→      'generateFromSpec',
   275→      'generateFromTemplate',
   276→      'generateFromDescription',
   277→      'refactorCode',
   278→      'optimizeCode',
   279→    ];
   280→
   281→    if (!supportedActions.includes(action)) {
   282→      return this.createAgentOutput(
   283→        input.taskId,
   284→        false,
   285→        null,
   286→        `Unknown code generation action: ${action}. Supported: ${supportedActions.join(', ')}`,
   287→        startTime,
   288→      );
   289→    }
   290→
   291→    try {
   292→      const tool = this.getTool(action);
   293→      if (!tool) {
   294→        return this.createAgentOutput(
   295→          input.taskId,
   296→          false,
   297→          null,
   298→          `Tool not found: ${action}`,
   299→          startTime,
   300→        );
   301→      }
   302→
   303→      const result = await tool.execute(params);
   304→
   305→      this.generationHistory.push({
   306→        action,
   307→        language: params.language || 'unknown',
   308→        timestamp: new Date(),
   309→        success: true,
   310→      });
   311→
   312→      // Store generation result in session memory for context continuity
   313→      await this.storeInWorkingMemory(
   314→        `codegen:last:${action}`,
   315→        { params, result, timestamp: new Date() },
   316→        300000,
   317→      );
   318→
   319→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   320→    } catch (error) {
   321→      const msg = (error as Error).message;
   322→      this.logger.error(`CodeGeneration execution failed for ${action}: ${msg}`);
   323→
   324→      this.generationHistory.push({
   325→        action,
   326→        language: params.language || 'unknown',
   327→        timestamp: new Date(),
   328→        success: false,
   329→      });
   330→
   331→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   332→    }
   333→  }

   343→  private async generateFromSpec(params: {
   344→    spec: string;
   345→    language: string;
   346→    framework?: string;
   347→    style?: string;
   348→  }): Promise<{
   349→    code: string;
   350→    language: string;
   351→    files: GeneratedFile[];
   352→    warnings: string[];
   353→  }> {
   354→    const { spec, language, framework, style } = params;
   355→
   356→    if (!spec || typeof spec !== 'string') {
   357→      throw new Error('A valid specification string is required');
   358→    }
   359→    if (!language || typeof language !== 'string') {
   360→      throw new Error('A valid programming language is required');
   361→    }
   362→
   363→    const warnings: string[] = [];
   364→
   365→    // Parse specification into structured sections
   366→    const specSections = this.parseSpec(spec);
   367→    const normalizedLang = this.normalizeLanguage(language);
   368→
   369→    // Validate framework compatibility
   370→    if (framework) {
   371→      const isCompatible = this.validateFrameworkCompatibility(normalizedLang, framework);
   372→      if (!isCompatible) {
   373→        warnings.push(
   374→          `Framework "${framework}" may not be fully compatible with ${normalizedLang}`,
   375→        );
   376→      }
   377→    }
   378→
   379→    // Generate code from each spec section
   380→    const files: GeneratedFile[] = [];
   381→    for (const section of specSections) {
   382→      const generatedCode = this.synthesizeCode(section, normalizedLang, framework, style);
   383→      const fileName = this.inferFileName(section.name, normalizedLang, framework);
   384→
   385→      files.push({
   386→        name: fileName,
   387→        path: `./src/${section.name}/${fileName}`,
   388→        content: generatedCode,
   389→        language: normalizedLang,
   390→      });
   391→    }
   392→
   393→    // Produce a main combined code output
   394→    const code = files.map((f) => f.content).join('\n\n');
   395→
   396→    this.logger.log(
   397→      `Generated code from spec: ${files.length} file(s), language=${normalizedLang}, framework=${framework || 'none'}`,
   398→    );
   399→
   400→    return { code, language: normalizedLang, files, warnings };
   401→  }

   403→  private async generateFromTemplate(params: {
   404→    templateId: string;
   405→    variables: Record<string, string>;
   406→    outputFileName?: string;
   407→  }): Promise<{
   408→    code: string;
   409→    fileName: string;
   410→    templateId: string;
   411→    substitutedVars: string[];
   412→  }> {
   413→    const { templateId, variables, outputFileName } = params;
   414→
   415→    if (!templateId || typeof templateId !== 'string') {
   416→      throw new Error('A valid templateId is required');
   417→    }
   418→    if (!variables || typeof variables !== 'object') {
   419→      throw new Error('Variables must be a key-value object');
   420→    }
   421→
   422→    const template = this.templates.get(templateId);
   423→    if (!template) {
   424→      const available = Array.from(this.templates.keys()).join(', ');
   425→      throw new Error(`Template not found: ${templateId}. Available: ${available}`);
   426→    }
   427→
   428→    // Perform variable substitution
   429→    let code = template.content;
   430→    const substitutedVars: string[] = [];
   431→
   432→    for (const [key, value] of Object.entries(variables)) {
   433→      const placeholder = `{{${key}}}`;
   434→      if (code.includes(placeholder)) {
   435→        code = code.replaceAll(placeholder, value);
   436→        substitutedVars.push(key);
   437→      }
   438→    }
   439→
   440→    // Check for unsubstituted variables
   441→    const remainingPlaceholders = code.match(/\{\{(\w+)\}\}/g);
   442→    if (remainingPlaceholders) {
   443→      const missingVars = remainingPlaceholders.map((p) => p.replace(/[{}]/g, ''));
   444→      throw new Error(
   445→        `Unsubstituted template variables: ${missingVars.join(', ')}. Provide values for all placeholders.`,
   446→      );
   447→    }
   448→
   449→    const fileName =
   450→      outputFileName || `${templateId}.generated.${this.getExtension(template.language)}`;
   451→
   452→    this.logger.log(
   453→      `Generated code from template: ${templateId}, substituted ${substitutedVars.length} variable(s)`,
   454→    );
   455→
   456→    return { code, fileName, templateId, substitutedVars };
   457→  }

   500→  private async refactorCode(params: {
   501→    code: string;
   502→    language: string;
   503→    goals?: string[];
   504→    preserveBehavior?: boolean;
   505→  }): Promise<{
   506→    refactoredCode: string;
   507→    changes: RefactoringChange[];
   508→    improvedMetrics: Record<string, number>;
   509→  }> {
   510→    const { code, language, goals = [], preserveBehavior = true } = params;
   511→
   512→    if (!code || typeof code !== 'string') {
   513→      throw new Error('Source code is required for refactoring');
   514→    }
   515→    if (!language || typeof language !== 'string') {
   516→      throw new Error('Programming language is required');
   517→    }
   518→
   519→    const normalizedLang = this.normalizeLanguage(language);
   520→    const changes: RefactoringChange[] = [];
   521→
   522→    let refactoredCode = code;
   523→    const originalLines = code.split('\n');
   524→
   525→    // Apply refactoring patterns
   526→    const patterns = this.getRefactoringPatterns(goals, normalizedLang);
   527→
   528→    for (const pattern of patterns) {
   529→      const result = this.applyRefactoringPattern(refactoredCode, pattern, preserveBehavior);
   530→      if (result.modified) {
   531→        changes.push(result.change);
   532→        refactoredCode = result.code;
   533→      }
   534→    }
   535→
   536→    // Calculate improved metrics
   537→    const refactoredLines = refactoredCode.split('\n');
   538→    const improvedMetrics: Record<string, number> = {
   539→      linesReduced: originalLines.length - refactoredLines.length,
   540→      complexityReduction: this.estimateComplexityReduction(originalLines, refactoredLines),
   541→      duplicateCodeReduction: this.estimateDuplicateReduction(code, refactoredCode),
   542→    };
   543→
   544→    this.logger.log(`Refactored code: ${changes.length} change(s), language=${normalizedLang}`);
   545→
   546→    return { refactoredCode, changes, improvedMetrics };
   547→  }

   549→  private async optimizeCode(params: {
   550→    code: string;
   551→    language: string;
   552→    target: string;
   553→    level?: string;
   554→  }): Promise<{
   555→    optimizedCode: string;
   556→    improvements: Array<{ type: string; description: string; impact: string }>;
   557→    estimatedGain: string;
   558→    tradeoffs: string[];
   559→  }> {
   560→    const { code, language, target, level = 'safe' } = params;
   561→
   562→    if (!code || typeof code !== 'string') {
   563→      throw new Error('Source code is required for optimization');
   564→    }
   565→    if (!language || typeof language !== 'string') {
   566→      throw new Error('Programming language is required');
   567→    }
   568→    const validTargets = ['performance', 'memory', 'readability', 'bundle-size'];
   569→    if (!validTargets.includes(target)) {
   570→      throw new Error(`Invalid optimization target: ${target}. Valid: ${validTargets.join(', ')}`);
   571→    }
   572→    const validLevels = ['safe', 'moderate', 'aggressive'];
   573→    if (!validLevels.includes(level)) {
   574→      throw new Error(`Invalid optimization level: ${level}. Valid: ${validLevels.join(', ')}`);
   575→    }
   576→
   577→    const normalizedLang = this.normalizeLanguage(language);
   578→    const improvements: Array<{ type: string; description: string; impact: string }> = [];
   579→    const tradeoffs: string[] = [];
   580→
   581→    let optimizedCode = code;
   582→
   583→    // Apply optimization strategies based on target and level
   584→    const strategies = this.getOptimizationStrategies(target, level, normalizedLang);
   585→
   586→    for (const strategy of strategies) {
   587→      const result = this.applyOptimizationStrategy(optimizedCode, strategy, normalizedLang);
   588→      if (result.modified) {
   589→        optimizedCode = result.code;
   590→        improvements.push({
   591→          type: strategy.type,
   592→          description: strategy.description,
   593→          impact: result.impact,
   594→        });
   595→
   596→        if (result.tradeoff) {
   597→          tradeoffs.push(result.tradeoff);
   598→        }
   599→      }
   600→    }
   601→
   602→    // Estimate overall gain
   603→    const estimatedGain = this.estimateOptimizationGain(improvements, target, level);
   604→
   605→    this.logger.log(
   606→      `Optimized code: target=${target}, level=${level}, ${improvements.length} improvement(s)`,
   607→    );
   608→
   609→    return { optimizedCode, improvements, estimatedGain, tradeoffs };
   610→  }

   745→  private parseSpec(
   746→    spec: string,
   747→  ): Array<{ name: string; type: string; description: string; methods?: string[] }> {
   748→    const sections: Array<{ name: string; type: string; description: string; methods?: string[] }> =
   749→      [];
   750→    const lines = spec.split('\n');
   751→    let currentSection: (typeof sections)[0] | null = null;
   752→
   753→    for (const line of lines) {
   754→      const trimmed = line.trim();
   755→      if (trimmed.startsWith('# ') || trimmed.startsWith('## ')) {
   756→        if (currentSection) {
   757→          sections.push(currentSection);
   758→        }
   759→        const name = trimmed.replace(/^#+\s*/, '');
   760→        currentSection = {
   761→          name: this.toPascalCase(name),
   762→          type: name.toLowerCase().includes('service')
   763→            ? 'service'
   764→            : name.toLowerCase().includes('controller')
   765→              ? 'controller'
   766→              : 'module',
   767→          description: name,
   768→        };
   769→      } else if (currentSection && trimmed.length > 0) {
   770→        if (!currentSection.methods) currentSection.methods = [];
   771→        if (trimmed.startsWith('- ') || trimmed.startsWith('* ')) {
   772→          currentSection.methods.push(trimmed.replace(/^[-*]\s*/, ''));
   773→        }
   774→      }
   775→    }
   776→
   777→    if (currentSection) sections.push(currentSection);
   778→
   779→    // Default section if spec has no headers
   780→    if (sections.length === 0) {
   781→      sections.push({
   782→        name: 'GeneratedModule',
   783→        type: 'module',
   784→        description: spec.substring(0, 100),
   785→      });
   786→    }
   787→
   788→    return sections;
   789→  }

  1239→  private applyOptimizationStrategy(
  1240→    code: string,

  1242→    language: string,
  1243→  ): { modified: boolean; code: string; impact: string; tradeoff?: string } {

  1252→        const pattern = /if\s*\(([^)]+)\)\s*\{\s*return\s+([^;]+);\s*\}\s*else\s*\{/g;
  1253→        const newCode = code.replace(pattern, 'if ($1) {\n    return $2;\n  }\n  {');

  1260→      }
  1261→      case 'caching': {

  1264→          const cacheComment = '// OPTIMIZE: Consider adding memoization to expensive functions\n';
  1265→          if (!code.includes('OPTIMIZE:')) {
  1266→            optimizedCode = cacheComment + code;
  1267→            modified = true;
  1268→            impact = 'medium';
  1269→            tradeoff = 'Memoization increases memory usage proportional to unique inputs';
  1270→          }
  1271→        }
  1272→        break;
  1273→      }
  1274→      case 'lazy-loading': {
  1275→        if ((language === 'typescript' || language === 'javascript') && code.includes('import ')) {
  1276→          optimizedCode = code.replace(
  1277→            /import\s+(\w+)\s+from\s+(['"]([^'"]+)['"])/g,
  1278→            '// OPTIMIZE: Consider dynamic import: const $1 = await import($2)\nimport $1 from $2',
  1279→          );
  1280→          modified = optimizedCode !== code;
  1281→          impact = 'high';
  1282→          tradeoff = 'Dynamic imports add async overhead on first load';
  1283→        }
  1284→        break;
  1285→      }
  1286→      default: {
  1287→        // Generic strategy: add optimization suggestions as comments
  1288→        const comment = `// OPTIMIZE [${strategy.type}]: ${strategy.description}\n`;
  1289→        if (!code.includes(`OPTIMIZE [${strategy.type}]`)) {
  1290→          optimizedCode = comment + code;
  1291→          modified = true;
  1292→          impact = 'low';
  1293→        }
  1294→        break;
  1295→      }
  1296→    }
  1297→
  1298→    return { modified, code: optimizedCode, impact, tradeoff };
  1299→  }

  1301→  private estimateComplexityReduction(originalLines: string[], refactoredLines: string[]): number {
  1302→    // Simplified cyclomatic complexity estimation based on branch keywords
  1303→    const countBranches = (lines: string[]) =>
  1304→      lines.reduce((count, line) => {
  1305→        const branchKeywords = ['if ', 'else ', 'switch ', 'case ', 'for ', 'while ', 'catch '];
  1306→        return count + branchKeywords.filter((kw) => line.includes(kw)).length;
  1307→      }, 0);
  1308→
  1309→    const originalBranches = countBranches(originalLines);
  1310→    const refactoredBranches = countBranches(refactoredLines);
  1311→
  1312→    if (originalBranches === 0) return 0;
  1313→    return Math.round(((originalBranches - refactoredBranches) / originalBranches) * 100);
  1314→  }

  1316→  private estimateDuplicateReduction(originalCode: string, refactoredCode: string): number {
  1317→    // Simplified duplicate estimation by comparing line frequencies
  1318→    const countLineFrequencies = (code: string): number => {
  1319→      const lines = code
  1320→        .split('\n')
  1321→        .map((l) => l.trim())
  1322→        .filter((l) => l.length > 0);
  1323→      const freq = new Map<string, number>();
  1324→      for (const line of lines) {
  1325→        freq.set(line, (freq.get(line) || 0) + 1);
  1326→      }
  1327→      let duplicates = 0;
  1328→      for (const count of freq.values()) {
  1329→        if (count > 1) duplicates += count - 1;
  1330→      }
  1331→      return duplicates;
  1332→    };
  1333→
  1334→    const originalDuplicates = countLineFrequencies(originalCode);
  1335→    const refactoredDuplicates = countLineFrequencies(refactoredCode);
  1336→
  1337→    if (originalDuplicates === 0) return 0;
  1338→    return Math.round(((originalDuplicates - refactoredDuplicates) / originalDuplicates) * 100);
  1339→  }

  1341→  private estimateOptimizationGain(
  1342→    improvements: Array<{ type: string; description: string; impact: string }>,
  1343→    target: string,
  1344→    level: string,
  1345→  ): string {
  1346→    if (improvements.length === 0) return 'No measurable improvement';
  1347→
  1348→    const highImpact = improvements.filter((i) => i.impact === 'high').length;
  1349→    const mediumImpact = improvements.filter((i) => i.impact === 'medium').length;
  1350→
  1351→    const baseGain =
  1352→      highImpact * 20 + mediumImpact * 10 + (improvements.length - highImpact - mediumImpact) * 5;
  1353→    const levelMultiplier = level === 'aggressive' ? 1.5 : level === 'moderate' ? 1.2 : 1.0;
  1354→
  1355→    const estimatedPercent = Math.min(90, Math.round(baseGain * levelMultiplier));
  1356→
  1357→    return `~${estimatedPercent}% estimated improvement in ${target}`;
  1358→  }
