   363→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   364→    const startTime = Date.now();
   365→    const { action, ...params } = input.payload;
   366→
   367→    if (!action) {
   368→      return this.createAgentOutput(
   369→        input.taskId,
   370→        false,
   371→        null,
   372→        'Missing required parameter: action',
   373→        startTime,
   374→      );
   375→    }
   376→
   377→    const supportedActions = [
   378→      'listDependencies',
   379→      'checkVulnerabilities',
   380→      'updateDependency',
   381→      'auditDependencies',
   382→      'resolveConflict',
   383→    ];
   384→
   385→    if (!supportedActions.includes(action)) {
   386→      return this.createAgentOutput(
   387→        input.taskId,
   388→        false,
   389→        null,
   390→        `Unknown dependency action: ${action}. Supported: ${supportedActions.join(', ')}`,
   391→        startTime,
   392→      );
   393→    }
   394→
   395→    try {
   396→      const tool = this.getTool(action);
   397→      if (!tool) {
   398→        return this.createAgentOutput(
   399→          input.taskId,
   400→          false,
   401→          null,
   402→          `Tool not found: ${action}`,
   403→          startTime,
   404→        );
   405→      }
   406→
   407→      const result = await tool.execute(params);
   408→
   409→      await this.storeInWorkingMemory(
   410→        `deps:last:${action}`,
   411→        { params, result, timestamp: new Date() },
   412→        300000,
   413→      );
   414→
   415→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   416→    } catch (error) {
   417→      const msg = (error as Error).message;
   418→      this.logger.error(`Dependency execution failed for ${action}: ${msg}`);
   419→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   420→    }
   421→  }
   422→
   423→  protected async onDestroy(): Promise<void> {
   424→    this.auditHistory = [];

   426→  }

   430→  private async listDependencies(params: {
   431→    packageJson: Record<string, any>;
   432→    includeDev?: boolean;
   433→    includeTransitive?: boolean;
   434→    filter?: string;
   435→  }): Promise<{
   436→    dependencies: DependencyInfo[];
   437→    totalCount: number;
   438→    outdatedCount: number;
   439→  }> {
   440→    const { packageJson, includeDev = true, includeTransitive = false, filter } = params;
   441→
   442→    if (!packageJson || typeof packageJson !== 'object') {
   443→      throw new Error('Valid package.json object is required');
   444→    }
   445→
   446→    const dependencies: DependencyInfo[] = [];
   447→
   448→    // Parse production dependencies
   449→    const prodDeps = packageJson.dependencies || {};
   450→    for (const [name, version] of Object.entries(prodDeps)) {
   451→      if (filter && !name.includes(filter)) continue;
   452→      dependencies.push(this.createDependencyInfo(name, version as string, 'production'));
   453→    }
   454→
   455→    // Parse dev dependencies
   456→    if (includeDev) {
   457→      const devDeps = packageJson.devDependencies || {};
   458→      for (const [name, version] of Object.entries(devDeps)) {
   459→        if (filter && !name.includes(filter)) continue;
   460→        dependencies.push(this.createDependencyInfo(name, version as string, 'development'));
   461→      }
   462→    }
   463→
   464→    // Parse peer dependencies
   465→    const peerDeps = packageJson.peerDependencies || {};
   466→    for (const [name, version] of Object.entries(peerDeps)) {
   467→      if (filter && !name.includes(filter)) continue;
   468→      dependencies.push(this.createDependencyInfo(name, version as string, 'peer'));
   469→    }
   470→
   471→    // Parse optional dependencies
   472→    const optDeps = packageJson.optionalDependencies || {};
   473→    for (const [name, version] of Object.entries(optDeps)) {
   474→      if (filter && !name.includes(filter)) continue;
   475→      dependencies.push(this.createDependencyInfo(name, version as string, 'optional'));
   476→    }
   477→
   478→    const outdatedCount = dependencies.filter((d) => d.outdated).length;
   479→
   480→    this.logger.log(`Listed dependencies: ${dependencies.length} total, ${outdatedCount} outdated`);
   481→
   482→    return { dependencies, totalCount: dependencies.length, outdatedCount };
   483→  }

   485→  private async checkVulnerabilities(params: {
   486→    packageJson: Record<string, any>;
   487→    severityThreshold?: string;
   488→    includeDev?: boolean;
   489→  }): Promise<{
   490→    vulnerabilities: VulnerabilityInfo[];
   491→    totalVulnerabilities: number;
   492→    riskScore: number;
   493→    summary: string;
   494→  }> {
   495→    const { packageJson, severityThreshold = 'low', includeDev = true } = params;
   496→
   497→    if (!packageJson || typeof packageJson !== 'object') {
   498→      throw new Error('Valid package.json object is required');
   499→    }
   500→
   501→    const vulnerabilities: VulnerabilityInfo[] = [];
   502→    const allDeps: Record<string, string> = { ...(packageJson.dependencies || {}) };
   503→    if (includeDev) {
   504→      Object.assign(allDeps, packageJson.devDependencies || {});
   505→    }
   506→
   507→    const severityOrder: Record<string, number> = { low: 0, moderate: 1, high: 2, critical: 3 };
   508→    const minSeverity = severityOrder[severityThreshold] || 0;
   509→
   510→    // Check each dependency against known vulnerability database
   511→    for (const [name, version] of Object.entries(allDeps)) {
   512→      const cleanVersion = this.cleanVersion(version);
   513→      const knownVulns = KNOWN_VULNERABILITIES.filter((v) => v.package === name);
   514→
   515→      for (const vuln of knownVulns) {
   516→        if (this.isVersionVulnerable(cleanVersion, vuln.versions)) {
   517→          const severity = severityOrder[vuln.severity] || 0;
   518→          if (severity >= minSeverity) {
   519→            vulnerabilities.push({
   520→              package: name,
   521→              severity: vuln.severity,
   522→              title: vuln.title,
   523→              cve: vuln.cve,
   524→              vulnerableVersions: vuln.versions,
   525→              patchedVersions: vuln.patched,
   526→              recommendation: `Upgrade ${name} to ${vuln.patched}`,
   527→              url: `https://nvd.nist.gov/vuln/detail/${vuln.cve}`,
   528→            });
   529→          }
   530→        }
   531→      }
   532→    }
   533→
   534→    // Simulate additional vulnerability checks
   535→    this.simulateAdditionalVulnerabilityChecks(allDeps, vulnerabilities, minSeverity);
   536→
   537→    // Calculate risk score
   538→    const riskScore = this.calculateVulnerabilityRiskScore(vulnerabilities);
   539→
   540→    // Generate summary
   541→    const critical = vulnerabilities.filter((v) => v.severity === 'critical').length;
   542→    const high = vulnerabilities.filter((v) => v.severity === 'high').length;
   543→    const moderate = vulnerabilities.filter((v) => v.severity === 'moderate').length;
   544→    const low = vulnerabilities.filter((v) => v.severity === 'low').length;
   545→
   546→    const summary =
   547→      `Found ${vulnerabilities.length} vulnerabilities: ` +
   548→      `${critical} critical, ${high} high, ${moderate} moderate, ${low} low. ` +
   549→      `Risk score: ${riskScore}/100.`;
   550→
   551→    this.logger.log(
   552→      `Vulnerability check: ${vulnerabilities.length} found, risk score=${riskScore}`,
   553→    );
   554→
   555→    return { vulnerabilities, totalVulnerabilities: vulnerabilities.length, riskScore, summary };
   556→  }

   558→  private async updateDependency(params: {
   559→    packageName: string;
   560→    targetVersion: string;
   561→    packageJson: Record<string, any>;
   562→    currentVersion?: string;
   563→    isDevDependency?: boolean;
   564→  }): Promise<{
   565→    updatedPackageJson: Record<string, any>;
   566→    changeDescription: string;
   567→    breakingChanges: string[];
   568→    migrationSteps: string[];
   569→  }> {
   570→    const {
   571→      packageName,
   572→      targetVersion,
   573→      packageJson,
   574→      currentVersion,
   575→      isDevDependency = false,
   576→    } = params;
   577→
   578→    if (!packageName || typeof packageName !== 'string') {
   579→      throw new Error('Package name is required');
   580→    }
   581→    if (!targetVersion || typeof targetVersion !== 'string') {
   582→      throw new Error('Target version is required');
   583→    }
   584→    if (!packageJson || typeof packageJson !== 'object') {
   585→      throw new Error('Valid package.json object is required');
   586→    }
   587→
   588→    // Find the current version
   589→    const resolvedCurrentVersion =
   590→      currentVersion ||
   591→      packageJson.dependencies?.[packageName] ||
   592→      packageJson.devDependencies?.[packageName];
   593→
   594→    if (!resolvedCurrentVersion) {
   595→      throw new Error(`Package "${packageName}" not found in dependencies`);
   596→    }
   597→
   598→    // Determine the target version
   599→    const resolvedTarget =
   600→      targetVersion === 'latest' ? this.simulateLatestVersion(packageName) : targetVersion;
   601→
   602→    // Create updated package.json
   603→    const updatedPackageJson = { ...packageJson };
   604→
   605→    if (isDevDependency || packageJson.devDependencies?.[packageName]) {
   606→      if (!updatedPackageJson.devDependencies) updatedPackageJson.devDependencies = {};
   607→      updatedPackageJson.devDependencies[packageName] = `^${resolvedTarget}`;
   608→    } else {
   609→      if (!updatedPackageJson.dependencies) updatedPackageJson.dependencies = {};
   610→      updatedPackageJson.dependencies[packageName] = `^${resolvedTarget}`;
   611→    }
   612→
   613→    // Detect breaking changes
   614→    const breakingChanges = this.detectBreakingChanges(
   615→      packageName,
   616→      this.cleanVersion(resolvedCurrentVersion),
   617→      resolvedTarget,
   618→    );
   619→
   620→    // Generate migration steps
   621→    const migrationSteps = this.generateMigrationSteps(
   622→      packageName,
   623→      resolvedCurrentVersion,
   624→      resolvedTarget,
   625→      breakingChanges,
   626→    );
   627→
   628→    const changeDescription = `Updated ${packageName} from ${resolvedCurrentVersion} to ^${resolvedTarget}`;
   629→
   630→    this.logger.log(`Updated dependency: ${changeDescription}`);
   631→
   632→    return {
   633→      updatedPackageJson,
   634→      changeDescription,
   635→      breakingChanges,
   636→      migrationSteps,
   637→    };
   638→  }

   640→  private async auditDependencies(params: {
   641→    packageJson: Record<string, any>;
   642→    checkUnused?: boolean;
   643→    checkOutdated?: boolean;
   644→    checkLicenses?: boolean;
   645→  }): Promise<{
   646→    unusedDependencies: string[];
   647→    outdatedDependencies: OutdatedDependency[];
   648→    licenseIssues: LicenseIssue[];
   649→    healthScore: number;
   650→    recommendations: string[];
   651→  }> {
   652→    const { packageJson, checkUnused = true, checkOutdated = true, checkLicenses = true } = params;
   653→
   654→    if (!packageJson || typeof packageJson !== 'object') {
   655→      throw new Error('Valid package.json object is required');
   656→    }
   657→
   658→    const unusedDependencies: string[] = [];
   659→    const outdatedDependencies: OutdatedDependency[] = [];
   660→    const licenseIssues: LicenseIssue[] = [];
   661→    const recommendations: string[] = [];
   662→
   663→    // Check for unused dependencies
   664→    if (checkUnused) {
   665→      const allDeps = {
   666→        ...(packageJson.dependencies || {}),
   667→        ...(packageJson.devDependencies || {}),
   668→      };
   669→
   670→      for (const name of Object.keys(allDeps)) {
   671→        // Simulate unused detection based on common patterns
   672→        if (this.isLikelyUnused(name, packageJson)) {
   673→          unusedDependencies.push(name);
   674→        }
   675→      }
   676→
   677→      if (unusedDependencies.length > 0) {
   678→        recommendations.push(
   679→          `Found ${unusedDependencies.length} potentially unused dependencies. Review and remove them to reduce bundle size.`,
   680→        );
   681→      }
   682→    }
   683→
   684→    // Check for outdated dependencies
   685→    if (checkOutdated) {
   686→      const allDeps = {
   687→        ...(packageJson.dependencies || {}),
   688→        ...(packageJson.devDependencies || {}),
   689→      };
   690→
   691→      for (const [name, version] of Object.entries(allDeps)) {
   692→        const cleanVer = this.cleanVersion(version as string);
   693→        const latest = this.simulateLatestVersion(name);
   694→        const currentMajor = parseInt(cleanVer.split('.')[0], 10);
   695→        const latestMajor = parseInt(latest.split('.')[0], 10);
   696→
   697→        if (latestMajor > currentMajor) {
   698→          outdatedDependencies.push({
   699→            name,
   700→            current: cleanVer,
   701→            wanted: `${currentMajor}.${this.simulateMinorVersion()}.0`,
   702→            latest,
   703→            type: packageJson.dependencies?.[name] ? 'production' : 'development',
   704→          });
   705→        }
   706→      }
   707→
   708→      if (outdatedDependencies.length > 0) {
   709→        recommendations.push(
   710→          `Found ${outdatedDependencies.length} outdated dependencies. Consider updating them for security and feature improvements.`,
   711→        );
   712→      }
   713→    }
   714→
   715→    // Check license issues
   716→    if (checkLicenses) {
   717→      const allDeps = {
   718→        ...(packageJson.dependencies || {}),
   719→        ...(packageJson.devDependencies || {}),
   720→      };
   721→
   722→      const problematicLicenses = ['GPL-3.0', 'AGPL-3.0', 'GPL-2.0', 'SSPL-1.0'];
   723→
   724→      for (const name of Object.keys(allDeps)) {
   725→        const simulatedLicense = this.simulateLicense(name);
   726→        if (problematicLicenses.includes(simulatedLicense)) {
   727→          licenseIssues.push({
   728→            package: name,
   729→            license: simulatedLicense,
   730→            issue: `${simulatedLicense} is a copyleft license that may require source code disclosure`,
   731→            severity: 'high',
   732→          });
   733→        } else if (simulatedLicense === 'UNLICENSED') {
   734→          licenseIssues.push({
   735→            package: name,
   736→            license: simulatedLicense,
   737→            issue: 'Package has no license specified',
   738→            severity: 'medium',
   739→          });
   740→        }
   741→      }
   742→
   743→      if (licenseIssues.length > 0) {
   744→        recommendations.push(
   745→          `Found ${licenseIssues.length} license issue(s). Review and ensure compliance with your project's license policy.`,
   746→        );
   747→      }
   748→    }
   749→
   750→    // Calculate health score
   751→    const healthScore = this.calculateDependencyHealthScore(
   752→      unusedDependencies.length,
   753→      outdatedDependencies.length,
   754→      licenseIssues.length,
   755→      Object.keys({ ...(packageJson.dependencies || {}), ...(packageJson.devDependencies || {}) })
   756→        .length,
   757→    );
   758→
   759→    // General recommendations
   760→    if (Object.keys(packageJson.dependencies || {}).length > 30) {
   761→      recommendations.push(
   762→        'Project has many production dependencies. Consider reducing dependency count for better maintainability.',
   763→      );
   764→    }
   765→
   766→    this.auditHistory.push({ timestamp: new Date(), healthScore });
   767→
   768→    this.logger.log(
   769→      `Dependency audit: ${unusedDependencies.length} unused, ${outdatedDependencies.length} outdated, ${licenseIssues.length} license issues, health=${healthScore}`,
   770→    );
   771→
   772→    return {
   773→      unusedDependencies,
   774→      outdatedDependencies,
   775→      licenseIssues,
   776→      healthScore,
   777→      recommendations,
   778→    };
   779→  }

   781→  private async resolveConflict(params: {
   782→    conflicts: Array<{ package: string; requiredBy: string[]; versions: string[] }>;
   783→    packageJson: Record<string, any>;
   784→    strategy?: string;
   785→  }): Promise<{
   786→    resolvedPackageJson: Record<string, any>;
   787→    resolutions: ConflictResolution[];
   788→    unresolved: Array<{ package: string; reason: string }>;
   789→  }> {
   790→    const { conflicts, packageJson, strategy = 'semver-compatible' } = params;
   791→
   792→    if (!conflicts || !Array.isArray(conflicts) || conflicts.length === 0) {
   793→      throw new Error('At least one conflict description is required');
   794→    }
   795→    if (!packageJson || typeof packageJson !== 'object') {
   796→      throw new Error('Valid package.json object is required');
   797→    }
   798→
   799→    const resolutions: ConflictResolution[] = [];
   800→    const unresolved: Array<{ package: string; reason: string }> = [];
   801→    const resolvedPackageJson = { ...packageJson };
   802→
   803→    for (const conflict of conflicts) {
   804→      if (!conflict.package || !conflict.versions || conflict.versions.length < 2) {
   805→        unresolved.push({
   806→          package: conflict.package || 'unknown',
   807→          reason: 'Invalid conflict description: must specify package and at least 2 versions',
   808→        });
   809→        continue;
   810→      }
   811→
   812→      try {
   813→        const resolution = this.resolveVersionConflict(conflict, strategy);
   814→        resolutions.push(resolution);
   815→
   816→        // Apply resolution to package.json
   817→        if (resolvedPackageJson.dependencies?.[conflict.package]) {
   818→          resolvedPackageJson.dependencies[conflict.package] = `^${resolution.resolvedVersion}`;
   819→        }
   820→        if (resolvedPackageJson.devDependencies?.[conflict.package]) {
   821→          resolvedPackageJson.devDependencies[conflict.package] = `^${resolution.resolvedVersion}`;
   822→        }
   823→
   824→        // Add resolutions/overrides field if needed
   825→        if (!resolvedPackageJson.resolutions) {
   826→          resolvedPackageJson.resolutions = {};
   827→        }
   828→        resolvedPackageJson.resolutions[conflict.package] = resolution.resolvedVersion;
   829→      } catch (error) {
   830→        unresolved.push({
   831→          package: conflict.package,
   832→          reason: (error as Error).message,
   833→        });
   834→      }
   835→    }
   836→
   837→    this.logger.log(
   838→      `Resolved conflicts: ${resolutions.length} resolved, ${unresolved.length} unresolved, strategy=${strategy}`,
   839→    );
   840→
   841→    return { resolvedPackageJson, resolutions, unresolved };
   842→  }

   872→  private isVersionVulnerable(version: string, vulnerableRange: string): boolean {
   873→    // Simplified version comparison
   874→    const rangeMatch = vulnerableRange.match(/([<>=]+)([\d.]+)/);
   875→    if (!rangeMatch) return false;
   876→
   877→    const operator = rangeMatch[1];
   878→    const targetVersion = rangeMatch[2];
   879→
   880→    const vParts = version.split('.').map(Number);
   881→    const tParts = targetVersion.split('.').map(Number);
   882→
   883→    for (let i = 0; i < Math.max(vParts.length, tParts.length); i++) {
   884→      const v = vParts[i] || 0;
   885→      const t = tParts[i] || 0;
   886→
   887→      if (v < t) return operator.includes('<') || operator.includes('<=');
   888→      if (v > t) return operator.includes('>') || operator.includes('>=');
   889→    }
   890→
   891→    return operator.includes('=');
   892→  }

  1073→  private calculateDependencyHealthScore(
  1074→    unusedCount: number,
  1075→    outdatedCount: number,
  1076→    licenseIssueCount: number,
  1077→    totalDeps: number,
  1078→  ): number {
  1079→    let score = 100;
  1080→
  1081→    // Deduct for unused dependencies
  1082→    score -= Math.min(20, unusedCount * 5);
  1083→
  1084→    // Deduct for outdated dependencies
  1085→    score -= Math.min(30, outdatedCount * 5);
  1086→
  1087→    // Deduct for license issues
  1088→    score -= Math.min(25, licenseIssueCount * 10);
  1089→
  1090→    // Penalty for too many dependencies
  1091→    if (totalDeps > 50) score -= 5;
  1092→    if (totalDeps > 100) score -= 10;
  1093→
  1094→    return Math.max(0, Math.min(100, score));
  1095→  }

  1097→  private resolveVersionConflict(
  1098→    conflict: { package: string; requiredBy: string[]; versions: string[] },
  1099→    strategy: string,
  1100→  ): ConflictResolution {
  1101→    const versions = conflict.versions.map((v) => this.cleanVersion(v));
  1102→
  1103→    let resolvedVersion: string;
  1104→    let reason: string;
  1105→
  1106→    switch (strategy) {
  1107→      case 'newest': {
  1108→        resolvedVersion = this.findNewestVersion(versions);
  1109→        reason = `Selected newest version ${resolvedVersion} among conflicting versions: ${versions.join(', ')}`;
  1110→        break;
  1111→      }
  1112→      case 'oldest': {
  1113→        resolvedVersion = this.findOldestVersion(versions);
  1114→        reason = `Selected oldest compatible version ${resolvedVersion} for maximum compatibility`;
  1115→        break;
  1116→      }
  1117→      case 'semver-compatible': {
  1118→        resolvedVersion = this.findSemverCompatibleVersion(versions);
  1119→        reason = `Found semver-compatible version ${resolvedVersion} that satisfies all requirements`;
  1120→        break;
  1121→      }
  1122→      case 'manual': {
  1123→        throw new Error(
  1124→          `Manual resolution required for ${conflict.package}. Provide explicit version.`,
  1125→        );
  1126→      }
  1127→      default: {
  1128→        resolvedVersion = this.findSemverCompatibleVersion(versions);
  1129→        reason = `Defaulted to semver-compatible strategy: resolved to ${resolvedVersion}`;
  1130→      }
  1131→    }
  1132→
  1133→    return {
  1134→      package: conflict.package,
  1135→      conflictVersions: conflict.versions,
  1136→      resolvedVersion,
  1137→      strategy,
  1138→      reason,
  1139→    };
  1140→  }
  1141→
  1142→  private findNewestVersion(versions: string[]): string {
  1143→    return versions.sort((a, b) => {
  1144→      const aParts = a.split('.').map(Number);
  1145→      const bParts = b.split('.').map(Number);
  1146→      for (let i = 0; i < 3; i++) {
  1147→        if ((aParts[i] || 0) !== (bParts[i] || 0)) return (bParts[i] || 0) - (aParts[i] || 0);
  1148→      }
  1149→      return 0;
  1150→    })[0];
  1151→  }
  1152→
  1153→  private findOldestVersion(versions: string[]): string {
  1154→    return versions.sort((a, b) => {
  1155→      const aParts = a.split('.').map(Number);
  1156→      const bParts = b.split('.').map(Number);
  1157→      for (let i = 0; i < 3; i++) {
  1158→        if ((aParts[i] || 0) !== (bParts[i] || 0)) return (aParts[i] || 0) - (bParts[i] || 0);
  1159→      }
  1160→      return 0;
  1161→    })[0];
  1162→  }
  1163→
  1164→  private findSemverCompatibleVersion(versions: string[]): string {
  1165→    // Find the highest version that is semver-compatible with all requirements
  1166→    const newest = this.findNewestVersion(versions);
  1167→
  1168→    // If all versions share the same major, the newest is compatible
  1169→    const majorGroups = new Set(versions.map((v) => v.split('.')[0]));
  1170→    if (majorGroups.size === 1) {
  1171→      return newest;
  1172→    }
  1173→
  1174→    // If mixed majors, use the lowest common major version
  1175→    const lowestMajor = Math.min(...Array.from(majorGroups).map(Number));
  1176→    const compatible = versions.filter((v) => v.startsWith(`${lowestMajor}.`));
  1177→    return this.findNewestVersion(compatible);
  1178→  }
  1179→}
