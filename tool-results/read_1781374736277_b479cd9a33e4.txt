     1→     1→/**
     2→     2→ * AENEWS Software Factory — Browser Connector
     3→     3→ *
     4→     4→ * Maps browser.* capabilities to real Playwright invocations:
     5→     5→ *   browser.login      → Playwright: navigate + fill credentials + submit
     6→     6→ *   browser.navigation → Playwright: goto URL, wait for load
     7→     7→ *   browser.search     → Playwright: navigate to search engine + type query
     8→     8→ *   browser.form       → Playwright: fill form fields + submit
     9→     9→ *   browser.upload     → Playwright: file upload via input[type=file]
    10→    10→ *   browser.download   → Playwright: download file
    11→    11→ *   browser.screenshot → Playwright: take full-page screenshot
    12→    12→ *   browser.vision     → Playwright: screenshot + LLM visual analysis
    13→    13→ *   browser.session    → Playwright: manage cookies/storage
    14→    14→ *   browser.cookie     → Playwright: get/set cookies
    15→    15→ *   browser.popup      → Playwright: handle popup/new tab
    16→    16→ *   browser.ocr        → Playwright: screenshot + OCR via LLM
    17→    17→ *
    18→    18→ * Tools: playwright, z-ai-web-dev-sdk (for vision/OCR)
    19→    19→ */
    20→    20→
    21→    21→import { Injectable, Logger } from '@nestjs/common';
    22→    22→import * as fs from 'fs';
    23→    23→import * as path from 'path';
    24→    24→import {
    25→    25→  CapabilityId,
    26→    26→  CapabilityPack,
    27→    27→  BrowserCapability,
    28→    28→} from '../interfaces';
    29→    29→import {
    30→    30→  ICapabilityConnector,
    31→    31→  ConnectorInput,
    32→    32→  ConnectorOutput,
    33→    33→  GeneratedArtifact,
    34→    34→} from './connector.interface';
    35→    35→import { LLMHelper } from './llm-helper';
    36→    36→
    37→    37→@Injectable()
    38→    38→export class BrowserConnector implements ICapabilityConnector {
    39→    39→  readonly supportedPack = CapabilityPack.BROWSER;
    40→    40→  private readonly logger = new Logger(BrowserConnector.name);
    41→    41→  private readonly llm = new LLMHelper();
    42→    42→
    43→    43→  private static readonly BROWSER_CAPABILITIES = new Set<string>(Object.values(BrowserCapability));
    44→    44→
    45→    45→  supports(capabilityId: CapabilityId): boolean {
    46→    46→    return BrowserConnector.BROWSER_CAPABILITIES.has(capabilityId as string);
    47→    47→  }
    48→    48→
    49→    49→  async execute(capabilityId: CapabilityId, input: ConnectorInput): Promise<ConnectorOutput> {
    50→    50→    const startTime = Date.now();
    51→    51→    const capId = capabilityId as BrowserCapability;
    52→    52→
    53→    53→    this.logger.log(`Browser connector executing: ${capId} for mission ${input.missionId}`);
    54→    54→
    55→    55→    try {
    56→    56→      let result: ConnectorOutput;
    57→    57→
    58→    58→      switch (capId) {
    59→    59→        case BrowserCapability.SCREENSHOT:
    60→    60→          result = await this.executeScreenshot(input);
    61→    61→          break;
    62→    62→        case BrowserCapability.NAVIGATION:
    63→    63→          result = await this.executeNavigation(input);
    64→    64→          break;
    65→    65→        case BrowserCapability.LOGIN:
    66→    66→          result = await this.executeLogin(input);
    67→    67→          break;
    68→    68→        case BrowserCapability.SEARCH:
    69→    69→          result = await this.executeSearch(input);
    70→    70→          break;
    71→    71→        case BrowserCapability.FORM:
    72→    72→          result = await this.executeForm(input);
    73→    73→          break;
    74→    74→        case BrowserCapability.VISION:
    75→    75→          result = await this.executeVision(input);
    76→    76→          break;
    77→    77→        case BrowserCapability.OCR:
    78→    78→          result = await this.executeOCR(input);
    79→    79→          break;
    80→    80→        case BrowserCapability.DOWNLOAD:
    81→    81→          result = await this.executeDownload(input);
    82→    82→          break;
    83→    83→        case BrowserCapability.UPLOAD:
    84→    84→          result = await this.executeUpload(input);
    85→    85→          break;
    86→    86→        case BrowserCapability.SESSION:
    87→    87→          result = await this.executeSession(input);
    88→    88→          break;
    89→    89→        case BrowserCapability.COOKIE:
    90→    90→          result = await this.executeCookie(input);
    91→    91→          break;
    92→    92→        case BrowserCapability.POPUP:
    93→    93→          result = await this.executePopup(input);
    94→    94→          break;
    95→    95→        default:
    96→    96→          result = await this.executeGenericBrowser(capId, input);
    97→    97→      }
    98→    98→
    99→    99→      result.durationMs = Date.now() - startTime;
   100→   100→      return result;
   101→   101→    } catch (error: any) {
   102→   102→      this.logger.error(`Browser connector failed for ${capId}: ${error.message}`);
   103→   103→      return {
   104→   104→        success: false,
   105→   105→        artifacts: [],
   106→   106→        output: { error: error.message },
   107→   107→        costUsd: 0,
   108→   108→        durationMs: Date.now() - startTime,
   109→   109→        error: error.message,
   110→   110→      };
   111→   111→    }
   112→   112→  }

   114→   114→  // ═══════════════════════════════════════════════════════════
   115→   115→  //  browser.screenshot → Playwright full-page screenshot
   116→   116→  // ═══════════════════════════════════════════════════════════
   117→   117→
   118→   118→  private async executeScreenshot(input: ConnectorInput): Promise<ConnectorOutput> {
   119→   119→    const url = input.parameters.url || input.parameters.target;
   120→   120→    if (!url) {
   121→   121→      return {
   122→   122→        success: false,
   123→   123→        artifacts: [],
   124→   124→        output: { error: 'No URL provided for screenshot' },
   125→   125→        costUsd: 0,
   126→   126→        durationMs: 0,
   127→   127→        error: 'Missing url parameter',
   128→   128→      };
   129→   129→    }
   130→   130→
   131→   131→    const screenshotDir = path.join(input.workspaceDir, 'screenshots');
   132→   132→    fs.mkdirSync(screenshotDir, { recursive: true });
   133→   133→    const screenshotPath = path.join(screenshotDir, `screenshot-${Date.now()}.png`);
   134→   134→
   135→   135→    try {
   136→   136→      const { chromium } = await import('playwright');
   137→   137→      const browser = await chromium.launch({ headless: true });
   138→   138→      const page = await browser.newPage();
   139→   139→      await page.goto(url, { waitUntil: 'networkidle', timeout: 30000 });
   140→   140→      await page.screenshot({ path: screenshotPath, fullPage: true });
   141→   141→      await browser.close();
   142→   142→
   143→   143→      const stats = fs.statSync(screenshotPath);
   144→   144→      this.logger.log(`Screenshot saved: ${screenshotPath} (${stats.size} bytes)`);
   145→   145→
   146→   146→      return {
   147→   147→        success: true,
   148→   148→        artifacts: [this.makeArtifact(path.basename(screenshotPath), 'screenshot', screenshotPath, stats.size)],
   149→   149→        output: { url, screenshotPath, sizeBytes: stats.size },
   150→   150→        costUsd: 0.01,
   151→   151→        durationMs: 0,
   152→   152→      };
   153→   153→    } catch (error: any) {
   154→   154→      this.logger.warn(`Playwright screenshot failed: ${error.message}`);
   155→   155→      // Fallback: generate a report about the attempt
   156→   156→      const reportPath = path.join(input.workspaceDir, 'docs', 'browser-report.md');
   157→   157→      const report = `# Browser Screenshot Report\n\nURL: ${url}\nStatus: FAILED\nError: ${error.message}\n\nNote: Playwright may not be installed. Run: npx playwright install chromium\n`;
   158→   158→      fs.mkdirSync(path.dirname(reportPath), { recursive: true });
   159→   159→      fs.writeFileSync(reportPath, report, 'utf-8');
   160→   160→
   161→   161→      return {
   162→   162→        success: false,
   163→   163→        artifacts: [this.makeArtifact('browser-report.md', 'report', reportPath, report)],
   164→   164→        output: { url, error: error.message },
   165→   165→        costUsd: 0,
   166→   166→        durationMs: 0,
   167→   167→        error: error.message,
   168→   168→      };
   169→   169→    }
   170→   170→  }

   172→   172→  // ═══════════════════════════════════════════════════════════
   173→   173→  //  browser.navigation → Playwright navigate + extract content
   174→   174→  // ═══════════════════════════════════════════════════════════
   175→   175→
   176→   176→  private async executeNavigation(input: ConnectorInput): Promise<ConnectorOutput> {
   177→   177→    const url = input.parameters.url || input.parameters.target;
   178→   178→    if (!url) {
   179→   179→      return this.missingParamResult('url', 'navigation');
   180→   180→    }
   181→   181→
   182→   182→    try {
   183→   183→      const { chromium } = await import('playwright');
   184→   184→      const browser = await chromium.launch({ headless: true });
   185→   185→      const page = await browser.newPage();
   186→   186→      await page.goto(url, { waitUntil: 'networkidle', timeout: 30000 });
   187→   187→
   188→   188→      const title = await page.title();
   189→   189→      const content = await page.content();
   190→   190→
   191→   191→      // Save page content
   192→   192→      const contentDir = path.join(input.workspaceDir, 'browser-data');
   193→   193→      fs.mkdirSync(contentDir, { recursive: true });
   194→   194→
   195→   195→      const htmlPath = path.join(contentDir, 'page.html');
   196→   196→      fs.writeFileSync(htmlPath, content, 'utf-8');
   197→   197→
   198→   198→      // Extract text content
   199→   199→      const textContent = await page.evaluate(() => document.body?.innerText || '');
   200→   200→      const textPath = path.join(contentDir, 'page-text.txt');
   201→   201→      fs.writeFileSync(textPath, textContent, 'utf-8');
   202→   202→
   203→   203→      await browser.close();
   204→   204→
   205→   205→      return {
   206→   206→        success: true,
   207→   207→        artifacts: [
   208→   208→          this.makeArtifact('page.html', 'source', htmlPath, content),
   209→   209→          this.makeArtifact('page-text.txt', 'document', textPath, textContent),
   210→   210→        ],
   211→   211→        output: { url, title, textLength: textContent.length },
   212→   212→        costUsd: 0.01,
   213→   213→        durationMs: 0,
   214→   214→      };
   215→   215→    } catch (error: any) {
   216→   216→      return this.playwrightFallback(input, 'navigation', url, error);
   217→   217→    }
   218→   218→  }

   220→   220→  // ═══════════════════════════════════════════════════════════
   221→   221→  //  browser.login → Playwright login flow
   222→   222→  // ═══════════════════════════════════════════════════════════
   223→   223→
   224→   224→  private async executeLogin(input: ConnectorInput): Promise<ConnectorOutput> {
   225→   225→    const url = input.parameters.url || input.parameters.loginUrl;
   226→   226→    const username = input.parameters.username || input.parameters.email;
   227→   227→    const password = input.parameters.password;
   228→   228→
   229→   229→    if (!url || !username || !password) {
   230→   230→      return this.missingParamResult('url, username, password', 'login');
   231→   231→    }
   232→   232→
   233→   233→    try {
   234→   234→      const { chromium } = await import('playwright');
   235→   235→      const browser = await chromium.launch({ headless: true });
   236→   236→      const page = await browser.newPage();
   237→   237→      await page.goto(url, { waitUntil: 'networkidle', timeout: 30000 });
   238→   238→
   239→   239→      // Try common selectors for username/email and password fields
   240→   240→      const usernameSelectors = ['input[type="email"]', 'input[name="username"]', 'input[name="email"]', 'input[id="username"]', 'input[id="email"]'];
   241→   241→      const passwordSelectors = ['input[type="password"]', 'input[name="password"]', 'input[id="password"]'];
   242→   242→
   243→   243→      let filledUsername = false;
   244→   244→      for (const selector of usernameSelectors) {
   245→   245→        try {
   246→   246→          const element = await page.$(selector);
   247→   247→          if (element) {
   248→   248→            await element.fill(username);
   249→   249→            filledUsername = true;
   250→   250→            break;
   251→   251→          }
   252→   252→        } catch { /* try next selector */ }
   253→   253→      }
   254→   254→
   255→   255→      let filledPassword = false;
   256→   256→      for (const selector of passwordSelectors) {
   257→   257→        try {
   258→   258→          const element = await page.$(selector);
   259→   259→          if (element) {
   260→   260→            await element.fill(password);
   261→   261→            filledPassword = true;
   262→   262→            break;
   263→   263→          }
   264→   264→        } catch { /* try next selector */ }
   265→   265→      }
   266→   266→
   267→   267→      // Submit form
   268→   268→      if (filledUsername && filledPassword) {
   269→   269→        await page.keyboard.press('Enter');
   270→   270→        await page.waitForLoadState('networkidle', { timeout: 15000 }).catch(() => {});
   271→   271→      }
   272→   272→
   273→   273→      const resultTitle = await page.title();
   274→   274→      const resultUrl = page.url();
   275→   275→
   276→   276→      // Screenshot of post-login state
   277→   277→      const screenshotDir = path.join(input.workspaceDir, 'screenshots');
   278→   278→      fs.mkdirSync(screenshotDir, { recursive: true });
   279→   279→      const screenshotPath = path.join(screenshotDir, 'post-login.png');
   280→   280→      await page.screenshot({ path: screenshotPath }).catch(() => {});
   281→   281→
   282→   282→      await browser.close();
   283→   283→
   284→   284→      return {
   285→   285→        success: filledUsername && filledPassword,
   286→   286→        artifacts: [this.makeArtifact('post-login.png', 'screenshot', screenshotPath, fs.existsSync(screenshotPath) ? fs.statSync(screenshotPath).size : 0)],
   287→   287→        output: { url, resultUrl, resultTitle, usernameFilled: filledUsername, passwordFilled: filledPassword },
   288→   288→        costUsd: 0.02,
   289→   289→        durationMs: 0,
   290→   290→      };
   291→   291→    } catch (error: any) {
   292→   292→      return this.playwrightFallback(input, 'login', url, error);
   293→   293→    }
   294→   294→  }
   295→   295→
   296→   296→  // ═══════════════════════════════════════════════════════════
   297→   297→  //  browser.search → Playwright search
   298→   298→  // ═══════════════════════════════════════════════════════════
   299→   299→
   300→   300→  private async executeSearch(input: ConnectorInput): Promise<ConnectorOutput> {
   301→   301→    const query = input.parameters.query || input.parameters.searchQuery;
   302→   302→    const engine = input.parameters.engine || 'google';
   303→   303→
   304→   304→    if (!query) {
   305→   305→      return this.missingParamResult('query', 'search');
   306→   306→    }
   307→   307→
   308→   308→    const searchUrls: Record<string, string> = {
   309→   309→      google: `https://www.google.com/search?q=${encodeURIComponent(query)}`,
   310→   310→      duckduckgo: `https://duckduckgo.com/?q=${encodeURIComponent(query)}`,
   311→   311→      bing: `https://www.bing.com/search?q=${encodeURIComponent(query)}`,
   312→   312→    };
   313→   313→
   314→   314→    const url = searchUrls[engine.toLowerCase()] || searchUrls.google;
   315→   315→
   316→   316→    try {
   317→   317→      const { chromium } = await import('playwright');
   318→   318→      const browser = await chromium.launch({ headless: true });
   319→   319→      const page = await browser.newPage();
   320→   320→      await page.goto(url, { waitUntil: 'networkidle', timeout: 30000 });
   321→   321→
   322→   322→      // Extract search results
   323→   323→      const results = await page.evaluate(() => {
   324→   324→        const items: any[] = [];
   325→   325→        document.querySelectorAll('h3').forEach((h3, i) => {
   326→   326→          if (i < 10) {
   327→   327→            const link = h3.closest('a');
   328→   328→            items.push({
   329→   329→              title: h3.textContent || '',
   330→   330→              url: link?.href || '',
   331→   331→            });
   332→   332→          }
   333→   333→        });
   334→   334→        return items;
   335→   335→      });
   336→   336→
   337→   337→      // Save results
   338→   338→      const dataDir = path.join(input.workspaceDir, 'browser-data');
   339→   339→      fs.mkdirSync(dataDir, { recursive: true });
   340→   340→      const resultsPath = path.join(dataDir, 'search-results.json');
   341→   341→      fs.writeFileSync(resultsPath, JSON.stringify(results, null, 2), 'utf-8');
   342→   342→
   343→   343→      // Screenshot
   344→   344→      const screenshotDir = path.join(input.workspaceDir, 'screenshots');
   345→   345→      fs.mkdirSync(screenshotDir, { recursive: true });
   346→   346→      const screenshotPath = path.join(screenshotDir, 'search-results.png');
   347→   347→      await page.screenshot({ path: screenshotPath, fullPage: true }).catch(() => {});
   348→   348→
   349→   349→      await browser.close();
   350→   350→
   351→   351→      return {
   352→   352→        success: results.length > 0,
   353→   353→        artifacts: [
   354→   354→          this.makeArtifact('search-results.json', 'source', resultsPath, JSON.stringify(results)),
   355→   355→          this.makeArtifact('search-results.png', 'screenshot', screenshotPath, fs.existsSync(screenshotPath) ? fs.statSync(screenshotPath).size : 0),
   356→   356→        ],
   357→   357→        output: { query, engine, resultCount: results.length, results: results.slice(0, 5) },
   358→   358→        costUsd: 0.02,
   359→   359→        durationMs: 0,
   360→   360→      };
   361→   361→    } catch (error: any) {
   362→   362→      return this.playwrightFallback(input, 'search', url, error);
   363→   363→    }
   364→   364→  }

   366→   366→  // ═══════════════════════════════════════════════════════════
   367→   367→  //  browser.form → Playwright fill form
   368→   368→  // ═══════════════════════════════════════════════════════════
   369→   369→
   370→   370→  private async executeForm(input: ConnectorInput): Promise<ConnectorOutput> {
   371→   371→    const url = input.parameters.url;
   372→   372→    const fields = input.parameters.fields || {};
   373→   373→
   374→   374→    if (!url) {
   375→   375→      return this.missingParamResult('url', 'form');
   376→   376→    }
   377→   377→
   378→   378→    try {
   379→   379→      const { chromium } = await import('playwright');
   380→   380→      const browser = await chromium.launch({ headless: true });
   381→   381→      const page = await browser.newPage();
   382→   382→      await page.goto(url, { waitUntil: 'networkidle', timeout: 30000 });
   383→   383→
   384→   384→      // Fill each field
   385→   385→      const filledFields: string[] = [];
   386→   386→      for (const [selector, value] of Object.entries(fields)) {
   387→   387→        try {
   388→   388→          await page.fill(selector, value as string);
   389→   389→          filledFields.push(selector);
   390→   390→        } catch {
   391→   391→          // Try by name attribute
   392→   392→          try {
   393→   393→            await page.fill(`[name="${selector}"]`, value as string);
   394→   394→            filledFields.push(`[name="${selector}"]`);
   395→   395→          } catch { /* skip this field */ }
   396→   396→        }
   397→   397→      }
   398→   398→
   399→   399→      // Submit
   400→   400→      if (input.parameters.submit !== false) {
   401→   401→        await page.keyboard.press('Enter');
   402→   402→        await page.waitForLoadState('networkidle', { timeout: 15000 }).catch(() => {});
   403→   403→      }
   404→   404→
   405→   405→      const resultUrl = page.url();
   406→   406→
   407→   407→      await browser.close();
   408→   408→
   409→   409→      return {
   410→   410→        success: filledFields.length > 0,
   411→   411→        artifacts: [],
   412→   412→        output: { url, resultUrl, filledFields, totalFields: Object.keys(fields).length },
   413→   413→        costUsd: 0.01,
   414→   414→        durationMs: 0,
   415→   415→      };
   416→   416→    } catch (error: any) {
   417→   417→      return this.playwrightFallback(input, 'form', url, error);
   418→   418→    }
   419→   419→  }
   420→   420→
   421→   421→  // ═══════════════════════════════════════════════════════════
   422→   422→  //  browser.vision → Screenshot + LLM visual analysis
   423→   423→  // ═══════════════════════════════════════════════════════════
   424→   424→
   425→   425→  private async executeVision(input: ConnectorInput): Promise<ConnectorOutput> {
   426→   426→    const url = input.parameters.url || input.parameters.target;
   427→   427→    if (!url) {
   428→   428→      return this.missingParamResult('url', 'vision');
   429→   429→    }
   430→   430→
   431→   431→    // First take a screenshot
   432→   432→    const screenshotResult = await this.executeScreenshot(input);
   433→   433→
   434→   434→    if (!screenshotResult.success) {
   435→   435→      return screenshotResult;
   436→   436→    }
   437→   437→
   438→   438→    // Then analyze with LLM
   439→   439→    try {
   440→   440→      const llmResult = await this.llm.call({
   441→   441→        systemPrompt: 'You are a visual UI/UX analyst. Describe what you see in a webpage screenshot and provide insights.',
   442→   442→        userPrompt: `Analyze this webpage screenshot from ${url}. The mission is: "${input.instruction}". Describe the layout, identify any UI issues, and suggest improvements.`,
   443→   443→        maxTokens: 2048,
   444→   444→      });
   445→   445→
   446→   446→      const analysisPath = path.join(input.workspaceDir, 'browser-data', 'visual-analysis.md');
   447→   447→      fs.mkdirSync(path.dirname(analysisPath), { recursive: true });
   448→   448→      fs.writeFileSync(analysisPath, llmResult.content, 'utf-8');
   449→   449→
   450→   450→      return {
   451→   451→        success: true,
   452→   452→        artifacts: [
   453→   453→          ...screenshotResult.artifacts,
   454→   454→          this.makeArtifact('visual-analysis.md', 'document', analysisPath, llmResult.content),
   455→   455→        ],
   456→   456→        output: { url, analysis: llmResult.content.substring(0, 1000) },
   457→   457→        costUsd: screenshotResult.costUsd + llmResult.costUsd,
   458→   458→        durationMs: 0,
   459→   459→      };
   460→   460→    } catch (error: any) {
   461→   461→      return {
   462→   462→        ...screenshotResult,
   463→   463→        output: { ...screenshotResult.output, visionError: error.message },
   464→   464→      };
   465→   465→    }
   466→   466→  }

   472→   472→  private async executeOCR(input: ConnectorInput): Promise<ConnectorOutput> {

   479→   479→    const navResult = await this.executeNavigation(input);
   480→   480→
   481→   481→    if (navResult.success && navResult.output?.textLength > 0) {
   482→   482→      return {
   483→   483→        success: true,
   484→   484→        artifacts: navResult.artifacts,
   485→   485→        output: {
   486→   486→          url,
   487→   487→          extractedText: navResult.output.textContent?.substring(0, 2000),
   488→   488→          method: 'DOM extraction (more reliable than OCR for web)',
   489→   489→        },
   490→   490→        costUsd: navResult.costUsd,
   491→   491→        durationMs: 0,
   492→   492→      };
   493→   493→    }
   494→   494→
   495→   495→    return navResult;
   496→   496→  }

   501→   501→
   502→   502→  private async executeDownload(input: ConnectorInput): Promise<ConnectorOutput> {
   503→   503→    const url = input.parameters.url;
   504→   504→    const outputPath = input.parameters.outputPath;
   505→   505→
   506→   506→    if (!url) {
   507→   507→      return this.missingParamResult('url', 'download');
   508→   508→    }
   509→   509→
   510→   510→    try {
   511→   511→      const { chromium } = await import('playwright');
   512→   512→      const browser = await chromium.launch({ headless: true });
   513→   513→      const page = await browser.newPage();
   514→   514→
   515→   515→      const downloadDir = path.join(input.workspaceDir, 'downloads');
   516→   516→      fs.mkdirSync(downloadDir, { recursive: true });
   517→   517→
   518→   518→      const download = await page.waitForEvent('download', { timeout: 30000 });
   519→   519→      const fileName = outputPath || download.suggestedFilename() || `download-${Date.now()}`;
   520→   520→      const savePath = path.join(downloadDir, fileName);
   521→   521→      await download.saveAs(savePath);
   522→   522→
   523→   523→      await browser.close();
   524→   524→
   525→   525→      const stats = fs.statSync(savePath);
   526→   526→      return {
   527→   527→        success: true,
   528→   528→        artifacts: [this.makeArtifact(fileName, 'source', savePath, stats.size)],
   529→   529→        output: { url, savedTo: savePath, sizeBytes: stats.size },
   530→   530→        costUsd: 0.01,
   531→   531→        durationMs: 0,
   532→   532→      };
   533→   533→    } catch (error: any) {
   534→   534→      return this.playwrightFallback(input, 'download', url, error);
   535→   535→    }
   536→   536→  }

   542→   542→  private async executeUpload(input: ConnectorInput): Promise<ConnectorOutput> {
   543→   543→    const url = input.parameters.url;
   544→   544→    const filePath = input.parameters.filePath;
   545→   545→    const selector = input.parameters.selector || 'input[type="file"]';
   546→   546→
   547→   547→    if (!url || !filePath) {
   548→   548→      return this.missingParamResult('url, filePath', 'upload');

   553→   553→      const browser = await chromium.launch({ headless: true });
   554→   554→      const page = await browser.newPage();
   555→   555→      await page.goto(url, { waitUntil: 'networkidle', timeout: 30000 });
   556→   556→
   557→   557→      await page.setInputFiles(selector, filePath);
   558→   558→
   559→   559→      // Submit if auto-submit not triggered
   560→   560→      if (input.parameters.autoSubmit !== true) {
   561→   561→        await page.keyboard.press('Enter').catch(() => {});
   562→   562→      }
   563→   563→
   564→   564→      await page.waitForLoadState('networkidle', { timeout: 15000 }).catch(() => {});
   565→   565→
   566→   566→      await browser.close();
   567→   567→
   568→   568→      return {
   569→   569→        success: true,
   570→   570→        artifacts: [],
   571→   571→        output: { url, uploadedFile: filePath },
   572→   572→        costUsd: 0.01,
   573→   573→        durationMs: 0,
   574→   574→      };
   575→   575→    } catch (error: any) {
   576→   576→      return this.playwrightFallback(input, 'upload', url, error);
   577→   577→    }

   666→   666→  private async executePopup(input: ConnectorInput): Promise<ConnectorOutput> {
   667→   667→    const url = input.parameters.url;
   668→   668→    if (!url) {
   669→   669→      return this.missingParamResult('url', 'popup');
   670→   670→    }
   671→   671→
   672→   672→    try {
   673→   673→      const { chromium } = await import('playwright');
   674→   674→      const browser = await chromium.launch({ headless: true });
   675→   675→      const page = await browser.newPage();
   676→   676→      await page.goto(url, { waitUntil: 'networkidle', timeout: 30000 });
   677→   677→
   678→   678→      // Listen for popups
   679→   679→      const popupPromise = page.waitForEvent('popup', { timeout: 10000 }).catch(() => null);
   680→   680→
   681→   681→      // Click the trigger if provided
   682→   682→      if (input.parameters.triggerSelector) {
   683→   683→        await page.click(input.parameters.triggerSelector).catch(() => {});
   684→   684→      }
   685→   685→
   686→   686→      const popup = await popupPromise;
   687→   687→      let popupUrl = '';
   688→   688→      let popupTitle = '';
   689→   689→
   690→   690→      if (popup) {
   691→   691→        popupUrl = popup.url();
   692→   692→        popupTitle = await popup.title();
   693→   693→      }
   694→   694→
   695→   695→      await browser.close();
   696→   696→
   697→   697→      return {
   698→   698→        success: true,
   699→   699→        artifacts: [],
   700→   700→        output: { url, popupUrl, popupTitle, popupDetected: !!popup },
   701→   701→        costUsd: 0.01,
   702→   702→        durationMs: 0,
   703→   703→      };
   704→   704→    } catch (error: any) {
   705→   705→      return this.playwrightFallback(input, 'popup', url, error);
   706→   706→    }
   707→   707→  }

   756→   756→  private playwrightFallback(input: ConnectorInput, action: string, url: string, error: Error): ConnectorOutput {

   762→   762→    fs.writeFileSync(reportPath, report, 'utf-8');
   763→   763→
   764→   764→    return {
   765→   765→      success: false,
   766→   766→      artifacts: [this.makeArtifact(`browser-${action}-report.md`, 'report', reportPath, report)],
