    36→// ─── Overview Tab ────────────────────────────────────────────────
    37→function OverviewTab() {
    38→  const { data, isLoading } = useDashboardOverview();
    39→
    40→  if (isLoading || !data) {
    41→    return <div className="animate-shimmer h-96 rounded-xl" />;
    42→  }
    43→
    44→  const { kpis, agents, missions } = data;
    45→
    46→  // Agent status distribution
    47→  const statusData = [
    48→    { name: 'Running', value: kpis.activeAgents, color: '#22c55e' },
    49→    { name: 'Idle', value: agents.filter((a: Agent) => a.status === 'idle').length, color: '#94a3b8' },
    50→    { name: 'Error', value: kpis.errorAgents, color: '#ef4444' },
    51→    { name: 'Paused', value: agents.filter((a: Agent) => a.status === 'paused').length, color: '#f59e0b' },
    52→  ];
    53→
    54→  // Mission state distribution
    55→  const missionData = Object.entries(
    56→    missions.reduce((acc: Record<string, number>, m: Mission) => {
    57→      acc[m.state] = (acc[m.state] || 0) + 1;
    58→      return acc;
    59→    }, {})
    60→  ).map(([state, count]) => ({ state, count }));
    61→
    62→  // NOTE: Historical time-series data requires a dedicated API endpoint.
    63→  // Until one is available, we show only the current snapshot value at the most recent slot
    64→  // to avoid misleading users with synthetic/fake trend lines.
    65→  const tsData = Array.from({ length: 24 }, (_, i) => ({
    66→    time: `${23 - i}h`,
    67→    agents: i === 0 ? kpis.activeAgents : 0,
    68→    missions: i === 0 ? kpis.activeMissions : 0,
    69→  })).reverse();
    70→
    71→  return (
    72→    <div className="space-y-6">
    73→      {/* KPI Cards */}
    74→      <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
    75→        {[
    76→          { label: 'Total Agents', value: kpis.totalAgents, icon: Bot, color: 'text-primary' },
    77→          { label: 'Active Missions', value: kpis.activeMissions, icon: Rocket, color: 'text-orange-400' },
    78→          { label: 'Uptime', value: `${kpis.uptimePercent}%`, icon: CheckCircle2, color: 'text-emerald-400' },
    79→          { label: 'Errors', value: kpis.errorAgents, icon: AlertTriangle, color: 'text-red-400' },
    80→        ].map((kpi) => (
    81→          <div key={kpi.label} className="rounded-xl border border-border bg-card p-4">
    82→            <div className="flex items-center justify-between">
    83→              <div>
    84→                <p className="text-[10px] font-medium uppercase tracking-wider text-muted-foreground">{kpi.label}</p>
    85→                <p className="mt-1 text-2xl font-bold text-foreground">{kpi.value}</p>
    86→              </div>
    87→              <div className={cn('rounded-lg p-2', kpi.color.replace('text-', 'bg-') + '/15')}>
    88→                <kpi.icon className={cn('h-4 w-4', kpi.color)} />
    89→              </div>
    90→            </div>
    91→          </div>
    92→        ))}
    93→      </div>
    94→
    95→      {/* Charts */}
    96→      <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
    97→        {/* Activity Time Series */}
    98→        <div className="lg:col-span-2 rounded-xl border border-border bg-card p-4">
    99→          <h4 className="text-xs font-semibold text-foreground uppercase tracking-wider mb-3">Activity Timeline</h4>
   100→          <ResponsiveContainer width="100%" height={200}>
   101→            <AreaChart data={tsData}>
   102→              <defs>
   103→                <linearGradient id="adminAgentGrad" x1="0" y1="0" x2="0" y2="1">
   104→                  <stop offset="5%" stopColor="#3b82f6" stopOpacity={0.3} />

   248→// ─── Missions Tab ────────────────────────────────────────────────
   249→function MissionsTab() {
   250→  const [filter, setFilter] = useState<string>('all');
   251→  const [missions, setMissions] = useState<Mission[]>([]);
   252→  const [, setLoadingMissions] = useState(true);
   253→
   254→  useEffect(() => {
   255→    async function fetchMissions() {
   256→      try {
   257→        const result = await api.getMissions({ limit: 50 });
   258→        setMissions(result.data || []);
   259→      } catch {
   260→        setMissions([]);
   261→      } finally {
   262→        setLoadingMissions(false);
   263→      }
   264→    }
   265→    fetchMissions();
   266→  }, []);
   267→
   268→  const filtered = filter === 'all' ? missions : missions.filter(m => m.state === filter);
   269→
   270→  return (
   271→    <div className="space-y-4">
   272→      <div className="flex items-center gap-3">
   273→        <select
   274→          value={filter}
   275→          onChange={(e) => setFilter(e.target.value)}
   276→          className="rounded-lg border border-border bg-background px-3 py-2 text-sm text-foreground focus:border-primary/50 focus:outline-none"
   277→        >
   278→          <option value="all">All States</option>
   279→          {Object.values(MissionState).map(s => <option key={s} value={s}>{s}</option>)}
   280→        </select>
   281→        <Link href="/missions" className="flex items-center gap-1.5 rounded-lg bg-primary px-3 py-2 text-xs font-medium text-primary-foreground hover:bg-primary/90 transition-colors">
   282→          <Rocket className="h-3.5 w-3.5" /> New Mission
   283→        </Link>
   284→      </div>
   285→
   286→      <div className="space-y-2">
   287→        {filtered.map((mission) => (
   288→          <div key={mission.id} className="rounded-xl border border-border bg-card p-4">
   289→            <div className="flex items-start justify-between">
   290→              <div>
   291→                <h4 className="text-sm font-semibold text-foreground">{mission.name}</h4>
   292→                <p className="text-xs text-muted-foreground mt-0.5">{mission.description}</p>
   293→              </div>
   294→              <span className={cn(
   295→                'shrink-0 rounded-full border px-2 py-0.5 text-[10px] font-semibold uppercase ml-2',
   296→                mission.state === 'COMPLETED' ? 'bg-emerald-500/20 text-emerald-400 border-emerald-500/30' :
   297→                mission.state === 'FAILED' ? 'bg-red-500/20 text-red-400 border-red-500/30' :
   298→                mission.state === 'BUILDING' ? 'bg-orange-500/20 text-orange-400 border-orange-500/30' :
   299→                'bg-blue-500/20 text-blue-400 border-blue-500/30'
   300→              )}>
   301→                {mission.state}
   302→              </span>
   303→            </div>
   304→            <div className="mt-3">
   305→              <div className="flex items-center justify-between text-[10px] mb-1">
   306→                <span className="text-muted-foreground">Progress</span>
   307→                <span className="text-foreground font-medium">{mission.progress}%</span>
   308→              </div>
   309→              <div className="h-1.5 rounded-full bg-border overflow-hidden">
   310→                <div className={cn(
   311→                  'h-full rounded-full transition-all duration-500',
   312→                  mission.state === 'COMPLETED' ? 'bg-emerald-400' :
   313→                  mission.state === 'FAILED' ? 'bg-red-400' : 'bg-primary'
   314→                )} style={{ width: `${mission.progress}%` }} />
   315→              </div>
   316→            </div>
   317→            <div className="mt-3 flex items-center gap-2">
   318→              {mission.state !== 'COMPLETED' && mission.state !== 'FAILED' && mission.state !== 'CANCELLED' && (
   319→                <>
   320→                  <button disabled title="Mission pause coming soon" className="p-1 rounded text-amber-400/40 cursor-not-allowed">
   321→                    <Pause className="h-3.5 w-3.5" />
   322→                  </button>

   325→                  </button>
   326→                </>
   327→              )}
   328→              <button disabled title="Mission details coming soon" className="p-1 rounded text-muted-foreground/40 cursor-not-allowed">
   329→                <Eye className="h-3.5 w-3.5" />
   330→              </button>
   331→            </div>
   332→          </div>
   333→        ))}
   334→      </div>
   335→    </div>
   336→  );
   337→}
