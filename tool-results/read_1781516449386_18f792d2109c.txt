     2→     2→
     3→     3→const API_BASE = '/api/v1';
     4→     4→const REQUEST_TIMEOUT_MS = 30000;
     5→     5→const MAX_RETRIES = 2;
     6→     6→
     7→     7→/**
     8→     8→ * Build standard auth headers for raw fetch() calls.
     9→     9→ * Use this when you can't use the ApiClient but still need authenticated requests.
    10→    10→ *
    11→    11→ * Reads the access token from the Zustand in-memory store (NOT localStorage).
    12→    12→ */
    13→    13→export function getAuthHeaders(): Record<string, string> {
    14→    14→  const headers: Record<string, string> = {
    15→    15→    'Content-Type': 'application/json',
    16→    16→  };
    17→    17→  const token = useAuthStore.getState().token;
    18→    18→  if (token) {
    19→    19→    headers['Authorization'] = `Bearer ${token}`;
    20→    20→  }
    21→    21→  return headers;
    22→    22→}
    23→    23→
    24→    24→class ApiClient {
    25→    25→  private baseUrl: string;
    26→    26→  private refreshing: Promise<string | null> | null = null;
    27→    27→
    28→    28→  constructor() {
    29→    29→    this.baseUrl = API_BASE;
    30→    30→  }
    31→    31→
    32→    32→  private getHeaders(): HeadersInit {
    33→    33→    const headers: HeadersInit = {
    34→    34→      'Content-Type': 'application/json',
    35→    35→    };
    36→    36→    const token = useAuthStore.getState().token;
    37→    37→    if (token) {
    38→    38→      headers['Authorization'] = `Bearer ${token}`;
    39→    39→    }
    40→    40→    return headers;
    41→    41→  }
    42→    42→
    43→    43→  /**
    44→    44→   * Attempt to refresh the access token using the httpOnly refresh cookie.
    45→    45→   * Ensures only one refresh request is in-flight at a time.
    46→    46→   *
    47→    47→   * @returns The new access token, or null if refresh failed
    48→    48→   */
    49→    49→  private async tryRefresh(): Promise<string | null> {
    50→    50→    // Reuse in-flight refresh promise to prevent concurrent refreshes
    51→    51→    if (!this.refreshing) {
    52→    52→      this.refreshing = useAuthStore.getState().refreshAuth().finally(() => {
    53→    53→        this.refreshing = null;
    54→    54→      });
    55→    55→    }
    56→    56→    return this.refreshing;
    57→    57→  }
    58→    58→
    59→    59→  /**
    60→    60→   * Handle session expiry — try refreshing before redirecting to login.
    61→    61→   */
    62→    62→  private async handleUnauthorized(originalEndpoint: string, originalOptions?: RequestInit): Promise<boolean> {
    63→    63→    const newToken = await this.tryRefresh();
    64→    64→
    65→    65→    if (newToken) {
    66→    66→      // Token refreshed successfully — the caller should retry the request
    67→    67→      return true;
    68→    68→    }
    69→    69→
    70→    70→    // Refresh also failed — redirect to login
    71→    71→    if (typeof window !== 'undefined' && !window.location.pathname.startsWith('/login')) {
    72→    72→      window.location.href = '/login';
    73→    73→    }
    74→    74→    return false;
    75→    75→  }
    76→    76→
    77→    77→  /**
    78→    78→   * Determine if a request should be retried.
    79→    79→   * - 5xx: retry with backoff
    80→    80→   * - 4xx: never retry (client error)
    81→    81→   * - Network error: retry
    82→    82→   */
    83→    83→  private shouldRetry(status: number | null, attempt: number): boolean {
    84→    84→    if (attempt >= MAX_RETRIES) return false;
    85→    85→    // 4xx = client error, never retry (except 401 which is handled separately)
    86→    86→    if (status !== null && status >= 400 && status < 500) return false;
    87→    87→    // 5xx or network failure: retry
    88→    88→    return true;
    89→    89→  }
    90→    90→
    91→    91→  private async request<T>(endpoint: string, options?: RequestInit, _attempt: number = 0, _isRetryAfterRefresh: boolean = false): Promise<T> {
    92→    92→    const url = `${this.baseUrl}${endpoint}`;
    93→    93→    const controller = new AbortController();
    94→    94→    const timeoutId = setTimeout(() => controller.abort(), REQUEST_TIMEOUT_MS);
    95→    95→
    96→    96→    try {
    97→    97→      const response = await fetch(url, {
    98→    98→        ...options,
    99→    99→        signal: controller.signal,
   100→   100→        credentials: 'include', // Required for sending httpOnly cookies
   101→   101→        headers: {
   102→   102→          ...this.getHeaders(),
   103→   103→          ...options?.headers,
   104→   104→        },
   105→   105→      });
   106→   106→
   107→   107→      if (response.status === 401) {
   108→   108→        // On 401, try refreshing the token before giving up
   109→   109→        if (!_isRetryAfterRefresh) {
   110→   110→          const refreshed = await this.handleUnauthorized(endpoint, options);
   111→   111→          if (refreshed) {
   112→   112→            // Retry the original request with the new token
   113→   113→            return this.request<T>(endpoint, options, 0, true);
   114→   114→          }
   115→   115→        }
   116→   116→        throw new Error('Session expired. Please log in again.');
   117→   117→      }
   118→   118→
   119→   119→      if (!response.ok) {
   120→   120→        // Retry on 5xx
   121→   121→        if (this.shouldRetry(response.status, _attempt)) {
   122→   122→          const delay = Math.min(1000 * Math.pow(2, _attempt), 5000);
   123→   123→          await new Promise((resolve) => setTimeout(resolve, delay));
   124→   124→          return this.request<T>(endpoint, options, _attempt + 1, _isRetryAfterRefresh);
   125→   125→        }
   126→   126→
   127→   127→        const error = await response.json().catch(() => ({ message: 'Request failed' }));
   128→   128→        throw new Error(error.message || `HTTP ${response.status}`);
   129→   129→      }
   130→   130→
   131→   131→      // Handle 204 No Content
   132→   132→      if (response.status === 204) {
   133→   133→        return undefined as T;
   134→   134→      }
   135→   135→
   136→   136→      const json = await response.json();
   137→   137→
   138→   138→      // Unwrap the TransformInterceptor response: { success, data, meta }
   139→   139→      // If the response has a `data` field at the top level and a `success` field,
   140→   140→      // return just the `data` portion. Otherwise, return the full response.
   141→   141→      if (json && typeof json === 'object' && 'success' in json && 'data' in json) {
   142→   142→        return json.data as T;
   143→   143→      }
   144→   144→
   145→   145→      return json as T;
   146→   146→    } catch (err) {
   147→   147→      // Retry on network errors (abort, fetch failure) unless it's an auth redirect
   148→   148→      if (err instanceof Error && err.message === 'Session expired. Please log in again.') {
   149→   149→        throw err;
   150→   150→      }
   151→   151→
   152→   152→      if (err instanceof DOMException && err.name === 'AbortError') {
   153→   153→        if (this.shouldRetry(null, _attempt)) {
   154→   154→          const delay = Math.min(1000 * Math.pow(2, _attempt), 5000);
   155→   155→          await new Promise((resolve) => setTimeout(resolve, delay));
   156→   156→          return this.request<T>(endpoint, options, _attempt + 1, _isRetryAfterRefresh);
   157→   157→        }
   158→   158→        throw new Error('Request timed out. Please try again.');
   159→   159→      }
   160→   160→
   161→   161→      // Network error — retry
   162→   162→      if (err instanceof TypeError && this.shouldRetry(null, _attempt)) {
   163→   163→        const delay = Math.min(1000 * Math.pow(2, _attempt), 5000);
   164→   164→        await new Promise((resolve) => setTimeout(resolve, delay));
   165→   165→        return this.request<T>(endpoint, options, _attempt + 1, _isRetryAfterRefresh);
   166→   166→      }
   167→   167→
   168→   168→      throw err;
   169→   169→    } finally {
   170→   170→      clearTimeout(timeoutId);
   171→   171→    }
   172→   172→  }
