     1→     1→/**

     8→     8→ *   dev.database      → LLM: generate schema + write files
     9→     9→ *   dev.api           → LLM: generate API code + write files
    10→    10→ *   dev.devops        → LLM: generate CI/CD configs + write files
    11→    11→ *   dev.docker        → Generate Dockerfile + docker-compose
    12→    12→ *   dev.kubernetes    → Generate K8s manifests
    13→    13→ *   dev.qa            → Shell: run tests + LLM: analyze
    14→    14→ *   dev.test          → LLM: generate test code + write files
    15→    15→ *   dev.debug         → LLM: analyze errors + suggest fixes
    16→    16→ *   dev.documentation → LLM: generate docs + write files
    17→    17→ *
    18→    18→ * Tools: z-ai-web-dev-sdk (LLM), child_process (shell), fs (files)
    19→    19→ */

    46→    46→  supports(capabilityId: CapabilityId): boolean {

    49→    49→
    50→    50→  async execute(capabilityId: CapabilityId, input: ConnectorInput): Promise<ConnectorOutput> {
    51→    51→    const startTime = Date.now();
    52→    52→    const capId = capabilityId as DevCapability;
    53→    53→
    54→    54→    this.logger.log(`Dev connector executing: ${capId} for mission ${input.missionId}`);
    55→    55→
    56→    56→    // Ensure workspace directories exist
    57→    57→    this.ensureWorkspace(input.workspaceDir);
    58→    58→
    59→    59→    try {
    60→    60→      let result: ConnectorOutput;
    61→    61→
    62→    62→      switch (capId) {
    63→    63→        case DevCapability.ARCHITECTURE:
    64→    64→          result = await this.executeArchitecture(input);
    65→    65→          break;
    66→    66→        case DevCapability.FRONTEND:
    67→    67→          result = await this.executeFrontend(input);
    68→    68→          break;
    69→    69→        case DevCapability.BACKEND:
    70→    70→          result = await this.executeBackend(input);
    71→    71→          break;
    72→    72→        case DevCapability.DATABASE:
    73→    73→          result = await this.executeDatabase(input);
    74→    74→          break;
    75→    75→        case DevCapability.API:
    76→    76→          result = await this.executeApi(input);
    77→    77→          break;
    78→    78→        case DevCapability.DEVOPS:
    79→    79→          result = await this.executeDevOps(input);
    80→    80→          break;
    81→    81→        case DevCapability.DOCKER:
    82→    82→          result = await this.executeDocker(input);
    83→    83→          break;
    84→    84→        case DevCapability.KUBERNETES:
    85→    85→          result = await this.executeKubernetes(input);
    86→    86→          break;
    87→    87→        case DevCapability.QA:
    88→    88→          result = await this.executeQA(input);
    89→    89→          break;
    90→    90→        case DevCapability.TEST:
    91→    91→          result = await this.executeTest(input);
    92→    92→          break;
    93→    93→        case DevCapability.DEBUG:
    94→    94→          result = await this.executeDebug(input);
    95→    95→          break;
    96→    96→        case DevCapability.DOCUMENTATION:
    97→    97→          result = await this.executeDocumentation(input);
    98→    98→          break;
    99→    99→        default:
   100→   100→          result = await this.executeGenericDev(capId, input);
   101→   101→      }
   102→   102→
   103→   103→      result.durationMs = Date.now() - startTime;
   104→   104→      return result;
   105→   105→    } catch (error: any) {
   106→   106→      this.logger.error(`Dev connector failed for ${capId}: ${error.message}`);
   107→   107→      return {
   108→   108→        success: false,
   109→   109→        artifacts: [],
   110→   110→        output: { error: error.message },
   111→   111→        costUsd: 0,
   112→   112→        durationMs: Date.now() - startTime,
   113→   113→        error: error.message,
   114→   114→      };
   115→   115→    }
   116→   116→  }

   122→   122→  private async executeArchitecture(input: ConnectorInput): Promise<ConnectorOutput> {
   123→   123→    const systemPrompt = `You are a senior software architect. Analyze the mission and produce a complete technical architecture document.
   124→   124→Include: tech stack, project structure, component diagram, data models, API design, security considerations.
   125→   125→Output in markdown format.`;
   126→   126→
   127→   127→    const userPrompt = `Design the architecture for: "${input.instruction}"
   128→   128→${input.parameters.context ? `Context: ${JSON.stringify(input.parameters.context)}` : ''}
   129→   129→Produce a detailed ARCHITECTURE.md with all design decisions.`;
   130→   130→
   131→   131→    const llmResult = await this.llm.call({ systemPrompt, userPrompt, maxTokens: 4096 });
   132→   132→
   133→   133→    const filePath = path.join(input.workspaceDir, 'docs', 'ARCHITECTURE.md');
   134→   134→    fs.mkdirSync(path.dirname(filePath), { recursive: true });
   135→   135→    fs.writeFileSync(filePath, llmResult.content, 'utf-8');
   136→   136→
   137→   137→    return {
   138→   138→      success: true,
   139→   139→      artifacts: [this.makeArtifact('ARCHITECTURE.md', 'document', filePath, llmResult.content)],
   140→   140→      output: { architecture: llmResult.content.substring(0, 1000) },
   141→   141→      costUsd: llmResult.costUsd,
   142→   142→      durationMs: 0,

   167→   167→    // If LLM didn't produce parseable files, generate a template

   172→   172→      artifacts.push(...this.writeFiles(templateFiles, input.workspaceDir));
   173→   173→    }
   174→   174→
   175→   175→    return {
   176→   176→      success: artifacts.length > 0,
   177→   177→      artifacts,
   178→   178→      output: { filesGenerated: artifacts.map(a => a.name) },
   179→   179→      costUsd: llmResult.costUsd,

   394→   394→  // ═══════════════════════════════════════════════════════════
   395→   395→
   396→   396→  private async executeQA(input: ConnectorInput): Promise<ConnectorOutput> {
   397→   397→    const results: any[] = [];
   398→   398→    let allPassed = true;
   399→   399→
   400→   400→    // Try running existing test files
   401→   401→    const testDir = path.join(input.workspaceDir, 'tests');
   402→   402→    if (fs.existsSync(testDir)) {
   403→   403→      const testFiles = fs.readdirSync(testDir).filter(f => f.endsWith('.js') || f.endsWith('.mjs'));
   404→   404→      for (const testFile of testFiles.slice(0, 5)) {
   405→   405→        try {
   406→   406→          const output = execSync(`node "${path.join(testDir, testFile)}" 2>&1`, {
   407→   407→            timeout: 30000,
   408→   408→            cwd: input.workspaceDir,
   409→   409→          }).toString();
   410→   410→          results.push({ file: testFile, passed: true, output: output.slice(0, 500) });
   411→   411→        } catch (err: any) {
   412→   412→          results.push({ file: testFile, passed: false, output: (err.stdout || err.message || '').toString().slice(0, 500) });
   413→   413→          allPassed = false;
   414→   414→        }
   415→   415→      }
   416→   416→    }
   417→   417→
   418→   418→    // LLM-based quality analysis
   419→   419→    let llmAnalysis = '';
   420→   420→    try {
   421→   421→      const srcFiles = this.collectSourceFiles(input.workspaceDir);
   422→   422→      if (srcFiles.length > 0) {
   423→   423→        const llmResult = await this.llm.call({
   424→   424→          systemPrompt: 'You are a QA engineer. Analyze code for bugs and quality issues.',
   425→   425→          userPrompt: `Analyze this code for quality:\n${srcFiles.slice(0, 5).map(f => `--- ${f.name} ---\n${f.content?.slice(0, 500) || ''}`).join('\n\n')}\n\nReply in JSON: {"passed": true/false, "analysis": "brief analysis", "bugs": []}`,
   426→   426→          maxTokens: 2048,
   427→   427→        });
   428→   428→        llmAnalysis = llmResult.content;
   429→   429→      }
   430→   430→    } catch { /* optional */ }
   431→   431→
   432→   432→    // Write QA report
   433→   433→    const reportPath = path.join(input.workspaceDir, 'docs', 'QA-REPORT.md');
   434→   434→    fs.mkdirSync(path.dirname(reportPath), { recursive: true });

   591→   591→  private writeFiles(files: Map<string, string>, workspaceDir: string): GeneratedArtifact[] {
   592→   592→    const artifacts: GeneratedArtifact[] = [];
   593→   593→    files.forEach((content, filePath) => {
   594→   594→      const fullPath = path.join(workspaceDir, filePath);
   595→   595→      fs.mkdirSync(path.dirname(fullPath), { recursive: true });
   596→   596→      fs.writeFileSync(fullPath, content, 'utf-8');
   597→   597→
   598→   598→      let type: GeneratedArtifact['type'] = 'source';
   599→   599→      if (filePath.includes('test') || filePath.includes('spec')) type = 'test';
   600→   600→      else if (filePath.endsWith('.md') || filePath.endsWith('.txt')) type = 'document';
   601→   601→      else if (filePath.endsWith('.json') || filePath.endsWith('.yml') || filePath.endsWith('.yaml') || filePath.endsWith('Dockerfile') || filePath.includes('.config')) type = 'config';
   602→   602→
   603→   603→      artifacts.push(this.makeArtifact(path.basename(filePath), type, fullPath, content));
   604→   604→      this.logger.log(`  Created: ${filePath} (${Buffer.byteLength(content)} bytes)`);
   605→   605→    });
   606→   606→    return artifacts;
   607→   607→  }

   619→   619→  private getExistingFilesContext(input: ConnectorInput): string {
   620→   620→    const existing = input.previousResults;
   621→   621→    if (!existing || existing.size === 0) return '';
   622→   622→
   623→   623→    const summaries: string[] = [];
   624→   624→    existing.forEach((output, capId) => {
   625→   625→      if (output.artifacts?.length > 0) {
   626→   626→        summaries.push(`${capId}: ${output.artifacts.map((a: any) => a.name).join(', ')}`);
   627→   627→      }
   628→   628→    });
   629→   629→    return summaries.length > 0 ? `Previously generated: ${summaries.join('; ')}` : '';
   630→   630→  }

   632→   632→  private collectSourceFiles(workspaceDir: string): { name: string; content?: string }[] {
   633→   633→    const files: { name: string; content?: string }[] = [];
   634→   634→    const extensions = ['.js', '.ts', '.html', '.css', '.py'];
   635→   635→
   636→   636→    const walkDir = (dir: string) => {
   637→   637→      if (!fs.existsSync(dir)) return;
   638→   638→      for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
   639→   639→        const fullPath = path.join(dir, entry.name);
   640→   640→        if (entry.isDirectory() && !entry.name.startsWith('.') && entry.name !== 'node_modules') {
   641→   641→          walkDir(fullPath);
   642→   642→        } else if (entry.isFile() && extensions.some(ext => entry.name.endsWith(ext))) {
   643→   643→          try {
   644→   644→            const content = fs.readFileSync(fullPath, 'utf-8').slice(0, 1000);
   645→   645→            files.push({ name: path.relative(workspaceDir, fullPath), content });
   646→   646→          } catch { /* skip */ }
   647→   647→        }
   648→   648→      }
   649→   649→    };
   650→   650→
   651→   651→    walkDir(workspaceDir);
   652→   652→    return files;
   653→   653→  }
