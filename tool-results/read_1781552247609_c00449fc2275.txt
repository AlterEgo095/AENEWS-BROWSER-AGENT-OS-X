     1→/**
     2→ * AENEWS Agent OS X — Swarm Intelligence Service
     3→ *
     4→ * Phase 10 — True swarm intelligence with stigmergy, dynamic spawning,
     5→ * emergent behavior detection, and swarm size optimization.
     6→ *
     7→ * Stigmergy Model:
     8→ *   Agents leave digital "pheromones" in a shared Redis environment.
     9→ *   Pheromones have: type (exploration, success, failure, warning),
    10→ *   strength (decays over time), and spatial coordinates in capability space.
    11→ *   Other agents sense pheromones to guide their behavior.
    12→ *
    13→ * Dynamic Spawning:
    14→ *   Based on workload and pheromone density, the swarm can request
    15→ *   new agent instances. Spawning is bounded by maxSwarmSize and
    16→ *   requires capacity checks with AgentHealthService.
    17→ *
    18→ * Emergent Behavior Detection:
    19→ *   Monitors swarm metrics (convergence, divergence, oscillation, stagnation)
    20→ *   and triggers alerts and adaptive responses.
    21→ *
    22→ * Swarm Size Optimization:
    23→ *   Uses gradient-descent-inspired optimization to find ideal agent count.
    24→ *   Starts with initialSize, adjusts based on throughput and coordination overhead.
    25→ */

   241→  /**
   242→   * Execute a swarm mission — agents explore, sense pheromones, and converge.
   243→   */
   244→  async executeSwarm(swarmId: string): Promise<SwarmResult> {
   245→    const config = this.swarms.get(swarmId);
   246→    if (!config) {
   247→      throw new Error(`Swarm ${swarmId} not found`);
   248→    }
   249→
   250→    const startTime = Date.now();
   251→    const agents = this.swarmAgents.get(swarmId) || [];
   252→
   253→    this.logger.log(`Executing swarm ${swarmId} with ${agents.length} agents`);
   254→
   255→    try {
   256→      // Phase 1: Initial exploration — each agent explores independently
   257→      await this.eventBus.publish({
   258→        type: AgentEventType.CUSTOM,
   259→        source: 'SwarmIntelligenceService',
   260→        data: { event: 'swarm.phase', swarmId, phase: 'exploration' },
   261→        timestamp: new Date(),
   262→      });
   263→
   264→      const explorationPromises = agents
   265→        .filter(a => a.status === 'active')
   266→        .map(agent => this.executeAgentExploration(swarmId, agent));
   267→
   268→      const explorationResults = await Promise.allSettled(explorationPromises);
   269→
   270→      // Process exploration results and deposit pheromones
   271→      for (const result of explorationResults) {
   272→        if (result.status === 'fulfilled' && result.value) {
   273→          await this.depositPheromone(swarmId, {
   274→            id: `pher-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   275→            type: result.value.success ? 'success' : 'failure',
   276→            strength: result.value.success ? 1.0 : 0.3,
   277→            coordinates: result.value.capabilityCoordinates || {},
   278→            agentId: result.value.agentId,
   279→            collaborationId: swarmId,
   280→            timestamp: Date.now(),
   281→            decayRate: config.pheromoneDecayRate,
   282→            metadata: result.value.metadata,
   283→          });
   284→        }
   285→      }
   286→
   287→      // Phase 2: Pheromone-guided exploration — agents sense trails and adjust
   288→      await this.eventBus.publish({
   289→        type: AgentEventType.CUSTOM,
   290→        source: 'SwarmIntelligenceService',
   291→        data: { event: 'swarm.phase', swarmId, phase: 'pheromone_guided' },
   292→        timestamp: new Date(),
   293→      });
   294→
   295→      const guidedPromises = agents
   296→        .filter(a => a.status === 'active')
   297→        .map(agent => this.executePheromoneGuidedExploration(swarmId, agent));
   298→
   299→      const guidedResults = await Promise.allSettled(guidedPromises);
   300→
   301→      for (const result of guidedResults) {
   302→        if (result.status === 'fulfilled' && result.value) {
   303→          await this.depositPheromone(swarmId, {
   304→            id: `pher-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   305→            type: result.value.success ? 'success' : 'exploration',
   306→            strength: result.value.success ? 0.8 : 0.5,
   307→            coordinates: result.value.capabilityCoordinates || {},
   308→            agentId: result.value.agentId,
   309→            collaborationId: swarmId,
   310→            timestamp: Date.now(),
   311→            decayRate: config.pheromoneDecayRate,
   312→          });
   313→        }
   314→      }
   315→
   316→      // Phase 3: Convergence — agents evaluate and validate findings
   317→      await this.eventBus.publish({
   318→        type: AgentEventType.CUSTOM,
   319→        source: 'SwarmIntelligenceService',
   320→        data: { event: 'swarm.phase', swarmId, phase: 'convergence' },
   321→        timestamp: new Date(),
   322→      });
   323→
   324→      const findings = this.collectFindings(swarmId);
   325→      const validatedFindings = await this.validateFindings(swarmId, findings);
   326→
   327→      // Detect emergent behaviors
   328→      if (config.enableEmergentDetection) {
   329→        await this.detectEmergentBehaviors(swarmId);
   330→      }
   331→
   332→      // Optimize swarm size
   333→      if (config.enableDynamicSpawning) {
   334→        await this.optimizeSwarmSize(swarmId);
   335→      }
   336→
   337→      // Decay pheromones
   338→      this.decayPheromones(swarmId);
   339→
   340→      // Update final metrics
   341→      const metrics = this.calculateMetrics(swarmId);
   342→      metrics.elapsedTimeMs = Date.now() - startTime;
   343→      this.swarmMetrics.set(swarmId, metrics);
   344→
   345→      const result: SwarmResult = {
   346→        swarmId,
   347→        status: 'completed',
   348→        findings: validatedFindings,
   349→        metrics,
   350→        durationMs: Date.now() - startTime,
   351→        agentsUsed: agents.map(a => a.agentId),
   352→        pheromoneTrail: this.pheromoneTrails.get(swarmId) || [],
   353→      };
   354→
   355→      this.swarmResults.set(swarmId, result);
   356→
   357→      // Store in memory for future learning
   358→      await this.memoryService.store(
   359→        `swarm:${swarmId}`,
   360→        MemoryTier.LONG_TERM,
   361→        'result',
   362→        JSON.stringify(result),
   363→        86400, // 24h TTL
   364→      );
   365→
   366→      await this.eventBus.publish({
   367→        type: AgentEventType.CUSTOM,
   368→        source: 'SwarmIntelligenceService',
   369→        data: { event: 'swarm.completed', swarmId, findingCount: validatedFindings.length, durationMs: result.durationMs },
   370→        timestamp: new Date(),
   371→      });
   372→
   373→      return result;
   374→    } catch (error: any) {
   375→      this.logger.error(`Swarm ${swarmId} execution failed: ${error.message}`, error.stack);
   376→
   377→      const failedResult: SwarmResult = {
   378→        swarmId,
   379→        status: 'failed',
   380→        findings: [],
   381→        metrics: this.swarmMetrics.get(swarmId)!,
   382→        durationMs: Date.now() - startTime,
   383→        agentsUsed: agents.map(a => a.agentId),
   384→        pheromoneTrail: this.pheromoneTrails.get(swarmId) || [],
   385→      };
   386→
   387→      this.swarmResults.set(swarmId, failedResult);
   388→
   389→      await this.eventBus.publish({
   390→        type: AgentEventType.CUSTOM,
   391→        source: 'SwarmIntelligenceService',
   392→        data: { event: 'swarm.failed', swarmId, error: error.message },
   393→        timestamp: new Date(),
   394→      });
   395→
   396→      return failedResult;
   397→    }
   398→  }

   426→  /**
   427→   * Deposit a pheromone in the environment.
   428→   */
   429→  async depositPheromone(swarmId: string, pheromone: Pheromone): Promise<void> {
   430→    const trail = this.pheromoneTrails.get(swarmId) || [];
   431→    trail.push(pheromone);
   432→
   433→    // Limit trail size to prevent unbounded growth
   434→    if (trail.length > 500) {
   435→      // Remove weakest pheromones
   436→      trail.sort((a, b) => b.strength - a.strength);
   437→      trail.splice(400);
   438→    }
   439→
   440→    this.pheromoneTrails.set(swarmId, trail);
   441→
   442→    // Update metrics
   443→    const metrics = this.swarmMetrics.get(swarmId);
   444→    if (metrics) {
   445→      metrics.totalPheromones = trail.length;
   446→      metrics.pheromoneDistribution[pheromone.type] =
   447→        (metrics.pheromoneDistribution[pheromone.type] || 0) + 1;
   448→    }
   449→
   450→    await this.eventBus.publish({
   451→      type: AgentEventType.CUSTOM,
   452→      source: 'SwarmIntelligenceService',
   453→      data: { event: 'pheromone.deposited', swarmId, type: pheromone.type, strength: pheromone.strength },
   454→      timestamp: new Date(),
   455→    });
   456→  }

   458→  /**
   459→   * Sense pheromones near given capability coordinates.
   460→   */
   461→  async sensePheromones(
   462→    swarmId: string,
   463→    coordinates: Record<string, number>,
   464→    radius: number = 0.5,
   465→  ): Promise<Pheromone[]> {
   466→    const trail = this.pheromoneTrails.get(swarmId) || [];
   467→
   468→    return trail.filter(p => {
   469→      // Calculate Euclidean distance in capability space
   470→      const distance = this.capabilityDistance(p.coordinates, coordinates);
   471→      return distance <= radius && p.strength > 0.05; // ignore negligible pheromones
   472→    }).sort((a, b) => b.strength - a.strength); // strongest first
   473→  }

   475→  /**
   476→   * Decay all pheromones (called periodically).
   477→   */
   478→  private decayPheromones(swarmId: string): void {
   479→    const trail = this.pheromoneTrails.get(swarmId);
   480→    if (!trail) return;
   481→
   482→    const now = Date.now();
   483→    for (const pheromone of trail) {
   484→      const ageSeconds = (now - pheromone.timestamp) / 1000;
   485→      pheromone.strength = Math.max(0, pheromone.strength - pheromone.decayRate * ageSeconds);
   486→    }
   487→
   488→    // Remove fully decayed pheromones
   489→    const alive = trail.filter(p => p.strength > 0.01);
   490→    this.pheromoneTrails.set(swarmId, alive);
   491→  }

   495→  /**
   496→   * Deploy agents to a swarm from available clusters.
   497→   */
   498→  private async deployAgents(swarmId: string, count: number): Promise<SwarmAgent[]> {
   499→    const config = this.swarms.get(swarmId);
   500→    if (!config) return [];
   501→
   502→    const deployed: SwarmAgent[] = [];
   503→    const existingAgents = this.swarmAgents.get(swarmId) || [];
   504→
   505→    // Don't exceed max size
   506→    const availableSlots = config.maxSize - existingAgents.length;
   507→    const toDeploy = Math.min(count, availableSlots);
   508→
   509→    for (let i = 0; i < toDeploy; i++) {
   510→      // Determine which cluster to pull from
   511→      const clusterType = config.preferredClusters[i % config.preferredClusters.length]
   512→        || ClusterType.META_INTELLIGENCE;
   513→
   514→      const role = this.assignRole(i, toDeploy);
   515→
   516→      const agent: SwarmAgent = {
   517→        agentId: `swarm-${swarmId}-agent-${i}`,
   518→        clusterType,
   519→        role,
   520→        capabilities: config.requiredCapabilities,
   521→        joinedAt: Date.now(),
   522→        pheromonesSensed: 0,
   523→        tasksCompleted: 0,
   524→        status: 'active',
   525→      };
   526→
   527→      deployed.push(agent);
   528→    }
   529→
   530→    const allAgents = [...existingAgents, ...deployed];
   531→    this.swarmAgents.set(swarmId, allAgents);
   532→
   533→    // Update metrics
   534→    const metrics = this.swarmMetrics.get(swarmId);
   535→    if (metrics) {
   536→      metrics.currentSize = allAgents.length;
   537→    }
   538→
   539→    return deployed;
   540→  }

   542→  /**
   543→   * Assign a role to an agent based on its position in the swarm.
   544→   */
   545→  private assignRole(index: number, total: number): SwarmAgent['role'] {
   546→    if (index === 0) return 'coordinator';
   547→    if (index < Math.ceil(total * 0.6)) return 'explorer';
   548→    if (index < Math.ceil(total * 0.85)) return 'evaluator';
   549→    return 'specialist';
   550→  }

   552→  /**
   553→   * Execute an agent's initial exploration phase.
   554→   */
   555→  private async executeAgentExploration(
   556→    swarmId: string,
   557→    agent: SwarmAgent,
   558→  ): Promise<{ agentId: string; success: boolean; capabilityCoordinates: Record<string, number>; metadata?: any } | null> {
   559→    try {
   560→      const config = this.swarms.get(swarmId);
   561→      if (!config) return null;
   562→
   563→      // Build context for the agent
   564→      const context: AgentContext = {
   565→        missionId: swarmId,
   566→        agentId: agent.agentId,
   567→        tenantId: 'swarm',
   568→        taskDescription: config.mission,
   569→        config: {
   570→          objectives: config.objectives,
   571→          capabilities: agent.capabilities,
   572→          role: agent.role,
   573→          phase: 'exploration',
   574→        },
   575→        metadata: { swarmId, clusterType: agent.clusterType },
   576→      };
   577→
   578→      // Use LLM for intelligent exploration if available
   579→      let result: any;
   580→      if (this.llmService) {
   581→        try {
   582→          const llmResponse = await this.llmService.chat(
   583→            [
   584→              {
   585→                role: 'system',
   586→                content: `You are a swarm agent with role "${agent.role}" in cluster "${agent.clusterType}". Explore the problem space for: ${config.mission}. Report findings with confidence and capability coordinates.`,
   587→              },
   588→              {
   589→                role: 'user',
   590→                content: `Objectives: ${config.objectives.join(', ')}. Capabilities: ${agent.capabilities.join(', ')}. What do you discover?`,
   591→              },
   592→            ],
   593→            { temperature: 0.7, maxTokens: 500 },
   594→          );
   595→
   596→          result = {
   597→            success: true,
   598→            finding: llmResponse.content,
   599→            capabilityCoordinates: this.inferCapabilityCoordinates(agent.capabilities, config.requiredCapabilities),
   600→          };
   601→        } catch {
   602→          result = {
   603→            success: false,
   604→            finding: 'LLM exploration failed',
   605→            capabilityCoordinates: {},
   606→          };
   607→        }
   608→      } else {
   609→        // Fallback: simulation-based exploration
   610→        result = {
   611→          success: Math.random() > 0.3,
   612→          finding: `Explored ${agent.role} aspect of mission`,
   613→          capabilityCoordinates: this.inferCapabilityCoordinates(agent.capabilities, config.requiredCapabilities),
   614→        };
   615→      }
   616→
   617→      agent.tasksCompleted++;
   618→      return { agentId: agent.agentId, ...result };
   619→    } catch (error: any) {
   620→      this.logger.warn(`Agent ${agent.agentId} exploration failed: ${error.message}`);
   621→      agent.status = 'failed';
   622→      return { agentId: agent.agentId, success: false, capabilityCoordinates: {} };
   623→    }
   624→  }

   626→  /**
   627→   * Execute pheromone-guided exploration for an agent.
   628→   */
   629→  private async executePheromoneGuidedExploration(
   630→    swarmId: string,
   631→    agent: SwarmAgent,
   632→  ): Promise<{ agentId: string; success: boolean; capabilityCoordinates: Record<string, number>; metadata?: any } | null> {
   633→    try {
   634→      const config = this.swarms.get(swarmId);
   635→      if (!config) return null;
   636→
   637→      // Sense nearby pheromones
   638→      const currentCoords = this.inferCapabilityCoordinates(agent.capabilities, config.requiredCapabilities);
   639→      const sensedPheromones = await this.sensePheromones(swarmId, currentCoords, config.pheromoneSensingRadius);
   640→
   641→      agent.pheromonesSensed = sensedPheromones.length;
   642→
   643→      // Determine direction based on pheromone signals
   644→      const direction = this.computePheromoneDirection(sensedPheromones, currentCoords);
   645→
   646→      // Move agent's focus toward strongest signals
   647→      const newCoords = this.moveInCapabilitySpace(currentCoords, direction, 0.3);
   648→
   649→      // Execute guided exploration
   650→      const context: AgentContext = {
   651→        missionId: swarmId,
   652→        agentId: agent.agentId,
   653→        tenantId: 'swarm',
   654→        taskDescription: config.mission,
   655→        config: {
   656→          objectives: config.objectives,
   657→          sensedPheromones: sensedPheromones.map(p => ({ type: p.type, strength: p.strength })),
   658→          direction,
   659→          phase: 'pheromone_guided',
   660→        },
   661→        metadata: { swarmId },
   662→      };
   663→
   664→      let result: any;
   665→      if (this.llmService && sensedPheromones.length > 0) {
   666→        try {
   667→          const successSignals = sensedPheromones.filter(p => p.type === 'success');
   668→          const failureSignals = sensedPheromones.filter(p => p.type === 'failure');
   669→
   670→          const llmResponse = await this.llmService.chat(
   671→            [
   672→              {
   673→                role: 'system',
   674→                content: `You are a swarm agent following pheromone trails. There are ${successSignals.length} success signals and ${failureSignals.length} failure signals nearby. ${successSignals.length > failureSignals.length ? 'Move toward success areas.' : 'Avoid failure areas.'}`,
   675→              },
   676→              {
   677→                role: 'user',
   678→                content: `Mission: ${config.mission}. Based on pheromone signals, what is your next finding?`,
   679→              },
   680→            ],
   681→            { temperature: 0.5, maxTokens: 400 },
   682→          );
   683→
   684→          result = {
   685→            success: true,
   686→            finding: llmResponse.content,
   687→            capabilityCoordinates: newCoords,
   688→          };
   689→        } catch {
   690→          result = { success: false, finding: 'LLM guided exploration failed', capabilityCoordinates: newCoords };
   691→        }
   692→      } else {
   693→        result = {
   694→          success: sensedPheromones.some(p => p.type === 'success'),
   695→          finding: `Guided exploration following ${sensedPheromones.length} pheromone signals`,
   696→          capabilityCoordinates: newCoords,
   697→        };
   698→      }
   699→
   700→      agent.tasksCompleted++;
   701→      return { agentId: agent.agentId, ...result };
   702→    } catch (error: any) {
   703→      this.logger.warn(`Agent ${agent.agentId} guided exploration failed: ${error.message}`);
   704→      return { agentId: agent.agentId, success: false, capabilityCoordinates: {} };
   705→    }
   706→  }

   710→  /**
   711→   * Detect emergent behaviors in the swarm.
   712→   */
   713→  private async detectEmergentBehaviors(swarmId: string): Promise<EmergentBehaviorType[]> {
   714→    const metrics = this.swarmMetrics.get(swarmId);
   715→    if (!metrics) return ['none'];
   716→
   717→    const behaviors: EmergentBehaviorType[] = [];
   718→    const trail = this.pheromoneTrails.get(swarmId) || [];
   719→    const agents = this.swarmAgents.get(swarmId) || [];
   720→
   721→    // Convergence: high concentration of success pheromones in one area
   722→    if (metrics.convergenceScore > 0.85) {
   723→      behaviors.push('convergence');
   724→    }
   725→
   726→    // Divergence: agents exploring very different areas
   727→    if (metrics.explorationCoverage > 0.8 && metrics.convergenceScore < 0.3) {
   728→      behaviors.push('divergence');
   729→    }
   730→
   731→    // Oscillation: agents switching between approaches
   732→    const successCount = trail.filter(p => p.type === 'success').length;
   733→    const failureCount = trail.filter(p => p.type === 'failure').length;
   734→    if (successCount > 3 && failureCount > 3 && Math.abs(successCount - failureCount) < 2) {
   735→      behaviors.push('oscillation');
   736→    }
   737→
   738→    // Stagnation: no progress for extended period
   739→    if (metrics.stagnationCounter > 3) {
   740→      behaviors.push('stagnation');
   741→    }
   742→
   743→    // Cascade: one pheromone triggered many follow-ons
   744→    const recentPheromones = trail.filter(p => Date.now() - p.timestamp < 5000);
   745→    if (recentPheromones.length > agents.length * 2) {
   746→      behaviors.push('cascade');
   747→    }
   748→
   749→    // Flocking: agents grouping around a successful approach
   750→    const highStrengthSuccess = trail.filter(p => p.type === 'success' && p.strength > 0.7);
   751→    if (highStrengthSuccess.length > agents.length * 0.6) {
   752→      behaviors.push('flocking');
   753→    }
   754→
   755→    if (behaviors.length === 0) {
   756→      behaviors.push('none');
   757→    }
   758→
   759→    // Store in history
   760→    const history = this.emergentHistory.get(swarmId) || [];
   761→    history.push(...behaviors);
   762→    this.emergentHistory.set(swarmId, history);
   763→
   764→    // Update metrics
   765→    metrics.emergentBehaviors = behaviors;
   766→
   767→    // Publish event
   768→    await this.eventBus.publish({
   769→      type: AgentEventType.CUSTOM,
   770→      source: 'SwarmIntelligenceService',
   771→      data: { event: 'emergent.detected', swarmId, behaviors },
   772→      timestamp: new Date(),
   773→    });
   774→
   775→    return behaviors;
   776→  }

   778→  // ─── Swarm Size Optimization ──────────────────────────────────
   779→
   780→  /**
   781→   * Optimize swarm size using gradient-descent-inspired approach.
   782→   */
   783→  private async optimizeSwarmSize(swarmId: string): Promise<number> {
   784→    const config = this.swarms.get(swarmId);
   785→    const metrics = this.swarmMetrics.get(swarmId);
   786→    if (!config || !metrics) return metrics?.currentSize ?? config?.initialSize ?? 5;
   787→
   788→    const optState = this.sizeOptState.get(swarmId) || {
   789→      previousThroughput: 0,
   790→      previousSize: metrics.currentSize,
   791→      gradientHistory: [],
   792→    };
   793→
   794→    // Calculate current throughput (tasks per second per agent)
   795→    const currentThroughput = metrics.agentThroughput;
   796→
   797→    // Estimate gradient: how does throughput change with size?
   798→    const gradient = optState.previousSize > 0
   799→      ? (currentThroughput - optState.previousThroughput) / (metrics.currentSize - optState.previousSize || 1)
   800→      : 0;
   801→
   802→    optState.gradientHistory.push(gradient);
   803→    if (optState.gradientHistory.length > 10) {
   804→      optState.gradientHistory.shift();
   805→    }
   806→
   807→    // Average gradient for smoother optimization
   808→    const avgGradient = optState.gradientHistory.reduce((a, b) => a + b, 0) / optState.gradientHistory.length;
   809→
   810→    // Calculate coordination overhead
   811→    const coordinationPenalty = metrics.coordinationOverhead * 0.1;
   812→
   813→    // Determine size adjustment
   814→    let optimalSize = metrics.currentSize;
   815→    if (avgGradient > 0.1 && coordinationPenalty < 0.5) {
   816→      // More agents improve throughput — grow
   817→      optimalSize = Math.min(config.maxSize, metrics.currentSize + 1);
   818→    } else if (avgGradient < -0.1 || coordinationPenalty > 0.8) {
   819→      // More agents hurt — shrink
   820→      optimalSize = Math.max(config.minSize, metrics.currentSize - 1);
   821→    }
   822→
   823→    // Apply the adjustment
   824→    if (optimalSize > metrics.currentSize) {
   825→      await this.deployAgents(swarmId, optimalSize - metrics.currentSize);
   826→      this.logger.log(`Swarm ${swarmId}: growing from ${metrics.currentSize} to ${optimalSize} agents (gradient: ${avgGradient.toFixed(3)})`);
   827→    } else if (optimalSize < metrics.currentSize) {
   828→      // Remove idle agents
   829→      const agents = this.swarmAgents.get(swarmId) || [];
   830→      const toRemove = metrics.currentSize - optimalSize;
   831→      let removed = 0;
   832→      for (const agent of agents) {
   833→        if (agent.status === 'idle' && removed < toRemove) {
   834→          agent.status = 'departed';
   835→          removed++;
   836→        }
   837→      }
   838→      this.logger.log(`Swarm ${swarmId}: shrinking from ${metrics.currentSize} to ${optimalSize} agents`);
   839→    }
   840→
   841→    // Update optimization state
   842→    optState.previousThroughput = currentThroughput;
   843→    optState.previousSize = metrics.currentSize;
   844→    this.sizeOptState.set(swarmId, optState);
   845→
   846→    metrics.optimalSize = optimalSize;
   847→    metrics.currentSize = (this.swarmAgents.get(swarmId) || []).filter(a => a.status === 'active').length;
   848→
   849→    return optimalSize;
   850→  }

   896→
   897→  /**
   898→   * Calculate comprehensive swarm metrics.
   899→   */
   900→  private calculateMetrics(swarmId: string): SwarmMetrics {
   901→    const existing = this.swarmMetrics.get(swarmId);
   902→    const trail = this.pheromoneTrails.get(swarmId) || [];
   903→    const agents = this.swarmAgents.get(swarmId) || [];
   904→    const activeAgents = agents.filter(a => a.status === 'active');
   905→
   906→    // Convergence score: how concentrated are success pheromones?
   907→    const successPheromones = trail.filter(p => p.type === 'success');
   908→    let convergenceScore = 0;
   909→    if (successPheromones.length > 1) {
   910→      const avgCoords = this.averageCoordinates(successPheromones.map(p => p.coordinates));
   911→      const distances = successPheromones.map(p => this.capabilityDistance(p.coordinates, avgCoords));
   912→      const avgDistance = distances.reduce((a, b) => a + b, 0) / distances.length;
   913→      convergenceScore = Math.max(0, 1 - avgDistance);
   914→    }
   915→
   916→    // Exploration coverage: how many unique capability areas have been explored?
   917→    const uniqueAreas = new Set<string>();
   918→    for (const p of trail) {
   919→      const key = Object.entries(p.coordinates)
   920→        .map(([k, v]) => `${k}:${Math.round(v * 10) / 10}`)
   921→        .join('|');
   922→      uniqueAreas.add(key);
   923→    }
   924→    const explorationCoverage = Math.min(1, uniqueAreas.size / 20); // normalized to 20 areas
   925→
   926→    // Agent throughput
   927→    const totalTasks = activeAgents.reduce((sum, a) => sum + a.tasksCompleted, 0);
   928→    const elapsedMs = existing?.elapsedTimeMs || 1;
   929→    const agentThroughput = totalTasks / (elapsedMs / 1000) / Math.max(1, activeAgents.length);
   930→
   931→    // Coordination overhead
   932→    const totalPheromonesSensed = activeAgents.reduce((sum, a) => sum + a.pheromonesSensed, 0);
   933→    const coordinationOverhead = totalTasks > 0 ? totalPheromonesSensed / totalTasks : 0;
   934→
   935→    const metrics: SwarmMetrics = {
   936→      totalPheromones: trail.length,
   937→      pheromoneDistribution: existing?.pheromoneDistribution || {
   938→        exploration: 0, success: 0, failure: 0, warning: 0, resource: 0,
   939→      },
   940→      agentThroughput,
   941→      coordinationOverhead,
   942→      convergenceScore,
   943→      explorationCoverage,
   944→      stagnationCounter: existing?.stagnationCounter || 0,
   945→      emergentBehaviors: existing?.emergentBehaviors || ['none'],
   946→      currentSize: activeAgents.length,
   947→      optimalSize: existing?.optimalSize ?? activeAgents.length,
   948→      elapsedTimeMs: elapsedMs,
   949→    };
   950→
   951→    this.swarmMetrics.set(swarmId, metrics);
   952→    return metrics;
   953→  }

   957→  /**
   958→   * Calculate Euclidean distance between two coordinate sets in capability space.
   959→   */
   960→  private capabilityDistance(a: Record<string, number>, b: Record<string, number>): number {
   961→    const allKeys = new Set([...Object.keys(a), ...Object.keys(b)]);
   962→    let sumSquares = 0;
   963→    for (const key of allKeys) {
   964→      const diff = (a[key] || 0) - (b[key] || 0);
   965→      sumSquares += diff * diff;
   966→    }
   967→    return Math.sqrt(sumSquares);
   968→  }

   970→  /**
   971→   * Compute direction vector from pheromone signals.
   972→   */
   973→  private computePheromoneDirection(
   974→    pheromones: Pheromone[],
   975→    currentCoords: Record<string, number>,
   976→  ): Record<string, number> {
   977→    if (pheromones.length === 0) return currentCoords;
   978→
   979→    const direction: Record<string, number> = {};
   980→    for (const pheromone of pheromones) {
   981→      const weight = pheromone.type === 'success' ? 1.0
   982→        : pheromone.type === 'failure' ? -0.5
   983→        : pheromone.type === 'warning' ? -0.3
   984→        : 0.2;
   985→
   986→      for (const [key, value] of Object.entries(pheromone.coordinates)) {
   987→        direction[key] = (direction[key] || 0) + (value - (currentCoords[key] || 0)) * weight * pheromone.strength;
   988→      }
   989→    }
   990→
   991→    return direction;
   992→  }

   994→  /**
   995→   * Move in capability space toward a direction.
   996→   */
   997→  private moveInCapabilitySpace(
   998→    current: Record<string, number>,
   999→    direction: Record<string, number>,
  1000→    stepSize: number,
  1001→  ): Record<string, number> {
  1002→    const result = { ...current };
  1003→    for (const [key, dir] of Object.entries(direction)) {
  1004→      result[key] = (result[key] || 0) + dir * stepSize;
  1005→    }
  1006→    return result;
  1007→  }

  1012→  private inferCapabilityCoordinates(

  1020→    return coords;

  1023→  /**
  1024→   * Calculate average coordinates from an array of coordinate sets.
  1025→   */
  1026→  private averageCoordinates(coordsArray: Record<string, number>[]): Record<string, number> {
  1027→    if (coordsArray.length === 0) return {};
  1028→    const result: Record<string, number> = {};
  1029→    const allKeys = new Set(coordsArray.flatMap(Object.keys));
  1030→
  1031→    for (const key of allKeys) {
  1032→      const values = coordsArray.map(c => c[key] || 0);
  1033→      result[key] = values.reduce((a, b) => a + b, 0) / values.length;
  1034→    }
  1035→    return result;
  1036→  }
