   146→// ─── Service ─────────────────────────────────────────────────
   147→
   148→/**
   149→ * Agent Orchestrator — implements the full mission pipeline:
   150→ *
   151→ *   Decompose → Plan → Execute → Critique → Repair → Validate → Deliver
   152→ *
   153→ * Each step uses the AgentRegistryService to discover capable agents
   154→ * and emits events via AgentEventBusService for observability.
   155→ *
   156→ * Enhanced features:
   157→ *   - Timeout protection per pipeline step (configurable via env vars)
   158→ *   - findBestAgent for intelligent agent selection with health + load awareness
   159→ *   - Improved validation with schema, quality, completeness, constraints, and safety checks
   160→ *   - Concurrent mission support (deduplication of active pipelines)
   161→ *   - Structured logging for observability
   162→ *   - Pipeline metrics tracking
   163→ *
   164→ * When LLM is available, the orchestrator uses it for intelligent
   165→ * decomposition, planning, critiquing, and repair. When LLM is
   166→ * unavailable, it falls back to heuristic methods.
   167→ */
   168→@Injectable()
   169→export class AgentOrchestratorService {
   170→  private readonly logger = new Logger(AgentOrchestratorService.name);
   171→
   172→  /** Per-mission pipeline state tracking */
   173→  private readonly pipelineStates = new Map<string, PipelineState>();
   174→
   175→  /** Concurrent mission deduplication: active pipeline promises */
   176→  private readonly activePipelines = new Map<string, Promise<DeliveryPackage>>();
   177→
   178→  /** Pipeline execution metrics history (in-memory, last 1000 missions) */
   179→  private readonly metricsHistory: PipelineMetrics[] = [];
   180→  private readonly MAX_METRICS_HISTORY = 1000;
   181→
   182→  /** Step-level agent dispatch counter (reset per pipeline) */
   183→  private currentStepDispatchCount = 0;
   184→
   185→  /** Configurable pipeline timeouts */
   186→  private readonly pipelineTimeouts: Record<string, number>;
   187→
   188→  constructor(
   189→    private readonly registry: AgentRegistryService,
   190→    private readonly eventBus: AgentEventBusService,
   191→    private readonly memory: AgentMemoryService,
   192→    @Optional() private readonly llmService: LLMService,
   193→    @Optional() private readonly healthService: AgentHealthService,
   194→    @Optional() private readonly configService: ConfigService,
   195→    @Optional() private readonly circuitBreakerService: CircuitBreakerService,
   196→  ) {
   197→    // Load timeouts from env vars with fallback to defaults
   198→    this.pipelineTimeouts = { ...DEFAULT_PIPELINE_TIMEOUTS };
   199→    if (this.configService) {
   200→      for (const step of Object.keys(DEFAULT_PIPELINE_TIMEOUTS)) {
   201→        const envKey = `PIPELINE_TIMEOUT_${step.toUpperCase()}`;
   202→        const envVal = this.configService.get<string>(envKey);
   203→        if (envVal) {
   204→          const parsed = parseInt(envVal, 10);
   205→          if (!isNaN(parsed) && parsed > 0) {
   206→            this.pipelineTimeouts[step] = parsed;
   207→          }
   208→        }
   209→      }
   210→    }
   211→    this.logger.log({
   212→      msg: 'AgentOrchestrator initialized with pipeline timeouts',
   213→      timeouts: this.pipelineTimeouts,
   214→    });
   215→  }
   216→
   217→  // ─── Full Pipeline ──────────────────────────────────────────
   218→
   219→  /**
   220→   * Execute a complete mission through the full pipeline.
   221→   *
   222→   * Concurrent mission support: if this mission already has an active
   223→   * pipeline running, return its promise instead of starting a duplicate.
   224→   *
   225→   * Circuit Breaker: The entire pipeline is wrapped in an orchestrator-level
   226→   * circuit breaker (`orchestrator:pipeline`). If the circuit is OPEN, a
   227→   * cached/simplified result is returned. Individual steps are also wrapped
   228→   * in per-step circuit breakers (`pipeline:{stepName}`).
   229→   */
   230→  async executeMission(mission: Mission): Promise<DeliveryPackage> {
   231→    // Deduplicate: if this mission already has an active pipeline, await it
   232→    if (this.activePipelines.has(mission.id)) {
   233→      this.logger.log({
   234→        step: 'executeMission',
   235→        missionId: mission.id,
   236→        msg: 'Mission already has an active pipeline — returning existing promise',
   237→      });
   238→      return this.activePipelines.get(mission.id)!;
   239→    }
   240→
   241→    const pipelinePromise = this.runPipelineWithCircuitBreaker(mission).finally(() => {
   242→      this.activePipelines.delete(mission.id);
   243→    });
   244→
   245→    this.activePipelines.set(mission.id, pipelinePromise);
   246→    return pipelinePromise;
   247→  }
   248→
   249→  /**
   250→   * Wrap the pipeline execution in the orchestrator-level circuit breaker.
   251→   * If the orchestrator circuit is OPEN, returns a failed delivery package
   252→   * with a meaningful error instead of throwing.
   253→   */
   254→  private async runPipelineWithCircuitBreaker(mission: Mission): Promise<DeliveryPackage> {
   255→    if (this.circuitBreakerService) {
   256→      const circuitKey = `${CIRCUIT_KEY_PREFIX.ORCHESTRATOR}:pipeline`;
   257→      return this.circuitBreakerService.execute<DeliveryPackage>(
   258→        circuitKey,
   259→        () => this.runPipeline(mission),
   260→        // Fallback: return a failed delivery package when circuit is OPEN
   261→        async () => ({
   262→          missionId: mission.id,
   263→          status: 'failed',
   264→          results: [],
   265→          critiques: [],
   266→          validations: [],
   267→          totalDuration: 0,
   268→          summary: `Pipeline circuit breaker is OPEN — mission "${mission.description}" could not be executed. Will retry after circuit recovers.`,
   269→        }),
   270→      );
   271→    }
   272→
   273→    return this.runPipeline(mission);
   274→  }
   275→
   276→  /**
   277→   * Execute a pipeline step with its own circuit breaker.
   278→   * On circuit OPEN: skip to the next step or return a cached/default result.
   279→   * On HALF_OPEN: allow a probe request through.
   280→   */
   281→  private async executeStepWithCircuitBreaker<T>(
   282→    stepName: string,
   283→    fn: () => Promise<T>,
   284→    fallback?: () => Promise<T>,
   285→  ): Promise<T> {
   286→    if (!this.circuitBreakerService) {
   287→      return fn();
   288→    }
   289→
   290→    const circuitKey = `${CIRCUIT_KEY_PREFIX.PIPELINE}:${stepName}`;
   291→    return this.circuitBreakerService.execute<T>(
   292→      circuitKey,
   293→      fn,
   294→      fallback,
   295→    );
   296→  }
   297→
   298→  /**
   299→   * Internal pipeline execution. Wrapped by executeMission for deduplication.
   300→   */
   301→  private async runPipeline(mission: Mission): Promise<DeliveryPackage> {
   302→    const pipelineStartTime = Date.now();
   303→    const stepMetrics: StepMetrics[] = [];
   304→    this.currentStepDispatchCount = 0;
   305→
   306→    this.logStructured('pipeline_start', mission.id, {
   307→      description: mission.description,
   308→      priority: mission.priority || 'medium',
   309→    });
   310→
   311→    try {
   312→      // 1. Decompose
   313→      this.setState(mission.id, 'decomposing');
   314→      this.eventBus.emit(AgentEventType.AGENT_STARTED, mission.id, {
   315→        phase: 'decompose',
   316→      });
   317→      const decomposeStart = Date.now();
   318→      let llmUsedDecompose = false;
   319→      let subtasks: Subtask[];
   320→      try {
   321→        subtasks = await this.withTimeout(
   322→          this.decompose(mission),
   323→          this.pipelineTimeouts.decompose,
   324→          'decompose',
   325→        );
   326→        llmUsedDecompose = this.llmService?.isAnyAvailable() ?? false;
   327→      } catch (error: any) {
   328→        stepMetrics.push({
   329→          stepName: 'decompose',
   330→          durationMs: Date.now() - decomposeStart,
   331→          llmUsed: false,
   332→          agentDispatchCount: 0,
   333→          success: false,
   334→          errorMessage: error.message,
   335→        });
   336→        throw error;
   337→      }
   338→      stepMetrics.push({
   339→        stepName: 'decompose',
   340→        durationMs: Date.now() - decomposeStart,
   341→        llmUsed: llmUsedDecompose,
   342→        agentDispatchCount: 0,
   343→        success: true,
   344→      });
   345→      await this.memory.store(mission.id, MemoryTier.WORKING, 'subtasks', subtasks);
   346→
   347→      this.logStructured('decompose', mission.id, {
   348→        subtaskCount: subtasks.length,
   349→        durationMs: Date.now() - decomposeStart,
   350→        llmUsed: llmUsedDecompose,
   351→      });
   352→
   353→      // 2. Plan
   354→      this.setState(mission.id, 'planning');
   355→      const planStart = Date.now();
   356→      let llmUsedPlan = false;
   357→      let plan: ExecutionPlan;
   358→      try {
   359→        plan = await this.withTimeout(
   360→          this.plan(subtasks),
   361→          this.pipelineTimeouts.plan,
   362→          'plan',
   363→        );
   364→        llmUsedPlan = this.llmService?.isAnyAvailable() ?? false;
   365→      } catch (error: any) {
   366→        stepMetrics.push({
   367→          stepName: 'plan',
   368→          durationMs: Date.now() - planStart,
   369→          llmUsed: false,
   370→          agentDispatchCount: 0,
   371→          success: false,
   372→          errorMessage: error.message,
   373→        });
   374→        throw error;
   375→      }
   376→      stepMetrics.push({
   377→        stepName: 'plan',
   378→        durationMs: Date.now() - planStart,
   379→        llmUsed: llmUsedPlan,
   380→        agentDispatchCount: 0,
   381→        success: true,
   382→      });
   383→      await this.memory.store(mission.id, MemoryTier.WORKING, 'plan', plan);
   384→
   385→      this.logStructured('plan', mission.id, {
   386→        waveCount: plan.executionOrder.length,
   387→        subtaskCount: plan.subtasks.length,
   388→        durationMs: Date.now() - planStart,
   389→        llmUsed: llmUsedPlan,
   390→      });
   391→
   392→      // 3. Execute
   393→      this.setState(mission.id, 'executing');
   394→      const execStart = Date.now();
   395→      const dispatchCountBeforeExecute = this.currentStepDispatchCount;
   396→      let results: ExecutionResult[];
   397→      try {
   398→        results = await this.withTimeout(
   399→          this.execute(plan),
   400→          this.pipelineTimeouts.execute,
   401→          'execute',
   402→        );
   403→      } catch (error: any) {
   404→        stepMetrics.push({
   405→          stepName: 'execute',
   406→          durationMs: Date.now() - execStart,
   407→          llmUsed: false,
   408→          agentDispatchCount: this.currentStepDispatchCount - dispatchCountBeforeExecute,
   409→          success: false,
   410→          errorMessage: error.message,
   411→        });
   412→        throw error;
   413→      }
   414→      const executeDispatchCount = this.currentStepDispatchCount - dispatchCountBeforeExecute;
   415→      stepMetrics.push({
   416→        stepName: 'execute',
   417→        durationMs: Date.now() - execStart,
   418→        llmUsed: false,
   419→        agentDispatchCount: executeDispatchCount,
   420→        success: true,
   421→      });
   422→      await this.memory.store(mission.id, MemoryTier.WORKING, 'results', results);
   423→
   424→      this.logStructured('execute', mission.id, {
   425→        resultCount: results.length,
   426→        successCount: results.filter((r) => r.success).length,
   427→        failCount: results.filter((r) => !r.success).length,
   428→        agentDispatchCount: executeDispatchCount,
   429→        durationMs: Date.now() - execStart,
   430→      });
   431→
   432→      // 4. Critique
   433→      this.setState(mission.id, 'critiquing');
   434→      const critiqueStart = Date.now();
   435→      let llmUsedCritique = false;
   436→      let critiques: CritiqueResult[];
   437→      try {
   438→        critiques = await this.withTimeout(
   439→          this.critique(results),
   440→          this.pipelineTimeouts.critique,
   441→          'critique',
   442→        );
   443→        llmUsedCritique = this.llmService?.isAnyAvailable() ?? false;
   444→      } catch (error: any) {
   445→        stepMetrics.push({
   446→          stepName: 'critique',
   447→          durationMs: Date.now() - critiqueStart,
   448→          llmUsed: false,
   449→          agentDispatchCount: 0,
   450→          success: false,
   451→          errorMessage: error.message,
   452→        });
   453→        throw error;
   454→      }
   455→      stepMetrics.push({
   456→        stepName: 'critique',
   457→        durationMs: Date.now() - critiqueStart,
   458→        llmUsed: llmUsedCritique,
   459→        agentDispatchCount: 0,
   460→        success: true,
   461→      });
   462→      await this.memory.store(mission.id, MemoryTier.WORKING, 'critiques', critiques);
   463→
   464→      this.logStructured('critique', mission.id, {
   465→        passedCount: critiques.filter((c) => c.passed).length,
   466→        failedCount: critiques.filter((c) => !c.passed).length,
   467→        durationMs: Date.now() - critiqueStart,
   468→        llmUsed: llmUsedCritique,
   469→      });
   470→
   471→      // 5. Repair (only for failed/critiqued subtasks)
   472→      const needsRepair = critiques.filter((c) => !c.passed);
   473→      let repairResults: RepairResult[] = [];
   474→      if (needsRepair.length > 0) {
   475→        this.setState(mission.id, 'repairing');
   476→        const repairStart = Date.now();
   477→        const dispatchCountBeforeRepair = this.currentStepDispatchCount;
   478→        let llmUsedRepair = false;
   479→        try {
   480→          repairResults = await this.withTimeout(
   481→            this.repair({ critiques: needsRepair, results }),
   482→            this.pipelineTimeouts.repair,
   483→            'repair',
   484→          );
   485→          llmUsedRepair = this.llmService?.isAnyAvailable() ?? false;
   486→        } catch (error: any) {
   487→          stepMetrics.push({
   488→            stepName: 'repair',
   489→            durationMs: Date.now() - repairStart,
   490→            llmUsed: false,
   491→            agentDispatchCount: this.currentStepDispatchCount - dispatchCountBeforeRepair,
   492→            success: false,
   493→            errorMessage: error.message,
   494→          });
   495→          throw error;
   496→        }
   497→        const repairDispatchCount = this.currentStepDispatchCount - dispatchCountBeforeRepair;
   498→        stepMetrics.push({
   499→          stepName: 'repair',
   500→          durationMs: Date.now() - repairStart,
   501→          llmUsed: llmUsedRepair,
   502→          agentDispatchCount: repairDispatchCount,
   503→          success: true,
   504→        });
   505→        await this.memory.store(mission.id, MemoryTier.WORKING, 'repairs', repairResults);
   506→
   507→        this.logStructured('repair', mission.id, {
   508→          repairCount: needsRepair.length,
   509→          repairedCount: repairResults.filter((r) => r.repaired).length,
   510→          agentDispatchCount: repairDispatchCount,
   511→          durationMs: Date.now() - repairStart,
   512→          llmUsed: llmUsedRepair,
   513→        });
   514→      }
   515→
   516→      // 6. Validate
   517→      this.setState(mission.id, 'validating');
   518→      const validateStart = Date.now();
   519→      let validations: ValidationResult[];
   520→      try {
   521→        validations = await this.withTimeout(
   522→          this.validate(
   523→            repairResults.length > 0 ? repairResults : (results as any[]),
   524→            mission,
   525→          ),
   526→          this.pipelineTimeouts.validate,
   527→          'validate',
   528→        );
   529→      } catch (error: any) {
   530→        stepMetrics.push({
   531→          stepName: 'validate',
   532→          durationMs: Date.now() - validateStart,
   533→          llmUsed: false,
   534→          agentDispatchCount: 0,
   535→          success: false,
   536→          errorMessage: error.message,
   537→        });
   538→        throw error;
   539→      }
   540→      stepMetrics.push({
   541→        stepName: 'validate',
   542→        durationMs: Date.now() - validateStart,
   543→        llmUsed: false,
   544→        agentDispatchCount: 0,
   545→        success: true,
   546→      });
   547→      await this.memory.store(mission.id, MemoryTier.WORKING, 'validations', validations);
   548→
   549→      this.logStructured('validate', mission.id, {
   550→        validCount: validations.filter((v) => v.valid).length,
   551→        invalidCount: validations.filter((v) => !v.valid).length,
   552→        avgScore: validations.length > 0
   553→          ? validations.reduce((sum, v) => sum + v.score, 0) / validations.length
   554→          : 0,
   555→        durationMs: Date.now() - validateStart,
   556→      });
   557→
   558→      // 7. Deliver
   559→      this.setState(mission.id, 'delivering');
   560→      const deliverStart = Date.now();
   561→      let delivery: DeliveryPackage;
   562→      try {
   563→        delivery = await this.withTimeout(
   564→          Promise.resolve(this.deliver(validations)),
   565→          this.pipelineTimeouts.deliver,
   566→          'deliver',
   567→        );
   568→      } catch (error: any) {
   569→        stepMetrics.push({
   570→          stepName: 'deliver',
   571→          durationMs: Date.now() - deliverStart,
   572→          llmUsed: false,
   573→          agentDispatchCount: 0,
   574→          success: false,
   575→          errorMessage: error.message,
   576→        });
   577→        throw error;
   578→      }
   579→      stepMetrics.push({
   580→        stepName: 'deliver',
   581→        durationMs: Date.now() - deliverStart,
   582→        llmUsed: false,
   583→        agentDispatchCount: 0,
   584→        success: true,
   585→      });
   586→
   587→      delivery.missionId = mission.id;
   588→      delivery.results = results;
   589→      delivery.critiques = critiques;
   590→      delivery.validations = validations;
   591→      delivery.totalDuration = Date.now() - pipelineStartTime;
   592→
   593→      this.setState(mission.id, 'completed');
   594→      this.eventBus.emit(AgentEventType.AGENT_COMPLETED, mission.id, {
   595→        status: delivery.status,
   596→        duration: delivery.totalDuration,
   597→      });
   598→
   599→      // Persist to long-term memory
   600→      await this.memory.store(mission.id, MemoryTier.LONG_TERM, 'delivery', delivery);
   601→
   602→      // Record pipeline metrics
   603→      const totalExecutions = results.length;
   604→      const repairableCount = needsRepair.length;
   605→      const metrics = this.buildMetrics(mission, pipelineStartTime, stepMetrics, delivery);
   606→      this.recordMetrics(metrics);
   607→
   608→      // Record execution results to health service
   609→      if (this.healthService) {
   610→        for (const result of results) {
   611→          if (result.agentKey) {
   612→            this.healthService.recordExecution(
   613→              result.agentKey,
   614→              result.duration || 0,
   615→              result.success,
   616→              result.error,
   617→            );
   618→          }
   619→        }
   620→      }
   621→
   622→      this.logStructured('pipeline_complete', mission.id, {
   623→        status: delivery.status,
   624→        totalDurationMs: delivery.totalDuration,
   625→        totalAgentDispatches: metrics.totalAgentDispatches,
   626→        repairRate: metrics.repairRate,
   627→      });
   628→
   629→      return delivery;
   630→    } catch (error: any) {
   631→      this.setState(mission.id, 'failed');
   632→      this.eventBus.emit(AgentEventType.AGENT_FAILED, mission.id, {
   633→        error: error.message,
   634→      });
   635→
   636→      // Record failed pipeline metrics
   637→      const metrics = this.buildMetrics(mission, pipelineStartTime, stepMetrics, null);
   638→      this.recordMetrics(metrics);
   639→
   640→      this.logStructured('pipeline_failed', mission.id, {
   641→        error: error.message,
   642→        totalDurationMs: Date.now() - pipelineStartTime,
   643→      });
   644→
   645→      return {
   646→        missionId: mission.id,
   647→        status: 'failed',
   648→        results: [],
   649→        critiques: [],
   650→        validations: [],
   651→        totalDuration: Date.now() - pipelineStartTime,
   652→        summary: `Mission failed: ${error.message}`,
   653→      };
   654→    }
   655→  }
   656→
   657→  // ─── Timeout Protection ─────────────────────────────────────
   658→
   659→  /**
   660→   * Wrap a promise with a timeout. If the promise does not resolve
   661→   * within the specified duration, it is rejected with a timeout error.
   662→   */
   663→  private withTimeout<T>(promise: Promise<T>, ms: number, stepName: string): Promise<T> {
   664→    return Promise.race([
   665→      promise,
   666→      new Promise<never>((_, reject) =>
   667→        setTimeout(
   668→          () => reject(new Error(`Pipeline step "${stepName}" timed out after ${ms}ms`)),
   669→          ms,
   670→        ),
   671→      ),
   672→    ]);
   673→  }

   677→  /**
   678→   * Step 1: Decompose a mission into subtasks.
   679→   * Uses LLM when available for intelligent decomposition,
   680→   * falls back to simple heuristic when LLM is unavailable.
   681→   */
   682→  @Trace('pipeline.decompose')
   683→  async decompose(mission: Mission): Promise<Subtask[]> {
   684→    this.logger.debug(`Decomposing mission: ${mission.id}`);
   685→
   686→    // Try LLM-powered decomposition first
   687→    if (this.llmService?.isAnyAvailable()) {
   688→      try {
   689→        return await this.decomposeWithLLM(mission);

   692→          `LLM decomposition failed, falling back to heuristic: ${error.message}`,
   693→        );
   694→      }
   695→    }
   696→
   697→    // Fallback: simple heuristic decomposition
   698→    return this.decomposeHeuristic(mission);
   699→  }
   700→
   701→  /**
   702→   * Step 2: Create an execution plan from subtasks.
   703→   * Uses LLM when available for intelligent planning,
   704→   * falls back to simple topological ordering.
   705→   */
   706→  @Trace('pipeline.plan')
   707→  async plan(subtasks: Subtask[]): Promise<ExecutionPlan> {
   708→    this.logger.debug(`Planning execution for ${subtasks.length} subtasks`);
   709→
   710→    // Try LLM-powered planning first
   711→    if (this.llmService?.isAnyAvailable()) {
   712→      try {
   713→        return await this.planWithLLM(subtasks);
   714→      } catch (error: any) {
   715→        this.logger.warn(
   716→          `LLM planning failed, falling back to heuristic: ${error.message}`,
   717→        );
   718→      }
   719→    }
   720→
   721→    // Fallback: simple topological ordering
   722→    return this.planHeuristic(subtasks);
   723→  }
   724→
   725→  /**
   726→   * Step 3: Execute the plan by dispatching subtasks to appropriate agents.
   727→   * Uses findBestAgent for intelligent agent selection.
   728→   */
   729→  @Trace('pipeline.execute')
   730→  async execute(plan: ExecutionPlan): Promise<ExecutionResult[]> {
   731→    this.logger.debug(
   732→      `Executing plan with ${plan.executionOrder.length} wave(s)`,
   733→    );
   734→
   735→    const results: ExecutionResult[] = [];
   736→
   737→    for (const wave of plan.executionOrder) {
   738→      // Execute each wave in parallel
   739→      const waveResults = await Promise.all(
   740→        wave.map(async (subtaskId) => {
   741→          const subtask = plan.subtasks.find((s) => s.id === subtaskId);
   742→          if (!subtask) {
   743→            return {
   744→              subtaskId,
   745→              success: false,
   746→              error: `Subtask not found: ${subtaskId}`,
   747→            };
   748→          }
   749→
   750→          return this.executeSubtask(subtask);
   751→        }),
   752→      );
   753→
   754→      results.push(...waveResults);
   755→    }
   756→
   757→    return results;
   758→  }

   813→  /**
   814→   * Step 6: Validate repaired results against quality criteria.
   815→   *
   816→   * Enhanced validation checks:
   817→   *   1. Success/failure check
   818→   *   2. Data presence check
   819→   *   3. Schema validation (if expectedSchema provided in mission)
   820→   *   4. Quality threshold (minimum score from critique/validation)
   821→   *   5. Completeness check (all objectives addressed, if provided)
   822→   *   6. No remaining issues (for RepairResult)
   823→   *   7. Constraint compliance (all constraints satisfied, if provided)
   824→   *   8. Safety check (no harmful outputs detected)
   825→   */
   826→  @Trace('pipeline.validate')
   827→  async validate(
   828→    items: Array<RepairResult | ExecutionResult>,
   829→    mission?: Mission,
   830→  ): Promise<ValidationResult[]> {
   831→    this.logger.debug(`Validating ${items.length} result(s)`);
   832→
   833→    return items.map((item) => {
   834→      const checks: Array<{ name: string; passed: boolean; detail?: string }> =
   835→        [];
   836→
   837→      // Check 1: Success/failure
   838→      const isSuccess = 'success' in item ? item.success : 'repaired' in item ? item.repaired : true;
   839→      checks.push({
   840→        name: 'success',
   841→        passed: isSuccess,
   842→        detail: isSuccess ? undefined : 'Execution did not succeed',
   843→      });
   844→
   845→      // Check 2: Data presence
   846→      const hasData = !!(item as any).data;
   847→      checks.push({
   848→        name: 'has_data',
   849→        passed: hasData,
   850→        detail: hasData ? undefined : 'No data returned',
   851→      });
   852→
   853→      // Check 3: Schema validation
   854→      if (mission?.expectedSchema && hasData) {
   855→        const schemaValid = this.validateSchema(
   856→          (item as any).data,
   857→          mission.expectedSchema,
   858→        );
   859→        checks.push({
   860→          name: 'schema_validation',
   861→          passed: schemaValid,
   862→          detail: schemaValid ? undefined : 'Result data does not match expected schema',
   863→        });
   864→      }
   865→
   866→      // Check 4: Quality threshold
   867→      if (mission?.qualityThreshold !== undefined) {
   868→        // Use the success status and data quality as proxy for quality score
   869→        const qualityScore = isSuccess ? (hasData ? 1.0 : 0.6) : 0.1;
   870→        const meetsThreshold = qualityScore >= mission.qualityThreshold;
   871→        checks.push({
   872→          name: 'quality_threshold',
   873→          passed: meetsThreshold,
   874→          detail: meetsThreshold
   875→            ? undefined
   876→            : `Quality score ${qualityScore.toFixed(2)} below threshold ${mission.qualityThreshold}`,
   877→        });
   878→      }
   879→
   880→      // Check 5: Completeness — all objectives addressed
   881→      if (mission?.objectives && mission.objectives.length > 0 && hasData) {
   882→        const dataStr = JSON.stringify((item as any).data).toLowerCase();
   883→        const addressedObjectives = mission.objectives.filter((obj) =>
   884→          dataStr.includes(obj.toLowerCase()),
   885→        );
   886→        const allAddressed = addressedObjectives.length === mission.objectives.length;
   887→        checks.push({
   888→          name: 'completeness',
   889→          passed: allAddressed,
   890→          detail: allAddressed
   891→            ? undefined
   892→            : `${mission.objectives.length - addressedObjectives.length} objective(s) not addressed: ` +
   893→              mission.objectives.filter((obj) => !addressedObjectives.includes(obj)).join(', '),
   894→        });
   895→      }
   896→
   897→      // Check 6: No remaining issues (for RepairResult)
   898→      if ('remainingIssues' in item) {
   899→        const noRemaining = item.remainingIssues.length === 0;
   900→        checks.push({
   901→          name: 'no_remaining_issues',
   902→          passed: noRemaining,
   903→          detail: noRemaining
   904→            ? undefined
   905→            : `${item.remainingIssues.length} issue(s) remain`,
   906→        });
   907→      }
   908→
   909→      // Check 7: Constraint compliance
   910→      if (mission?.constraints) {
   911→        const constraintIssues = this.checkConstraints(
   912→          (item as any).data,
   913→          mission.constraints,
   914→        );
   915→        checks.push({
   916→          name: 'constraint_compliance',
   917→          passed: constraintIssues.length === 0,
   918→          detail: constraintIssues.length > 0
   919→            ? `Constraint violations: ${constraintIssues.join('; ')}`
   920→            : undefined,
   921→        });
   922→      }
   923→
   924→      // Check 8: Safety check
   925→      if (hasData) {
   926→        const safetyResult = this.safetyCheck((item as any).data);
   927→        checks.push({
   928→          name: 'safety_check',
   929→          passed: safetyResult.safe,
   930→          detail: safetyResult.safe
   931→            ? undefined
   932→            : `Safety concerns detected: ${safetyResult.flags.join(', ')}`,
   933→        });
   934→      }
   935→
   936→      const score =
   937→        checks.filter((c) => c.passed).length / checks.length;
   938→      const valid = score >= 0.6; // at least 60% of checks must pass (raised from 50%)
   939→
   940→      return {
   941→        subtaskId: (item as any).subtaskId || 'unknown',
   942→        valid,
   943→        score,
   944→        checks,
   945→      };
   946→    });
   947→  }

   966→        ? `${validations.filter((v) => v.valid).length}/${validations.length} subtask(s) validated`
   967→        : 'No subtasks passed validation';
   968→
   969→    return {
   970→      missionId: '',
   971→      status,
   972→      results: [],
   973→      critiques: [],

   975→      totalDuration: 0,

   977→    };

   998→  // ─── Validation Helpers ─────────────────────────────────────
   999→
  1000→  /**
  1001→   * Basic schema validation: checks that all required top-level keys
  1002→   * exist in the data object. This is a lightweight check — full JSON
  1003→   * Schema validation would require a library like ajv.
  1004→   */
  1005→  private validateSchema(
  1006→    data: any,
  1007→    schema: Record<string, any>,
  1008→  ): boolean {
  1009→    if (!data || typeof data !== 'object') return false;
  1010→
  1011→    try {
  1012→      const required = schema.required || [];
  1013→      if (Array.isArray(required)) {
  1014→        for (const key of required) {
  1015→          if (!(key in data)) return false;
  1016→        }
  1017→      }
  1018→
  1019→      // Check type hints if provided in schema.properties
  1020→      const properties = schema.properties || {};
  1021→      for (const [key, propSchema] of Object.entries(properties)) {
  1022→        const prop = propSchema as any;
  1023→        if (key in data && prop.type) {
  1024→          const actualType = typeof data[key];
  1025→          const expectedType = prop.type;
  1026→          if (expectedType === 'string' && actualType !== 'string') return false;
  1027→          if (expectedType === 'number' && actualType !== 'number') return false;
  1028→          if (expectedType === 'boolean' && actualType !== 'boolean') return false;
  1029→          if (expectedType === 'object' && (actualType !== 'object' || Array.isArray(data[key]))) return false;
  1030→          if (expectedType === 'array' && !Array.isArray(data[key])) return false;
  1031→        }
  1032→      }
  1033→
  1034→      return true;
  1035→    } catch {
  1036→      return false;
  1037→    }
  1038→  }

  1040→  /**
  1041→   * Check constraint compliance against mission constraints.
  1042→   * Returns an array of violation descriptions.
  1043→   */
  1044→  private checkConstraints(
  1045→    data: any,
  1046→    constraints: Record<string, any>,
  1047→  ): string[] {
  1048→    const violations: string[] = [];
  1049→
  1050→    if (!data || typeof data !== 'object') return violations;
  1051→
  1052→    try {
  1053→      // Check max length constraints
  1054→      if (constraints.maxLength && typeof data === 'string') {
  1055→        if (data.length > constraints.maxLength) {
  1056→          violations.push(`Data exceeds max length of ${constraints.maxLength}`);
  1057→        }
  1058→      }
  1059→
  1060→      // Check allowed values
  1061→      if (constraints.allowedValues && Array.isArray(constraints.allowedValues)) {
  1062→        const dataStr = JSON.stringify(data);
  1063→        const hasAllowed = constraints.allowedValues.some((v: any) =>
  1064→          dataStr.includes(String(v)),
  1065→        );
  1066→        if (!hasAllowed) {
  1067→          violations.push('Data does not contain any of the allowed values');
  1068→        }
  1069→      }
  1070→
  1071→      // Check forbidden patterns
  1072→      if (constraints.forbiddenPatterns && Array.isArray(constraints.forbiddenPatterns)) {
  1073→        const dataStr = JSON.stringify(data);
  1074→        for (const pattern of constraints.forbiddenPatterns) {
  1075→          if (dataStr.includes(pattern)) {
  1076→            violations.push(`Data contains forbidden pattern: ${pattern}`);
  1077→          }
  1078→        }
  1079→      }
  1080→
  1081→      // Check required fields in data
  1082→      if (constraints.requiredFields && Array.isArray(constraints.requiredFields)) {
  1083→        for (const field of constraints.requiredFields) {
  1084→          if (!(field in data)) {
  1085→            violations.push(`Missing required field: ${field}`);
  1086→          }
  1087→        }
  1088→      }
  1089→
  1090→      // Check numeric range constraints
  1091→      if (constraints.numericRange) {
  1092→        const range = constraints.numericRange;
  1093→        for (const [field, bounds] of Object.entries(range)) {
  1094→          const b = bounds as { min?: number; max?: number };
  1095→          if (field in data && typeof data[field] === 'number') {
  1096→            if (b.min !== undefined && data[field] < b.min) {
  1097→              violations.push(`Field ${field} is below minimum ${b.min}`);
  1098→            }
  1099→            if (b.max !== undefined && data[field] > b.max) {
  1100→              violations.push(`Field ${field} exceeds maximum ${b.max}`);
  1101→            }
  1102→          }
  1103→        }
  1104→      }
  1105→    } catch {
  1106→      violations.push('Constraint validation encountered an error');
  1107→    }
  1108→
  1109→    return violations;
  1110→  }

  1112→  /**
  1113→   * Safety check: scan data for potentially harmful patterns.
  1114→   */
  1115→  private safetyCheck(data: any): { safe: boolean; flags: string[] } {
  1116→    const flags: string[] = [];
  1117→
  1118→    try {
  1119→      const dataStr = typeof data === 'string' ? data : JSON.stringify(data);
  1120→
  1121→      for (const { pattern, label } of SAFETY_PATTERNS) {
  1122→        if (pattern.test(dataStr)) {
  1123→          flags.push(label);
  1124→        }
  1125→      }
  1126→    } catch {
  1127→      // If we can't serialize, assume safe
  1128→    }
  1129→
  1130→    return { safe: flags.length === 0, flags };
  1131→  }

  1135→  /**
  1136→   * Build pipeline metrics from the collected step metrics.
  1137→   */
  1138→  private buildMetrics(
  1139→    mission: Mission,
  1140→    pipelineStartTime: number,
  1141→    stepMetrics: StepMetrics[],
  1142→    delivery: DeliveryPackage | null,
  1143→  ): PipelineMetrics {
  1144→    const totalAgentDispatches = stepMetrics.reduce(
  1145→      (sum, sm) => sum + sm.agentDispatchCount,
  1146→      0,
  1147→    );
  1148→
  1149→    // Calculate repair rate
  1150→    const executeStep = stepMetrics.find((s) => s.stepName === 'execute');
  1151→    const repairStep = stepMetrics.find((s) => s.stepName === 'repair');
  1152→    const totalExecutions = delivery?.results?.length || 0;
  1153→    const repairCount = repairStep ? delivery?.critiques?.filter((c) => !c.passed).length || 0 : 0;
  1154→    const repairRate = totalExecutions > 0 ? repairCount / totalExecutions : 0;
  1155→
  1156→    // Calculate success rate
  1157→    let successRate = 0;
  1158→    if (delivery) {
  1159→      if (delivery.status === 'success') successRate = 1;
  1160→      else if (delivery.status === 'partial') successRate = 0.5;
  1161→      else successRate = 0;
  1162→    }
  1163→
  1164→    return {
  1165→      missionId: mission.id,
  1166→      missionDescription: mission.description,
  1167→      missionPriority: mission.priority || 'medium',
  1168→      totalDurationMs: Date.now() - pipelineStartTime,
  1169→      stepMetrics,
  1170→      totalAgentDispatches,
  1171→      repairRate,
  1172→      successRate,
  1173→      timestamp: new Date().toISOString(),
  1174→    };
  1175→  }

  1177→  /**
  1178→   * Record pipeline metrics. Keeps the last MAX_METRICS_HISTORY entries.
  1179→   */
  1180→  private recordMetrics(metrics: PipelineMetrics): void {
  1181→    this.metricsHistory.push(metrics);
  1182→    if (this.metricsHistory.length > this.MAX_METRICS_HISTORY) {
  1183→      this.metricsHistory.shift();
  1184→    }
  1185→  }

  1187→  /**
  1188→   * Get aggregated pipeline metrics.
  1189→   */
  1190→  getMetrics(): {
  1191→    totalMissions: number;
  1192→    recentMetrics: PipelineMetrics[];
  1193→    aggregateRepairRate: number;
  1194→    aggregateSuccessRate: number;
  1195→    averageDurationMs: number;
  1196→  } {
  1197→    const total = this.metricsHistory.length;
  1198→    if (total === 0) {
  1199→      return {
  1200→        totalMissions: 0,
  1201→        recentMetrics: [],
  1202→        aggregateRepairRate: 0,
  1203→        aggregateSuccessRate: 0,
  1204→        averageDurationMs: 0,
  1205→      };
  1206→    }
  1207→
  1208→    const aggregateRepairRate =
  1209→      this.metricsHistory.reduce((sum, m) => sum + m.repairRate, 0) / total;
  1210→    const aggregateSuccessRate =
  1211→      this.metricsHistory.reduce((sum, m) => sum + m.successRate, 0) / total;
  1212→    const averageDurationMs =
  1213→      this.metricsHistory.reduce((sum, m) => sum + m.totalDurationMs, 0) / total;
  1214→
  1215→    return {
  1216→      totalMissions: total,
  1217→      recentMetrics: this.metricsHistory.slice(-20), // last 20 missions
  1218→      aggregateRepairRate,
  1219→      aggregateSuccessRate,
  1220→      averageDurationMs,
  1221→    };
  1222→  }
