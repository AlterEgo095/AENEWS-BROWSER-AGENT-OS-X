   262→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   263→    const startTime = Date.now();
   264→    const { action, ...params } = input.payload;
   265→
   266→    if (!action) {
   267→      return this.createAgentOutput(
   268→        input.taskId,
   269→        false,
   270→        null,
   271→        'Missing required parameter: action',
   272→        startTime,
   273→      );
   274→    }
   275→
   276→    const supportedActions = [
   277→      'generateFromSpec',
   278→      'generateFromTemplate',
   279→      'generateFromDescription',
   280→      'refactorCode',
   281→      'optimizeCode',
   282→    ];
   283→
   284→    if (!supportedActions.includes(action)) {
   285→      return this.createAgentOutput(
   286→        input.taskId,
   287→        false,
   288→        null,
   289→        `Unknown code generation action: ${action}. Supported: ${supportedActions.join(', ')}`,
   290→        startTime,
   291→      );
   292→    }
   293→
   294→    try {
   295→      const tool = this.getTool(action);
   296→      if (!tool) {
   297→        return this.createAgentOutput(
   298→          input.taskId,
   299→          false,
   300→          null,
   301→          `Tool not found: ${action}`,
   302→          startTime,
   303→        );
   304→      }
   305→
   306→      const result = await tool.execute(params);
   307→
   308→      this.generationHistory.push({
   309→        action,
   310→        language: params.language || 'unknown',
   311→        timestamp: new Date(),
   312→        success: true,
   313→      });
   314→
   315→      // Store generation result in session memory for context continuity
   316→      await this.storeInWorkingMemory(
   317→        `codegen:last:${action}`,
   318→        { params, result, timestamp: new Date() },
   319→        300000,
   320→      );
   321→
   322→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   323→    } catch (error) {
   324→      const msg = (error as Error).message;
   325→      this.logger.error(`CodeGeneration execution failed for ${action}: ${msg}`);
   326→
   327→      this.generationHistory.push({
   328→        action,
   329→        language: params.language || 'unknown',
   330→        timestamp: new Date(),
   331→        success: false,
   332→      });
   333→
   334→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   335→    }
   336→  }

   346→  private async generateFromSpec(params: {
   347→    spec: string;
   348→    language: string;
   349→    framework?: string;
   350→    style?: string;
   351→  }): Promise<{
   352→    code: string;
   353→    language: string;
   354→    files: GeneratedFile[];
   355→    warnings: string[];
   356→  }> {
   357→    const { spec, language, framework, style } = params;
   358→
   359→    if (!spec || typeof spec !== 'string') {
   360→      throw new Error('A valid specification string is required');
   361→    }
   362→    if (!language || typeof language !== 'string') {
   363→      throw new Error('A valid programming language is required');
   364→    }
   365→
   366→    const warnings: string[] = [];
   367→
   368→    // Parse specification into structured sections
   369→    const specSections = this.parseSpec(spec);
   370→    const normalizedLang = this.normalizeLanguage(language);
   371→
   372→    // Validate framework compatibility
   373→    if (framework) {
   374→      const isCompatible = this.validateFrameworkCompatibility(normalizedLang, framework);
   375→      if (!isCompatible) {
   376→        warnings.push(`Framework "${framework}" may not be fully compatible with ${normalizedLang}`);
   377→      }
   378→    }
   379→
   380→    // Generate code from each spec section
   381→    const files: GeneratedFile[] = [];
   382→    for (const section of specSections) {
   383→      const generatedCode = this.synthesizeCode(section, normalizedLang, framework, style);
   384→      const fileName = this.inferFileName(section.name, normalizedLang, framework);
   385→
   386→      files.push({
   387→        name: fileName,
   388→        path: `./src/${section.name}/${fileName}`,
   389→        content: generatedCode,
   390→        language: normalizedLang,
   391→      });
   392→    }
   393→
   394→    // Produce a main combined code output
   395→    const code = files.map((f) => f.content).join('\n\n');
   396→
   397→    this.logger.log(
   398→      `Generated code from spec: ${files.length} file(s), language=${normalizedLang}, framework=${framework || 'none'}`,
   399→    );
   400→
   401→    return { code, language: normalizedLang, files, warnings };
   402→  }

   404→  private async generateFromTemplate(params: {
   405→    templateId: string;
   406→    variables: Record<string, string>;
   407→    outputFileName?: string;
   408→  }): Promise<{
   409→    code: string;
   410→    fileName: string;
   411→    templateId: string;
   412→    substitutedVars: string[];
   413→  }> {
   414→    const { templateId, variables, outputFileName } = params;
   415→
   416→    if (!templateId || typeof templateId !== 'string') {
   417→      throw new Error('A valid templateId is required');
   418→    }
   419→    if (!variables || typeof variables !== 'object') {
   420→      throw new Error('Variables must be a key-value object');
   421→    }
   422→
   423→    const template = this.templates.get(templateId);
   424→    if (!template) {
   425→      const available = Array.from(this.templates.keys()).join(', ');
   426→      throw new Error(`Template not found: ${templateId}. Available: ${available}`);
   427→    }
   428→
   429→    // Perform variable substitution
   430→    let code = template.content;
   431→    const substitutedVars: string[] = [];
   432→
   433→    for (const [key, value] of Object.entries(variables)) {
   434→      const placeholder = `{{${key}}}`;
   435→      if (code.includes(placeholder)) {
   436→        code = code.replaceAll(placeholder, value);
   437→        substitutedVars.push(key);
   438→      }
   439→    }
   440→
   441→    // Check for unsubstituted variables
   442→    const remainingPlaceholders = code.match(/\{\{(\w+)\}\}/g);
   443→    if (remainingPlaceholders) {
   444→      const missingVars = remainingPlaceholders.map((p) => p.replace(/[{}]/g, ''));
   445→      throw new Error(
   446→        `Unsubstituted template variables: ${missingVars.join(', ')}. Provide values for all placeholders.`,
   447→      );
   448→    }
   449→
   450→    const fileName = outputFileName || `${templateId}.generated.${this.getExtension(template.language)}`;
   451→
   452→    this.logger.log(
   453→      `Generated code from template: ${templateId}, substituted ${substitutedVars.length} variable(s)`,
   454→    );
   455→
   456→    return { code, fileName, templateId, substitutedVars };
   457→  }

   500→  private async refactorCode(params: {
   501→    code: string;
   502→    language: string;
   503→    goals?: string[];
   504→    preserveBehavior?: boolean;
   505→  }): Promise<{
   506→    refactoredCode: string;
   507→    changes: RefactoringChange[];
   508→    improvedMetrics: Record<string, number>;
   509→  }> {
   510→    const { code, language, goals = [], preserveBehavior = true } = params;
   511→
   512→    if (!code || typeof code !== 'string') {
   513→      throw new Error('Source code is required for refactoring');
   514→    }
   515→    if (!language || typeof language !== 'string') {
   516→      throw new Error('Programming language is required');
   517→    }
   518→
   519→    const normalizedLang = this.normalizeLanguage(language);
   520→    const changes: RefactoringChange[] = [];
   521→
   522→    let refactoredCode = code;
   523→    const originalLines = code.split('\n');
   524→
   525→    // Apply refactoring patterns
   526→    const patterns = this.getRefactoringPatterns(goals, normalizedLang);
   527→
   528→    for (const pattern of patterns) {
   529→      const result = this.applyRefactoringPattern(refactoredCode, pattern, preserveBehavior);
   530→      if (result.modified) {
   531→        changes.push(result.change);
   532→        refactoredCode = result.code;
   533→      }
   534→    }
   535→
   536→    // Calculate improved metrics
   537→    const refactoredLines = refactoredCode.split('\n');
   538→    const improvedMetrics: Record<string, number> = {
   539→      linesReduced: originalLines.length - refactoredLines.length,
   540→      complexityReduction: this.estimateComplexityReduction(originalLines, refactoredLines),
   541→      duplicateCodeReduction: this.estimateDuplicateReduction(code, refactoredCode),
   542→    };
   543→
   544→    this.logger.log(
   545→      `Refactored code: ${changes.length} change(s), language=${normalizedLang}`,
   546→    );
   547→
   548→    return { refactoredCode, changes, improvedMetrics };
   549→  }

   551→  private async optimizeCode(params: {
   552→    code: string;
   553→    language: string;
   554→    target: string;
   555→    level?: string;
   556→  }): Promise<{
   557→    optimizedCode: string;
   558→    improvements: Array<{ type: string; description: string; impact: string }>;
   559→    estimatedGain: string;
   560→    tradeoffs: string[];
   561→  }> {
   562→    const { code, language, target, level = 'safe' } = params;
   563→
   564→    if (!code || typeof code !== 'string') {
   565→      throw new Error('Source code is required for optimization');
   566→    }
   567→    if (!language || typeof language !== 'string') {
   568→      throw new Error('Programming language is required');
   569→    }
   570→    const validTargets = ['performance', 'memory', 'readability', 'bundle-size'];
   571→    if (!validTargets.includes(target)) {
   572→      throw new Error(`Invalid optimization target: ${target}. Valid: ${validTargets.join(', ')}`);
   573→    }
   574→    const validLevels = ['safe', 'moderate', 'aggressive'];
   575→    if (!validLevels.includes(level)) {
   576→      throw new Error(`Invalid optimization level: ${level}. Valid: ${validLevels.join(', ')}`);
   577→    }
   578→
   579→    const normalizedLang = this.normalizeLanguage(language);
   580→    const improvements: Array<{ type: string; description: string; impact: string }> = [];
   581→    const tradeoffs: string[] = [];
   582→
   583→    let optimizedCode = code;
   584→
   585→    // Apply optimization strategies based on target and level
   586→    const strategies = this.getOptimizationStrategies(target, level, normalizedLang);
   587→
   588→    for (const strategy of strategies) {
   589→      const result = this.applyOptimizationStrategy(optimizedCode, strategy, normalizedLang);
   590→      if (result.modified) {
   591→        optimizedCode = result.code;
   592→        improvements.push({
   593→          type: strategy.type,
   594→          description: strategy.description,
   595→          impact: result.impact,
   596→        });
   597→
   598→        if (result.tradeoff) {
   599→          tradeoffs.push(result.tradeoff);
   600→        }
   601→      }
   602→    }
   603→
   604→    // Estimate overall gain
   605→    const estimatedGain = this.estimateOptimizationGain(improvements, target, level);
   606→
   607→    this.logger.log(
   608→      `Optimized code: target=${target}, level=${level}, ${improvements.length} improvement(s)`,
   609→    );
   610→
   611→    return { optimizedCode, improvements, estimatedGain, tradeoffs };
   612→  }

   747→  private parseSpec(spec: string): Array<{ name: string; type: string; description: string; methods?: string[] }> {
   748→    const sections: Array<{ name: string; type: string; description: string; methods?: string[] }> = [];
   749→    const lines = spec.split('\n');
   750→    let currentSection: typeof sections[0] | null = null;
   751→
   752→    for (const line of lines) {
   753→      const trimmed = line.trim();
   754→      if (trimmed.startsWith('# ') || trimmed.startsWith('## ')) {
   755→        if (currentSection) {
   756→          sections.push(currentSection);
   757→        }
   758→        const name = trimmed.replace(/^#+\s*/, '');
   759→        currentSection = {
   760→          name: this.toPascalCase(name),
   761→          type: name.toLowerCase().includes('service') ? 'service' : name.toLowerCase().includes('controller') ? 'controller' : 'module',
   762→          description: name,
   763→        };
   764→      } else if (currentSection && trimmed.length > 0) {
   765→        if (!currentSection.methods) currentSection.methods = [];
   766→        if (trimmed.startsWith('- ') || trimmed.startsWith('* ')) {
   767→          currentSection.methods.push(trimmed.replace(/^[-*]\s*/, ''));
   768→        }
   769→      }
   770→    }
   771→
   772→    if (currentSection) sections.push(currentSection);
   773→
   774→    // Default section if spec has no headers
   775→    if (sections.length === 0) {
   776→      sections.push({
   777→        name: 'GeneratedModule',
   778→        type: 'module',
   779→        description: spec.substring(0, 100),
   780→      });
   781→    }
   782→
   783→    return sections;
   784→  }

  1172→  private applyOptimizationStrategy(
  1173→    code: string,
  1174→    strategy: { type: string; description: string; target: string },
  1175→    language: string,
  1176→  ): { modified: boolean; code: string; impact: string; tradeoff?: string } {
  1177→    let optimizedCode = code;
  1178→    let modified = false;
  1179→    let impact = 'low';
  1180→    let tradeoff: string | undefined;
  1181→
  1182→    switch (strategy.type) {
  1183→      case 'early-return': {
  1184→        // Add early return patterns in if-else chains
  1185→        const pattern = /if\s*\(([^)]+)\)\s*\{\s*return\s+([^;]+);\s*\}\s*else\s*\{/g;
  1186→        const newCode = code.replace(pattern, 'if ($1) {\n    return $2;\n  }\n  {');
  1187→        if (newCode !== code) {
  1188→          optimizedCode = newCode;
  1189→          modified = true;
  1190→          impact = 'medium';
  1191→        }
  1192→        break;
  1193→      }
  1194→      case 'caching': {
  1195→        // Add memoization comment
  1196→        if ((language === 'typescript' || language === 'javascript') && code.includes('function')) {
  1197→          const cacheComment = '// OPTIMIZE: Consider adding memoization to expensive functions\n';
  1198→          if (!code.includes('OPTIMIZE:')) {
  1199→            optimizedCode = cacheComment + code;
  1200→            modified = true;
  1201→            impact = 'medium';
  1202→            tradeoff = 'Memoization increases memory usage proportional to unique inputs';
  1203→          }
  1204→        }
  1205→        break;
  1206→      }
  1207→      case 'lazy-loading': {
  1208→        if ((language === 'typescript' || language === 'javascript') && code.includes('import ')) {
  1209→          optimizedCode = code.replace(
  1210→            /import\s+(\w+)\s+from\s+(['"]([^'"]+)['"])/g,
  1211→            '// OPTIMIZE: Consider dynamic import: const $1 = await import($2)\nimport $1 from $2',
  1212→          );
  1213→          modified = optimizedCode !== code;
  1214→          impact = 'high';
  1215→          tradeoff = 'Dynamic imports add async overhead on first load';
  1216→        }
  1217→        break;
  1218→      }
  1219→      default: {
  1220→        // Generic strategy: add optimization suggestions as comments
  1221→        const comment = `// OPTIMIZE [${strategy.type}]: ${strategy.description}\n`;
  1222→        if (!code.includes(`OPTIMIZE [${strategy.type}]`)) {
  1223→          optimizedCode = comment + code;
  1224→          modified = true;
  1225→          impact = 'low';
  1226→        }
  1227→        break;
  1228→      }
  1229→    }
  1230→
  1231→    return { modified, code: optimizedCode, impact, tradeoff };

  1234→  private estimateComplexityReduction(originalLines: string[], refactoredLines: string[]): number {
  1235→    // Simplified cyclomatic complexity estimation based on branch keywords
  1236→    const countBranches = (lines: string[]) =>
  1237→      lines.reduce((count, line) => {
  1238→        const branchKeywords = ['if ', 'else ', 'switch ', 'case ', 'for ', 'while ', 'catch '];
  1239→        return count + branchKeywords.filter((kw) => line.includes(kw)).length;
  1240→      }, 0);
  1241→
  1242→    const originalBranches = countBranches(originalLines);
  1243→    const refactoredBranches = countBranches(refactoredLines);
  1244→
  1245→    if (originalBranches === 0) return 0;
  1246→    return Math.round(((originalBranches - refactoredBranches) / originalBranches) * 100);
  1247→  }

  1249→  private estimateDuplicateReduction(originalCode: string, refactoredCode: string): number {
  1250→    // Simplified duplicate estimation by comparing line frequencies
  1251→    const countLineFrequencies = (code: string): number => {
  1252→      const lines = code.split('\n').map((l) => l.trim()).filter((l) => l.length > 0);
  1253→      const freq = new Map<string, number>();
  1254→      for (const line of lines) {
  1255→        freq.set(line, (freq.get(line) || 0) + 1);
  1256→      }
  1257→      let duplicates = 0;
  1258→      for (const count of freq.values()) {
  1259→        if (count > 1) duplicates += count - 1;
  1260→      }
  1261→      return duplicates;
  1262→    };
  1263→
  1264→    const originalDuplicates = countLineFrequencies(originalCode);
  1265→    const refactoredDuplicates = countLineFrequencies(refactoredCode);
  1266→
  1267→    if (originalDuplicates === 0) return 0;
  1268→    return Math.round(((originalDuplicates - refactoredDuplicates) / originalDuplicates) * 100);
  1269→  }

  1271→  private estimateOptimizationGain(
  1272→    improvements: Array<{ type: string; description: string; impact: string }>,
  1273→    target: string,
  1274→    level: string,
  1275→  ): string {
  1276→    if (improvements.length === 0) return 'No measurable improvement';
  1277→
  1278→    const highImpact = improvements.filter((i) => i.impact === 'high').length;
  1279→    const mediumImpact = improvements.filter((i) => i.impact === 'medium').length;
  1280→
  1281→    const baseGain = highImpact * 20 + mediumImpact * 10 + (improvements.length - highImpact - mediumImpact) * 5;
  1282→    const levelMultiplier = level === 'aggressive' ? 1.5 : level === 'moderate' ? 1.2 : 1.0;
  1283→
  1284→    const estimatedPercent = Math.min(90, Math.round(baseGain * levelMultiplier));
  1285→
  1286→    return `~${estimatedPercent}% estimated improvement in ${target}`;
  1287→  }
