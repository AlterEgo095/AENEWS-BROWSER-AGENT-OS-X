   107→function StatusBadge({ status }: { status: string }) {
   108→  const config: Record<string, { color: string; icon: React.ComponentType<{ className?: string }> }> = {
   109→    active: { color: 'bg-emerald-500/15 text-emerald-400', icon: CheckCircle2 },
   110→    running: { color: 'bg-emerald-500/15 text-emerald-400', icon: Activity },
   111→    converging: { color: 'bg-cyan-500/15 text-cyan-400', icon: Target },
   112→    completed: { color: 'bg-blue-500/15 text-blue-400', icon: CheckCircle2 },
   113→    failed: { color: 'bg-red-500/15 text-red-400', icon: XCircle },
   114→    voting: { color: 'bg-amber-500/15 text-amber-400', icon: Vote },
   115→    pending: { color: 'bg-slate-500/15 text-slate-400', icon: Clock },
   116→    timeout: { color: 'bg-orange-500/15 text-orange-400', icon: AlertTriangle },
   117→    paused: { color: 'bg-yellow-500/15 text-yellow-400', icon: Minus },
   118→    initializing: { color: 'bg-purple-500/15 text-purple-400', icon: Sparkles },
   119→    partial: { color: 'bg-amber-500/15 text-amber-400', icon: AlertTriangle },
   120→  };
   121→  const c = config[status] || config.pending;
   122→  const IconComp = c.icon;
   123→  return (
   124→    <span className={cn('inline-flex items-center gap-1 rounded-full px-2 py-0.5 text-[10px] font-semibold', c.color)}>
   125→      <IconComp className="h-3 w-3" />
   126→      {status}
   127→    </span>
   128→  );
   129→}

   147→function ConvergenceRing({ score }: { score: number }) {
   148→  const pct = Math.round(score * 100);
   149→  const color = score >= 0.8 ? 'text-emerald-400' : score >= 0.5 ? 'text-amber-400' : 'text-red-400';
   150→  return (
   151→    <div className="relative flex h-10 w-10 items-center justify-center">
   152→      <svg viewBox="0 0 36 36" className="h-10 w-10 -rotate-90">
   153→        <circle cx="18" cy="18" r="15.9" fill="none" stroke="currentColor" strokeWidth="3" className="text-white/10" />
   154→        <circle cx="18" cy="18" r="15.9" fill="none" stroke="currentColor" strokeWidth="3" strokeDasharray={`${pct} ${100 - pct}`} className={color} />
   155→      </svg>
   156→      <span className={cn('absolute text-[9px] font-bold', color)}>{pct}%</span>
   157→    </div>
   158→  );
   159→}
