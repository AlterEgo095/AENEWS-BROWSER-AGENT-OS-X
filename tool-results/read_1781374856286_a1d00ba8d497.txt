    71→  /** Small delay to avoid rate limiting */
    72→  private delay(ms: number): Promise<void> {
    73→    return new Promise(resolve => setTimeout(resolve, ms));
    74→  }

    76→  /** Delay between LLM calls (adaptive based on rate limit experience) */
    77→  private async rateLimitDelay(): Promise<void> {
    78→    // After each LLM call, wait a bit to avoid 429s
    79→    const delayMs = this.llmCallCount > 3 ? 2000 : 500;
    80→    await this.delay(delayMs);
    81→  }

    83→  /**
    84→   * Execute a complete mission from natural language instruction
    85→   */
    86→  async executeMission(instruction: string): Promise<RuntimeResult> {
    87→    const missionId = `mission-${uuidv4().slice(0, 8)}`;
    88→    const startTime = Date.now();
    89→    let totalCost = 0;
    90→    const phases: PhaseTiming[] = [];
    91→    const errors: string[] = [];
    92→    let retries = 0;
    93→
    94→    console.log(`\n${'═'.repeat(70)}`);
    95→    console.log(`  MISSION: ${missionId}`);
    96→    console.log(`  Instruction: "${instruction}"`);
    97→    console.log(`${'═'.repeat(70)}\n`);
    98→
    99→    // ─── Step 1: Create workspace ──────────────────────────
   100→    const workspaceDir = path.join(this.baseWorkspace, missionId);
   101→    fs.mkdirSync(workspaceDir, { recursive: true });
   102→    fs.mkdirSync(path.join(workspaceDir, 'src'), { recursive: true });
   103→    fs.mkdirSync(path.join(workspaceDir, 'tests'), { recursive: true });
   104→    fs.mkdirSync(path.join(workspaceDir, 'docs'), { recursive: true });
   105→
   106→    const artifacts: RuntimeArtifact[] = [];
   107→
   108→    // ─── Step 2: Analyze mission with LLM ──────────────────
   109→    let phaseStart = Date.now();
   110→    console.log('📋 Phase 1: Analyzing mission...');
   111→    let analysis: any;
   112→    try {
   113→      analysis = await this.analyzeMission(instruction);
   114→      totalCost += analysis.cost;
   115→      phases.push({ name: 'analyze', durationMs: Date.now() - phaseStart, success: true });
   116→      console.log(`   ✓ Plan: ${analysis.plan.phases?.length || 0} phases, ${analysis.plan.requiredCapabilities?.length || 0} capabilities`);
   117→    } catch (err: any) {
   118→      errors.push(`Analysis failed: ${err.message}`);
   119→      analysis = { plan: this.fallbackPlan(instruction), cost: 0 };
   120→      phases.push({ name: 'analyze', durationMs: Date.now() - phaseStart, success: false });
   121→      console.log(`   ✗ Analysis failed, using fallback plan`);
   122→    }
   123→
   124→    // ─── Step 3: Generate code with LLM ────────────────────
   125→    phaseStart = Date.now();
   126→    console.log('🔨 Phase 2: Building application...');
   127→    let buildResult: { artifacts: RuntimeArtifact[]; cost: number };
   128→    try {
   129→      buildResult = await this.executeBuild(instruction, analysis, workspaceDir);
   130→      totalCost += buildResult.cost;
   131→      artifacts.push(...buildResult.artifacts);
   132→      phases.push({ name: 'build', durationMs: Date.now() - phaseStart, success: true });
   133→      console.log(`   ✓ Generated ${buildResult.artifacts.length} files`);
   134→    } catch (err: any) {
   135→      errors.push(`Build failed: ${err.message}`);
   136→      phases.push({ name: 'build', durationMs: Date.now() - phaseStart, success: false });
   137→      console.log(`   ✗ Build failed: ${err.message}`);
   138→      // Try fallback with template code
   139→      const templateCode = this.generateTemplateCode(instruction, analysis.plan);
   140→      const files = this.parseGeneratedFiles(templateCode);
   141→      for (const [filePath, content] of files) {
   142→        this.writeFile(workspaceDir, filePath, content);
   143→        let type: RuntimeArtifact['type'] = 'source';
   144→        if (filePath.includes('test') || filePath.includes('spec')) type = 'test';
   145→        else if (filePath.endsWith('.md') || filePath.endsWith('.txt')) type = 'document';
   146→        else if (filePath.endsWith('.json') || filePath.endsWith('.yml') || filePath.endsWith('.yaml') || filePath.endsWith('Dockerfile') || filePath.includes('.config')) type = 'config';
   147→        artifacts.push({
   148→          name: path.basename(filePath),
   149→          type,
   150→          path: path.join(workspaceDir, filePath),
   151→          size: Buffer.byteLength(content),
   152→          content: content.substring(0, 200),
   153→        });
   154→      }
   155→      console.log(`   → Used template fallback: ${files.size} files`);
   156→      retries++;
   157→    }
   158→
   159→    // ─── Step 4: Execute tests ──────────────────────────────
   160→    await this.rateLimitDelay(); // Avoid rate limit
   161→    phaseStart = Date.now();
   162→    console.log('🧪 Phase 3: Testing...');
   163→    let testResult: { passed: boolean; results: any[]; cost: number };
   164→    try {
   165→      testResult = await this.executeTests(workspaceDir, analysis);
   166→      totalCost += testResult.cost;
   167→      phases.push({ name: 'test', durationMs: Date.now() - phaseStart, success: testResult.passed });
   168→      console.log(`   ${testResult.passed ? '✓' : '✗'} Tests: ${testResult.results.length} executed`);
   169→    } catch (err: any) {
   170→      testResult = { passed: false, results: [], cost: 0 };
   171→      phases.push({ name: 'test', durationMs: Date.now() - phaseStart, success: false });
   172→      console.log(`   ✗ Testing failed: ${err.message}`);
   173→    }
   174→
   175→    // ─── Step 5: Audit ─────────────────────────────────────
   176→    await this.rateLimitDelay(); // Avoid rate limit
   177→    phaseStart = Date.now();
   178→    console.log('🔍 Phase 4: Auditing...');
   179→    let auditResult: { passed: boolean; findings: string[]; cost: number };
   180→    try {
   181→      auditResult = await this.executeAudit(workspaceDir, artifacts);
   182→      totalCost += auditResult.cost;
   183→      phases.push({ name: 'audit', durationMs: Date.now() - phaseStart, success: auditResult.passed });
   184→      console.log(`   ${auditResult.passed ? '✓' : '✗'} Audit: ${auditResult.findings.length} findings`);
   185→      if (auditResult.findings.length > 0) {
   186→        auditResult.findings.forEach(f => console.log(`     - ${f}`));
   187→      }
   188→    } catch (err: any) {
   189→      auditResult = { passed: true, findings: [], cost: 0 };
   190→      phases.push({ name: 'audit', durationMs: Date.now() - phaseStart, success: false });
   191→    }
   192→
   193→    // ─── Step 6: Certification ──────────────────────────────
   194→    phaseStart = Date.now();
   195→    console.log('🎖️  Phase 5: Certification...');
   196→    const certResult = this.certify(artifacts, testResult, auditResult);
   197→    phases.push({ name: 'certify', durationMs: Date.now() - phaseStart, success: certResult.certified });
   198→    console.log(`   ${certResult.certified ? '✓' : '✗'} Score: ${certResult.qualityScore}/100`);
   199→
   200→    if (!certResult.certified && certResult.qualityScore >= 50) {
   201→      console.log(`   → Partial certification accepted (score ${certResult.qualityScore} >= 50)`);
   202→    }
   203→
   204→    // ─── Step 7: Generate README ────────────────────────────
   205→    await this.rateLimitDelay(); // Avoid rate limit
   206→    phaseStart = Date.now();
   207→    console.log('📖 Phase 6: Generating documentation...');
   208→    try {
   209→      const readme = await this.generateReadme(instruction, analysis, artifacts);
   210→      totalCost += readme.cost;
   211→      this.writeFile(workspaceDir, 'README.md', readme.content);
   212→      artifacts.push({
   213→        name: 'README.md',
   214→        type: 'document',
   215→        path: path.join(workspaceDir, 'README.md'),
   216→        size: Buffer.byteLength(readme.content),
   217→      });
   218→      phases.push({ name: 'readme', durationMs: Date.now() - phaseStart, success: true });
   219→      console.log(`   ✓ README.md generated (${Buffer.byteLength(readme.content)} bytes)`);
   220→    } catch (err: any) {
   221→      const fallbackReadme = `# ${instruction}\n\nGenerated by AENEWS Software Factory\n\n## Files\n\n${artifacts.map(a => `- ${a.name}`).join('\n')}\n`;
   222→      this.writeFile(workspaceDir, 'README.md', fallbackReadme);
   223→      artifacts.push({ name: 'README.md', type: 'document', path: path.join(workspaceDir, 'README.md'), size: Buffer.byteLength(fallbackReadme) });
   224→      phases.push({ name: 'readme', durationMs: Date.now() - phaseStart, success: false });
   225→    }
   226→
   227→    // ─── Step 8: Generate Dockerfile (ensure it exists) ────
   228→    const dockerfilePath = path.join(workspaceDir, 'Dockerfile');
   229→    if (!fs.existsSync(dockerfilePath)) {
   230→      const dockerfile = this.generateDockerfile(analysis.plan);
   231→      this.writeFile(workspaceDir, 'Dockerfile', dockerfile);
   232→      artifacts.push({ name: 'Dockerfile', type: 'config', path: dockerfilePath, size: Buffer.byteLength(dockerfile) });
   233→      console.log(`   ✓ Dockerfile generated`);
   234→    }
   235→
   236→    // ─── Step 9: Create ZIP ─────────────────────────────────
   237→    phaseStart = Date.now();
   238→    console.log('📦 Phase 7: Packaging delivery...');
   239→    const zipPath = await this.createZipArchive(missionId, workspaceDir);
   240→    if (zipPath) {
   241→      artifacts.push({
   242→        name: `${missionId}.zip`,
   243→        type: 'archive',
   244→        path: zipPath,
   245→        size: fs.statSync(zipPath).size,
   246→      });
   247→      phases.push({ name: 'delivery', durationMs: Date.now() - phaseStart, success: true });
   248→      console.log(`   ✓ ZIP: ${zipPath} (${fs.statSync(zipPath).size} bytes)`);
   249→    } else {
   250→      phases.push({ name: 'delivery', durationMs: Date.now() - phaseStart, success: false });
   251→      console.log(`   ✗ ZIP packaging failed`);
   252→    }
   253→
   254→    // ─── Step 10: Generate Report ───────────────────────────
   255→    const reportContent = this.generateReport(missionId, instruction, artifacts, certResult, testResult, auditResult, phases);
   256→    this.writeFile(workspaceDir, 'docs/REPORT.md', reportContent);
   257→    artifacts.push({ name: 'REPORT.md', type: 'report', path: path.join(workspaceDir, 'docs/REPORT.md'), size: Buffer.byteLength(reportContent) });
   258→
   259→    // ─── Final result ───────────────────────────────────────
   260→    const totalDuration = Date.now() - startTime;
   261→    const success = errors.length === 0 || artifacts.filter(a => a.type === 'source').length > 0;
   262→
   263→    console.log(`\n${'═'.repeat(70)}`);
   264→    console.log(`  RESULT: ${success ? '✅ SUCCESS' : '❌ FAILED'}`);
   265→    console.log(`  Artifacts: ${artifacts.length} files`);
   266→    console.log(`  Quality: ${certResult.qualityScore}/100 ${certResult.certified ? '(CERTIFIED)' : '(UNCERTIFIED)'}`);
   267→    console.log(`  Duration: ${(totalDuration / 1000).toFixed(1)}s`);
   268→    console.log(`  Cost: $${totalCost.toFixed(3)}`);
   269→    console.log(`  Retries: ${retries}`);
   270→    console.log(`  Workspace: ${workspaceDir}`);
   271→    console.log(`${'═'.repeat(70)}\n`);
   272→
   273→    const result: RuntimeResult = {
   274→      missionId,
   275→      success,
   276→      artifacts,
   277→      workspaceDir,
   278→      qualityScore: certResult.qualityScore,

   283→    };
   284→
   285→    // Track metrics
   286→    this.metrics.push({
   287→      missionId,
   288→      instruction,
   289→      success,
   290→      certified: certResult.certified,
   291→      qualityScore: certResult.qualityScore,
   292→      artifactCount: artifacts.length,
   293→      totalSizeBytes: artifacts.reduce((sum, a) => sum + a.size, 0),
   294→      durationMs: totalDuration,
   295→      costUsd: totalCost,
   296→      retries,
   297→      errors,
   298→      phases,
   299→    });
   300→
   301→    return result;
   302→  }

   304→  /**
   305→   * Get Mission Success Rate (MSR)
   306→   */
   307→  getMSR(): { rate: number; total: number; successes: number; certified: number } {
   308→    const total = this.metrics.length;
   309→    const successes = this.metrics.filter(m => m.success).length;
   310→    const certified = this.metrics.filter(m => m.certified).length;
   311→    return {
   312→      rate: total > 0 ? successes / total : 0,
   313→      total,
   314→      successes,
   315→      certified,
   316→    };
   317→  }

   319→  /**
   320→   * Print aggregate statistics
   321→   */
   322→  printStats(): void {
   323→    const msr = this.getMSR();
   324→    if (msr.total === 0) return;
   325→
   326→    const avgDuration = this.metrics.reduce((s, m) => s + m.durationMs, 0) / msr.total;
   327→    const avgQuality = this.metrics.reduce((s, m) => s + m.qualityScore, 0) / msr.total;
   328→    const avgCost = this.metrics.reduce((s, m) => s + m.costUsd, 0) / msr.total;
   329→    const totalRetries = this.metrics.reduce((s, m) => s + m.retries, 0);
   330→
   331→    console.log(`\n${'═'.repeat(70)}`);
   332→    console.log(`  AENEWS SOFTWARE FACTORY — AGGREGATE METRICS`);
   333→    console.log(`${'═'.repeat(70)}`);
   334→    console.log(`  Mission Success Rate (MSR): ${(msr.rate * 100).toFixed(1)}% (${msr.successes}/${msr.total})`);
   335→    console.log(`  Certification Rate:         ${((msr.certified / msr.total) * 100).toFixed(1)}% (${msr.certified}/${msr.total})`);
   336→    console.log(`  Average Quality Score:      ${avgQuality.toFixed(1)}/100`);
   337→    console.log(`  Average Duration:           ${(avgDuration / 1000).toFixed(1)}s`);
   338→    console.log(`  Average Cost:               $${avgCost.toFixed(3)}`);
   339→    console.log(`  Total Retries:              ${totalRetries}`);
   340→    console.log(`${'═'.repeat(70)}\n`);
   341→
   342→    // Per-mission summary
   343→    console.log('  Mission Details:');
   344→    for (const m of this.metrics) {
   345→      const status = m.certified ? '✅' : m.success ? '⚠️' : '❌';
   346→      console.log(`    ${status} ${m.missionId} — "${m.instruction.slice(0, 50)}" — Score: ${m.qualityScore} — ${(m.durationMs / 1000).toFixed(1)}s — ${m.artifactCount} artifacts`);
   347→    }
   348→  }

   354→  private async analyzeMission(instruction: string): Promise<{ plan: any; cost: number }> {
   355→    const prompt = `You are a software project planner. Analyze this mission and create a detailed execution plan.
   356→
   357→Mission: "${instruction}"
   358→
   359→Respond in JSON format:
   360→{
   361→  "objective": "clear statement of what needs to be built",
   362→  "techStack": ["list", "of", "technologies"],
   363→  "phases": [
   364→    {
   365→      "name": "phase name",
   366→      "tasks": ["task 1", "task 2"],
   367→      "capabilities": ["dev.frontend", "dev.backend"],
   368→      "estimatedMinutes": 30
   369→    }
   370→  ],
   371→  "requiredCapabilities": ["dev.frontend", "dev.backend"],
   372→  "deliverables": ["index.html", "style.css", "app.js", "tests/", "README.md", "Dockerfile"],
   373→  "risks": ["potential risk"],
   374→  "complexity": "low|medium|high"
   375→}
   376→
   377→Be specific and practical. List exact files to create.`;
   378→
   379→    try {
   380→      const response = await this.callLLM(prompt);
   381→      let plan: any;
   382→      try {
   383→        const jsonMatch = response.match(/\{[\s\S]*\}/);
   384→        plan = jsonMatch ? JSON.parse(jsonMatch[0]) : this.fallbackPlan(instruction);
   385→      } catch {
   386→        plan = this.fallbackPlan(instruction);
   387→      }
   388→      return { plan, cost: 0.02 };
   389→    } catch {
   390→      return { plan: this.fallbackPlan(instruction), cost: 0 };
   391→    }
   392→  }

   394→  private async executeBuild(
   395→    instruction: string,
   396→    analysis: { plan: any },
   397→    workspaceDir: string,
   398→  ): Promise<{ artifacts: RuntimeArtifact[]; cost: number }> {
   399→    const artifacts: RuntimeArtifact[] = [];
   400→    let totalCost = 0;
   401→
   402→    const codePrompt = `You are an expert software developer. Build the following project completely.
   403→
   404→Mission: "${instruction}"
   405→
   406→Plan: ${JSON.stringify(analysis.plan, null, 2)}
   407→
   408→Generate ALL the code needed. For each file, use this format:
   409→
   410→===FILE: path/to/file===
   411→(file content here)
   412→===ENDFILE===
   413→
   414→Include:
   415→1. All source files (HTML, CSS, JS, etc.)
   416→2. A package.json if it's a Node.js project
   417→3. A Dockerfile
   418→4. Test files in tests/ directory
   419→5. Any configuration files needed
   420→
   421→Make the code complete, functional, and production-ready.
   422→Write REAL, WORKING code — not stubs or placeholders.`;
   423→
   424→    let codeResponse: string;
   425→    let llmSucceeded = false;
   426→    try {
   427→      codeResponse = await this.callLLM(codePrompt);
   428→      totalCost += 0.10;
   429→      llmSucceeded = true;
   430→      console.log(`     LLM returned ${Buffer.byteLength(codeResponse)} bytes`);
   431→    } catch (err: any) {
   432→      console.log(`     LLM call failed: ${err.message}, using template fallback`);
   433→    }
   434→
   435→    // Parse and write files from LLM response
   436→    let files = new Map<string, string>();
   437→    if (llmSucceeded) {
   438→      files = this.parseGeneratedFiles(codeResponse!);
   439→      console.log(`     Parsed ${files.size} files from LLM response`);
   440→
   441→      // If parsing partially worked but missed some, try to extract code blocks
   442→      if (files.size < 2) {
   443→        const codeBlocks = this.extractCodeBlocks(codeResponse!);
   444→        if (codeBlocks.size > files.size) {
   445→          files = codeBlocks;
   446→          console.log(`     Extracted ${files.size} files from code blocks`);
   447→        }
   448→      }
   449→    }
   450→
   451→    // If LLM parsing still produced nothing, use template fallback
   452→    if (files.size === 0) {
   453→      console.log(`     Using template code generation fallback`);
   454→      const templateResponse = this.generateTemplateCode(instruction, analysis.plan);
   455→      files = this.parseGeneratedFiles(templateResponse);
   456→    }
   457→
   458→    for (const [filePath, content] of files) {
   459→      const fullPath = path.join(workspaceDir, filePath);
   460→      const dir = path.dirname(fullPath);
   461→      fs.mkdirSync(dir, { recursive: true });
   462→      fs.writeFileSync(fullPath, content, 'utf-8');
   463→
   464→      let type: RuntimeArtifact['type'] = 'source';
   465→      if (filePath.includes('test') || filePath.includes('spec')) type = 'test';
   466→      else if (filePath.endsWith('.md') || filePath.endsWith('.txt')) type = 'document';
   467→      else if (filePath.endsWith('.json') || filePath.endsWith('.yml') || filePath.endsWith('.yaml') || filePath.endsWith('Dockerfile') || filePath.includes('.config')) type = 'config';
   468→
   469→      artifacts.push({
   470→        name: path.basename(filePath),
   471→        type,
   472→        path: fullPath,
   473→        size: Buffer.byteLength(content),
   474→        content: content.substring(0, 500),
   475→      });
   476→
   477→      console.log(`     Created: ${filePath} (${Buffer.byteLength(content)} bytes)`);
   478→    }
   479→
   480→    // If still no files were created, something is very wrong
   481→    if (files.size === 0) {
   482→      console.log(`     WARNING: No files could be created!`);
   483→    }
   484→
   485→    // Ensure package.json exists for Node.js projects
   486→    if (analysis.plan.techStack?.some((t: string) => t.toLowerCase().includes('node') || t.toLowerCase().includes('javascript'))) {
   487→      const packageJsonPath = path.join(workspaceDir, 'package.json');
   488→      if (!fs.existsSync(packageJsonPath)) {
   489→        const packageJson = {
   490→          name: `aenews-${(analysis.plan.objective || 'project').toLowerCase().replace(/\s+/g, '-').slice(0, 30)}`,
   491→          version: '1.0.0',
   492→          description: instruction,
   493→          scripts: { start: 'node server.js', test: 'node tests/test.js' },
   494→        };
   495→        this.writeFile(workspaceDir, 'package.json', JSON.stringify(packageJson, null, 2));
   496→        artifacts.push({ name: 'package.json', type: 'config', path: packageJsonPath, size: 200 });
   497→      }
   498→    }
   499→
   500→    // Ensure Dockerfile exists
   501→    const dockerfilePath = path.join(workspaceDir, 'Dockerfile');
   502→    if (!fs.existsSync(dockerfilePath)) {

   504→      this.writeFile(workspaceDir, 'Dockerfile', dockerfile);

   506→    }
   507→
   508→    // Ensure test file exists — generate fallback if LLM didn't create one
   509→    const testArtifacts = artifacts.filter(a => a.type === 'test');
   510→    const testDir = path.join(workspaceDir, 'tests');
   511→    const testDirFiles = fs.existsSync(testDir) ? fs.readdirSync(testDir).filter(f => f.endsWith('.js') || f.endsWith('.mjs') || f.endsWith('.ts')) : [];
   512→    const rootTestFiles = fs.readdirSync(workspaceDir).filter(f => (f.includes('test') || f.includes('spec')) && (f.endsWith('.js') || f.endsWith('.ts')));
   513→    const hasTests = testArtifacts.length > 0 || testDirFiles.length > 0 || rootTestFiles.length > 0;

   523→      }
   524→    }
   525→
   526→    return { artifacts, cost: totalCost };
   527→  }

   621→  private certify(
   622→    artifacts: RuntimeArtifact[],
   623→    testResult: { passed: boolean; results: any[] },
   624→    auditResult: { passed: boolean; findings: string[] },
   625→  ): { certified: boolean; qualityScore: number; reasons: string[] } {
   626→    const reasons: string[] = [];
   627→    let score = 100;
   628→
   629→    // Test scoring — partial credit if some tests pass
   630→    if (!testResult.passed) {
   631→      const totalTests = testResult.results.length;
   632→      const passedTests = testResult.results.filter(r => r.passed).length;
   633→      if (totalTests > 0 && passedTests > 0) {
   634→        const passRate = passedTests / totalTests;
   635→        score -= Math.round(30 * (1 - passRate)); // Partial deduction
   636→        if (passRate < 0.5) reasons.push(`${passedTests}/${totalTests} tests passed`);
   637→      } else {
   638→        score -= 30;
   639→        reasons.push('Tests failed');
   640→      }
   641→    }
   642→
   643→    // Audit scoring — only critical findings reduce score significantly
   644→    const criticalFindings = auditResult.findings.filter(f => 
   645→      f.toLowerCase().includes('no source') || 
   646→      f.toLowerCase().includes('injection') ||
   647→      f.toLowerCase().includes('execute') ||
   648→      f.toLowerCase().includes('malicious')
   649→    );
   650→    const minorFindings = auditResult.findings.filter(f => !criticalFindings.includes(f));
   651→    
   652→    if (criticalFindings.length > 0) {
   653→      score -= 20;
   654→      reasons.push(...criticalFindings.slice(0, 3));
   655→    }
   656→    if (minorFindings.length > 0) {
   657→      score -= Math.min(10, minorFindings.length * 3); // Minor findings: -3 each, max -10
   658→      if (minorFindings.length <= 3) reasons.push(...minorFindings);
   659→      else reasons.push(`${minorFindings.length} minor findings`);
   660→    }
   661→
   662→    // Missing artifacts — these are critical
   663→    if (artifacts.filter(a => a.type === 'source').length === 0) {
   664→      score -= 40;
   665→      reasons.push('No source code');
   666→    }
   667→    if (!artifacts.find(a => a.name === 'README.md')) {
   668→      score -= 10;
   669→      reasons.push('No README');
   670→    }
   671→    if (!artifacts.find(a => a.name === 'Dockerfile')) {
   672→      score -= 10;
   673→      reasons.push('No Dockerfile');
   674→    }
   675→    if (!artifacts.some(a => a.type === 'test')) {
   676→      score -= 10;
   677→      reasons.push('No test files');
   678→    }
   679→
   680→    return { certified: score >= 60, qualityScore: Math.max(0, score), reasons };
   681→  }

   712→  private async callLLM(prompt: string): Promise<string> {
   713→    if (!this.zaiInstance) {
   714→      try {
   715→        const sdk = await import('z-ai-web-dev-sdk');
   716→        const ZAIClass = sdk.default || sdk;
   717→        this.zaiInstance = await ZAIClass.create();
   718→      } catch (err: any) {
   719→        throw new Error(`z-ai-web-dev-sdk not available: ${err.message}`);
   720→      }
   721→    }
   722→
   723→    // Retry with exponential backoff for rate limiting
   724→    const maxRetries = 3;
   725→    for (let attempt = 0; attempt < maxRetries; attempt++) {
   726→      try {
   727→        const completion = await this.zaiInstance.chat.completions.create({
   728→          messages: [
   729→            { role: 'system', content: 'You are an expert software engineer. Generate complete, working, production-ready code. Be thorough and practical.' },
   730→            { role: 'user', content: prompt },
   731→          ],
   732→          temperature: 0.3,
   733→          max_tokens: 4096,
   734→        });
   735→
   736→        const content = completion.choices?.[0]?.message?.content;
   737→        if (content) {
   738→          this.llmCallCount++;
   739→          return content;
   740→        }
   741→        throw new Error('Empty LLM response');
   742→      } catch (err: any) {
   743→        const isRateLimit = err.message?.includes('429') || err.message?.includes('rate');
   744→        if (isRateLimit && attempt < maxRetries - 1) {
   745→          const delayMs = Math.pow(2, attempt) * 3000; // 3s, 6s, 12s
   746→          console.log(`     Rate limited, retrying in ${delayMs / 1000}s... (attempt ${attempt + 1}/${maxRetries})`);
   747→          await new Promise(resolve => setTimeout(resolve, delayMs));
   748→          continue;
   749→        }
   750→        throw err;
   751→      }
   752→    }
   753→    throw new Error('Max retries exceeded');
   754→  }

   767→  private parseGeneratedFiles(response: string): Map<string, string> {
   768→    const files = new Map<string, string>();
   769→
   770→    // Try ===FILE: path=== format
   771→    const fileRegex = /===FILE:\s*(.+?)===\s*\n([\s\S]*?)===ENDFILE===/g;
   772→    let match;
   773→    while ((match = fileRegex.exec(response)) !== null) {
   774→      const filePath = match[1].trim();
   775→      const content = match[2].trim();
   776→      if (filePath && content) files.set(filePath, content);
   777→    }
   778→
   779→    if (files.size > 0) return files;
   780→
   781→    // Try ```filename format  
   782→    const codeBlockRegex = /```(\w*?)\s*\n([\s\S]*?)```/g;
   783→    while ((match = codeBlockRegex.exec(response)) !== null) {
   784→      const lang = match[1].trim();
   785→      const content = match[2].trim();
   786→      if (!content || content.length < 10) continue;
   787→      
   788→      // Try to find filename in the line before the code block
   789→      const beforeMatch = response.substring(Math.max(0, match.index - 200), match.index);
   790→      const nameMatch = beforeMatch.match(/(?:^|\n)[\s]*(?:\/\/|#|<!--)?\s*(?:file:\s*)?(\S+\.\w+)\s*(?:$|\n)/m);
   791→      
   792→      if (nameMatch) {
   793→        files.set(nameMatch[1], content);
   794→      } else {
   795→        // Infer filename from language
   796→        const langMap: Record<string, string> = {
   797→          'html': 'index.html', 'css': 'style.css', 'javascript': 'app.js', 'js': 'app.js',
   798→          'typescript': 'app.ts', 'ts': 'app.ts', 'python': 'app.py', 'py': 'app.py',
   799→          'json': 'package.json', 'yaml': 'docker-compose.yml', 'yml': 'docker-compose.yml',
   800→          'dockerfile': 'Dockerfile', 'bash': 'start.sh', 'sh': 'start.sh',
   801→          'sql': 'schema.sql', 'md': 'README.md',
   802→        };
   803→        const fileName = langMap[lang.toLowerCase()];
   804→        if (fileName && !files.has(fileName)) {
   805→          files.set(fileName, content);
   806→        }
   807→      }
   808→    }
   809→
   810→    return files;
   811→  }

   813→  /**
   814→   * Extract code blocks from LLM response with filename inference
   815→   */
   816→  private extractCodeBlocks(response: string): Map<string, string> {
   817→    const files = new Map<string, string>();
   818→    
   819→    // Split by markdown code blocks and look for filename patterns
   820→    const lines = response.split('\n');
   821→    let currentFile = '';
   822→    let currentContent: string[] = [];
   823→    let inCodeBlock = false;
   824→    let codeLang = '';
   825→    let fileCounter = 0;
   826→
   827→    for (let i = 0; i < lines.length; i++) {
   828→      const line = lines[i];
   829→      
   830→      if (line.trim().startsWith('```')) {
   831→        if (inCodeBlock) {
   832→          // End of code block
   833→          const content = currentContent.join('\n').trim();
   834→          if (content.length > 10) {
   835→            if (!currentFile) {
   836→              // Infer from language
   837→              const langMap: Record<string, string> = {
   838→                'html': 'index.html', 'css': 'style.css', 'javascript': 'app.js', 'js': 'app.js',
   839→                'typescript': 'app.ts', 'json': 'package.json', 'dockerfile': 'Dockerfile',
   840→                'python': 'app.py', 'bash': 'start.sh', 'sql': 'schema.sql', 'md': 'README.md',
   841→              };
   842→              currentFile = langMap[codeLang.toLowerCase()] || `file-${++fileCounter}.${codeLang || 'txt'}`;
   843→            }
   844→            if (!files.has(currentFile)) {
   845→              files.set(currentFile, content);
   846→            }
   847→          }
   848→          currentFile = '';
   849→          currentContent = [];
   850→          inCodeBlock = false;
   851→        } else {
   852→          // Start of code block
   853→          inCodeBlock = true;
   854→          codeLang = line.trim().replace('```', '').trim();
   855→          
   856→          // Look back for filename
   857→          const prevLines = lines.slice(Math.max(0, i - 3), i).join('\n');
   858→          const filePatterns = [
   859→            /(?:file|filename|path|create|save)\s*[:=]\s*`?(\S+\.\w+)`?/i,
   860→            /(\S+\.\w+)\s*[:=]/,
   861→            /`(\S+\.\w+)`/,
   862→            /(?:\/|\w\/)(\w+\.\w+)/,
   863→            /(\w+\.(?:html|css|js|ts|json|py|sql|sh|yml|yaml|md|txt|dockerfile))/i,
   864→          ];
   865→          
   866→          for (const pattern of filePatterns) {
   867→            const m = prevLines.match(pattern);
   868→            if (m) { currentFile = m[1]; break; }
   869→          }
   870→        }
   871→      } else if (inCodeBlock) {
   872→        currentContent.push(line);
   873→      }
   874→    }
   875→    
   876→    return files;
   877→  }

   879→  private collectSourceFiles(workspaceDir: string): { name: string; content?: string }[] {
   880→    const files: { name: string; content?: string }[] = [];
   881→    const extensions = ['.js', '.ts', '.html', '.css', '.json', '.py', '.jsx', '.tsx'];
   882→
   883→    const walkDir = (dir: string) => {
   884→      if (!fs.existsSync(dir)) return;
   885→      const entries = fs.readdirSync(dir, { withFileTypes: true });
   886→      for (const entry of entries) {
   887→        const fullPath = path.join(dir, entry.name);
   888→        if (entry.isDirectory() && !entry.name.startsWith('.') && entry.name !== 'node_modules') {
   889→          walkDir(fullPath);
   890→        } else if (entry.isFile() && extensions.some(ext => entry.name.endsWith(ext))) {
   891→          try {
   892→            const content = fs.readFileSync(fullPath, 'utf-8');
   893→            files.push({ name: path.relative(workspaceDir, fullPath), content: content.slice(0, 1000) });
   894→          } catch { /* skip */ }
   895→        }
   896→      }
   897→    };
   898→
   899→    walkDir(workspaceDir);
   900→    return files;
   901→  }

   907→  private async createZipArchive(missionId: string, workspaceDir: string): Promise<string | null> {
   908→    const zipPath = path.join(this.baseWorkspace, `${missionId}.zip`);
   909→    
   910→    // Try native zip first
   911→    try {
   912→      const { execSync } = await import('child_process');
   913→      const result = execSync(`cd "${workspaceDir}" && zip -r "${zipPath}" . -x "*.git*" 2>&1`, {
   914→        timeout: 60000,
   915→        encoding: 'utf-8',
   916→      });
   917→      if (fs.existsSync(zipPath) && fs.statSync(zipPath).size > 0) {
   918→        return zipPath;
   919→      }
   920→    } catch (err: any) {
   921→      console.log(`     zip command failed: ${err.message?.slice(0, 200)}`);
   922→    }
   923→    
   924→    // Fallback: use Node.js archiver
   925→    try {
   926→      const archiverModule: any = await import('archiver');
   927→      const archiverFn = archiverModule.default || archiverModule;
   928→      const output = fs.createWriteStream(zipPath);
   929→      const archive = archiverFn('zip', { zlib: { level: 9 } });
   930→      
   931→      await new Promise<void>((resolve, reject) => {
   932→        output.on('close', () => resolve());
   933→        archive.on('error', (err: Error) => reject(err));
   934→        archive.pipe(output);
   935→        archive.directory(workspaceDir, false);
   936→        archive.finalize();
   937→      });
   938→      
   939→      if (fs.existsSync(zipPath) && fs.statSync(zipPath).size > 0) {
   940→        return zipPath;
   941→      }
   942→    } catch (err: any) {
   943→      console.log(`     archiver failed: ${err.message?.slice(0, 200)}`);
   944→    }
   945→    
   946→    // Last fallback: create a simple tar.gz
   947→    try {
   948→      const { execSync } = await import('child_process');
   949→      const tarPath = path.join(this.baseWorkspace, `${missionId}.tar.gz`);
   950→      execSync(`cd "${workspaceDir}" && tar -czf "${tarPath}" . 2>&1`, { timeout: 60000 });
   951→      if (fs.existsSync(tarPath) && fs.statSync(tarPath).size > 0) {
   952→        console.log(`     Using tar.gz fallback: ${tarPath}`);
   953→        return tarPath;
   954→      }
   955→    } catch (err: any) {
   956→      console.log(`     tar failed: ${err.message?.slice(0, 200)}`);
   957→    }
   958→    
   959→    return null;
   960→  }
