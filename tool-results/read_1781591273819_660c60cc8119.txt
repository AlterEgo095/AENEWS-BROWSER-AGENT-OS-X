   353→          this.logger.log(`Invoice operation: ${operation}${invoiceId ? ` (ID: ${invoiceId})` : ''}`);
   354→
   355→          const llmResult = await this.executeWithLLM(
   356→            `You are an invoicing and billing expert. You create invoices with proper calculations, tax handling, and payment tracking.`,
   357→            `Create invoice for client ${clientId || clientName}. Items: ${JSON.stringify(items.slice(0, 5))}. Tax: ${taxRate}%. Discount: ${discount}%. Terms: ${paymentTerms}. Return JSON with: invoice {invoiceNumber, subtotal, taxAmount, discountAmount, totalAmount, balanceDue, paymentStatus, payments (array)}.`,
   358→            { responseFormat: 'json', temperature: 0.3, maxTokens: 1024 },
   359→          );
   360→
   361→          const parsed = this.safeJsonParse(llmResult);
   362→
   363→          if (parsed) {
   364→            this.emitEvent(AgentEventType.AGENT_COMPLETED, { action, llmUsed: true });
   365→            return {
   366→              success: true,
   367→              data: {
   368→                action, operation, invoiceId, clientId, clientName, items, dueDate, issueDate, currency, taxRate, discount, paymentTerms, notes, queryStatus: status_,
   369→                invoice: parsed.invoice || { invoiceNumber: '', subtotal: 0, taxAmount: 0, discountAmount: 0, totalAmount: 0, balanceDue: 0, paymentStatus: 'draft', payments: [] },
   370→                status: 'invoice_operation_complete', timestamp: new Date().toISOString(),
   371→              },
   372→              metadata: { duration: Date.now() - startTime, llmUsed: true },
   373→            };
   374→          }
   375→
   376→          // Intelligent fallback
   377→          const subtotal = items.reduce((sum: number, item: any) => sum + (item.amount || item.quantity * item.unitPrice || 0), 0) || 5000;
   378→          const taxAmt = Math.round(subtotal * taxRate / 100);
   379→          const discAmt = Math.round(subtotal * discount / 100);
   380→          this.emitEvent(AgentEventType.AGENT_COMPLETED, { action, llmUsed: false, fallback: true });
   381→          return {
   382→            success: true,
   383→            data: {
   384→              action, operation, invoiceId, clientId, clientName, items, dueDate, issueDate, currency, taxRate, discount, paymentTerms, notes, queryStatus: status_,
   385→              invoice: { invoiceNumber: `INV-${Date.now().toString(36).toUpperCase()}`, subtotal, taxAmount: taxAmt, discountAmount: discAmt, totalAmount: subtotal + taxAmt - discAmt, balanceDue: subtotal + taxAmt - discAmt, paymentStatus: 'draft', payments: [] },
   386→              status: 'invoice_operation_complete', timestamp: new Date().toISOString(),
   387→            },
   388→            metadata: { duration: Date.now() - startTime, llmUsed: false, fallback: true },
   389→          };
   390→        }
   391→
   392→        default:
   393→          return { success: false, error: `Unknown action: ${action}. Supported actions: budget, forecast, expense, revenue, report, invoice` };
   394→      }
   395→    } catch (error: any) {
   396→      this.emitEvent(AgentEventType.AGENT_FAILED, { error: error.message });
   397→      return { success: false, error: error.message };
   398→    }
   399→  }
