   265→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   266→    const startTime = Date.now();
   267→    const { action, ...params } = input.payload;
   268→    if (!action)
   269→      return this.createAgentOutput(
   270→        input.taskId,
   271→        false,
   272→        null,
   273→        'Missing required parameter: action',
   274→        startTime,
   275→      );
   276→    const supportedActions = [
   277→      'enforcePolicy',
   278→      'auditCompliance',
   279→      'reviewGovernance',
   280→      'updatePolicy',
   281→      'generateGovernanceReport',
   282→      'manageExceptions',
   283→    ];
   284→    if (!supportedActions.includes(action))
   285→      return this.createAgentOutput(
   286→        input.taskId,
   287→        false,
   288→        null,
   289→        `Unknown governance action: ${action}. Supported: ${supportedActions.join(', ')}`,
   290→        startTime,
   291→      );
   292→    try {
   293→      const tool = this.getTool(action);
   294→      if (!tool)
   295→        return this.createAgentOutput(
   296→          input.taskId,
   297→          false,
   298→          null,
   299→          `Tool not found: ${action}`,
   300→          startTime,
   301→        );
   302→      const result = await tool.execute(params);
   303→      await this.storeInWorkingMemory(
   304→        `governance:last:${action}`,
   305→        { params, result, timestamp: new Date() },
   306→        300000,
   307→      );
   308→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   309→    } catch (error) {
   310→      const msg = (error as Error).message;
   311→      this.logger.error(`MetaGovernance execution failed for ${action}: ${msg}`);
   312→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   313→    }
   314→  }

   323→  private async enforcePolicy(params: {
   324→    policyId: string;
   325→    target?: string;
   326→    action?: string;
   327→  }): Promise<{ enforced: boolean; policyId: string; violations: number; enforcementId: string }> {
   328→    const { policyId, target = 'all', action = 'check' } = params;
   329→    if (!policyId || typeof policyId !== 'string')
   330→      throw new Error('Valid policyId string is required');
   331→    const policy = this.policies.get(policyId);
   332→    if (!policy) throw new Error(`Policy not found: ${policyId}`);
   333→    if (!policy.active) throw new Error(`Policy is not active: ${policyId}`);
   334→    const enforcementId = this.generateId();
   335→
   336→    // Check for existing exceptions
   337→    const hasException = Array.from(this.exceptions.values()).some(
   338→      (e) => e.policyId === policyId && e.target === target && e.status === 'active',
   339→    );
   340→
   341→    if (hasException) {
   342→      this.logger.log(
   343→        `Policy enforcement skipped: exception exists for policy=${policyId}, target=${target}`,
   344→      );
   345→      return { enforced: true, policyId, violations: 0, enforcementId };
   346→    }
   347→
   348→    // Simulate violation detection
   349→    let violationCount = 0;
   350→    for (const rule of policy.rules) {
   351→      const violationChance = Math.random();
   352→      if (violationChance > 0.8) {
   353→        violationCount++;
   354→        this.violations.push({
   355→          id: this.generateId(),
   356→          policyId,
   357→          ruleId: rule.id,
   358→          target,
   359→          severity: rule.severity,
   360→          description: `Rule "${rule.condition}" violated for target "${target}"`,
   361→          timestamp: new Date(),
   362→          resolved: false,
   363→        });
   364→      }
   365→    }
   366→
   367→    if (action === 'enforce' && violationCount > 0) {
   368→      // Mark violations for resolution
   369→      const recentViolations = this.violations.filter(
   370→        (v) => !v.resolved && v.policyId === policyId,
   371→      );
   372→      for (const v of recentViolations.slice(-violationCount)) {
   373→        v.resolved = true;
   374→      }
   375→    }
   376→
   377→    this.logger.log(
   378→      `Policy enforced: policyId=${policyId}, target=${target}, violations=${violationCount}`,
   379→    );
   380→    return { enforced: true, policyId, violations: violationCount, enforcementId };
   381→  }

   383→  private async auditCompliance(params: { scope?: string; policies?: string[] }): Promise<{
   384→    auditId: string;
   385→    complianceScore: number;
   386→    violations: Array<{ policyId: string; severity: string; description: string }>;
   387→    compliant: boolean;
   388→  }> {
   389→    const { scope = 'all', policies: policyIds = [] } = params;
   390→    const auditId = this.generateId();
   391→    const targetPolicies =
   392→      policyIds.length > 0
   393→        ? Array.from(this.policies.values()).filter((p) => policyIds.includes(p.id))
   394→        : Array.from(this.policies.values()).filter((p) => p.active);
   395→
   396→    const recentViolations = this.violations.filter((v) => {
   397→      const isRecent = Date.now() - v.timestamp.getTime() < 24 * 60 * 60 * 1000;
   398→      const inScope = scope === 'all' || v.target.includes(scope);
   399→      const inPolicies = policyIds.length === 0 || policyIds.includes(v.policyId);
   400→      return isRecent && inScope && inPolicies;
   401→    });
   402→
   403→    const criticalViolations = recentViolations.filter(
   404→      (v) => v.severity === 'critical' && !v.resolved,
   405→    );
   406→    const warningViolations = recentViolations.filter((v) => v.severity === 'warning');
   407→
   408→    const complianceScore = Math.max(
   409→      0,
   410→      Math.round(100 - criticalViolations.length * 20 - warningViolations.length * 5),
   411→    );
   412→    const compliant = complianceScore >= 80 && criticalViolations.length === 0;
   413→
   414→    const violationSummary = recentViolations.slice(0, 10).map((v) => ({
   415→      policyId: v.policyId,
   416→      severity: v.severity,
   417→      description: v.description,
   418→    }));
   419→
   420→    this.logger.log(
   421→      `Compliance audited: scope=${scope}, score=${complianceScore}, violations=${recentViolations.length}, compliant=${compliant}`,
   422→    );
   423→    return { auditId, complianceScore, violations: violationSummary, compliant };
   424→  }

   426→  private async reviewGovernance(params: { scope?: string; depth?: string }): Promise<{
   427→    reviewId: string;
   428→    findings: Array<{ area: string; status: string; description: string; priority: string }>;
   429→    recommendations: string[];
   430→    overallRating: string;
   431→  }> {
   432→    const { scope = 'all', depth = 'standard' } = params;
   433→    const reviewId = this.generateId();
   434→    const findings: Array<{ area: string; status: string; description: string; priority: string }> =
   435→      [];
   436→    const recommendations: string[] = [];
   437→
   438→    // Policy coverage
   439→    const activePolicies = Array.from(this.policies.values()).filter((p) => p.active);
   440→    if (activePolicies.length < 5) {
   441→      findings.push({
   442→        area: 'policy-coverage',
   443→        status: 'warning',
   444→        description: `Only ${activePolicies.length} active policies; consider expanding coverage`,
   445→        priority: 'medium',
   446→      });
   447→      recommendations.push('Add policies for security, data handling, and resource management');
   448→    } else {
   449→      findings.push({
   450→        area: 'policy-coverage',
   451→        status: 'good',
   452→        description: `${activePolicies.length} active policies covering key areas`,
   453→        priority: 'low',
   454→      });
   455→    }
   456→
   457→    // Violation trends
   458→    const unresolvedViolations = this.violations.filter((v) => !v.resolved);
   459→    if (unresolvedViolations.length > 5) {
   460→      findings.push({
   461→        area: 'violation-resolution',
   462→        status: 'warning',
   463→        description: `${unresolvedViolations.length} unresolved violations`,
   464→        priority: 'high',
   465→      });
   466→      recommendations.push('Establish a violation resolution SLA and escalation process');
   467→    } else {
   468→      findings.push({
   469→        area: 'violation-resolution',
   470→        status: 'good',
   471→        description: 'Low number of unresolved violations',
   472→        priority: 'low',
   473→      });
   474→    }
   475→
   476→    // Exception management
   477→    const activeExceptions = Array.from(this.exceptions.values()).filter(
   478→      (e) => e.status === 'active',
   479→    );
   480→    if (activeExceptions.length > 10) {
   481→      findings.push({
   482→        area: 'exception-management',
   483→        status: 'warning',
   484→        description: `${activeExceptions.length} active exceptions may indicate governance gaps`,
   485→        priority: 'medium',
   486→      });
   487→      recommendations.push('Review and reduce policy exceptions; update policies instead');
   488→    } else {
   489→      findings.push({
   490→        area: 'exception-management',
   491→        status: 'good',
   492→        description: 'Exception count within acceptable range',
   493→        priority: 'low',
   494→      });
   495→    }
   496→
   497→    if (depth === 'comprehensive') {
   498→      findings.push({
   499→        area: 'policy-versioning',
   500→        status: 'good',
   501→        description: 'All policies have version tracking',
   502→        priority: 'low',
   503→      });
   504→      findings.push({
   505→        area: 'audit-trail',
   506→        status: 'good',
   507→        description: 'Violation audit trail is maintained',
   508→        priority: 'low',
   509→      });
   510→      recommendations.push('Implement automated compliance monitoring');
   511→      recommendations.push('Create policy change approval workflow');
   512→    }
   513→
   514→    const highPriorityFindings = findings.filter((f) => f.priority === 'high');
   515→    const overallRating =
   516→      highPriorityFindings.length > 0
   517→        ? 'needs-improvement'
   518→        : findings.some((f) => f.status === 'warning')
   519→          ? 'acceptable'
   520→          : 'excellent';
   521→
   522→    this.logger.log(
   523→      `Governance reviewed: findings=${findings.length}, rating=${overallRating}, scope=${scope}`,
   524→    );
   525→    return { reviewId, findings, recommendations, overallRating };
   526→  }

   528→  private async updatePolicy(params: {
   529→    policyId: string;
   530→    changes: Record<string, any>;
   531→    reason?: string;
   532→  }): Promise<{ updated: boolean; policyId: string; version: string; updateId: string }> {
   533→    const { policyId, changes, reason = '' } = params;
   534→    if (!policyId || typeof policyId !== 'string')
   535→      throw new Error('Valid policyId string is required');
   536→    if (!changes || typeof changes !== 'object')
   537→      throw new Error('Valid changes object is required');
   538→    const updateId = this.generateId();
   539→
   540→    const existing = this.policies.get(policyId);
   541→    if (existing) {
   542→      const versionParts = existing.version.split('.');
   543→      const minor = parseInt(versionParts[1] || '0', 10) + 1;
   544→      const newVersion = `${versionParts[0]}.${minor}`;
   545→
   546→      if (changes.name) existing.name = changes.name;
   547→      if (changes.description) existing.description = changes.description;
   548→      if (changes.rules) existing.rules = changes.rules;
   549→      if (changes.active !== undefined) existing.active = changes.active;
   550→      existing.version = newVersion;
   551→      existing.updatedAt = new Date();
   552→
   553→      this.logger.log(
   554→        `Policy updated: id=${policyId}, version=${newVersion}, reason="${reason.substring(0, 50)}"`,
   555→      );
   556→      return { updated: true, policyId, version: newVersion, updateId };
   557→    } else {
   558→      // Create new policy
   559→      const newPolicy: Policy = {
   560→        id: policyId,
   561→        name: changes.name || policyId,
   562→        description: changes.description || 'Auto-generated policy',
   563→        rules: changes.rules || [
   564→          { id: `${policyId}-rule-1`, condition: 'default', action: 'allow', severity: 'info' },
   565→        ],
   566→        version: '1.0',
   567→        active: changes.active !== undefined ? changes.active : true,
   568→        createdAt: new Date(),
   569→        updatedAt: new Date(),
   570→      };
   571→      this.policies.set(policyId, newPolicy);
   572→
   573→      this.logger.log(`Policy created: id=${policyId}, version=1.0`);
   574→      return { updated: true, policyId, version: '1.0', updateId };
   575→    }
   576→  }

   578→  private async generateGovernanceReport(params: {
   579→    period?: string;
   580→    includeViolations?: boolean;
   581→    includeRecommendations?: boolean;
   582→  }): Promise<{
   583→    reportId: string;
   584→    summary: Record<string, any>;
   585→    metrics: Record<string, number>;
   586→    recommendations: string[];
   587→  }> {
   588→    const { period = '30d', includeViolations = true, includeRecommendations = true } = params;
   589→    const reportId = this.generateId();
   590→
   591→    const periodMs =
   592→      period === '7d' ? 7 * 86400000 : period === '30d' ? 30 * 86400000 : 90 * 86400000;
   593→    const cutoff = new Date(Date.now() - periodMs);
   594→    const periodViolations = this.violations.filter((v) => v.timestamp >= cutoff);
   595→
   596→    const summary: Record<string, any> = {
   597→      period,
   598→      totalPolicies: this.policies.size,
   599→      activePolicies: Array.from(this.policies.values()).filter((p) => p.active).length,
   600→      totalViolations: periodViolations.length,
   601→      unresolvedViolations: periodViolations.filter((v) => !v.resolved).length,
   602→      activeExceptions: Array.from(this.exceptions.values()).filter((e) => e.status === 'active')
   603→        .length,
   604→      generatedAt: new Date().toISOString(),
   605→    };
   606→
   607→    if (includeViolations) {
   608→      summary.violationsBySeverity = {
   609→        critical: periodViolations.filter((v) => v.severity === 'critical').length,
   610→        warning: periodViolations.filter((v) => v.severity === 'warning').length,
   611→        info: periodViolations.filter((v) => v.severity === 'info').length,
   612→      };
   613→    }
   614→
   615→    const metrics: Record<string, number> = {
   616→      complianceRate:
   617→        periodViolations.length > 0
   618→          ? Math.round(
   619→              (1 - periodViolations.filter((v) => !v.resolved).length / periodViolations.length) *
   620→                100,
   621→            )
   622→          : 100,
   623→      policyCoverage:
   624→        this.policies.size > 0
   625→          ? Math.round(
   626→              (Array.from(this.policies.values()).filter((p) => p.active).length /
   627→                this.policies.size) *
   628→                100,
   629→            )
   630→          : 0,
   631→      violationResolutionRate:
   632→        periodViolations.length > 0
   633→          ? Math.round(
   634→              (periodViolations.filter((v) => v.resolved).length / periodViolations.length) * 100,
   635→            )
   636→          : 100,
   637→    };
   638→
   639→    const recommendations: string[] = [];
   640→    if (includeRecommendations) {
   641→      if (metrics.complianceRate < 80)
   642→        recommendations.push('Improve violation resolution processes to increase compliance rate');
   643→      if (summary.unresolvedViolations > 10)
   644→        recommendations.push('Prioritize resolution of unresolved violations');
   645→      if (summary.activeExceptions > 5)
   646→        recommendations.push('Review and reduce active policy exceptions');
   647→      if (summary.activePolicies < 3)
   648→        recommendations.push('Expand policy coverage for better governance');
   649→      if (recommendations.length === 0)
   650→        recommendations.push('Governance is in good standing; continue regular reviews');
   651→    }
   652→
   653→    this.logger.log(
   654→      `Governance report generated: id=${reportId}, period=${period}, violations=${periodViolations.length}`,
   655→    );
   656→    return { reportId, summary, metrics, recommendations };
   657→  }
   658→
   659→  private async manageExceptions(params: {
   660→    operation: string;
   661→    exceptionId?: string;
   662→    details?: Record<string, any>;
   663→  }): Promise<{ success: boolean; exceptionId: string; status: string; operation: string }> {
   664→    const { operation, exceptionId, details = {} } = params;
   665→    if (!operation || typeof operation !== 'string')
   666→      throw new Error('Valid operation string is required');
   667→
   668→    const validOperations = ['create', 'revoke', 'extend', 'list'];
   669→    if (!validOperations.includes(operation)) {
   670→      throw new Error(`Invalid operation: ${operation}. Supported: ${validOperations.join(', ')}`);
   671→    }
   672→
   673→    switch (operation) {
   674→      case 'create': {
   675→        const id = this.generateId();
   676→        const exception: Exception = {
   677→          id,
   678→          policyId: details.policyId || 'unknown',
   679→          target: details.target || 'all',
   680→          reason: details.reason || 'No reason provided',
   681→          grantedAt: new Date(),
   682→          expiresAt: new Date(Date.now() + (details.durationDays || 30) * 86400000),
   683→          status: 'active',
   684→        };
   685→        this.exceptions.set(id, exception);
   686→        this.logger.log(
   687→          `Exception created: id=${id}, policy=${exception.policyId}, target=${exception.target}`,
   688→        );
   689→        return { success: true, exceptionId: id, status: 'active', operation };
   690→      }
   691→
   692→      case 'revoke': {
   693→        if (!exceptionId) throw new Error('exceptionId is required for revoke operation');
   694→        const exception = this.exceptions.get(exceptionId);
   695→        if (!exception) throw new Error(`Exception not found: ${exceptionId}`);
   696→        exception.status = 'revoked';
   697→        this.logger.log(`Exception revoked: id=${exceptionId}`);
   698→        return { success: true, exceptionId, status: 'revoked', operation };
   699→      }
   700→
   701→      case 'extend': {
   702→        if (!exceptionId) throw new Error('exceptionId is required for extend operation');
   703→        const exception = this.exceptions.get(exceptionId);
   704→        if (!exception) throw new Error(`Exception not found: ${exceptionId}`);
   705→        const extendDays = details.durationDays || 30;
   706→        exception.expiresAt = new Date(exception.expiresAt.getTime() + extendDays * 86400000);
   707→        this.logger.log(
   708→          `Exception extended: id=${exceptionId}, new expiry=${exception.expiresAt.toISOString()}`,
   709→        );
   710→        return { success: true, exceptionId, status: exception.status, operation };
   711→      }
   712→
   713→      case 'list': {
   714→        const active = Array.from(this.exceptions.values()).filter((e) => e.status === 'active');
   715→        this.logger.log(`Exceptions listed: active=${active.length}`);
   716→        return {
   717→          success: true,
   718→          exceptionId: 'list',
   719→          status: `${active.length} active exceptions`,
   720→          operation,
   721→        };
   722→      }
   723→
   724→      default:
   725→        return {
   726→          success: false,
   727→          exceptionId: exceptionId || '',
   728→          status: 'unknown-operation',
   729→          operation,
   730→        };
   731→    }
   732→  }
