   244→  describe('POST auth/login — sets httpOnly refresh cookie', () => {
   245→    it('should set a refresh_token cookie on successful login', async () => {
   246→      const response = await request(app.getHttpServer())
   247→        .post('/auth/login')
   248→        .send({ email: 'test@example.com', password: TEST_PASSWORD })

   251→      // Check that the cookie was set

   253→      expect(cookies).toBeDefined();
   254→
   255→      const cookieList = Array.isArray(cookies) ? cookies : [cookies];

   259→      expect(refreshTokenCookie).toContain('HttpOnly');
   260→      expect(refreshTokenCookie).toContain('SameSite=Strict');

   263→
   264→    it('should return access token and user info in the response body', async () => {
   265→      const response = await request(app.getHttpServer())
   266→        .post('/auth/login')

   268→        .expect(201);
