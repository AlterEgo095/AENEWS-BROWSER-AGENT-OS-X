    71→container_is_running() {

    75→ensure_container_running() {
    76→    local container="$1"
    77→    local service="$2"
    78→
    79→    if ! container_is_running "$container"; then
    80→        log_warning "Container '${container}' is not running. Starting it..."
    81→        docker_compose up -d "$service"
    82→        sleep 5
    83→        local elapsed=0
    84→        while [ $elapsed -lt 60 ]; do
    85→            local status
    86→            status=$(docker inspect --format='{{.State.Health.Status}}' "$container" 2>/dev/null || echo "missing")
    87→            if [ "$status" = "healthy" ]; then
    88→                return 0
    89→            fi
    90→            sleep 5
    91→            elapsed=$((elapsed + 5))
    92→        done
    93→        log_error "Container '${container}' failed to start"
    94→        return 1
    95→    fi
    96→}

    98→get_env_value() {
    99→    local key="$1"
   100→    local default="${2:-}"
   101→    local val
   102→    val=$(docker exec "${PG_CONTAINER}" printenv "$key" 2>/dev/null || echo "")
   103→    if [ -z "$val" ] && [ -f "${PROJECT_ROOT}/.env" ]; then
   104→        val=$(grep "^${key}=" "${PROJECT_ROOT}/.env" 2>/dev/null | cut -d= -f2- || echo "")
   105→    fi
   106→    echo "${val:-$default}"
   107→}

   119→# ─── PostgreSQL Backup ─────────────────────────────────────────────────
   120→backup_postgres() {
   121→    log_step "Backing up PostgreSQL..."
   122→
   123→    ensure_container_running "${PG_CONTAINER}" "postgres"
   124→
   125→    local db_name="${POSTGRES_DB:-aenews_osx}"
   126→    local db_user="${POSTGRES_USER:-aenews}"
   127→    local db_password="${POSTGRES_PASSWORD:-aenews_secret_2024}"
   128→    local backup_file="${BACKUP_DIR}/postgres_${TIMESTAMP}.sql.gz"
   129→
   130→    log_info "Database: ${db_name}"
   131→
   132→    docker exec "${PG_CONTAINER}" pg_dump \
   133→        -U "${db_user}" \
   134→        -d "${db_name}" \
   135→        --format=plain \
   136→        --no-owner \
   137→        --no-acl \
   138→        --clean \
   139→        --if-exists \
   140→        2>/dev/null | gzip > "${backup_file}"
   141→
   142→    local size
   143→    size=$(du -h "${backup_file}" | cut -f1)
   144→    log_success "PostgreSQL backup completed (${size}): ${backup_file}"
   145→}

   147→# ─── PostgreSQL Restore ────────────────────────────────────────────────
   148→restore_postgres() {
   149→    local backup_file="$1"
   150→
   151→    if [ ! -f "$backup_file" ]; then
   152→        log_error "PostgreSQL backup file not found: ${backup_file}"
   153→        return 1
   154→    fi
   155→
   156→    log_step "Restoring PostgreSQL from ${backup_file}..."
   157→
   158→    ensure_container_running "${PG_CONTAINER}" "postgres"
   159→
   160→    local db_name="${POSTGRES_DB:-aenews_osx}"
   161→    local db_user="${POSTGRES_USER:-aenews}"
   162→
   163→    gunzip -c "${backup_file}" | docker exec -i "${PG_CONTAINER}" psql \
   164→        -U "${db_user}" \
   165→        -d "${db_name}" \
   166→        2>/dev/null
   167→
   168→    log_success "PostgreSQL restore completed"
   169→}

   171→# ─── Redis Backup ──────────────────────────────────────────────────────
   172→backup_redis() {
   173→    log_step "Backing up Redis..."
   174→
   175→    ensure_container_running "${REDIS_CONTAINER}" "redis"
   176→
   177→    local redis_password="${REDIS_PASSWORD:-aenews_redis_2024}"
   178→    local backup_file="${BACKUP_DIR}/redis_${TIMESTAMP}.rdb"
   179→
   180→    # Trigger BGSAVE
   181→    docker exec "${REDIS_CONTAINER}" redis-cli \
   182→        -a "${redis_password}" \
   183→        BGSAVE 2>/dev/null
   184→
   185→    # Wait for BGSAVE to complete
   186→    local elapsed=0
   187→    while [ $elapsed -lt 30 ]; do
   188→        local bgsave_status
   189→        bgsave_status=$(docker exec "${REDIS_CONTAINER}" redis-cli \
   190→            -a "${redis_password}" \
   191→            LASTSAVE 2>/dev/null)
   192→        sleep 2
   193→        elapsed=$((elapsed + 2))
   194→        # Check if save is done
   195→        local saving
   196→        saving=$(docker exec "${REDIS_CONTAINER}" redis-cli \
   197→            -a "${redis_password}" \
   198→            --no-auth-warning \
   199→            INFO persistence 2>/dev/null | grep "rdb_last_bgsave_status:ok" || echo "")
   200→        if [ -n "$saving" ]; then
   201→            break
   202→        fi
   203→    done
   204→
   205→    # Copy the dump.rdb file out of the container
   206→    docker cp "${REDIS_CONTAINER}:/data/dump.rdb" "${backup_file}" 2>/dev/null || {
   207→        # Fallback: try appendonly.aof
   208→        docker cp "${REDIS_CONTAINER}:/data/appendonly.aof" "${BACKUP_DIR}/redis_${TIMESTAMP}.aof" 2>/dev/null || true
   209→        log_warning "Could not copy Redis dump.rdb, attempted appendonly.aof"
   210→    }
   211→
   212→    if [ -f "${backup_file}" ]; then
   213→        local size
   214→        size=$(du -h "${backup_file}" | cut -f1)
   215→        log_success "Redis backup completed (${size}): ${backup_file}"
   216→    else
   217→        log_warning "Redis backup file not created (Redis may be empty or using AOF only)"
   218→    fi
   219→}

   242→# ─── Neo4j Backup ──────────────────────────────────────────────────────

   244→    log_step "Backing up Neo4j..."
   245→
   246→    ensure_container_running "${NEO4J_CONTAINER}" "neo4j"
   247→
   248→    local neo4j_password="${NEO4J_PASSWORD:-aenews_neo4j_2024}"
   249→    local backup_file="${BACKUP_DIR}/neo4j_${TIMESTAMP}.dump"
   250→
   251→    # Use neo4j-admin dump inside the container
   252→    docker exec "${NEO4J_CONTAINER}" neo4j-admin database dump neo4j \
   253→        --to-path=/tmp/ \
   254→        --overwrite-destination \
   255→        2>/dev/null || {
   256→            # Fallback: try older neo4j-admin syntax
   257→            docker exec "${NEO4J_CONTAINER}" neo4j-admin dump \
   258→                --database=neo4j \
   259→                --to=/tmp/neo4j.dump \
   260→                2>/dev/null || {
   261→                    log_warning "neo4j-admin dump failed. Attempting file copy..."
   262→                    # Fallback: copy data directory
   263→                    docker cp "${NEO4J_CONTAINER}:/data/" "${BACKUP_DIR}/neo4j_data_${TIMESTAMP}/" 2>/dev/null || true
   264→                    log_success "Neo4j backup completed (file copy): ${BACKUP_DIR}/neo4j_data_${TIMESTAMP}/"
   265→                    return 0
   266→                }
   267→        }
   268→
   269→    # Copy dump file from container
   270→    docker cp "${NEO4J_CONTAINER}:/tmp/neo4j.dump" "${backup_file}" 2>/dev/null || \
   271→        docker cp "${NEO4J_CONTAINER}:/tmp/neo4j_"*.dump "${backup_file}" 2>/dev/null || true
   272→
   273→    # Clean up temp file in container
   274→    docker exec "${NEO4J_CONTAINER}" rm -f /tmp/neo4j.dump 2>/dev/null || true
   275→
   276→    if [ -f "${backup_file}" ]; then
   277→        local size
   278→        size=$(du -h "${backup_file}" | cut -f1)
   279→        log_success "Neo4j backup completed (${size}): ${backup_file}"
   280→    else
   281→        log_warning "Neo4j dump file not created"
   282→    fi
   283→}

   285→# ─── Neo4j Restore ─────────────────────────────────────────────────────
   286→restore_neo4j() {
   287→    local backup_file="$1"
   288→
   289→    if [ ! -f "$backup_file" ]; then
   290→        log_error "Neo4j backup file not found: ${backup_file}"
   291→        return 1
   292→    fi
   293→
   294→    log_step "Restoring Neo4j from ${backup_file}..."
   295→
   296→    # Stop Neo4j for restore
   297→    docker_compose stop neo4j 2>/dev/null || true
   298→
   299→    # Copy dump into container
   300→    docker cp "${backup_file}" "${NEO4J_CONTAINER}:/tmp/neo4j.dump"
   301→
   302→    # Run neo4j-admin load
   303→    docker exec "${NEO4J_CONTAINER}" neo4j-admin database load neo4j \
   304→        --from-path=/tmp/ \
   305→        --overwrite-destination \
   306→        2>/dev/null || {
   307→            docker exec "${NEO4J_CONTAINER}" neo4j-admin load \
   308→                --database=neo4j \
   309→                --from=/tmp/neo4j.dump \
   310→                --force \
   311→                2>/dev/null || log_error "Neo4j restore failed"
   312→        }
   313→
   314→    # Clean up and start
   315→    docker exec "${NEO4J_CONTAINER}" rm -f /tmp/neo4j.dump 2>/dev/null || true
   316→    docker_compose start neo4j 2>/dev/null || docker_compose up -d neo4j 2>/dev/null
   317→
   318→    log_success "Neo4j restore completed"
   319→}

   321→# ─── Qdrant Backup ─────────────────────────────────────────────────────
   322→backup_qdrant() {
   323→    log_step "Backing up Qdrant..."
   324→
   325→    ensure_container_running "${QDRANT_CONTAINER}" "qdrant"
   326→
   327→    local backup_file="${BACKUP_DIR}/qdrant_${TIMESTAMP}.tar.gz"
   328→
   329→    # Create snapshot via Qdrant API
   330→    local collections
   331→    collections=$(curl -sf "http://localhost:6333/collections" 2>/dev/null | \
   332→        python3 -c "import sys,json; data=json.load(sys.stdin); [print(c['name']) for c in data.get('result',{}).get('collections',[])]" 2>/dev/null || echo "")
   333→
   334→    if [ -z "$collections" ]; then
   335→        log_warning "No Qdrant collections found or API unavailable. Copying storage directory."
   336→        docker cp "${QDRANT_CONTAINER}:/qdrant/storage/" "${BACKUP_DIR}/qdrant_storage_${TIMESTAMP}/" 2>/dev/null || true
   337→        cd "${BACKUP_DIR}" && tar czf "qdrant_${TIMESTAMP}.tar.gz" "qdrant_storage_${TIMESTAMP}/" 2>/dev/null && \
   338→            rm -rf "qdrant_storage_${TIMESTAMP}/"
   339→        log_success "Qdrant backup completed (storage copy)"
   340→        return 0
   341→    fi
   342→
   343→    # Create snapshots for each collection
   344→    for collection in $collections; do
   345→        log_info "Creating snapshot for collection: ${collection}"
   346→        curl -sf -X POST "http://localhost:6333/collections/${collection}/snapshots" 2>/dev/null || \
   347→            log_warning "Failed to create snapshot for collection: ${collection}"
   348→    done
   349→
   350→    # Copy snapshots out
   351→    docker cp "${QDRANT_CONTAINER}:/qdrant/storage/" "${BACKUP_DIR}/qdrant_storage_${TIMESTAMP}/" 2>/dev/null || true
   352→
   353→    # Compress
   354→    if [ -d "${BACKUP_DIR}/qdrant_storage_${TIMESTAMP}" ]; then
   355→        cd "${BACKUP_DIR}" && tar czf "qdrant_${TIMESTAMP}.tar.gz" "qdrant_storage_${TIMESTAMP}/" 2>/dev/null && \
   356→            rm -rf "qdrant_storage_${TIMESTAMP}/"
   357→    fi
   358→
   359→    if [ -f "${backup_file}" ]; then
   360→        local size
   361→        size=$(du -h "${backup_file}" | cut -f1)
   362→        log_success "Qdrant backup completed (${size}): ${backup_file}"
   363→    else
   364→        log_warning "Qdrant backup file not created"
   365→    fi
   366→}

   447→# ─── Full Backup ───────────────────────────────────────────────────────
   448→full_backup() {
   449→    log_header "AENEWS Agent OS X — Full Backup"
   450→
   451→    load_env
   452→
   453→    # Create backup directory
   454→    mkdir -p "${BACKUP_DIR}"
   455→
   456→    log_info "Backup directory: ${BACKUP_DIR}"
   457→    log_info "Timestamp: ${TIMESTAMP}"
   458→
   459→    # Run all backups
   460→    backup_postgres
   461→    backup_redis
   462→    backup_neo4j
   463→    backup_qdrant
   464→    backup_minio
   465→
   466→    # Create a compressed archive of the entire backup
   467→    log_step "Creating compressed archive..."
   468→    local archive_name="aenews_backup_${TIMESTAMP}.tar.gz"
   469→    cd "${BACKUP_BASE}" && tar czf "${archive_name}" "${TIMESTAMP}/" 2>/dev/null
   470→
   471→    local archive_size
   472→    archive_size=$(du -h "${BACKUP_BASE}/${archive_name}" | cut -f1)
   473→    log_success "Backup archive created (${archive_size}): ${BACKUP_BASE}/${archive_name}"
   474→
   475→    # Write manifest
   476→    cat > "${BACKUP_DIR}/manifest.json" <<EOF
   477→{
   478→    "timestamp": "${TIMESTAMP}",
   479→    "date": "$(date -Iseconds)",
   480→    "type": "full",
   481→    "services": ["postgres", "redis", "neo4j", "qdrant", "minio"],
   482→    "archive": "${archive_name}",
   483→    "generated_by": "aenews-backup-script"
   484→}
   485→EOF
   486→
   487→    # Rotate old backups
   488→    rotate_backups
   489→
   490→    log_header "Backup Complete!"
   491→    echo -e "${GREEN}${BOLD}  Archive: ${BACKUP_BASE}/${archive_name}${NC}"
   492→    echo -e "${GREEN}${BOLD}  Size: ${archive_size}${NC}"
   493→    echo -e "${GREEN}${BOLD}  Retained: ${KEEP_BACKUPS} most recent backups${NC}"
   494→}

   539→
   540→    load_env
   541→
   542→    local backup_path="${1:-}"
   543→

   548→        echo ""
   549→        echo -e "${RED}${BOLD}Enter backup timestamp to restore (YYYYMMDD_HHMMSS): ${NC}"
   550→        read -r backup_ts
   551→
   552→        if [ -z "$backup_ts" ]; then
   553→            log_error "No backup timestamp provided"
   554→            exit 1
   555→        fi
   556→
   557→        backup_path="${BACKUP_BASE}/${backup_ts}"
   558→    fi
   559→
   560→    # Check if it's a compressed archive or a directory
   561→    if [ -f "${backup_path}" ] && [[ "${backup_path}" == *.tar.gz ]]; then
   562→        log_step "Extracting archive: ${backup_path}"
   563→        local extract_dir="${backup_path%.tar.gz}"
   564→        mkdir -p "${extract_dir}"
   565→        tar xzf "${backup_path}" -C "${extract_dir}" --strip-components=1
   566→        backup_path="${extract_dir}"
   567→    fi
   568→
   569→    if [ ! -d "${backup_path}" ]; then
   570→        log_error "Backup directory not found: ${backup_path}"
   571→        exit 1
   572→    fi
   573→
   574→    # Show manifest if available
   575→    if [ -f "${backup_path}/manifest.json" ]; then
   576→        log_info "Backup manifest:"
   577→        cat "${backup_path}/manifest.json"
   578→        echo ""
   579→    fi
   580→
   581→    # Confirmation
   582→    echo -e "${RED}${BOLD}⚠  WARNING: This will overwrite current data!  ⚠${NC}"
   583→    echo -e "${RED}${BOLD}   Make sure you have a current backup before proceeding.${NC}"
   584→    echo ""
   585→    read -rp "$(echo -e ${RED}${BOLD}Type 'RESTORE' to confirm: ${NC})" confirm
   586→
   587→    if [ "$confirm" != "RESTORE" ]; then
   588→        log_info "Restore cancelled"
   589→        exit 0
   590→    fi
   591→
   592→    # Restore each service
   593→    local ts_dir
   594→    ts_dir=$(basename "${backup_path}")
   595→
   596→    # Find backup files
   597→    local pg_file redis_file neo4j_file qdrant_file minio_file
   598→
   599→    pg_file=$(find "${backup_path}" -name "postgres_*.sql.gz" -print -quit 2>/dev/null || echo "")
   600→    redis_file=$(find "${backup_path}" -name "redis_*.rdb" -print -quit 2>/dev/null || echo "")
   601→    neo4j_file=$(find "${backup_path}" -name "neo4j_*.dump" -print -quit 2>/dev/null || echo "")
   602→    qdrant_file=$(find "${backup_path}" -name "qdrant_*.tar.gz" -print -quit 2>/dev/null || echo "")
   603→    minio_file=$(find "${backup_path}" -name "minio_*.tar.gz" -print -quit 2>/dev/null || echo "")
   604→
   605→    # Selective restore
   606→    echo ""
   607→    echo -e "${BOLD}Select services to restore:${NC}"
   608→    echo "  1) All services"
   609→    echo "  2) PostgreSQL only"
   610→    echo "  3) Redis only"
   611→    echo "  4) Neo4j only"
   612→    echo "  5) Qdrant only"
   613→    echo "  6) MinIO only"
   614→    echo "  7) Custom selection"
   615→    read -rp "$(echo -e ${BOLD}Choice [1-7]: ${NC})" choice
   616→
   617→    case "$choice" in
   618→        1)
   619→            [ -n "$pg_file" ] && restore_postgres "$pg_file"
   620→            [ -n "$redis_file" ] && restore_redis "$redis_file"
   621→            [ -n "$neo4j_file" ] && restore_neo4j "$neo4j_file"
   622→            [ -n "$qdrant_file" ] && restore_qdrant "$qdrant_file"
   623→            [ -n "$minio_file" ] && restore_minio "$minio_file"
   624→            ;;
   625→        2) [ -n "$pg_file" ] && restore_postgres "$pg_file" ;;
   626→        3) [ -n "$redis_file" ] && restore_redis "$redis_file" ;;
   627→        4) [ -n "$neo4j_file" ] && restore_neo4j "$neo4j_file" ;;
   628→        5) [ -n "$qdrant_file" ] && restore_qdrant "$qdrant_file" ;;
   629→        6) [ -n "$minio_file" ] && restore_minio "$minio_file" ;;
   630→        7)
   631→            read -rp "Restore PostgreSQL? [y/N]: " r_pg
   632→            read -rp "Restore Redis? [y/N]: " r_redis
   633→            read -rp "Restore Neo4j? [y/N]: " r_neo4j
   634→            read -rp "Restore Qdrant? [y/N]: " r_qdrant
   635→            read -rp "Restore MinIO? [y/N]: " r_minio
   636→
   637→            [[ "$r_pg" == [yY]* ]] && [ -n "$pg_file" ] && restore_postgres "$pg_file"
   638→            [[ "$r_redis" == [yY]* ]] && [ -n "$redis_file" ] && restore_redis "$redis_file"
   639→            [[ "$r_neo4j" == [yY]* ]] && [ -n "$neo4j_file" ] && restore_neo4j "$neo4j_file"
   640→            [[ "$r_qdrant" == [yY]* ]] && [ -n "$qdrant_file" ] && restore_qdrant "$qdrant_file"
   641→            [[ "$r_minio" == [yY]* ]] && [ -n "$minio_file" ] && restore_minio "$minio_file"
   642→            ;;
   643→        *)
   644→            log_error "Invalid choice"
   645→            exit 1
   646→            ;;
   647→    esac
   648→
   649→    log_header "Restore Complete!"
   650→}

   652→# ─── List Backups ──────────────────────────────────────────────────────
   653→list_backups() {
   654→    if [ ! -d "${BACKUP_BASE}" ]; then
   655→        log_info "No backups directory found at ${BACKUP_BASE}"
   656→        return 0
   657→    fi
   658→
   659→    echo -e "${BOLD}Available Backups:${NC}\n"
   660→
   661→    # List directories (raw backups)
   662→    for dir in "${BACKUP_BASE}"/[0-9]*; do
   663→        if [ -d "$dir" ]; then
   664→            local name
   665→            name=$(basename "$dir")
   666→            local manifest="${dir}/manifest.json"
   667→            if [ -f "$manifest" ]; then
   668→                local btype bdate services
   669→                btype=$(python3 -c "import json; d=json.load(open('${manifest}')); print(d.get('type','unknown'))" 2>/dev/null || echo "unknown")
   670→                bdate=$(python3 -c "import json; d=json.load(open('${manifest}')); print(d.get('date','unknown'))" 2>/dev/null || echo "unknown")
   671→                services=$(python3 -c "import json; d=json.load(open('${manifest}')); print(','.join(d.get('services',[])))" 2>/dev/null || echo "unknown")
   672→                echo -e "  ${GREEN}${name}${NC}  type=${btype}  date=${bdate}  services=[${services}]"
   673→            else
   674→                local dir_size
   675→                dir_size=$(du -sh "$dir" 2>/dev/null | cut -f1)
   676→                echo -e "  ${YELLOW}${name}${NC}  size=${dir_size}  (no manifest)"
   677→            fi
   678→        fi
   679→    done
   680→
   681→    # List archives
   682→    for archive in "${BACKUP_BASE}"/aenews_*.tar.gz; do
   683→        if [ -f "$archive" ]; then
   684→            local name
   685→            name=$(basename "$archive")
   686→            local size
   687→            size=$(du -h "$archive" | cut -f1)
   688→            echo -e "  ${CYAN}${name}${NC}  size=${size}  (archive)"
   689→        fi
   690→    done
   691→
   692→    echo ""
   693→    local total_count
   694→    total_count=$(find "${BACKUP_BASE}" -maxdepth 1 -name "[0-9]*" -type d 2>/dev/null | wc -l | tr -d ' ')
   695→    local total_size
   696→    total_size=$(du -sh "${BACKUP_BASE}" 2>/dev/null | cut -f1)
   697→    echo -e "  Total: ${total_count} backups, ${total_size} disk usage"
   698→}

   700→# ─── Rotate Backups ────────────────────────────────────────────────────
   701→rotate_backups() {
   702→    log_step "Rotating backups (keeping last ${KEEP_BACKUPS})..."
   703→
   704→    local count
   705→    count=$(find "${BACKUP_BASE}" -maxdepth 1 -name "[0-9]*" -type d | sort | wc -l | tr -d ' ')
   706→
   707→    if [ "$count" -le "${KEEP_BACKUPS}" ]; then
   708→        log_info "Only ${count} backups found. No rotation needed."
   709→        return 0
   710→    fi
   711→
   712→    local to_delete
   713→    to_delete=$((count - KEEP_BACKUPS))
   714→
   715→    log_info "Removing ${to_delete} old backup(s)..."
   716→
   717→    find "${BACKUP_BASE}" -maxdepth 1 -name "[0-9]*" -type d | sort | head -n "${to_delete}" | while read -r dir; do
   718→        local name
   719→        name=$(basename "$dir")
   720→        log_info "Removing old backup: ${name}"
   721→        rm -rf "$dir"
   722→
   723→        # Also remove the corresponding archive if it exists
   724→        rm -f "${BACKUP_BASE}/aenews_backup_${name}.tar.gz" 2>/dev/null
   725→        rm -f "${BACKUP_BASE}/aenews_db_backup_${name}.tar.gz" 2>/dev/null
   726→    done
   727→
   728→    log_success "Backup rotation complete. Kept last ${KEEP_BACKUPS} backups."
   729→}
