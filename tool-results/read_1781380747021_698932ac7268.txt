   347→  protected async onExecute(input: AgentInput): Promise<AgentOutput> {
   348→    const startTime = Date.now();
   349→    const { action, ...params } = input.payload;
   350→
   351→    if (!action) {
   352→      return this.createAgentOutput(
   353→        input.taskId,
   354→        false,
   355→        null,
   356→        'Missing required parameter: action',
   357→        startTime,
   358→      );
   359→    }
   360→
   361→    const supportedActions = [
   362→      'getCpuUsage',
   363→      'getMemoryUsage',
   364→      'getDiskUsage',
   365→      'getNetworkStats',
   366→      'getSystemInfo',
   367→      'monitorResource',
   368→    ];
   369→
   370→    if (!supportedActions.includes(action)) {
   371→      return this.createAgentOutput(
   372→        input.taskId,
   373→        false,
   374→        null,
   375→        `Unknown system monitor action: ${action}. Supported: ${supportedActions.join(', ')}`,
   376→        startTime,
   377→      );
   378→    }
   379→
   380→    try {
   381→      const tool = this.getTool(action);
   382→      if (!tool) {
   383→        return this.createAgentOutput(
   384→          input.taskId,
   385→          false,
   386→          null,
   387→          `Tool not found: ${action}`,
   388→          startTime,
   389→        );
   390→      }
   391→
   392→      const result = await tool.execute(params);
   393→      return this.createAgentOutput(input.taskId, true, result, undefined, startTime);
   394→    } catch (error) {
   395→      const msg = (error as Error).message;
   396→      this.logger.error(`SystemMonitor execution failed for ${action}: ${msg}`);
   397→      return this.createAgentOutput(input.taskId, false, null, msg, startTime);
   398→    }
   399→  }

   409→  private async getCpuUsage(
   410→    perCore: boolean = false,
   411→    interval: number = 1000,
   412→  ): Promise<{
   413→    overall: number;
   414→    cores?: CpuCore[];
   415→    userPercent: number;
   416→    systemPercent: number;
   417→    idlePercent: number;
   418→    sampledAt: string;
   419→  }> {
   420→    // Fluctuate CPU usage
   421→    this.fluctuateCpu();
   422→
   423→    const userPercent = Math.round(this.systemState.currentCpuUsage * 0.6 * 100) / 100;
   424→    const systemPercent = Math.round(this.systemState.currentCpuUsage * 0.3 * 100) / 100;
   425→    const idlePercent = Math.round((100 - this.systemState.currentCpuUsage) * 100) / 100;
   426→
   427→    const sample: CpuSample = {
   428→      overall: Math.round(this.systemState.currentCpuUsage * 100) / 100,
   429→      userPercent,
   430→      systemPercent,
   431→      idlePercent,
   432→      sampledAt: new Date().toISOString(),
   433→    };
   434→
   435→    if (perCore) {
   436→      sample.cores = [];
   437→      for (let i = 0; i < this.systemState.cpuCores; i++) {
   438→        const coreUsage = Math.max(
   439→          0,
   440→          Math.min(100, this.systemState.currentCpuUsage + (Math.random() - 0.5) * 30),
   441→        );
   442→        sample.cores.push({
   443→          coreIndex: i,
   444→          usagePercent: Math.round(coreUsage * 100) / 100,
   445→          userPercent: Math.round(coreUsage * 0.6 * 100) / 100,
   446→          systemPercent: Math.round(coreUsage * 0.3 * 100) / 100,
   447→          idlePercent: Math.round((100 - coreUsage) * 100) / 100,
   448→        });
   449→      }
   450→    }
   451→
   452→    // Store in history
   453→    this.historicalCpu.push(sample);
   454→    this.trimHistory(this.historicalCpu);
   455→
   456→    this.logger.log(`CPU usage: ${sample.overall}%`);
   457→    return {
   458→      overall: sample.overall,
   459→      cores: sample.cores,
   460→      userPercent,
   461→      systemPercent,
   462→      idlePercent,
   463→      sampledAt: sample.sampledAt,
   464→    };
   465→  }

   467→  private async getMemoryUsage(detailed: boolean = false): Promise<{
   468→    total: number;
   469→    used: number;
   470→    free: number;
   471→    usedPercent: number;
   472→    swapTotal: number;
   473→    swapUsed: number;
   474→    details?: Record<string, number>;
   475→  }> {
   476→    // Fluctuate memory
   477→    this.fluctuateMemory();
   478→
   479→    const total = this.systemState.totalMemoryMb;
   480→    const used = Math.round(this.systemState.currentMemoryUsedMb);
   481→    const free = total - used;
   482→    const usedPercent = Math.round((used / total) * 10000) / 100;
   483→    const swapTotal = 4096;
   484→    const swapUsed = Math.round(this.systemState.currentSwapUsedMb);
   485→
   486→    const result: any = {
   487→      total,
   488→      used,
   489→      free,
   490→      usedPercent,
   491→      swapTotal,
   492→      swapUsed,
   493→    };
   494→
   495→    if (detailed) {
   496→      result.details = {
   497→        buffers: Math.round(used * 0.08),
   498→        cached: Math.round(used * 0.25),
   499→        shared: Math.round(used * 0.05),
   500→        available: Math.round(free + used * 0.25),
   501→        slab: Math.round(used * 0.03),
   502→        kernelStack: Math.round(used * 0.01),
   503→        pageTables: Math.round(used * 0.02),
   504→      };
   505→    }
   506→
   507→    // Store in history
   508→    const sample: MemorySample = {
   509→      totalMb: total,
   510→      usedMb: used,
   511→      freeMb: free,
   512→      usedPercent,
   513→      swapTotalMb: swapTotal,
   514→      swapUsedMb: swapUsed,
   515→      sampledAt: new Date().toISOString(),
   516→    };
   517→    this.historicalMemory.push(sample);
   518→    this.trimHistory(this.historicalMemory);
   519→
   520→    this.logger.log(`Memory usage: ${used}/${total} MB (${usedPercent}%)`);
   521→    return result;
   522→  }

   651→    if (interfaceName) {
   652→      filteredInterfaces = interfaces.filter((iface) => iface.name === interfaceName);
   653→      if (filteredInterfaces.length === 0) {
   654→        throw new Error(
   655→          `Network interface not found: ${interfaceName}. Available: ${interfaces.map((i) => i.name).join(', ')}`,
   656→        );
   657→      }
   658→    }
   659→
   660→    const totalBytesReceived = filteredInterfaces.reduce(
   661→      (sum, iface) => sum + iface.bytesReceived,
   662→      0,
   663→    );
   664→    const totalBytesSent = filteredInterfaces.reduce((sum, iface) => sum + iface.bytesSent, 0);
   665→
   666→    this.logger.log(
   667→      `Network stats: ${filteredInterfaces.length} interfaces, ${(totalBytesReceived / 1024 / 1024).toFixed(1)}MB in, ${(totalBytesSent / 1024 / 1024).toFixed(1)}MB out`,
   668→    );
   669→    return { interfaces: filteredInterfaces, totalBytesReceived, totalBytesSent };
   670→  }

   672→  private async getSystemInfo(): Promise<{
   673→    hostname: string;
   674→    os: string;
   675→    kernel: string;
   676→    arch: string;
   677→    cpuModel: string;
   678→    cpuCores: number;
   679→    totalMemory: number;
   680→    uptime: number;
   681→    loadAverage: number[];
   682→    systemTime: string;
   683→  }> {
   684→    const uptimeMs = Date.now() - this.systemState.bootTime.getTime();
   685→    const uptimeSec = Math.round(uptimeMs / 1000);
   686→
   687→    this.logger.log(`System info: ${this.systemState.hostname}, uptime: ${uptimeSec}s`);
   688→    return {
   689→      hostname: this.systemState.hostname,
   690→      os: this.systemState.os,
   691→      kernel: this.systemState.kernel,
   692→      arch: this.systemState.arch,
   693→      cpuModel: this.systemState.cpuModel,
   694→      cpuCores: this.systemState.cpuCores,
   695→      totalMemory: this.systemState.totalMemoryMb,
   696→      uptime: uptimeSec,
   697→      loadAverage: [
   698→        Math.round((this.systemState.currentCpuUsage / 100) * this.systemState.cpuCores * 100) /
   699→          100,
   700→        Math.round(
   701→          (this.systemState.currentCpuUsage / 100) * this.systemState.cpuCores * 0.9 * 100,
   702→        ) / 100,
   703→        Math.round(
   704→          (this.systemState.currentCpuUsage / 100) * this.systemState.cpuCores * 0.8 * 100,
   705→        ) / 100,
   706→      ],
   707→      systemTime: new Date().toISOString(),
   708→    };
   709→  }

   711→  private async monitorResource(
   712→    resource: string,
   713→    duration: number = 30000,
   714→    interval: number = 2000,
   715→    alertThreshold?: number,
   716→  ): Promise<{
   717→    resource: string;
   718→    samples: MonitorSample[];
   719→    avgUsage: number;
   720→    maxUsage: number;
   721→    minUsage: number;
   722→    alerts: MonitorAlert[];
   723→    monitoredDuration: number;
   724→  }> {
   725→    const validResources = ['cpu', 'memory', 'disk', 'network', 'all'];
   726→    if (!validResources.includes(resource)) {
   727→      throw new Error(
   728→        `Invalid resource: ${resource}. Must be one of: ${validResources.join(', ')}`,
   729→      );
   730→    }
   731→
   732→    if (duration < 1000 || duration > 300000) {
   733→      throw new Error('Duration must be between 1000ms and 300000ms (5 minutes)');
   734→    }
   735→
   736→    if (interval < 500 || interval > 30000) {
   737→      throw new Error('Interval must be between 500ms and 30000ms');
   738→    }
   739→
   740→    const samples: MonitorSample[] = [];
   741→    const alerts: MonitorAlert[] = [];
   742→    const monitorStart = Date.now();
   743→    const sampleCount = Math.min(Math.floor(duration / interval), 150); // Cap at 150 samples
   744→    const effectiveInterval = Math.min(interval, 200); // Compressed for simulation
   745→
   746→    for (let i = 0; i < sampleCount; i++) {
   747→      const sample: MonitorSample = {
   748→        timestamp: new Date().toISOString(),
   749→      };
   750→
   751→      if (resource === 'cpu' || resource === 'all') {
   752→        this.fluctuateCpu();
   753→        sample.cpu = Math.round(this.systemState.currentCpuUsage * 100) / 100;
   754→
   755→        if (alertThreshold && sample.cpu > alertThreshold) {
   756→          alerts.push({
   757→            timestamp: sample.timestamp,
   758→            resource: 'cpu',
   759→            value: sample.cpu,
   760→            threshold: alertThreshold,
   761→            message: `CPU usage ${sample.cpu}% exceeds threshold ${alertThreshold}%`,
   762→          });
   763→        }
   764→      }
   765→
   766→      if (resource === 'memory' || resource === 'all') {
   767→        this.fluctuateMemory();
   768→        const memPercent =
   769→          Math.round(
   770→            (this.systemState.currentMemoryUsedMb / this.systemState.totalMemoryMb) * 10000,
   771→          ) / 100;
   772→        sample.memory = memPercent;
   773→
   774→        if (alertThreshold && memPercent > alertThreshold) {
   775→          alerts.push({
   776→            timestamp: sample.timestamp,
   777→            resource: 'memory',
   778→            value: memPercent,
   779→            threshold: alertThreshold,
   780→            message: `Memory usage ${memPercent}% exceeds threshold ${alertThreshold}%`,
   781→          });
   782→        }
   783→      }
   784→
   785→      if (resource === 'disk' || resource === 'all') {
   786→        sample.disk = 23 + Math.random() * 2;
   787→      }
   788→
   789→      if (resource === 'network' || resource === 'all') {
   790→        this.fluctuateNetwork();
   791→        sample.networkBytesIn = Math.round(Math.random() * 100000);
   792→        sample.networkBytesOut = Math.round(Math.random() * 50000);
   793→      }
   794→
   795→      samples.push(sample);
   796→
   797→      if (i < sampleCount - 1) {
   798→        await this.sleep(effectiveInterval);
   799→      }
   800→    }
   801→
   802→    const monitoredDuration = Date.now() - monitorStart;
   803→
   804→    // Calculate statistics for the primary resource
   805→    const values = samples.map((s) => s.cpu ?? s.memory ?? s.disk ?? 0).filter((v) => v > 0);
   806→
   807→    const avgUsage =
   808→      values.length > 0
   809→        ? Math.round((values.reduce((a, b) => a + b, 0) / values.length) * 100) / 100
   810→        : 0;
   811→    const maxUsage = values.length > 0 ? Math.round(Math.max(...values) * 100) / 100 : 0;
   812→    const minUsage = values.length > 0 ? Math.round(Math.min(...values) * 100) / 100 : 0;
   813→
   814→    // Store monitoring result
   815→    await this.storeInWorkingMemory(
   816→      `sysmon:last:${resource}`,
   817→      {
   818→        avgUsage,
   819→        maxUsage,
   820→        minUsage,
   821→        alertCount: alerts.length,
   822→        sampledAt: new Date().toISOString(),
   823→      },
   824→      300000,
   825→    );
   826→
   827→    this.logger.log(
   828→      `Monitored ${resource}: ${samples.length} samples over ${monitoredDuration}ms, ` +
   829→        `avg: ${avgUsage}%, max: ${maxUsage}%, alerts: ${alerts.length}`,
   830→    );
   831→
   832→    return {
   833→      resource,
   834→      samples,
   835→      avgUsage,
   836→      maxUsage,
   837→      minUsage,
   838→      alerts,
   839→      monitoredDuration,
   840→    };
   841→  }

   845→  private fluctuateCpu(): void {
   846→    const delta = (Math.random() - 0.5) * 10;
   847→    this.systemState.currentCpuUsage = Math.max(
   848→      1,
   849→      Math.min(95, this.systemState.currentCpuUsage + delta),
   850→    );
   851→  }
   852→
   853→  private fluctuateMemory(): void {
   854→    const delta = (Math.random() - 0.5) * 200;
   855→    this.systemState.currentMemoryUsedMb = Math.max(
   856→      1024,
   857→      Math.min(this.systemState.totalMemoryMb * 0.9, this.systemState.currentMemoryUsedMb + delta),
   858→    );
   859→    const swapDelta = (Math.random() - 0.5) * 50;
   860→    this.systemState.currentSwapUsedMb = Math.max(
   861→      0,
   862→      Math.min(2048, this.systemState.currentSwapUsedMb + swapDelta),
   863→    );
   864→  }
   865→
   866→  private fluctuateNetwork(): void {
   867→    this.systemState.networkBytesReceived += Math.round(Math.random() * 500000);
   868→    this.systemState.networkBytesSent += Math.round(Math.random() * 250000);
   869→  }

   871→  private trimHistory<T>(history: T[]): void {
   872→    if (history.length > this.maxHistorySize) {
   873→      history.splice(0, history.length - this.maxHistorySize);
   874→    }
   875→  }
