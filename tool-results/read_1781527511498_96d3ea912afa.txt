   186→/**
   187→ * Generate a valid JWT Authorization header for a given role.
   188→ */
   189→function generateAuthHeader(role: UserRole = UserRole.SUPER_ADMIN): string {
   190→  const jwtService = new JwtService({ secret: 'test-jwt-secret' });
   191→  const token = jwtService.sign({
   192→    sub: 'user-123',
   193→    email: 'admin@aenews.io',
   194→    role,
   195→    tenantId: 'tenant-123',
   196→  });
   197→  return `Bearer ${token}`;
   198→}
