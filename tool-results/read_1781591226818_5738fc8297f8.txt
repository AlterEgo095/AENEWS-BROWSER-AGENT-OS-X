    81→// Status icon helper
    82→function StatusIcon({ status }: { status: string }) {
    83→  switch (status) {
    84→    case 'healthy':
    85→    case 'connected':
    86→    case 'completed':
    87→    case 'active':
    88→      return <CheckCircle2 className="h-4 w-4 text-emerald-400" />;
    89→    case 'degraded':
    90→    case 'initializing':
    91→    case 'coordinating':
    92→    case 'running':
    93→      return <AlertTriangle className="h-4 w-4 text-amber-400" />;
    94→    case 'critical':
    95→    case 'error':
    96→    case 'failed':
    97→      return <XCircle className="h-4 w-4 text-red-400" />;
    98→    case 'offline':
    99→    case 'disconnected':
   100→    case 'passive':
   101→      return <Unplug className="h-4 w-4 text-gray-400" />;
   102→    default:
   103→      return <Activity className="h-4 w-4 text-slate-400" />;
   104→  }
   105→}

   299→// ---------- History Item ----------
   300→function HistoryItem({ item }: { item: OrchestrationHistoryItem }) {

   306→  const statusColorMap: Record<string, string> = {
   307→    completed: 'text-emerald-400',
   308→    running: 'text-amber-400',
   309→    failed: 'text-red-400',
   310→    coordinating: 'text-amber-400',

   314→    <div className="flex items-center gap-3 rounded-lg border border-border bg-background/50 px-3 py-2.5 transition-colors hover:border-primary/20">

   316→        <div className="flex items-center gap-2">
   317→          <p className="text-sm font-medium text-foreground truncate">{item.description}</p>
   318→        </div>
   319→        <div className="flex items-center gap-2 mt-1">
   320→          <span className={cn('rounded-full border px-2 py-0.5 text-[9px] font-semibold uppercase', typeColors[item.type])}>
   321→            {item.type}
   322→          </span>
   323→          {item.pattern && (
   324→            <span className={cn('rounded-full border px-2 py-0.5 text-[9px] font-semibold uppercase', collaborationPatternColors[item.pattern])}>
   325→              {collaborationPatternIcons[item.pattern]} {patternLabels[item.pattern]}
   326→            </span>
   327→          )}
   328→          <span className={cn('text-[10px] font-medium', statusColorMap[item.status] || 'text-slate-400')}>
   329→            {item.status}
   330→          </span>
   331→        </div>
   332→      </div>
   333→      <span className="text-[10px] text-muted-foreground whitespace-nowrap shrink-0">
   334→        {formatRelativeTime(item.createdAt)}
   335→      </span>
   336→    </div>
   337→  );
   338→}

   360→// ---------- Main Page ----------
   361→export default function OrchestrationPage() {
   362→  // Data states
   363→  const [clusterHealth, setClusterHealth] = useState<ClusterHealthInfo[]>([]);
   364→  const [connectors, setConnectors] = useState<UnifiedConnectorInfo[]>([]);
   365→  const [stats, setStats] = useState<OrchestrationStatistics | null>(null);
   366→  const [history, setHistory] = useState<OrchestrationHistoryItem[]>([]);
   367→  const [loading, setLoading] = useState(true);
   368→
   369→  // Form states
   370→  const [decomposeDesc, setDecomposeDesc] = useState('');
   371→  const [decomposeObjectives, setDecomposeObjectives] = useState('');
   372→  const [decomposeResult, setDecomposeResult] = useState<DecompositionResult | null>(null);
   373→  const [decomposing, setDecomposing] = useState(false);
   374→
   375→  const [collabPattern, setCollabPattern] = useState<CollaborationPattern>('pipeline');
   376→  const [collabDesc, setCollabDesc] = useState('');
   377→  const [collabObjectives, setCollabObjectives] = useState('');
   378→  const [collabResult, setCollabResult] = useState<CollaborationResult | null>(null);
   379→  const [launchingCollab, setLaunchingCollab] = useState(false);
   380→
   381→  // Active section for tab-like navigation
   382→  const [activeSection, setActiveSection] = useState<'overview' | 'decompose' | 'collaborate'>('overview');
   383→
   384→  // WebSocket for live updates
   385→  const { subscribe, unsubscribe, lastEvent } = useWebSocket();
   386→
   387→  const handleOrchestrationEvent = useCallback((event: { type: string; payload: Record<string, unknown> }) => {
   388→    if (event.type === 'orchestration:collaboration' || event.type === 'orchestration:coordination') {
   389→      // Refresh history on live events
   390→      fetchHistory();
   391→    }
   392→  }, []);
   393→
   394→  useEffect(() => {
   395→    subscribe('orchestration:collaboration', handleOrchestrationEvent);
   396→    subscribe('orchestration:coordination', handleOrchestrationEvent);
   397→    return () => {
   398→      unsubscribe('orchestration:collaboration', handleOrchestrationEvent);
   399→      unsubscribe('orchestration:coordination', handleOrchestrationEvent);
   400→    };
   401→  }, [subscribe, unsubscribe, handleOrchestrationEvent]);
   402→
   403→  // Log last event for debugging
   404→  useEffect(() => {
   405→    if (lastEvent?.type?.startsWith('orchestration:')) {
   406→      // Live event received — could show toast
   407→    }
   408→  }, [lastEvent]);
   409→
   410→  useEffect(() => {
   411→    fetchAllData();
   412→  }, []);
   413→
   414→  async function fetchAllData() {
   415→    setLoading(true);
   416→    try {
   417→      const results = await Promise.allSettled([
   418→        api.orchestration.getClusterHealth(),
   419→        api.orchestration.getConnectors(),
   420→        api.orchestration.getStatistics(),
   421→        api.orchestration.getHistory(undefined, 10),
   422→      ]);
   423→
   424→      if (results[0].status === 'fulfilled') setClusterHealth(results[0].value);
   425→
   426→      if (results[1].status === 'fulfilled') setConnectors(results[1].value);
   427→
   428→      if (results[2].status === 'fulfilled') setStats(results[2].value);
   429→
   430→      if (results[3].status === 'fulfilled') setHistory(results[3].value);
   431→    } catch {
   432→    } finally {
   433→      setLoading(false);
   434→    }
   435→  }
   436→
   437→  async function fetchHistory() {
   438→    try {
   439→      const result = await api.orchestration.getHistory(undefined, 10);
   440→      setHistory(result);
   441→    } catch {
   442→      // keep existing
   443→    }
   444→  }
   445→
   446→  async function handleDecompose() {
   447→    if (!decomposeDesc.trim()) return;
   448→    setDecomposing(true);
   449→    setDecomposeResult(null);
   450→    try {
   451→      const objectives = decomposeObjectives
   452→        .split('\n')
   453→        .map((o) => o.trim())
   454→        .filter(Boolean);
   455→      const result = await api.orchestration.decompose(null, decomposeDesc, objectives);
   456→      setDecomposeResult(result);
   457→    } catch {
   458→      // Decomposition failed
   459→      setDecomposeResult(null);
   460→    } finally {
   461→      setDecomposing(false);
   462→    }
   463→  }
   464→
   465→  async function handleCollaborate() {
   466→    if (!collabDesc.trim()) return;
   467→    setLaunchingCollab(true);
   468→    setCollabResult(null);
   469→    try {
   470→      const objectives = collabObjectives
   471→        .split('\n')
   472→        .map((o) => o.trim())
   473→        .filter(Boolean);
   474→      const result = await api.orchestration.collaborate(collabPattern, collabDesc, objectives);
   475→      setCollabResult(result);
   476→    } catch {
   477→      // Collaboration failed
   478→      setCollabResult(null);
   479→    } finally {
   480→      setLaunchingCollab(false);
   481→    }
   482→  }
   483→
   484→  const healthyClusters = clusterHealth.filter((c) => c.status === 'healthy').length;
   485→  const connectedConnectors = connectors.filter((c) => c.status === 'connected').length;
   486→  const maxPatternUsage = stats ? Math.max(...Object.values(stats.patternUsage)) : 1;
   487→
   488→  if (loading) {
   489→    return (
   490→      <div className="space-y-6">
   491→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
   492→          {[1, 2, 3, 4].map((i) => (
   493→            <div key={i} className="h-32 animate-shimmer rounded-xl" />
   494→          ))}
   495→        </div>
   496→        <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
   497→          {[1, 2, 3].map((i) => (
   498→            <div key={i} className="h-64 animate-shimmer rounded-xl" />
   499→          ))}
   500→        </div>
   501→      </div>
   502→    );
   503→  }
   504→
   505→  return (
   506→    <div className="space-y-6 animate-slide-in">
   507→      {/* Header */}
   508→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   509→        <div>
   510→          <h1 className="text-2xl font-bold text-foreground">Intelligent Orchestration</h1>
   511→          <p className="text-sm text-muted-foreground">
   512→            Phase 8 — Cluster health, connector management & multi-agent collaboration
   513→          </p>
   514→        </div>
   515→        <button
   516→          onClick={fetchAllData}
   517→          className="flex items-center gap-2 rounded-lg border border-border bg-card px-4 py-2 text-sm font-medium text-foreground transition-colors hover:bg-white/5"
   518→        >
   519→          <RefreshCw className={cn('h-4 w-4', loading && 'animate-spin')} />
   520→          Refresh
   521→        </button>
   522→      </div>
   523→
   524→      {/* Section Tabs */}
   525→      <div className="flex gap-1 rounded-lg border border-border bg-card p-1">
   526→        {([
   527→          { key: 'overview', label: 'Overview', icon: Network },
   528→          { key: 'decompose', label: 'Decompose', icon: GitBranch },
   529→          { key: 'collaborate', label: 'Collaborate', icon: Layers },
   530→        ] as const).map((tab) => (
   531→          <button
   532→            key={tab.key}
   533→            onClick={() => setActiveSection(tab.key)}
   534→            className={cn(
